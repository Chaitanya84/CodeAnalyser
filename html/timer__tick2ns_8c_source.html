28unsigned long
29timer_tick2ns(unsigned long ticks) {
30 int power = (-9) - lsp.qtime.p->timer_scale;
31 unsigned long period = lsp.qtime.p->timer_period;
32 unsigned long ns = 0;
33
34 //
35 //Tricky bit. Try to maintain maximum accuracy without overflowing
36 //
37 for( ;; ) {
38 if(ns == 0) {
39 ns = period * ticks;
40 if((ns / period) != ticks) {
41 //Overflow
42 ns = 0;
43 }
44 }
45 if(power <= 0) break;
46 ns /= 10;
47 period /= 10;
48 --power;
49 }
50 while(power < 0) {
51 ns *= 10;
52 ++power;
53 }
54 return(ns);
55}
