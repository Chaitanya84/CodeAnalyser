40char *find_next_arg(char *str, char ** const end, const unsigned terminate) {
41 while (*str == ' ') {
42 ++str;
43 }
44 if (*str == '\0') {
45 *end = str;
46 return str;
47 }
48
49 char *next = str;
50 while ((*next != ' ') && (*next != '\0')) {
51 ++next;
52 }
53
54 if (*next != '\0') {
55 if (terminate) {
56 *next = '\0';
57 }
58 ++next;
59 }
60 *end = next;
61 return str;
62}

64void cpu_tweak_cmdline(struct bootargs_entry *bap, const char * const name) {
65 // Override this function if you want to adjust the command line arguments
66 // for the given bootstrap exectuable name - see:
67 // bootstrap_arg_adjust()
68 // bootstrap_env_adjust()
69 const static char *next_arg;
70
71 if (!next_arg) { // only do this once
72 char *str = mbox_get_cmdline();
73
74 if (str == NULL) {
75 return;
76 }
77
78 // skip vpu-inserted arguments up to dash or double-space
79 do {
80 next_arg = find_next_arg(str, &str, 0);
81 if (next_arg[0] == '\0') break;
82 if (next_arg[0] == '-') break;
83 if ((next_arg[0] == ' ') && (next_arg[1] == ' ')) break;
84 } while (1);
85
86 for (;;) {
87 if (next_arg[0] == '\0') break;
88 if ((next_arg[0] == '-') && (next_arg[1] == '-')) break;
89 bootstrap_arg_adjust(bap, NULL, next_arg);
90 next_arg = find_next_arg(str, &str, 1);
91 }
92 }
93}
