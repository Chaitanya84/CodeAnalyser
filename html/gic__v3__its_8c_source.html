27/*
28 ===============================================================================
29 its_params
30
31 'its_params' holds the initialization parameters used by gic_v3_its_initialize()
32 for various ITS settings. A set of APIs exist which can be called prior to
33 gic_v3_its_initialize() in order to modify/customize these parameters
34
35 NOTE:
36
37 Although more than one ITS instance is supported by the APIs, implementation
38 wise, we currently support only 1 ITS. Many of the variables and the callouts
39 will all need updates if more than one is supported but most of the other
40 infrastructure is in place
41 Because we have not seen a multi-ITS implementation, we don't even know what
42 the end result would look like
43
44*/
45static arm_gic_its_params_t its_params[1] =
46{
47 [0 ... (NUM_ELTS(its_params) - 1)] = ARM_GIC_ITS_PARAMS_DFLT_INIT(),
48};

85void gic_v3_its_initialize(const unsigned its_num)
86{
87 ASSERT(its_num < NUM_ELTS(its_params));
88 ASSERT(its_params[its_num].gits_paddr_base != 0);
89
90 const paddr_t gits_paddr_base = its_params[its_num].gits_paddr_base;
91
92 its_params[its_num].gits_vaddr_base = callout_io_map(0x10000, gits_paddr_base);
93
94 if (debug_flag > 1)
95 {
96 kprintf("Setting up ITS based LPIs\nGITS @ %L, callout vaddr %L\n",
97 gits_paddr_base, its_params[its_num].gits_vaddr_base);
98 }
99 /*
100 * Current assumptions and actions
101 * - Only 1 ITS in the system will be configured
102 * - A zeroed Device Table will be allocated to hold a predefined
103 * number of Device IDs
104 * - A collection Table will be created with an entry for each PE
105 * as required (may not require memory based collection table)
106 * - An ITS command queue will be created and initialized to empty
107 */
108 /* set up ITS Device Table */
109 /* make sure the ITS is disabled and quiescent */
110 _Uint32t gits_ctlr = in32(gits_paddr_base + 0);
111 gits_ctlr &= ~((_Uint32t)1 << 0); /* disabled */
112 out32(gits_paddr_base + 0, gits_ctlr);
113 do
114 {
115 gits_ctlr = in32(gits_paddr_base + 0);
116 } while ((gits_ctlr & ((_Uint32t)1 << 31)) == 0);
117
118 unsigned dt_entry_size = 0; /* in bytes */
119 int gits_baser_dt_idx = -1;
120 unsigned ct_entry_size = 0; /* in bytes */
121 int gits_baser_ct_idx = -1;
122
123 const _Uint64t gits_typer = in64(gits_paddr_base + 8);
124 const uint_t devid_bits = (((gits_typer >> 13) & 0x1Fu) + 1);
125 ASSERT(devid_bits < 32); /* protect the next calculation. Will need a larger type otherwise */
126 const unsigned num_device_ids = (_Uint32t)1 << devid_bits;
127
128 /*
129 * process each GITS_BASER register to obtain entry sizes for each table
130 * type. Currently we only look for the Device and Collection Tables
131 */
132 for (unsigned i=0; i<8; i++)
133 {
134 const _Uint64t gits_baser = in64(gits_paddr_base + 0x100 + (i * 8));
135 const _Uint64t table_type = (gits_baser >> 56) & 0x7;
136
137 if (table_type == 1)
138 {
139 /* Device Table */
140 ASSERT(dt_entry_size == 0); /* can only be one of these table types */
141 dt_entry_size = (unsigned)((gits_baser >> 48) & 0x1Fu) + 1;
142 gits_baser_dt_idx = i;
143 if (debug_flag > 1) kprintf("ITS Device Table at GITS_BASER%d\n", gits_baser_dt_idx);
144 }
145 else if (table_type == 4)
146 {
147 /* Collection Table */
148 ASSERT(ct_entry_size == 0); /* can only be one of these table types */
149 ct_entry_size = (unsigned)((gits_baser >> 48) & 0x1Fu) + 1;
150 gits_baser_ct_idx = i;
151 if (debug_flag > 1) kprintf("ITS Collection Table at GITS_BASER%d\n", gits_baser_ct_idx);
152 }
153 if ((gits_baser_dt_idx > 0) && (gits_baser_ct_idx > 0)) break;
154 }
155 const unsigned dt_page_size = its_params[its_num].device_tbl.page_size; /* 4K, 16K or 64K */
156 const unsigned num_dt_pages = max(1, (num_device_ids * dt_entry_size) / dt_page_size);
157 /* only 8 bits available for num pages. If this asserts, increase device_tbl.page_size
158 * by calling gic_v3_its_set_dt_page_size() before calling gic_v3_initialize() */
159 ASSERT(num_dt_pages <= 0x100);
160
161 _Uint64t dt = calloc_ram(num_dt_pages * dt_page_size, dt_page_size);
162
163 /* make sure the address is valid */
164 ASSERT(dt != NULL_PADDR_STARTUP);
165 ASSERT((dt & (_Uint64t)0xFFFFFFFFF000) == dt);
166
167 if (debug_flag > 1)
168 {
169 kprintf("Device Table @ %L, size 0x%x; num_device_ids: %d, entry_size: %d\n",
170 dt, num_dt_pages * dt_page_size, num_device_ids, dt_entry_size);
171 }
172
173 _Uint64t dt_page_size_bits;
174 if (dt_page_size == 4096) dt_page_size_bits = 0;
175 else if (dt_page_size == 16384) dt_page_size_bits = 1;
176 else if (dt_page_size == 65536) dt_page_size_bits = 2;
177 else {ASSERT(0);}
178
179 /* set page size bits */
180 dt |= (dt_page_size_bits << 8);
181 /* set the number of pages */
182 dt |= (((_Uint64t)(num_dt_pages - 1) & 0xFFu) << 0);
183 /*
184 * double check the size of the 'dt' again. The page_size (bits [9:8] of the
185 * BASER register) might be readonly and so even if the ASSERT(num_dt_pages <= 0x100)
186 * above doesn't fire, we may not be able to use a larger page_size. If not then
187 * the number of device IDs will need to be limited some other way
188 */
189 {
190 const unsigned page_size_bits = (dt >> 8) & (_Uint64t)0x3;
191 const unsigned page_size = 0x1000 << (page_size_bits * 2);
192 const unsigned num_pages = ((dt >> 0) & (_Uint64t)0xFF) + 1;
193 /* make sure what we programmed is what we got */
194 ASSERT((page_size * num_pages) == (num_dt_pages * dt_page_size));
195 }
196
197 /*
198 * Note that in some implementations, the shareability bits are non-modifiable
199 * and if they are 0b00 or 0b11, then cacheability should be set to
200 * non-cacheable so we check whether shareability is settable and adjust
201 * the parameters accordingly
202 */
203 unsigned dt_share_bits = (in64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8)) >> 10) & 0x3;
204 const unsigned flipped_cur_share_bits = ~dt_share_bits & 0x3;
205 out64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8), (_Uint64t)flipped_cur_share_bits << 10);
206 dt_share_bits = (in64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8)) >> 10) & 0x3;
207 const unsigned dt_share_bits_settable = (dt_share_bits == flipped_cur_share_bits);
208
209 if (dt_share_bits_settable == 0)
210 {
211 if (debug_flag > 1)
212 {
213 kprintf("GITS_BASER<%d> shareability bits non-settable, use non-cached attributes\n",
214 gits_baser_dt_idx);
215 }
216 its_params[its_num].device_tbl.params.shareability = arm_gic_tbl_shareability_NONE;
217 its_params[its_num].device_tbl.params.cacheability.inner = arm_gic_tbl_cacheability_NONE;
218 its_params[its_num].device_tbl.params.cacheability.outer = arm_gic_tbl_cacheability_OUTER_IS_NORMAL_INNER;
219 }
220 /* set shareability and cacheability */
221 dt |= ((_Uint64t)its_params[its_num].device_tbl.params.shareability << 10);
222 dt |= ((_Uint64t)its_params[its_num].device_tbl.params.cacheability.inner << 59);
223 dt |= ((_Uint64t)its_params[its_num].device_tbl.params.cacheability.outer << 53);
224
225 /* indirect (1 level, flat) table */
226 dt &= ~((_Uint64t)1 << 62);
227 /* mark it valid */
228 dt |= ((_Uint64t)1 << 63);
229
230 /* write the GITS_BASER for the Device Table */
231 DCACHE_FLUSH_FOR_GIC_ITS(MAKE_1TO1_PTR(dt), num_dt_pages * dt_page_size);
232 out64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8), dt);
233 /* the Page_Size field could be implementation defined read-only so assert
234 * if we do not get what we expect */
235 ASSERT(((in64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8)) >> 8) & 3) == dt_page_size_bits);
236 if (debug_flag > 1)
237 {
238 kprintf("GITS_BASER<%d> for Device Table set to %L\n",
239 gits_baser_dt_idx, in64(gits_paddr_base + 0x100 + (gits_baser_dt_idx * 8)));
240 }
241
242 /* handle the collection table */
243 if (debug_flag > 1) kprintf("Memory based collection table %ssupported\n", (gits_baser_ct_idx < 0) ? "not " : " ");
244
245 /* collect some information from the GITS_TYPER */
246 const unsigned num_hw_collections = (gits_typer >> 24) & 0xFFu;
247 its_params[its_num].num_collections = num_hw_collections;
248
249 if (gits_baser_ct_idx >= 0)
250 {
251 const unsigned cil = (gits_typer >> 36) & 1;
252 const unsigned cid_bits = (cil == 1) ? (((gits_typer >> 32) & 0xF) + 1) : 16;
253 /*
254 * I suppose we can have more collection IDs than there are PE/CPUs in
255 * the system but currently we use a 1:1 mapping of ICID to CPU so
256 * there is no point in creating more than we need and also avoids sending
257 * MAPC commands to non-existent PEs
258 */
259 const unsigned num_mem_collections = min((1u << cid_bits), lsp.syspage.p->num_cpu);
260
261 its_params[its_num].collection_tbl.num_collections = num_mem_collections;
262 its_params[its_num].num_collections += num_mem_collections;
263
264 if (debug_flag > 1)
265 {
266 if (lsp.syspage.p->num_cpu < (1u << cid_bits))
267 {
268 kprintf("Memory collections supported by CIDbits is %u but limiting to num CPUs of %u\n",
269 (1u << cid_bits), lsp.syspage.p->num_cpu);
270 }
271 kprintf("%u memory and %u hardware based collections supported, total %u\n",
272 num_mem_collections, num_hw_collections, its_params[its_num].num_collections);
273 }
274 /*
275 * Allocate memory for the Collection Table
276 * Note, since memory collections are supported, the CCT bit (2) in
277 * conjunction with 'num_hw_collections' indicates whether collections
278 * are in hardware and memory, or just memory.
279 * I have only seen implementations which support one or the other and
280 * am not sure what to do in case of when they are in both so we'll
281 * assert if this occurs
282 */
283 ASSERT(num_hw_collections == 0); /* don't really care about CCT bit */
284
285 const unsigned ct_page_size = its_params[its_num].collection_tbl.page_size; /* 4K, 16K or 64K */
286 const unsigned num_ct_pages = max(1, (num_mem_collections * ct_entry_size) / ct_page_size);
287 /* only 8 bits available for num pages. If this asserts, increase collection_tbl.page_size
288 * by calling gic_v3_its_set_ct_page_size() before calling gic_v3_initialize() */
289 ASSERT(num_ct_pages <= 0x100);
290
291 _Uint64t ct = calloc_ram(num_ct_pages * ct_page_size, ct_page_size);
292
293 /* make sure the address is valid */
294 ASSERT(ct != NULL_PADDR_STARTUP);
295 ASSERT((ct & (_Uint64t)0xFFFFFFFFF000) == ct);
296
297 if (debug_flag > 1)
298 {
299 kprintf("Collection Table @ %L, size 0x%x; num_mem_collections: %d, entry_size: %d\n",
300 ct, num_ct_pages * ct_page_size, num_mem_collections, ct_entry_size);
301 }
302
303 _Uint64t ct_page_size_bits;
304 if (ct_page_size == 4096) ct_page_size_bits = 0;
305 else if (ct_page_size == 16384) ct_page_size_bits = 1;
306 else if (ct_page_size == 65536) ct_page_size_bits = 2;
307 else {ASSERT(0);}
308
309 /* set page size bits */
310 ct |= (ct_page_size_bits << 8);
311 /* set the number of pages */
312 ct |= (((_Uint64t)(num_ct_pages - 1) & 0xFFu) << 0);
313 /*
314 * double check the size of the 'ct' again. The page_size (bits [9:8] of the
315 * BASER register) might be readonly and so even if the ASSERT(num_ct_pages <= 0x100)
316 * above doesn't fire, we may not be able to use a larger page_size. If not then
317 * the number of collection IDs will need to be limited some other way
318 */
319 {
320 const unsigned page_size_bits = (ct >> 8) & (_Uint64t)0x3;
321 const unsigned page_size = 0x1000 << (page_size_bits * 2);
322 const unsigned num_pages = ((ct >> 0) & (_Uint64t)0xFF) + 1;
323 /* make sure what we programmed is what we got */
324 ASSERT((page_size * num_pages) == (num_ct_pages * ct_page_size));
325 }
326 /*
327 * Note that in some implementations, the shareability bits are non-modifiable
328 * and if they are 0b00 or 0b11, then cacheability should be set to
329 * non-cacheable so we check whether shareability is settable and adjust
330 * the parameters accordingly
331 */
332 unsigned ct_share_bits = (in64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8)) >> 10) & 0x3;
333 const unsigned flipped_cur_share_bits = ~ct_share_bits & 0x3;
334 out64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8), (_Uint64t)flipped_cur_share_bits << 10);
335 ct_share_bits = (in64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8)) >> 10) & 0x3;
336 const unsigned ct_share_bits_settable = (ct_share_bits == flipped_cur_share_bits);
337
338 if (ct_share_bits_settable == 0)
339 {
340 if (debug_flag > 1)
341 {
342 kprintf("GITS_BASER<%d> shareability bits non-settable, use non-cached attributes\n",
343 gits_baser_ct_idx);
344 }
345 its_params[its_num].collection_tbl.params.shareability = arm_gic_tbl_shareability_NONE;
346 its_params[its_num].collection_tbl.params.cacheability.inner = arm_gic_tbl_cacheability_NONE;
347 its_params[its_num].collection_tbl.params.cacheability.outer = arm_gic_tbl_cacheability_OUTER_IS_NORMAL_INNER;
348 }
349 /* set shareability and cacheability */
350 ct |= ((_Uint64t)its_params[its_num].collection_tbl.params.shareability << 10);
351 ct |= ((_Uint64t)its_params[its_num].collection_tbl.params.cacheability.inner << 59);
352 ct |= ((_Uint64t)its_params[its_num].collection_tbl.params.cacheability.outer << 53);
353
354 /* indirect (1 level, flat) table */
355 ct &= ~((_Uint64t)1 << 62);
356 /* mark it valid */
357 ct |= ((_Uint64t)1 << 63);
358
359 /* write the GITS_BASER for the Collection Table */
360 DCACHE_FLUSH_FOR_GIC_ITS(MAKE_1TO1_PTR(ct), num_ct_pages * ct_page_size);
361 out64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8), ct);
362 /* the Page_Size field could be implementation defined read-only so assert
363 * if we do not get what we expect */
364 ASSERT(((in64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8)) >> 8) & 3) == ct_page_size_bits);
365 if (debug_flag > 1)
366 {
367 kprintf("GITS_BASER<%d> for Collection Table set to %L\n",
368 gits_baser_ct_idx, in64(gits_paddr_base + 0x100 + (gits_baser_ct_idx * 8)));
369 }
370 }
371 ASSERT(its_params[its_num].num_collections > 0);
372 /*
373 * we need to issue the MAPC command to map ICIDs to CPUs. That will be
374 * done from gic_v3_gicc_init() when we also invalidate any entries in the
375 * redistributors
376 */
377
378 /*
379 * Create the ITS command queue. It must be a multiple of 4KB and
380 * aligned to a 64KB boundary. One page provides for 128 entries
381 * The first time a command queue is allocated, we'll also allocate an
382 * extra page at the end of it for implementing the per ITS command queue
383 * spinlocks. Each lock will exist in its own 128 byte block ensuring no
384 * cache thrashing should there be more than one ITS instance
385 */
386 const size_t extra = (cmd_q_lock_base_vaddr == NULL) ? 4096 : 0;
387 const size_t its_cmd_q_size = its_params[its_num].cmd_q.num_4k_pages * 4096;
388 _Uint64t its_cmd_q = calloc_ram(its_cmd_q_size + extra, 65536);
389
390 /* make sure the address is valid */
391 ASSERT(its_cmd_q != NULL_PADDR_STARTUP);
392 ASSERT((its_cmd_q & (_Uint64t)0xFFFFFFFFF0000) == its_cmd_q);
393
394 its_params[its_num].cmd_q.vaddr = (_Uintptrt)callout_memory_map(its_cmd_q_size, its_cmd_q, its_params[its_num].cmd_q.mmap_flags);
395
396 if (debug_flag > 1)
397 {
398 kprintf("its_cmd_q @ %L, callout vaddr: %L, %u entries\n", its_cmd_q,
399 its_params[its_num].cmd_q.vaddr, its_cmd_q_size / 32);
400 }
401
402 /* lock page must be cachable */
403 if (cmd_q_lock_base_vaddr == NULL)
404 {
405 /*
406 * put the lock page for all ITS instances at the end of the command
407 * queue of the first instance
408 */
409 ASSERT(cmd_q_lock_base_paddr == NULL_PADDR);
410 cmd_q_lock_base_paddr = its_cmd_q + its_cmd_q_size;
411 ASSERT(extra != 0);
412 cmd_q_lock_base_vaddr = callout_memory_map(extra, cmd_q_lock_base_paddr, PROT_READ | PROT_WRITE);
413 if (debug_flag > 1)
414 {
415 kprintf("cmd_q_lock page @ %L, vaddr: %L allocated with ITS %u\n",
416 its_cmd_q + its_cmd_q_size, cmd_q_lock_base_vaddr, its_num);
417 }
418 }
419 its_params[its_num].cmd_q.lock_vaddr = (void *)((_Uintptrt)cmd_q_lock_base_vaddr + (its_num * 128));
420 arm_gic_its_lock_t *lock = (arm_gic_its_lock_t*)( cmd_q_lock_base_paddr + (its_num * 128) ) ;
421 lock->version = ITS_LOCK_VERSION;
422
423 if (debug_flag > 1)
424 {
425 kprintf("cmd_q_lock_base_vaddr @ %L, callout vaddr: %L for ITS %u\n",
426 cmd_q_lock_base_vaddr, its_params[its_num].cmd_q.lock_vaddr, its_num);
427 }
428
429 _Uint64t gits_cbaser = in64(gits_paddr_base + 0x80);
430 gits_cbaser &= ~((_Uint64t)0xFFFFFFFFFF000); /* align check already done in above ASSERT */
431 gits_cbaser |= its_cmd_q;
432 /* set the number of pages */
433 gits_cbaser &= ~((_Uint64t)0xFF << 0);
434 gits_cbaser |= ((_Uint64t)(its_params[its_num].cmd_q.num_4k_pages - 1) << 0); /* num pages field is less 1 */
435 /*
436 * set the shareability as outer shareable.
437 * Note that in some implementations, these bits are non-modifiable
438 * and if they are 0b00 or 0b11, then cacheability should be set to
439 * non-cacheable
440 */
441 gits_cbaser &= ~((_Uint64t)3 << 10);
442 /* check whether shareability is settable */
443 unsigned cmd_q_share_bits = (in64(gits_paddr_base + 0x80) >> 10) & 0x3;
444 const unsigned flipped_cmd_q_share_bits = ~cmd_q_share_bits & 0x3;
445 out64(gits_paddr_base + 0x80, (_Uint64t)flipped_cmd_q_share_bits << 10);
446 cmd_q_share_bits = (in64(gits_paddr_base + 0x80) >> 10) & 0x3;
447 const unsigned cmd_q_share_bits_settable = (cmd_q_share_bits == flipped_cmd_q_share_bits);
448
449 if (cmd_q_share_bits_settable == 0)
450 {
451 if (debug_flag > 1) kprintf("CBASER shareability bits non-settable, use non-cached attributes\n");
452
453 its_params[its_num].cmd_q.params.shareability = arm_gic_tbl_shareability_NONE;
454 its_params[its_num].cmd_q.params.cacheability.inner = arm_gic_tbl_cacheability_NONE;
455 its_params[its_num].cmd_q.params.cacheability.outer = arm_gic_tbl_cacheability_OUTER_IS_NORMAL_INNER;
456 }
457 gits_cbaser |= ((_Uint64t)its_params[its_num].cmd_q.params.shareability << 10);
458 /* set the inner and outer cacheability the same and as read/write allocate, write-back */
459 gits_cbaser &= ~((_Uint64t)7 << 59);
460 gits_cbaser |= ((_Uint64t)its_params[its_num].cmd_q.params.cacheability.inner << 59);
461 gits_cbaser &= ~((_Uint64t)7 << 53);
462 gits_cbaser |= ((_Uint64t)its_params[its_num].cmd_q.params.cacheability.outer << 53);
463
464 /* mark the table as valid */
465 gits_cbaser |= ((_Uint64t)1 << 63);
466
467 DCACHE_FLUSH_FOR_GIC_ITS(MAKE_1TO1_PTR(its_cmd_q), its_cmd_q_size + extra);
468
469 /* before writing out the GITS_CBASER value, make sure queue looks empty
470 * Note that CREADR will be zeroed when CBASER is written */
471 out64(gits_paddr_base + 0x88, 0);
472 out64(gits_paddr_base + 0x80, gits_cbaser);
473
474 /*
475 * Finally, create the HWINFO entry for the ITS. For each ITS instance, we
476 * add an entry with the name 'GIC_ITS' followed by 2 location tags, one
477 * for the paddr of the GIC ITS registers and one for the paddr of the
478 * command queue lock
479 */
480 const unsigned hwi_its_off = hwidev_add("GIC_ITS", 0, HWI_NULL_OFF);
481 if (hwi_its_off != HWI_NULL_OFF)
482 {
483 hwitag_add_location(hwi_its_off, gits_paddr_base, 0x20000, 0, 0);
484 const paddr_t its_cmd_q_lock_addr = cmd_q_lock_base_paddr + (its_num * 128);
485 hwitag_add_location(hwi_its_off, its_cmd_q_lock_addr, 4, 0, 0);
486 if (debug_flag > 2)
487 {
488 kprintf("added HWI tag for GIC_ITS/lock = %L/%L\n",
489 gits_paddr_base, its_cmd_q_lock_addr);
490 }
491#if 0 /* test for > 1 entry */
492 {
493 const unsigned hwi_its_off = hwidev_add("GIC_ITS", 0, HWI_NULL_OFF);
494 if (hwi_its_off != HWI_NULL_OFF)
495 {
496 hwitag_add_location(hwi_its_off, gits_paddr_base + 0x4444, 0x20000, 0, 0);
497 const paddr_t its_cmd_q_lock_addr = cmd_q_lock_base_paddr + ((its_num + 1) * 128);
498 hwitag_add_location(hwi_its_off, its_cmd_q_lock_addr, 4, 0, 0);

725static unsigned next_cmd_q_slot(unsigned * const cur_slot, const unsigned num_slots, _Uintptrt * const q_addr, const _Uintptrt q_base)
726{
727 const unsigned next_slot = (*cur_slot + 1) % num_slots;
728 if (next_slot == 0) *q_addr = q_base;
729//kprintf("cur_slot: %u, next_slot: %u, q_addr: 0x%L\n", *cur_slot, next_slot, *q_addr);

733void gic_v3_its_mapc(const unsigned its_num, const unsigned cpu, const unsigned gicr_base_for_cpu)
734{
735 ASSERT(its_num < NUM_ELTS(its_params));
736 ASSERT(its_params[its_num].gits_paddr_base != 0);
737
738 /* enable the ITS as required */
739 const paddr_t gits_paddr_base = its_params[its_num].gits_paddr_base;
740 _Uint32t gits_ctlr = in32(gits_paddr_base + 0);
741 if ((gits_ctlr & 0x1) == 0)
742 {
743 if (debug_flag > 0) kprintf("Enabling ITS %u\n", its_num);
744 /* wait for the ITS to be quiescent */
745 while ((gits_ctlr & ((_Uint32t)1 << 31)) == 0)
746 {
747 gits_ctlr = in32(gits_paddr_base + 0);
748 }
749 gits_ctlr |= ((_Uint32t)1 << 0); /* enabled */
750 out32(gits_paddr_base + 0, gits_ctlr);
751 }
752 else if (debug_flag > 0) kprintf("ITS %u already Enabled\n", its_num);
753
754 const unsigned its_cmd_q_num_slots = (its_params[its_num].cmd_q.num_4k_pages * 4096) / 32;
755 const _Uintptrt tmp_its_cmd_q_base = (_Uintptrt)MAKE_1TO1_PTR(in64(gits_paddr_base + 0x80) & 0xFFFFFFFFFF000);
756 if (debug_flag > 0) kprintf("ITS queue at %L, num slots: %u\n", (_Uint64t)tmp_its_cmd_q_base, its_cmd_q_num_slots);
757
758/*
759 * these 2 macros presuppose the existence of 'its_cmd_q_num_slots', 'tmp_its_cmd_q_base'
760 * and 'gits_paddr_base' variables as above. They are a convenience for issuing multiple
761 * commands to the queue and then executing them
762*/
763#define NEXT_CMD_Q_SLOT(_cur_slot_, _q_addr_) \
764 next_cmd_q_slot(&(_cur_slot_), its_cmd_q_num_slots, (_Uintptrt *)(&(_q_addr_)), tmp_its_cmd_q_base)
765
766#define EXECUTE_CMD_Q(_cur_slot_, _q_addr_) \
767 do { \
768 const unsigned next_slot = NEXT_CMD_Q_SLOT((_cur_slot_), (_q_addr_)); \
769 if (debug_flag > 0) kprintf("update CWRITER to 0x%x\n", next_slot * 32); \
770 /* set the Retry bit (bit 0) in case the ITS is stalled. No harm if its not */ \
771 out64(gits_paddr_base + 0x88, (next_slot * 32) | 1); \
772 } while(0)
773
774
775 const _Uint64t gits_typer = in64(gits_paddr_base + 8);
776 const unsigned cil_bit = (gits_typer >> 36) & 1;
777 const unsigned icid_bits = (cil_bit == 1) ? (((gits_typer >> 32) & 0xFu) + 1) : 16;
778 const _Uint64t icid_mask = ((_Uint64t)1 << icid_bits) - 1;
779 const unsigned pta_bit = (gits_typer >> 19) & 1;
780 const _Uint64t RDbase_mask = (((_Uint64t)1 << 52) - 1) ^ 0xFFFFu; /* 64KB aligned */
781 const unsigned processor_num = (in64(gicr_base_for_cpu + ARM_GICR_TYPER) >> 8) & 0xFFFF;
782 const _Uint64t RDbase = (pta_bit == 1) ? gicr_base_for_cpu : (processor_num << max(16, icid_bits));
783 const unsigned icid = cpu; /* 1:1 relationship */
784
785/*
786kprintf("CPU %u, Processor # %u, GICR base: %L, GITS_typer: %L, icid_mask: %L, icid: %u, RDbase_mask: %L, RDbase: %L\n",
787 cpu, processor_num, gicr_base_for_cpu, gits_typer, icid_mask, icid, RDbase_mask, RDbase);
788*/
789 /*
790 * Initialize the 2 required variables (cur_slot and tmp_its_cmd_q) for
791 * issuing the MAPC, SYNC and INVALL commands based on current ITS settings
792 */
793 unsigned cur_slot = in64(gits_paddr_base + 0x88) / 32;
794 ASSERT(cur_slot < its_cmd_q_num_slots);
795 _Uint64t *tmp_its_cmd_q = (_Uint64t *)(tmp_its_cmd_q_base + (cur_slot * 32));
796 const _Uint64t *addr_for_flush = MAKE_1TO1_PTR(tmp_its_cmd_q);
797 /*
798 * now we need to issue the MAPC command to map the ICID to RDbase
799 * If GITS_TYPER[19] is 0, the RDbase is the Processor number otherwise as
800 * provided in the GICR_TYPER register otherwise its the physical address of
801 * the distributor for the processor. Regardless we will always create a 1:1
802 * mapping between the ICID and the CPU number. That is, ICID0 --> CPU0,
803 * ICID1 --> CPU1, etc.
804 */
805 if (debug_flag > 0) kprintf("Issue MAPC/SYNC/INVALL commands for ICID %u\n", icid);
806
807 *tmp_its_cmd_q++ = 0x9; /* MAPC */
808 *tmp_its_cmd_q++ = 0;
809 *tmp_its_cmd_q++ = (icid & icid_mask) | (RDbase & RDbase_mask) | ((_Uint64t)1 << 63);
810 *tmp_its_cmd_q++ = 0;
811
812 NEXT_CMD_Q_SLOT(cur_slot, tmp_its_cmd_q);
813
814 *tmp_its_cmd_q++ = 0x5; /* SYNC */
815 *tmp_its_cmd_q++ = 0;
816 *tmp_its_cmd_q++ = (RDbase & RDbase_mask);
817 *tmp_its_cmd_q++ = 0;
818
819 NEXT_CMD_Q_SLOT(cur_slot, tmp_its_cmd_q);
820
821 *tmp_its_cmd_q++ = 0xD; /* INVALL */
822 *tmp_its_cmd_q++ = 0;
823 *tmp_its_cmd_q++ = icid & icid_mask;
824 *tmp_its_cmd_q++ = 0;
825
826 DCACHE_FLUSH_FOR_GIC_ITS(addr_for_flush, (uintptr_t)tmp_its_cmd_q - (uintptr_t)addr_for_flush);
827 /* execute the commands */
828 EXECUTE_CMD_Q(cur_slot, tmp_its_cmd_q);
829
830 if (debug_flag > 0) kprintf("Waiting for all commands to be processed");
831
832 _Uint8t loop = 0;
833 while(++loop != 0) /* give it (1u << (8 * sizeof(loop))) -1 tries */
834 {
835 const _Uint64t creader = in64(gits_paddr_base + 0x90);
836 const _Uint64t cwriter = in64(gits_paddr_base + 0x88);
837 if (debug_flag > 1)
838 {
839 kprintf("%s%u GITS_CWRITER/CREADER at offsets %d/%d",
840 (loop == 1) ? "\n\t" : "\r\t", loop, cwriter, creader);
841 }
842 if (creader == cwriter) break;
843 else if ((creader & 1) != 0) out64(gits_paddr_base + 0x88, cwriter | 1);
844 }
845 if (debug_flag > 0)
846 {
847 if (loop == 0) kprintf(" ... Failed in %u tries\n", (1u << (8 * sizeof(loop))) - 1);
848 else kprintf(" ... Done in %u tries\n", loop);
849 }
850 ASSERT(loop != 0);
851}
