35int pwm_fan_set(pwm_fan_dev_t *dev , const uint32_t pwm_ctrl)
36{
37 if (pwm_ctrl > PWM_CTRL_MAX_VAL) {
38 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Invalid value: %u", __func__, pwm_ctrl);
39 return EINVAL;
40 }
41
42 if (pwm_ctrl == 0U) {
43 /* if fan is on, turn it off */
44 if (dev->pwm_ctrl != 0U) {
45 out32(dev->vbase + RP1_PWM_CHAN3_DUTY, 0);
46 dev->pwm_ctrl = 0;
47 if (verbose > 0) {
48 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm channel3 duty register: 0", __func__);
49 }
50 }
51 }
52 else {
53 uint32_t val;
54 dev->pwm_ctrl = pwm_ctrl;
55 val = (pwm_ctrl * dev->range) / PWM_CTRL_MAX_VAL;
56 out32(dev->vbase + RP1_PWM_CHAN3_DUTY, val);
57 if (verbose > 0) {
58 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm channel3 duty register: 0x%x", __func__, val);
59 }
60 }
61 return EOK;
62}

71static int pwm_clock_init(void)
72{
73 uintptr_t vbase;
74 /*
75 * set RP1 PWM1 clock
76 */
77 vbase = (uintptr_t)mmap_device_io(RP1_CLOCK_MAIN_SIZE, RP1_CLOCK_MAIN_BASE);
78 if (vbase == (uintptr_t)MAP_FAILED) {
79 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Couldn't mmap rp1 clock, errno: %d", __func__, errno);
80 return errno;
81 }
82
83 out32(vbase + RP1_CLK_PWM1_CTRL, 0x11000840);
84 if (verbose > 0) {
85 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm1 clock reg: 0x%x", __func__, 0x11000840);
86 }
87 (void)munmap_device_io(vbase, RP1_CLOCK_MAIN_SIZE);
88
89 return EOK;
90}

92int pwm_fan_init(pwm_fan_dev_t *dev)
93{
94 int ret;
95 uint32_t val;
96 const uint32_t pwm_clk_period_ns = RP1_PWM_CLK_CYCLE_NS;
97
98 ret = pwm_clock_init();
99 if (ret != EOK) {
100 return ret;
101 }
102
103 dev->vbase = (uintptr_t)mmap_device_io(dev->reg_size, dev->base);
104 if (dev->vbase == (uintptr_t)MAP_FAILED) {
105 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Couldn't mmap rp1 pwm, errno: %d", __func__, errno);
106 return errno;
107 }
108
109 /*
110 * The user can enter range 0 ~ 255, while 0 means off
111 */
112
113 dev->range = RP1_PWM_CHAN3_RANGE_DEFAULT / pwm_clk_period_ns;
114 out32(dev->vbase + RP1_PWM_CHAN3_RANGE, dev->range);
115 out32(dev->vbase + RP1_PWM_CHAN3_PHASE, 0);
116 val = dev->range / 2U; // chose 1/2 of range as default
117 dev->pwm_ctrl = PWM_CTRL_MAX_VAL / 2U;
118 out32(dev->vbase + RP1_PWM_CHAN3_DUTY, val);
119 /* write PWM channel control register */
120 out32(dev->vbase + RP1_PWM_CHAN3_CTRL, RP1_PWM_CHAN3_CTRL_DEFAULT);
121 /* write PWM global control register */
122 out32(dev->vbase + RP1_PWM_GLOBAL_CTRL, (PWM_GLOBAL_CTRL_UPDATE | PWM_GLOBAL_CTRL_CHANNEL_EN(3)));
123 if (verbose > 0) {
124 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm channel3 range-phase-duty register: 0x%x-0-0x%x",
125 __func__, dev->range, val);
126 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm channel3 ctrl register: 0x%x", __func__, RP1_PWM_CHAN3_CTRL_DEFAULT);
127 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Set pwm global ctrl register: 0x%x", __func__, (PWM_GLOBAL_CTRL_UPDATE | PWM_GLOBAL_CTRL_CHANNEL_EN(3)));
128 }
129 return EOK;
130}
