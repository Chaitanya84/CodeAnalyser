29//
30// Read a Motorola 146818 compatable RTC (IBM AT standard).
31//
32
33static int
34rdcmos(unsigned i) {
35 chip_write8(0, i); //CMOS addr reg
36 return(chip_read8(1)); //CMOS data reg
37}
38
39static int inline check_uip_clr (void)
40{
41 return (rdcmos(MC146818_SRA) & MC146818_SRA_UIP) == 0;
42}
43
44unsigned long
45rtc_time_mc146818(paddr_t base, unsigned reg_shift, int mmap, int cent_reg) {
46 struct tm tm;
47 unsigned save_hour;
48 unsigned reg_b;
49 unsigned cent;
50 unsigned char sra;
51
52 //Tell Neutrino what kind of chip for 'rtc' utility
53 hwi_add_rtc("mc146818", base, reg_shift, 2, mmap, cent_reg);
54
55 chip_access(base, reg_shift, mmap, 2);
56
57 // bail if the clock is not running.
58 sra = rdcmos(MC146818_SRA);
59 if ((sra & 0x60) != 0x20) {
60 chip_write8 (1, (sra | 0x60)); //Check for ATI IXP200 RTC - these bits are reserved
61 if ((rdcmos(MC146818_SRA) & 0x60) != 0) {
62 chip_write8 (1, sra); //restore old value
63 return(0L);
64 }
65 }
66 reg_b = rdcmos(MC146818_SRB);
67
68 while (1) {
69 if (check_uip_clr ()) {
70 tm.tm_sec = rdcmos(0);
71 tm.tm_min = rdcmos(2);
72 tm.tm_hour = rdcmos(4);
73 tm.tm_mday = rdcmos(7);
74 tm.tm_mon = rdcmos(8);
75 tm.tm_year = rdcmos(9);
76 if (check_uip_clr ())
77 break;
78 }
79
80 //Loop while time inconsistent
81 }
82
83 chip_done();
84
85 save_hour = tm.tm_hour;
86 tm.tm_hour &= ~0x80;
87
88 if(!(reg_b & MC146818_SRB_DM)) {
89 tm.tm_sec = bcd2bin(tm.tm_sec);
90 tm.tm_min = bcd2bin(tm.tm_min);
91 tm.tm_hour = bcd2bin(tm.tm_hour);
92 tm.tm_mday = bcd2bin(tm.tm_mday);
93 tm.tm_mon = bcd2bin(tm.tm_mon);
94 tm.tm_year = bcd2bin(tm.tm_year);
95 }
96 if(!(reg_b & MC146818_SRB_24_12) && (save_hour & 0x80)) {
97 //12 hour format & past 12pm
98 tm.tm_hour += 12;
99 }
100 tm.tm_mon -= 1;
101
102 if(cent_reg >= 0) {
103 cent = rdcmos(cent_reg); //century
104 if(!(reg_b & MC146818_SRB_DM)) {
105 cent = bcd2bin(cent_reg);
106 }
107 if(cent == 20) tm.tm_year += 100;
108 } else if(tm.tm_year < 70) {
109 tm.tm_year += 100; //21st century.
110 }
111
112 return(calc_time_t(&tm));
113}
