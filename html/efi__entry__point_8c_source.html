46EFI_STATUS
47efi_entry_point(EFI_HANDLE ImageHandle, EFI_SYSTEM_TABLE *SystemTable)
48{
49 SystemTable->ConOut->OutputString(SystemTable->ConOut, u"Entering startup...\r\n");
50
51 efi_system_table = SystemTable;
52
53 const EFI_BOOT_SERVICES * const bs = SystemTable->BootServices;
54
55 EFI_GUID LoadedImageProto = LOADED_IMAGE_PROTO_GUID;
56 EFI_LOADED_IMAGE_PROTOCOL *plip;
57 EFI_STATUS status = bs->HandleProtocol(ImageHandle, &LoadedImageProto, (void **)&plip);
58 if (status == EFI_SUCCESS) {
59 if (plip->LoadOptionsSize > 0) {
60 efi_command_line_size = plip->LoadOptionsSize;
61 efi_command_line = plip->LoadOptions;
62 }
63 }
64
65 _Uintn map_key;
66 _Uint32t desc_version;
67
68 efi_map_size = 0;
69 status = bs->GetMemoryMap(&efi_map_size, NULL, &map_key,
70 &efi_map_desc_size, &desc_version);
71 if (status == EFI_BUFFER_TOO_SMALL) {
72 for (;;) {
73 // Allocation is all but guaranteed to add an entry to the list
74 efi_map_size += efi_map_desc_size;
75 status = bs->AllocatePool(EfiLoaderData, efi_map_size, &efi_map);
76 if (status != EFI_SUCCESS) {
77 SystemTable->ConOut->OutputString(SystemTable->ConOut, u"AllocatePool\r\n");
78 break;
79 }
80 status = bs->GetMemoryMap(&efi_map_size, efi_map, &map_key,
81 &efi_map_desc_size, &desc_version);
82 if (status != EFI_BUFFER_TOO_SMALL) {
83 if (status != EFI_SUCCESS) {
84 SystemTable->ConOut->OutputString(SystemTable->ConOut, u"GetMemoryMap\r\n");
85 }
86 break;
87 }
88 }
89
90 if (status == EFI_SUCCESS) {
91 status = bs->ExitBootServices(ImageHandle, map_key);
92 if (status == EFI_SUCCESS) {
93 void cstart();
94 cstart();
95 // NOTREACHED
96 }
97 SystemTable->ConOut->OutputString(SystemTable->ConOut, u"ExitBootServices failed\r\n");
98 }
99 } else {
100 SystemTable->ConOut->OutputString(SystemTable->ConOut, u"Error getting memory map\r\n");
101 }
102 bs->Exit(ImageHandle, EFI_LOAD_ERROR, 0, NULL);
103 // NOTREACHED
104 return EFI_LOAD_ERROR;
105}

113static void
114heapify(_Uintn const count)
115{
116 for (_Uintn end = 1; end < count; ++end) {
117 _Uintn child = end;
118 while (child > 0) {
119 _Uintn const parent = (child-1)/2;
120 EFI_MEMORY_DESCRIPTOR * const pdesc = efi_map_entry(parent);
121 EFI_MEMORY_DESCRIPTOR * const cdesc = efi_map_entry(child);
122
123 if (pdesc->PhysicalStart >= cdesc->PhysicalStart) {
124 break;
125 }
126
127 EFI_MEMORY_DESCRIPTOR desc = *pdesc;
128 *pdesc = *cdesc;
129 *cdesc = desc;
130 child = parent;
131 }
132 }
133}

135static void
136efi_sort_map(void)
137{
138 _Uintn const ndescs = efi_map_size / efi_map_desc_size;
139
140 heapify(ndescs);
141 for (_Uintn end = ndescs - 1; end > 0; --end) {
142 EFI_MEMORY_DESCRIPTOR *pend = efi_map_entry(end);
143 EFI_MEMORY_DESCRIPTOR *p0 = efi_map_entry(0);
144 EFI_MEMORY_DESCRIPTOR desc = *pend;
145 *pend = *p0;
146 *p0 = desc;
147 heapify(end);
148 }
149}

151EFI_MEMORY_DESCRIPTOR *
152efi_walk_map(EFI_MEMORY_DESCRIPTOR * const previous)
153{
154 if (previous == NULL) {
155 static unsigned sorted = 0;
156 if (!sorted) {
157 efi_sort_map();
158 sorted = 1;
159 }
160 return efi_map;
161 }
162
163 _Uintptrt const next = (_Uintptrt)(void *)previous + efi_map_desc_size;
164 _Uintptrt const end = (_Uintptrt)(void *)efi_map + efi_map_size;
165 if (next >= end) return NULL;
166
167 return (EFI_MEMORY_DESCRIPTOR *)(void *)next;
168}

171efi_get_table(const EFI_GUID * const guid)
172{
173 for (_Uintn i = 0; i < efi_system_table->NumberOfTableEntries; ++i) {
174 if (memcmp(&efi_system_table->ConfigurationTable[i].VendorGuid,
175 guid, sizeof(EFI_GUID)) == 0) {

177 }

182char *
183efi_convert_command_line(void)
184{
185 if (efi_command_line_size == 0) return NULL;
186
187 char *ascii_str = (void *)efi_command_line;
188 size_t i;
189 for (i = 0; i < efi_command_line_size; ++i) {
190 // Crude conversion
191 const char c = (char)efi_command_line[i];
192 *ascii_str++ = c;
193 if (c == 0) break;
194 }
195 if (i == efi_command_line_size) {
196 // Make sure to terminate it
197 *ascii_str = 0;
198 }
199
200 return (void *)efi_command_line;
201}
