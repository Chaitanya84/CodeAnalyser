67static void
68setup_cmdline(void) {
69 char *args;
70 int i, argc, envc;
71
72 //
73 // Find and construct argument and environment vectors for ourselves
74 //
75
76 tweak_cmdline(&boot_args, "startup");
77
78 argc = envc = 0;
79 args = boot_args.args;
80 for(i = 0; i < boot_args.argc; ++i) {
81 if(i < sizeof argv / sizeof *argv - 1) argv[argc++] = args;
82 while(*args++) ;
83 }
84 argv[argc] = 0;
85
86 for(i = 0; i < boot_args.envc; ++i) {
87 if (i < sizeof envv / sizeof *envv - 1) envv[envc++] = args;
88 while(*args++) ;
89 }
90 envv[envc] = 0;
91 _argc = argc;
92 _argv = argv;
93
94}

102void
103_main(void) {
104 void *syspage_mem = NULL;
105
106 shdr = (struct startup_header *)(uintptr_t)boot_args.shdr_addr;
107 if(shdr == (void *)SHDR_ADDR_64) {
108 // The boot image was loaded above 4G, so the 32 bits of the shdr_addr
109 // field isn't enough to hold the address of the startup header. The
110 // mkifs program hides the 8 byte address of the structure following
111 // the args/env strings in this case/.
112 unsigned const boot_size = boot_args.size_lo | (boot_args.size_hi << 8);
113 uint64_t real_shdr;
114 // extract with memmove() to avoid alignment problems
115 memmove(&real_shdr, (uint8_t *)&boot_args + boot_size - sizeof(real_shdr), sizeof(real_shdr));
116 shdr = (struct startup_header *)(uintptr_t)real_shdr;
117 }
118
119 full_image_paddr = shdr->image_paddr + shdr->addr_off;
120 full_ram_paddr = shdr->ram_paddr + shdr->addr_off;
121 if(shdr->imagefs_paddr != 0) {
122 full_imagefs_paddr = shdr->imagefs_paddr + shdr->addr_off;
123 }

126 board_init();

128 setup_cmdline();
129
130 stack_check_init();
131
132 cpu_startup();

134 #define INIT_SYSPAGE_SIZE 0x600
135 syspage_mem = ws_alloc(INIT_SYSPAGE_SIZE);
136 if(!syspage_mem) {
137 crash("No memory for syspage.\n");
138 }
139 init_syspage_memory(syspage_mem, INIT_SYSPAGE_SIZE);
140
141 if(full_imagefs_paddr != 0) {
142 avoid_ram(full_imagefs_paddr, shdr->stored_size);
143 }

145 main(_argc, _argv, envv);

154 // Tell the AP's that that the syspage is now present.
155 //
156 smp_hook_rtn();
157
158 startnext();
159}
