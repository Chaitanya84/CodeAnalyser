30static int _sdio_synchronize( struct sdio_device * const device, int const io, int const ncmds )
31{
32 int ret = EOK;
33 ret = pthread_mutex_lock( &sdio_ctrl.mutex );
34 if (ret != EOK) {
35 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
36 abort();
37 }
38 if( !io ) {
39 atomic_set( &device->flags, DEV_FLAG_RMV_PENDING );
40 if( device->usage != 0 ) {
41 pthread_mutex_unlock( &sdio_ctrl.mutex );
42 return( EBUSY );
43 }
44 }
45 else if( ncmds >= 0 ) {
46 if( ( device->flags & DEV_FLAG_RMV_PENDING ) || ( device->dev->flags & DEV_FLAG_MEDIA_CHANGE ) ) {
47 pthread_mutex_unlock( &sdio_ctrl.mutex );
48 return( ENXIO );
49 }
50 device->usage += ncmds;
51 }
52 else {
53 device->usage += ncmds; // Note: add since ncmds is negative
54 if( ( device->usage == 0 ) && ( device->flags & DEV_FLAG_RMV_PENDING ) ) {
55 ret = pthread_cond_signal( &sdio_ctrl.cd_cond );
56 if (ret != EOK) {
57 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "%s: pthread_cond_signal failed: (%s)\n",
58 __func__, strerror(ret));
59 ret = -1;
60 }
61 }
62 }
63 pthread_mutex_unlock( &sdio_ctrl.mutex );
64 return( ret );
65}

67struct sdio_device *sdio_device_lookup( struct sdio_connection * const connection, sdio_device_instance_t * const instance )
68{
69 int ret = EOK;
70 struct sdio_device *device;
71
72 device = NULL;
73 ret = pthread_mutex_lock( &sdio_ctrl.mutex );
74 if (ret != EOK) {
75 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
76 abort();
77 }
78 for( device = TAILQ_FIRST( &sdio_ctrl.dlist ); device != NULL; device = TAILQ_NEXT( device, dlink ) ) {
79 if( !memcmp( &device->instance, instance, sizeof( sdio_device_instance_t ) ) ) {
80 break;
81 }
82 }
83 (void)pthread_mutex_unlock( &sdio_ctrl.mutex );
84
85 return( device );
86}

143struct sdio_cmd *sdio_alloc_cmd( void )
144{
145 struct sdio_cmd *cmd;
146
147 int ret = EOK;
148 ret = pthread_mutex_lock( &sdio_ctrl.mutex );
149 if (ret != EOK) {
150 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
151 abort();
152 }
153 if( ( cmd = TAILQ_FIRST( &sdio_ctrl.clist ) ) ) {
154 TAILQ_REMOVE( &sdio_ctrl.clist, cmd, clink );
155 memset( cmd, 0, sizeof( struct sdio_cmd ) );
156 }
157 else {
158 cmd = calloc( 1, sizeof( struct sdio_cmd ) );
159 if (cmd == NULL) {
160 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "%s: Memory allocation failed\n",
161 __func__);
162 }
163 }
164 pthread_mutex_unlock( &sdio_ctrl.mutex );
165 return( cmd );
166}

172 if (ret != EOK) {
173 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
174 abort();
175 }
176 TAILQ_INSERT_TAIL( &sdio_ctrl.clist, cmd, clink );

417int sdio_retune( struct sdio_device * const device )
418{
419 sdio_hc_t *hc;
420 int status;
421
422 hc = device->hc;
423
424 if( !( hc->flags & HC_FLAG_TUNE ) ) {
425 return( EOK );
426 }
427
428 if( !( hc->device.flags & DEV_FLAG_PRESENT ) || ( hc->flags & HC_FLAG_RST ) ) {
429 atomic_clr( &hc->flags, HC_FLAG_TUNE );
430 return( EOK );
431 }
432
433 status = _sdio_synchronize( device, !0, 1 );
434 if( status != EOK ) {
435 return( status );
436 }
437
438 status = _sdio_retune( hc );
439
440 _sdio_synchronize( device, !0, -1 );
441
442 return( status );
443}

445int sdio_hpi( struct sdio_device * const device, uint32_t const timeout )
446{
447 int status;
448 uint32_t rsp[4];
449
450 status = sdio_wait_card_status( device, rsp, CDS_READY_FOR_DATA | CDS_CUR_STATE_MSK, CDS_READY_FOR_DATA | CDS_CUR_STATE_TRAN, timeout );
451 if( status == ETIMEDOUT ) {
452 if( ( device->dev->caps & DEV_CAP_HPI_CMD12 ) ) {
453 status = sdio_stop_transmission( device, SDIO_TRUE );
454 }
455 else {
456 status = sdio_send_status( device, rsp, SDIO_TRUE );
457 }
458 if( status == EOK ) {
459 status = sdio_wait_card_status( device, rsp, CDS_READY_FOR_DATA | CDS_CUR_STATE_MSK, CDS_READY_FOR_DATA | CDS_CUR_STATE_TRAN, MMC_OUT_OF_INTERRUPT_TIMEOUT * (uint32_t)device->dev->raw_ecsd[ECSD_OUT_OF_INTERRUPT_TIME] );
460 }
461 }
462 return( status );
463}

484int sdio_dev_info( const struct sdio_device * const device, sdio_dev_info_t *info )
485{
486 const sdio_hc_t *hc;
487 const sdio_dev_t *dev;
488 const sdio_cid_t *cid;
489 const sdio_csd_t *csd;
490 const sdio_ecsd_t *ecsd;
491
492 dev = device->dev;
493 hc = dev->hc;
494 cid = &dev->cid;
495 csd = &dev->csd;
496 ecsd = &dev->ecsd;
497
498 memset( info, 0, sizeof( sdio_dev_info_t ) );
499
500 info->dtype = dev->dtype;
501 info->rca = dev->rca;
502
503 if( csd->write_protect || !( csd->ccc & CCC_BLOCK_WRITE ) || ( dev->flags & DEV_FLAG_WRITE_PROTECT ) ) {
504 info->flags |= DEV_FLAG_WP;
505 }
506
507 if( ( dev->flags & DEV_FLAG_LOCKED ) ) {
508 info->flags |= DEV_FLAG_CARD_LOCKED;
509 }
510
511 info->mid = cid->mid;
512 info->oid = cid->oid;
513 info->prv = cid->prv;
514 info->psn = cid->psn;
515 info->month = cid->month;
516 info->year = cid->year;
517 memcpy( info->pnm, cid->pnm, sizeof( cid->pnm ) );
518
519 info->caps = dev->caps;
520
521 info->sector_size = csd->blksz;
522 info->native_sector_size = SDIO_BLKSZ_4K;
523 info->sectors = csd->sectors;
524 info->super_page_size = SDIO_BLKSZ_4K;
525
526 if( dev->dtype == DEV_TYPE_MMC ) {
527 if( ( csd->spec_vers >= CSD_SPEC_VER_4 ) && ecsd->sectors ) {
528 info->sectors = ecsd->sectors;
529 info->sector_size = ecsd->blksz;
530 info->super_page_size = ecsd->acc_size;
531 }
532 info->spec_vers = dev->csd.spec_vers;
533 info->spec_rev = ecsd->ext_csd_rev;
534 }
535 else {
536 info->spec_vers = dev->scr.sd_spec;
537 info->security = dev->scr.sd_security;
538 info->speed_class = dev->sds.speed_class;
539 }
540
541 info->wp_size = dev->wp_size;
542 info->erase_size = dev->erase_size;
543
544 info->rel_wr_sec_c = dev->rel_wr_sec_c;
545
546 info->optimal_trim_size = info->super_page_size;
547 info->optimal_read_size = info->super_page_size;
548 info->optimal_write_size = info->super_page_size;
549
550 info->dtr = hc->clk;
551 info->timing = hc->timing;
552 info->bus_width = hc->bus_width;
553
554 return( EOK );
555}

724{

726 if( device->usage ) {

729
730 ret = pthread_mutex_lock( &sdio_ctrl.mutex );

732 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
733 abort();

744 sdio_device_instance_t *instance,
745 struct sdio_device **hdl,
746 void * const client_hdl )
747{
748 sdio_hc_t *hc;
749 struct sdio_device *device;
750 int ret = EOK;
751
752 device = NULL;
753
754 ret = pthread_mutex_lock( &sdio_ctrl.mutex );
755 if (ret != EOK) {
756 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
757 abort();
758 }
759 for( hc = TAILQ_FIRST( &sdio_ctrl.hlist ); hc; hc = TAILQ_NEXT( hc, hlink ) ) {
760 if( ( hc->path == instance->path ) && ( hc->device.flags & DEV_FLAG_PRESENT ) ) {
761 if( ( instance->generation == SDIO_CONNECT_WILDCARD ) ||
762 ( instance->generation == hc->generation ) ) {
763 break;
764 }
765 }
766 }
767
768 if( hc != NULL ) {
769 device = calloc( 1, sizeof( struct sdio_device ) );
770 if( device != NULL ) {
771 TAILQ_INSERT_TAIL( &sdio_ctrl.dlist, device, dlink );
772 instance->path = hc->path;
773 instance->func = 0;
774 instance->generation = hc->generation;
775 instance->ident.vid = 0;
776 instance->ident.did = 0;
777 instance->ident.ccd = 0;
778 instance->ident.dtype = (uint16_t)hc->device.dtype;
779
780 memcpy( &device->instance, instance, sizeof( sdio_device_instance_t ) );
781 device->user = client_hdl;
782 device->hc = hc;
783 device->dev = &hc->device;
784 *hdl = device;
785 }
786 }
787
788 (void)pthread_mutex_unlock( &sdio_ctrl.mutex );
789
790 return( device ? EOK : ENODEV );
