42/*
43** In the comments the following letters tell you who uses the info
44**
45** I - used by the QNX IPL
46** S - used by the startup program
47** IS - used by both the QNX IPL and the startup program
48*/
49struct startup_header {
50 _Uint32t signature; /*I Header sig, see below*/
51 _Uint16t version; /*I Header vers, see below*/
52 _Uint8t flags1; /*IS Misc flags, see below*/
53 _Uint8t flags2; /* No flags defined yet*/
54 _Uint16t header_size; /*S sizeof(struct startup_header)*/
55 _Uint16t machine; /*IS Machine type from sys/elf.h*/
56 _Uint32t startup_vaddr; /*I Virtual Address to transfer*/
57 /* to after IPL is done*/
58 _Uint32t paddr_bias; /*S Value to add to physical address*/
59 /* to get a value to put into a*/
60 /* pointer and indirected through*/
61 _Uint32t image_paddr; /*IS Physical address of image*/
62 _Uint32t ram_paddr; /*IS Physical address of RAM to copy*/
63 /* image to (startup_size bytes copied)*/
64 _Uint32t ram_size; /*S Amount of RAM used by the startup*/
65 /* program and executables contained*/
66 /* in the file system*/
67 _Uint32t startup_size; /*I Size of startup (never compressed)*/
68 _Uint32t stored_size; /*I Size of entire image*/
69 _Uint32t imagefs_paddr; /*IS Set by IPL to where the imagefs is when startup runs*/
70 _Uint32t imagefs_size; /*S Size of uncompressed imagefs*/
71 _Uint16t preboot_size; /*I Size of loaded before header*/
72 _Uint16t zero0; /* Zeros */
73 _Uint32t zero[1]; /* Zeros */
74 _Uint64t addr_off; /*S Offset to add to startup_vaddr, image_paddr, */
75 /* ram_paddr, and imagefs_paddr members */
76 _Uint32t info[48]; /*IS Array of startup_info* structures*/
77};

348/* NOTE: The checksums in this trailer will only be valid prior */
349/* to entering startup. Because the startup binary is executed */
350/* in-place, its data segment will change once the program is */
351/* running. Hence, any checksum validation would need to be done */
352/* by the boot loader / IFS. */
