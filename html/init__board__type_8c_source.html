28/*
29 * Please refer to https://www.raspberrypi.org/documentation/hardware/raspberrypi/revision-codes/README.md
30 * New-style revision codes:
31 * NOQuuuWuFMMMCCCCPPPPTTTTTTTTRRRR in which:
32 * N Overvoltage
33 * O OTP Program
34 * Q OTP Read
35 * uuu Unused
36 * W Warranty bit
37 * u Unused
38 * F New flag
39 * MMM Memory size
40 * CCCC Manufacturer
41 * PPPP Processor
42 * TTTTTTTT Type
43 * RRRR Revision
44 */

57const char *get_soc_name(const uint32_t board_rev)
58{
59 const char * const socs[] = {
60 [0] = "BCM2835",
61 [1] = "BCM2836",
62 [2] = "BCM2837",
63 [3] = "BCM2711",
64 [4] = "BCM2712",
65 };
66
67 const uint32_t idx = (board_rev >> BOARD_REV_SOC_SHIFT) & BOARD_REV_SOC_MASK;
68 if (idx < NUM_ELTS(socs)) {
69 return socs[idx];
70 }
71 return "Unknown";
72}
