38static void
39vmsg(const char *fmt, va_list args) {
40 char *vs;
41 uint64_t num;
42 unsigned radix;
43 int dig;
44 char buf[64];
45
46 for(; *fmt; ++fmt) {
47 if(*fmt != '%') {
48 one_char(*fmt);
49 continue;
50 }
51 radix = dig = 0;
52 switch(*++fmt) {
53 case 'b':
54 num = va_arg(args, unsigned);
55 dig = sizeof(uint8_t)*2;
56 radix = 16;
57 break;
58 case 'w':
59 num = va_arg(args, unsigned);
60 dig = sizeof(uint16_t)*2;
61 radix = 16;
62 break;
63 case 'P':
64 num = va_arg(args, paddr_t);
65 dig = sizeof(paddr_t)*2;
66 radix = 16;
67 break;
68 case 'x':
69 num = va_arg(args, unsigned);
70 dig = sizeof(unsigned)*2;
71 radix = 16;
72 break;
73 case 'l':
74 num = va_arg(args, unsigned long);
75 dig = sizeof(unsigned long)*2;
76 radix = 16;
77 break;
78 case 'L':
79 num = va_arg(args, uint64_t);
80 dig = sizeof(uint64_t)*2;
81 radix = 16;
82 break;
83 case 'v':/* abbreviated hex */
84 num = va_arg(args, uintptr_t);
85 dig = sizeof(uintptr_t)*2;
86 radix = 16;
87 break;
88 case 'd':
89 case 'u':
90 num = va_arg(args, unsigned);
91 radix = 10;
92 break;
93 case 's':
94 vs = va_arg(args, char *);
95 while(*vs) {
96 one_char(*vs++);
97 }
98 continue;
99 default:
100 one_char(*fmt);
101 continue;
102 }
103 vs = &buf[sizeof(buf)];
104
105 *--vs = '\0';
106 do {
107 *--vs = c[num % radix];
108 num /= radix;
109 } while(num);
110 for(dig -= &buf[sizeof(buf)-1] - vs; dig > 0; --dig) {
111 one_char('0');
112 }
113 while(*vs) {
114 one_char(*vs++);
115 }
116 }
117}

137static void
138string_print_n_char(const int ch) {
139 if(ksnprinf_len < ksnprintf_max_len) {
140 *string_ptr++ = (char)ch;
141 ksnprinf_len++;
142 }
