59static unsigned
60gic_v2_cpunum(void)
61{
62 unsigned mask;
63 unsigned i;
64
65 /*
66 * Figure out our GIC cpu interface number for sending IPIs.
67 * GICv2 supports only 8 processors and ITARGETSR0 byte 0 contains the
68 * target mask for this cpu's ID0 SGI interrupt.
69 */
70 mask = in32(gic_gicd + ARM_GICD_ITARGETSRn) & 0xff;
71 if ((mask == 0) && (lsp.syspage.p->num_cpu == 1)) {
72 // uniprocessor
73 return 0;
74 }
75 for (i = 0; i < 8; i++) {
76 if (mask == (1 << i)) {
77 return i;
78 }
79 }
80 crash("gic_v2_cpunum: no target cpu detected %x\n", in32(gic_gicd + ARM_GICD_ITARGETSRn));
81 return 0;
82}

84static void
85gic_v2_gicc_init(unsigned cpu)
86{
87 struct aarch64_gic_map_entry *gic_map = lsp.cpu.aarch64_gic_map.p;
88
89 if (gic_gicd == 0 || gic_gicc == 0) {
90 crash("gic not initialised");
91 }
92 gic_map->gic_cpu[cpu] = gic_v2_cpunum();
93 if (debug_flag) {
94 kprintf("cpu%d: GICv2 cpu interface %d\n", cpu, gic_map->gic_cpu[cpu]);
95 }
96
97 /* Set SGI/PPI priority (all 0xA0, except for SGI0 which is 0xC0) */
98 out32(gic_gicd + ARM_GICD_IPRIORITYn, 0xA0A0A0C0);
99 unsigned i;
100 for(i = 1; i < 8; ++i) {
101 out32(gic_gicd + ARM_GICD_IPRIORITYn + (i*4), 0xA0A0A0A0);
102 }
103
104 /*
105 * Disable all banked PPI interrupts
106 * Enable all SGI interrupts
107 */
108 out32(gic_gicd + ARM_GICD_ICENABLERn, 0xffff0000);
109 out32(gic_gicd + ARM_GICD_ISENABLERn, 0x0000ffff);
110
111 /*
112 * Set priority mask to allow all interrupts and enable cpu interface
113 */
114 out32(gic_gicc + ARM_GICC_PMR, 0xF0);
115 out32(gic_gicc + ARM_GICC_CTLR, ARM_GICC_CTLR_EN);
116}

118void
119gic_v2_init(paddr_t gicd, paddr_t gicc)
120{
121 struct aarch64_gic_map_entry *gic_map = lsp.cpu.aarch64_gic_map.p;
122 unsigned gic_cpu;
123 unsigned itn;
124 unsigned i;
125
126 unsigned const off = as_find(AS_NULL_OFF, "memory", NULL);
127 as_add64(gicd, gicd + 0xfffu, AS_ATTR_DEV, "gicd", off);
128 as_add64(gicc, gicc + 0x1fffu, AS_ATTR_DEV, "gicc", off);
129
130 gic_gicd = gicd;
131 gic_gicc = gicc;
132 gic_cpu_init = gic_v2_gicc_init;
133 gic_sendipi = &sendipi_gic_v2;
134
135
136 gic_gicd_vaddr = callout_io_map(0x1000, gicd);
137 gic_gicc_vaddr = callout_io_map(0x2000, gicc);
138
139 /*
140 * Initialise the GIC cpu map with invalid values.
141 * gic_v2_gicc_init() will set the mapping for each cpu as it comes up.
142 */
143 for (i = 0; i < lsp.syspage.p->num_cpu; i++) {
144 gic_map->gic_cpu[i] = ~0u;
145 }
146 gic_cpu = gic_v2_cpunum();
147
148 /*
149 * Disable distributor
150 */
151 out32(gicd + ARM_GICD_CTLR, 0);
152
153 /*
154 * Calculate number of interrupt lines.
155 */
156 gic_v2_ppi.num_vectors = 32U; // always 32
157
158 itn = ((in32(gicd + ARM_GICD_TYPER) & ARM_GICD_TYPER_ITLN) + 1) * 32;
159 gic_v2_spi.num_vectors = min(1020, itn) - gic_v2_ppi.num_vectors;
160 unsigned const last_spi = gic_v2_ppi.num_vectors + gic_v2_spi.num_vectors;
161
162 if (debug_flag) {
163 kprintf("GICv2: %d interrupts\n", itn);
164 }
165
166 /*
167 * Disable all SPI interrupts and clear pending state
168 */
169 for (i = 32; i < itn; i += 32) {
170 out32(gicd + ARM_GICD_ICENABLERn + (i * 4 / 32), 0xffffffff);
171 out32(gicd + ARM_GICD_ICPENDRn + (i * 4 / 32), 0xffffffff);
172 }
173
174 /*
175 * Set default priority of all SPI interrupts to 0xa0
176 */
177 // XXX shouldn't i start at 32 ?
178 for (i = 8; i < last_spi; i += 4) {
179 out32(gicd + ARM_GICD_IPRIORITYn + i, 0xa0a0a0a0);
180 }
181
182 /*
183 * Route all SPI interrupts to cpu0
184 */
185 if (debug_flag) {
186 kprintf("GICv2: routing SPIs to gic cpu %d\n", gic_cpu);
187 }
188 for (i = 32; i < last_spi; i += 4) {
189 out32(gicd + ARM_GICD_ITARGETSRn + i, (0x01010101u << gic_cpu));
190 }
191
192 /*
193 * Default all PPI/SPI interrupts as level triggered
194 */
195 for (i = 16; i < itn; i += 16) {
196 out32(gicd + ARM_GICD_ICFGRn + (i * 4 / 16), 0);
197 }
198
199 /*
200 * Enable distributor - cpu interface is initialised via init_cpuinfo().
201 */
202 out32(gicd + ARM_GICD_CTLR, ARM_GICD_CTLR_EN);
203
204 /*
205 * Add the interrupt callouts
206 */
207 add_interrupt(&gic_v2_ppi);
208 add_interrupt(&gic_v2_spi);
209}

224void
225gic_v2_set_intr_trig_mode(int irq, int edge)
226{
227
228 if (gic_gicd) {
229 if (irq >= 32 && irq < gic_v2_num_spis() + 32) {
230 uint32_t oval, val;
231 uint32_t mask = 0x2 << ((irq % 16) * 2);
232 uint32_t off = (irq / 16) * 4;
233
234 oval = val = in32(gic_gicd + ARM_GICD_ICFGRn + off);
235 if (edge) {
236 val |= mask;
237 } else {
238 val &= ~mask;
239 }
240
241 out32(gic_gicd + ARM_GICD_ICFGRn + off, val);
242
243 if (debug_flag > 5) {
244 kprintf("%s: irq %d type %s: 0x%x: 0x%x -> 0x%x\n", __FUNCTION__,
245 irq, edge ? "edge" : "level", gic_gicd + ARM_GICD_ICFGRn + off, oval, val);
246 }
247 } else {
248 kprintf("%s: irq %d type %s: invalid irq (32-%d)\n", __FUNCTION__, irq, edge ? "edge" : "level", gic_v2_num_spis() + 32 - 1);
249 }
250 }
