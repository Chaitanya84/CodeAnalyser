66static int io_read (resmgr_context_t *ctp, io_read_t *msg, RESMGR_OCB_T *ocb)
67{
68 size_t nleft;
69 size_t nbytes;
70 int nparts;
71 int status;
72 char buf[PWM_CTRL_MAX_STRLEN] = { 0 };
73
74 status = iofunc_read_verify (ctp, msg, ocb, NULL);
75 if (status != EOK) {
76 return (status);
77 }
78
79 if ((msg->i.xtype & _IO_XTYPE_MASK) != _IO_XTYPE_NONE) {
80 return (ENOSYS);
81 }
82
83 // Nothing to read.
84 if (msg->i.nbytes == 0U) {
85 return (EOK);
86 }
87
88
89 ocb->attr->nbytes = snprintf(buf, sizeof(buf), "%u\n", pwm_fan_get(resmgr.dev));
90 if (ocb->attr->nbytes < 0) {
91 return (EINVAL);
92 }
93
94 /*
95 * On all reads (first and subsequent), calculate
96 * how many bytes we can return to the client,
97 * based upon the number of bytes available (nleft)
98 * and the client's buffer size
99 */
100 nleft = ocb->attr->nbytes - ocb->offset;
101 nbytes = min(_IO_READ_GET_NBYTES(msg), nleft);
102 if (nbytes > 0) {
103 /* set up the return data IOV */
104 SETIOV (ctp->iov, buf + ocb->offset, nbytes);
105
106 /* set up the number of bytes (returned by client's read()) */
107 _IO_SET_READ_NBYTES (ctp, nbytes);
108
109 /* advance the offset by the number of bytes returned to the client. */
110 ocb->offset += (off_t) nbytes;
111 nparts = 1;
112 } else {
113 /* they've asked for zero bytes or they've already previously read everything */
114 _IO_SET_READ_NBYTES (ctp, 0);
115 ocb->offset = 0;
116 nparts = 0;
117 }
118
119 /* mark the access time as invalid (we just accessed it) */
120 ocb->attr->flags |= IOFUNC_ATTR_ATIME;
121
122 return (_RESMGR_NPARTS (nparts));
123}

125static int io_write (resmgr_context_t *ctp, io_write_t *msg, RESMGR_OCB_T *ocb)
126{
127 int status;
128 size_t nbytes;
129 uint32_t pwm_ctrl;
130 char buf[PWM_CTRL_MAX_STRLEN] = { 0 };
131 uint64_t val = 0ULL;
132
133 status = iofunc_write_verify(ctp, msg, ocb, NULL);
134 if (status != EOK) {
135 return (status);
136 }
137
138 if ((msg->i.xtype & _IO_XTYPE_MASK) != _IO_XTYPE_NONE) {
139 return (ENOSYS);
140 }
141
142 /* Extract the length of the client's message. */
143 nbytes = _IO_WRITE_GET_NBYTES(msg);
144
145 /* Filter out malicious write requests that attempt to write more
146 data than they provide in the message. */
147
148 if (nbytes > ((size_t)ctp->info.srcmsglen - (size_t)ctp->offset - sizeof(io_write_t))) {
149 return EBADMSG;
150 }
151 if (nbytes == 0U) {
152 _IO_SET_WRITE_NBYTES(ctp, 0);
153 return _RESMGR_NPARTS(0);
154 }
155
156 if (nbytes >= PWM_CTRL_MAX_STRLEN) {
157 return EINVAL;
158 }
159
160 const ssize_t nbytes_read = resmgr_msgget(ctp, buf, nbytes, sizeof(msg->i));
161 if (nbytes_read < 0) {
162 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s: resmgr_msgget failed: err=%d", __func__, errno);
163 return errno;
164 }
165 buf[nbytes] = '\0'; /* just in case the text is not NULL terminated */
166
167 errno = EOK;
168 val = strtoul(buf, NULL, 10);
169 if (errno == EOK) {
170 if (val > PWM_CTRL_MAX_VAL) {
171 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Invalid value, valid range: range: 0 ~ %u", __func__, PWM_CTRL_MAX_VAL);
172 return (EINVAL);
173 }
174 pwm_ctrl = (uint32_t)val;
175 }
176 else {
177 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Invalid number string, err=%d", __func__, errno);
178 return (EINVAL);
179 }
180
181 if (verbose > 0) {
182 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "%s Write: nbytes: %ld pwm_ctrl=%d", __func__, nbytes, pwm_ctrl);
183 }
184
185 status = pwm_fan_set(resmgr.dev, pwm_ctrl);
186 if (status != EOK) {
187 return status;
188 }
189
190 ocb->attr->flags |= IOFUNC_ATTR_ATIME | IOFUNC_ATTR_CTIME;
191
192 // Indicate how many bytes were written
193 _IO_SET_WRITE_NBYTES(ctp, nbytes);
194 return _RESMGR_NPARTS(0);
195}

197int resmgr_loop_start()
198{
199 /* allocate a context structure */
200 resmgr.ctp = dispatch_context_alloc(resmgr.dpp);
201
202 /* Start the resource manager loop */
203 while (!resmgr.done) {
204 if (resmgr.ctp == dispatch_block(resmgr.ctp)) {
205 (void)dispatch_handler(resmgr.ctp);
206 }
207 else if (errno != EFAULT) {
208 atomic_set(&resmgr.done, 1);
209 }
210 }
211 return EOK;
212}

225int resmgr_init(pwm_fan_dev_t *dev)
226{
227 resmgr.dev = dev;
228
229 /* Single_instance. Ensure that the driver is not already running */
230 if (name_attach(NULL, RESMGR_NAME, 0) == NULL) {
231 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Is '%s' already started?", __func__, RESMGR_NAME);
232 return errno;
233 }
234 resmgr.filename = strdup(FAN_DEV_NAME);
235
236 /* allocate and initialize a dispatch structure for use by our main loop */
237 resmgr.dpp = dispatch_create();
238 if (resmgr.dpp == NULL) {
239 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Couldn't dispatch_create, errno: %d", __func__, errno);
240 return errno;
241 }
242
243 /*
244 * Intialize the connect functions and I/O functions tables to their defaults and then
245 * override the defaults with the functions that we are providing.
246 */
247 iofunc_func_init(_RESMGR_CONNECT_NFUNCS, &resmgr.connect_funcs, _RESMGR_IO_NFUNCS,
248 &resmgr.io_funcs);
249
250 resmgr.io_funcs.read = io_read;
251 resmgr.io_funcs.write = io_write;
252
253 iofunc_attr_init(&resmgr.iofunc_attr, S_IFCHR | 0666, NULL, NULL);
254 resmgr.id = resmgr_attach(resmgr.dpp, /* dispatch handle */
255 NULL, /* resource manager attrs */
256 FAN_DEV_NAME, /* device name */
257 _FTYPE_ANY, /* open type */
258 0, /* flags */
259 &resmgr.connect_funcs, /* connect routines */
260 &resmgr.io_funcs, /* I/O routines */
261 &resmgr.iofunc_attr); /* handle */
262 if (resmgr.id == -1) {
263 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "%s Couldn't attach pathname, errno: %d", __func__, errno);
264 return errno;
265 }
266
267 sig_init();
268
269 return EOK;
270}
