123static void gic_set_msi_irq_edge(const uint_t irq)
124{
125 static uintptr_t gicd_icfg_base;
126 const uint32_t conf_mask = 0x2U << (((irq + RP1_PCIE_MSI_GIC_IRQ_BASE) % 16U) * 2U);
127 const uint32_t conf_off = ((irq + RP1_PCIE_MSI_GIC_IRQ_BASE) / 16U) * 4U;
128 uint32_t val, oldval;
129 if (irq > RP1_PCIE_MSIX_IRQ_SIZE) {
130 (void)fprintf(stderr, "Invalid MSIX irq: %u\n", irq);
131 return;
132 }
133
134 gicd_icfg_base = (uintptr_t) mmap_device_memory(NULL,
135 (RP1_PCIE_MSIX_IRQ_SIZE * sizeof(uint32_t)),
136 PROT_NOCACHE|PROT_READ|PROT_WRITE,
137 0,
138 (GICD_PADDR + GICD_ICFGR));
139 if (gicd_icfg_base == (uintptr_t) MAP_FAILED) {
140 (void)fprintf(stderr, "mmap GICD interrupt configuration failed: %s\n", strerror (errno));
141 return;
142 }
143
144 oldval = in32(gicd_icfg_base + conf_off);
145 val = oldval | conf_mask;
146 out32(gicd_icfg_base + conf_off, val);
147 if (verbose > 1) {
148 (void)fprintf(stderr, "write irq: %u offset: 0x%x from 0x%x to 0x%x\n", irq, conf_off, oldval, val);
149 }
150
151 (void)munmap_device_memory((void*)gicd_icfg_base, RP1_PCIE_MSIX_IRQ_SIZE * sizeof(uint32_t));
152}

312 ret = pci_device_read_cap(bdf, &cap, (uint_t)idx);
313
