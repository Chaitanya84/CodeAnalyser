225static void rp1_gpio_get_bank_offset(const uint32_t gpio, uint32_t *bank, uint32_t *offset)
226{
227 *bank = 0;
228 *offset = 0;
229 if (gpio >= GPIO_MAX)
230 {
231 (void)printf("Invalid gpio number %u\n", gpio);
232 exit (EXIT_FAILURE);
233 }
234
235 if (gpio < rp1_bank_base[1]) {
236 *bank = 0;
237 }
238 else if (gpio < rp1_bank_base[2]) {
239 *bank = 1;
240 }
241 else {
242 *bank = 2;
243 }
244
245 *offset = gpio - rp1_bank_base[*bank];
246}

294static void set_gpio_dir(const uint32_t gpio, const uint32_t dir)
295{
296 uint32_t bank, offset;
297
298 rp1_gpio_get_bank_offset(gpio, &bank, &offset);
299 if (dir == FUNC_IP) {
300 out32(gpio_base + gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET + RP1_CLR_OFFSET, (1u << offset));
301 if (debug > 0) {
302 (void)printf("%s gpio:%u bank-offset: %u-%u, write sys_rio oe register:0x%lx=0x%x\n", __func__, gpio, bank, offset,
303 (uint64_t)RP1_GPIO_BASE+ gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET + RP1_CLR_OFFSET, (1u << offset));
304 }
305 }
306 else if (dir == FUNC_OP) {
307 out32(gpio_base + gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET + RP1_SET_OFFSET, (1u << offset));
308 if (debug > 0) {
309 (void)printf("%s gpio:%u bank-offset: %u-%u, write sys_rio oe register:0x%lx=0x%x\n", __func__, gpio, bank, offset,
310 (uint64_t)RP1_GPIO_BASE + gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET + RP1_SET_OFFSET, (1u << offset));
311 }
312 }
313}

316static uint32_t get_gpio_dir(const uint32_t gpio)
317{
318 uint32_t bank, offset;
319 uint32_t sys_rio_oe_read;
320
321 rp1_gpio_get_bank_offset(gpio, &bank, &offset);
322 sys_rio_oe_read = in32(gpio_base + gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET);
323
324 if (debug > 0) {
325 (void)printf("%s gpio:%u bank-offset: %u-%u, read sys_rio oe register:0x%lx=0x%x\n", __func__, gpio, bank, offset,
326 (uint64_t)RP1_GPIO_BASE+ gpio_sys_rio_bank_offset[bank] + RP1_GPIO_SYS_RIO_REG_OE_OFFSET, sys_rio_oe_read);
327 }
328 if (sys_rio_oe_read & (1U << offset)) {
329 return FUNC_OP;
330 }
331 else {
332 return FUNC_IP;
333 }
334}

336static uint32_t get_gpio_fsel(const uint32_t gpio)
337{
338 uint32_t bank, offset;
339 uint32_t io_val;
340
341 rp1_gpio_get_bank_offset(gpio, &bank, &offset);
342 io_val = in32(gpio_base + gpio_io_bank_offset[bank] + RP1_GPIO_IO_REG_CTRL_OFFSET(offset));
343 if (debug > 0) {
344 (void)printf("%s gpio:%u bank-offset: %u-%u, io register:0x%lx=0x%x\n", __func__, gpio, bank, offset,
345 (uint64_t)RP1_GPIO_BASE+ gpio_io_bank_offset[bank] + RP1_GPIO_IO_REG_CTRL_OFFSET(offset), io_val);
346 }
347 io_val &= FUNC_SEL_MASK;
348 if (io_val == RP1_FSEL_SYS_RIO) {
349 return get_gpio_dir(gpio); /* will return FUNC_IP or FUNC_OP */
350 }
351 return io_val;
352}

354static void set_gpio_fsel(const uint32_t gpio, uint32_t func_sel)
355{
356 uint32_t bank, offset;
357 uint32_t fsel;
358 uint32_t io_ctrl_val;
359 uint32_t orig_pad_val, pad_val;
360
361 rp1_gpio_get_bank_offset(gpio, &bank, &offset);
362
363 if ((func_sel == FUNC_IP) || (func_sel == FUNC_OP)) {
364 set_gpio_dir(gpio, func_sel);
365 }
366
367 if ((func_sel == FUNC_IP) || (func_sel == FUNC_OP) || (func_sel == FUNC_GP)) {
368 fsel = RP1_FSEL_SYS_RIO;
369 }
370 else if (func_sel == FUNC_NO) {
371 fsel = FUNC_NULL;
372 }
373 else {
374 fsel = func_sel;
375 }
376
377 io_ctrl_val = in32(gpio_base + gpio_io_bank_offset[bank] + RP1_GPIO_IO_REG_CTRL_OFFSET(offset));
378 io_ctrl_val &= ~(FUNC_SEL_MASK);
379 io_ctrl_val |= (fsel & FUNC_SEL_MASK);
380 out32(gpio_base + gpio_io_bank_offset[bank] + RP1_GPIO_IO_REG_CTRL_OFFSET(offset), io_ctrl_val);
381
382 pad_val = in32(gpio_base + gpio_pads_bank_offset[bank] + RP1_GPIO_PADS_REG_OFFSET(offset));
383 if (debug > 0) {
384 (void)printf("%s gpio:%u bank-offset: %u-%u, read pad register:0x%lx=0x%x\n", __func__, gpio, bank, offset,
385 (uint64_t)RP1_GPIO_BASE + gpio_pads_bank_offset[bank] + RP1_GPIO_IO_REG_CTRL_OFFSET(offset), pad_val);
386 }
387 orig_pad_val = pad_val;
388 if (fsel == FUNC_NULL) {
389 // Disable input
390 pad_val &= ~RP1_PADS_IE_SET;
391 // Disable peripheral func output
392 pad_val |= RP1_PADS_OD_SET;
393 }
394 else {
395 // Enable input
396 pad_val |= RP1_PADS_IE_SET;
397 // Enable peripheral func output
398 pad_val &= ~RP1_PADS_OD_SET;
399 }
400
401 if (pad_val != orig_pad_val) {
402 out32(gpio_base + gpio_pads_bank_offset[bank] + RP1_GPIO_PADS_REG_OFFSET(offset), pad_val);
403 if (debug > 0) {

582static void gpio_set(const uint32_t pinnum, const uint32_t func_sel, const uint32_t drive, const uint32_t pull)
583{
