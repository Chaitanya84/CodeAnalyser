121static inline paddr64_t cam_mphys( const void *addr )
122{
123 off64_t offset;
124
125 if( mem_offset64( addr, -1, 1, &offset, NULL ) == -1 ) {
126 return( (paddr64_t) -1LL );
127 }
128
129 return( offset );
130}
