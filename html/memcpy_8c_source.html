31void *memcpy(void *dst, const void *src, size_t nbytes) {
32 void *ret = dst;
33 unsigned n;
34
35 /* Both addresses must be aligned to stuff in int size chunks */
36 if( nbytes >= sizeof(unsigned) &&
37 ((uintptr_t)src & (sizeof(unsigned) - 1)) == 0 &&
38 ((uintptr_t)dst & (sizeof(unsigned) - 1)) == 0) {
39 unsigned *d = (unsigned *)dst;
40 unsigned *s = (unsigned *)src;
41
42
43 n = ((nbytes>>2) + 15)/16;
44
45 switch ((nbytes>>2)%16){
46 case 0: do { *d++ = *s++;
47 case 15: *d++ = *s++;
48 case 14: *d++ = *s++;
49 case 13: *d++ = *s++;
50 case 12: *d++ = *s++;
51 case 11: *d++ = *s++;
52 case 10: *d++ = *s++;
53 case 9: *d++ = *s++;
54 case 8: *d++ = *s++;
55 case 7: *d++ = *s++;
56 case 6: *d++ = *s++;
57 case 5: *d++ = *s++;
58 case 4: *d++ = *s++;
59 case 3: *d++ = *s++;
60 case 2: *d++ = *s++;
61 case 1: *d++ = *s++;
62 } while (--n > 0);
63 }
64
65 // remainder
66 nbytes = nbytes & 0x3;
67
68
69 if(nbytes) {
70 dst = (unsigned char *)d;
71 src = (const unsigned char *)s;
72 }
73 }
74
75 /* Get the unaligned bytes, or the remaining bytes */
76 while(nbytes) {
77 *(unsigned char *)dst = *(const unsigned char *)src;
78 dst = (char *)dst + 1;
79 src = (const char *)src + 1;
80 --nbytes;
81 }
82
83 return ret;
84}
