40int
41cpu_handle_common_option(int opt)
42{
43 char *cp, *value;
44
45 char *const ext_opts[] = {
46 "sve",
47 NULL
48 };
49
50 switch(opt) {
51 case 'X':
52 cp = optarg;
53 while(*cp != '\0') {
54 switch(getsubopt(&cp, xopts, &value)) {
55 case 0:
56 cache_snooped = CACHE_FLAG_SNOOPED;
57 break;
58 case 1:
59 __page_size = strtoul( value, 0, 16 );
60 if((__page_size != 0x1000) && (__page_size != 0x10000)) {
61 crash("Unsupported pagesize=%v\n", __page_size);
62 }
63 break;
64 default:
65 break;
66 }
67 }
68 return 1;
69 case 'E': ; // erratas
70 cp = optarg;
71 while(*cp != '\0') {
72 switch(getsubopt(&cp, erra_opts, &value)) {
73 case 0:
74 meltdown_active = 1;
75 break;
76 case 1:
77 meltdown_active = 0;
78 break;
79 case 2:
80 spectre_v2_active = 1;
81 break;
82 case 3:
83 spectre_v2_active = 0;
84 break;
85 case 4:
86 ssbs_active = 1;
87 break;
88 case 5:
89 ssbs_active = 0;
90 break;
91 default:
92 break;
93 }
94 }
95 return 1;
96 case 'U':
97 pauth_active = 1;
98 return 1;
99 case 'u':
100 if ( strcmp("arg", optarg) == 0 ) {
101 fdt_boot_option = FDT_BOOTOPT_ADDRPASS_BY_ARGS;
102 }
103 else if ( strcmp("reg", optarg) == 0 ) {
104 fdt_boot_option = FDT_BOOTOPT_ADDRPASS_BY_REG;
105 }
106 else {
107 /* if no option string is specified, set default boot option */
108 fdt_boot_option = FDT_BOOTOPT_ADDRPASS_BY_REG;
109 }
110 fdt_init_bootopt();
111 return 1;
112 case 'Q':
113 if (strcmp(optarg, "enable") == 0) {
114 hypervisor_set_options(HYP_FLAG_ENABLED);
115 } else if (strcmp(optarg, "disable") == 0) {
116 hypervisor_set_options(HYP_FLAG_DISABLED);
117 } else if (strcmp(optarg, "enable,el1-host") == 0) {
118 hypervisor_set_options(HYP_FLAG_ENABLED | HYP_FLAG_EL1_HOST);
119 } else if (strcmp(optarg, "enable,el2-host") == 0) {
120 hypervisor_set_options(HYP_FLAG_ENABLED | HYP_FLAG_EL2_HOST);
121 } else {
122 hypervisor_set_options(HYP_FLAG_INVALID);
123 }
124 return 1;
125 case 'e': // cpu extensions
126 cp = optarg;
127 while(*cp != '\0') {
128 switch(getsubopt(&cp, ext_opts, &value)) {
129 case 0:
130 sve_requested = 1;
131 break;
132 default:
133 break;
134 }
135 }
136 return 1;
137 default:
138 return 0;
139 }
140
141}
