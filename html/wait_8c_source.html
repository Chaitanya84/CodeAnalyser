29int32_t dwc_i2c_wait_bus_not_busy(dwc_dev_t* const dev)
30{
31 int32_t timeout = DW_I2C_TIMEOUT * 10;
32
33 while ((i2c_reg_read32(dev, DW_IC_STATUS) & DW_IC_STATUS_ACTIVITY) != 0U) {
34 if (timeout <= 0) {
35 logerr("timeout waiting for bus ready\n");
36 return -ETIMEDOUT;
37 }
38 timeout--;
39 (void)usleep(100);
40 }
41
42 return 0;
43}

45static int32_t dwc_i2c_process_intr(dwc_dev_t* const dev)
46{
47 uint32_t status;
48 uint32_t numbers, reg;
49
50 status = i2c_reg_read32(dev, DW_IC_INTR_STAT);
51
52 if ((status & DW_IC_INTR_TX_ABRT) != 0U) {
53 /* Get the Abort source and clean the ABRT interrupt bit */
54 dev->abort_source = i2c_reg_read32(dev, DW_IC_TX_ABRT_SOURCE);
55 (void)i2c_reg_read32(dev, DW_IC_CLR_TX_ABRT);
56
57 /* Set status */
58 if ((dev->abort_source & DW_IC_TX_ARBT_LOST) != 0U) {
59 dev->status = (uint32_t)I2C_STATUS_DONE | (uint32_t)I2C_STATUS_ARBL;
60 }
61 else if ((dev->abort_source & DW_IC_TX_ABRT_ADDR_NOACK) != 0U) {
62 dev->status = (uint32_t)I2C_STATUS_DONE | (uint32_t)I2C_STATUS_ADDR_NACK;
63 }
64 else if ((dev->abort_source & DW_IC_TX_ABRT_TXDATA_NOACK) != 0U) {
65 dev->status = (uint32_t)I2C_STATUS_DONE | (uint32_t)I2C_STATUS_DATA_NACK;
66 }
67 else {
68 dev->status = (uint32_t)I2C_STATUS_DONE | (uint32_t)I2C_STATUS_ABORT;
69 }
70
71 /* Disable interrupt and return interrupt event */
72 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
73 return EOK;
74 }
75
76 if ((status & DW_IC_INTR_STOP_DET) != 0U) {
77 /* Clean the STOP_DET interrupt */
78 (void)i2c_reg_read32(dev, DW_IC_CLR_STOP_DET);
79
80 /* Read Rx FIFO if required */
81 if (dev->rdlen < dev->rxlen) {
82 /* Check the received data length */
83 numbers = i2c_reg_read32(dev, DW_IC_RXFLR);
84 if ((dev->rdlen + numbers) != dev->rxlen) {
85 /* Wrong Received Rx length */
86 dev->status = (uint32_t)I2C_STATUS_ERROR;
87 numbers = 0;
88 }
89
90 for ( ; numbers > 0U; numbers--) {
91 reg = i2c_reg_read32(dev, DW_IC_DATA_CMD);
92 *dev->rxbuf++ = (uint8_t)(reg & 0x00FFU);
93 dev->rdlen++;
94 }
95 }
96
97 /* Disable interrupt and return interrupt event */
98 dev->status |= (uint32_t)I2C_STATUS_DONE;
99 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
100 return EOK;
101 }
102
103 if ((status & DW_IC_INTR_RX_FULL) != 0U) {
104 /* Note: this bit will be cleaned only when the value of DW_IC_RXFLR less than the value of DW_IC_RX_TL */
105 if (dev->rdlen < dev->rxlen) {
106 /* How many valid data entries in the receive FIFO */
107 numbers = i2c_reg_read32(dev, DW_IC_RXFLR);
108
109 /* Check the received data length */
110 if ((dev->rdlen + numbers) > dev->rxlen) {
111 /* Unexpected received length */
112 numbers = dev->rxlen - dev->rdlen;
113 dev->status |= (uint32_t)I2C_STATUS_ERROR;
114
115 /* Disable DW_IC_INTR_RX_FULL interrupt */
116 reg = i2c_reg_read32(dev, DW_IC_INTR_MASK) & ~DW_IC_INTR_RX_FULL;
117 i2c_reg_write32(dev, DW_IC_INTR_MASK, reg);
118 }
119
120 for ( ; numbers > 0U; numbers--) {
121 reg = i2c_reg_read32(dev, DW_IC_DATA_CMD);
122 *dev->rxbuf++ = (uint8_t)(reg & 0x00FFU);
123 dev->rdlen++;
124 }
125 }
126 else {
127 /* Disable DW_IC_INTR_RX_FULL interrupt, it should never go here */
128 reg = i2c_reg_read32(dev, DW_IC_INTR_MASK) & ~DW_IC_INTR_RX_FULL;
129 i2c_reg_write32(dev, DW_IC_INTR_MASK, reg);
130 }
131 }
132
133 if ((status & DW_IC_INTR_TX_EMPTY) != 0U) {
134 /* Note: this bit should been cleaned only when the value of DW_IC_TXFLR large than the value of DW_IC_TX_TL */
135 for (uint32_t i = 0; i < dev->fifo_depth; i++) {
136 if (dev->wrlen >= dev->xlen) {
137 /* No more command left, disable R_TX_EMPTY interrupt */
138 reg = i2c_reg_read32(dev, DW_IC_INTR_MASK) & ~DW_IC_INTR_TX_EMPTY;
139 i2c_reg_write32(dev, DW_IC_INTR_MASK, reg);
140 break;
141 }
142
143 if (xmit_FIFO_is_full(dev)) {
144 /* TXFIFO FULL */
145 break;
146 }
147
148 if (dev->wrlen < dev->txlen) {
149 /* Command for master-transmit */
150 if (dev->wrlen == (dev->xlen - 1U)) {
151 /* If last byte for master-transmit, force set STOP condition */
152 reg = (uint32_t)*dev->txbuf++ | DW_IC_DATA_CMD_WRITE | DW_IC_DATA_CMD_STOP;
153 }
154 else {
155 reg = (uint32_t)*dev->txbuf++ | DW_IC_DATA_CMD_WRITE;
156 }
157 }
158 else if ((dev->wrlen == dev->txlen) && (0U != dev->txlen)) {
159 /* First command for master-receive of isendrecv , need restart cmd */
160 if (dev->wrlen == (dev->xlen - 1U)) {
161 /* If last byte for master-receive, force set STOP condition */
162 reg = DW_IC_DATA_CMD_READ | DW_IC_DATA_CMD_RESTART | DW_IC_DATA_CMD_STOP;
163 }
164 else {
165 reg = DW_IC_DATA_CMD_READ | DW_IC_DATA_CMD_RESTART;
166 }
167 }
168 else {
169 /* Command for master-receive */
170 if (dev->wrlen == (dev->xlen - 1U)) {
171 /* If last byte for master-receive, force set STOP condition */
172 reg = DW_IC_DATA_CMD_READ | DW_IC_DATA_CMD_STOP;
173 }
174 else {
175 reg = DW_IC_DATA_CMD_READ;
176 }
177 }
178
179 i2c_reg_write32(dev, DW_IC_DATA_CMD, reg);
180 dev->wrlen++;
181 }
182 }
183
184 return EAGAIN;
185}

187void dwc_i2c_reset(dwc_dev_t *const dev)
188{
189 int32_t ret;
190 int32_t reset_retry = RESET_RETRY;
191
192 /* Abort the I2C transfer */
193 i2c_reg_write32(dev, DW_IC_ENABLE, DW_IC_ENABLE_ABORT | DW_IC_ENABLE_STATUS_ENABLE);
194
195 (void)usleep(1000);
196
197 /* Disable the adapter */
198 (void)dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_DISABLE);
199
200 do {
201
202 /* Perform soft reset.
203 * Notes:
204 * If the driver isn’t using DMA, then it is OK to just leave iDMA in Reset.
205 * To be 100% sure the DMA is not being used, read register BAR0 + 0xB98
206 * (DMACFGREG) bit[0], which is the DMA enable bit. If it is '0', then no
207 * need to worry about any pending transactions.
208 * At the time this soft reset was added, DMACFGREG is 0. So DMA is NOT being
209 * used.
210 * Additionally, in case DMA support is added in the future, there is a need
211 * to check whether or not DMA transactions need to be stopped before performing
212 * soft reset.
213 */
214
215 logfyi("Assert soft reset. retry=%d\n", reset_retry);
216 i2c_reg_write32(dev, DW_IC_SOFT_RESET, DW_IC_SOFT_RESET_ASSERTED);
217 (void)usleep(25); //Just miniscule delay
218
219 logfyi("De-assert soft reset.\n");
220 i2c_reg_write32(dev, DW_IC_SOFT_RESET, DW_IC_SOFT_RESET_NOT_ASSERTED);
221
222 logfyi("Re-initializing...");
223 ret = dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_DISABLE);
224 if (ret == 0) {
225 dwc_i2c_init_registers(dev);
226 logfyi("I2C reinitialized.\n");
227 return;
228 }
229 else {
230 logerr("Failed to reinitialize!\n");
231 }
232 } while (--reset_retry > 0);
233
234 logerr("I2C failed to reset!\n");
235}

237i2c_status_t dwc_i2c_wait_complete(dwc_dev_t* const dev)
238{
239 int32_t err;
240
241 // Estimate transfer time in us... The calculated time is only used for
242 // the timeout, so it doesn't have to be that accurate. At higher clock
243 // rates, a calcuated time of 0 would mess-up the timeout calculation, so
244 // round up to 1 us per byte, and wait at least 2.5 ms for the xfer to complete.
245 const uint32_t xtime_us = max(2500U, max(1U, 10U * 1000U * 1000U / dev->scl_freq) * dev->xlen);
246
247 const uint64_t to_ns = (uint64_t)xtime_us * 1000U * 50U; // convert to ns and extend to 50 times of estimate time
248
249 while (true) {
250 (void)TimerTimeout(CLOCK_MONOTONIC, _NTO_TIMEOUT_RECEIVE, NULL, &to_ns, NULL);
251
252 err = InterruptWait(0, NULL);
253
254 if (err == -1) {
255 logerr("%s: PID_%d %s(%d), stat reg %x\n", __func__, getpid(),
256 strerror(errno), errno, i2c_reg_read32(dev, DW_IC_STATUS));
257 dwc_i2c_reset(dev);
258 break;
259 }
260
261 if (dwc_i2c_process_intr(dev) == EAGAIN) {
262 (void)InterruptUnmask(dev->irq, dev->iid);
263 continue;
264 }
265
266 if ((dev->status & ((uint32_t)I2C_STATUS_ARBL | (uint32_t)I2C_STATUS_ERROR)) != 0U) {
267 dwc_i2c_reset(dev);
268 }
269
270 /* Disabled interrupts */
271 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
272
273 (void)InterruptUnmask(dev->irq, dev->iid);
274
275 return (i2c_status_t)dev->status;
276 }
277
278 return I2C_STATUS_BUSY;
279}
