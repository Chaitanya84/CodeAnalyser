28#define BCM2712_DRAM_BASE (0UL)
29#define BCM2712_DRAM_STUB_SIZE KILO(512UL)
30#define BCM2712_DRAM_BASE_1 (0x100000000)
31#define BCM2712_DRAM_BASE_2 (0x200000000)
32
33/* |----------------| <-0x400000000
34 * | SDRAM |
35 * | (for ARM) |
36 * |----------------| <-0x100000000
37 * | ARM local |
38 * | peripherals |
39 * |----------------| <-0x0ff800000
40 * |Main peripherals|
41 * |----------------| <-0x0fc000000
42 * | SDRAM(for ARM)|
43 * |----------------| <-0x040000000
44 * | SDRAM(for VC) |
45 * |----------------|
46 * | SDRAM(for ARM) |
47 * |----------------| <-0x000000000
48 */
49void bcm2712_init_raminfo(void) {
50 const uint32_t vcram_size = mbox_get_vc_memory();
51 const paddr_t vcram_start = (paddr_t)(GIG(1UL) - vcram_size);
52 const uint32_t rev = get_board_revision();
53 uint64_t ramsize = 0UL;
54
55 /* Avoid touching the stub in startup */
56 avoid_ram(BCM2712_DRAM_BASE, BCM2712_DRAM_STUB_SIZE);
57 /* Avoid touching the VideoCore memory in startup */
58 avoid_ram(vcram_start, vcram_size);
59
60 ramsize = MEG(get_memory_size(rev));
61 if (ramsize == 0UL) {
62 ramsize = MEG(256UL);
63 kprintf("Warning: Failed to detect RAM size. Defaulting to %u MB.\n",
64 ramsize);
65 }
66
67 if (ramsize >= GIG(4UL)) {
68 add_ram(BCM2712_DRAM_BASE, GIG(4UL) - MEG(64UL)); // periphs are mapped to 0xfc000000
69 if (ramsize >= GIG(8UL)) {
70 add_ram(BCM2712_DRAM_BASE_1, GIG(4UL)); // for rpi5 8GB target
71 }
72 if (ramsize >= GIG(16UL)) {
73 add_ram(BCM2712_DRAM_BASE_2, GIG(8UL)); // for rpi5 16GB target
74 }
75 } else {
76 add_ram(BCM2712_DRAM_BASE, ramsize); // 256M, 512M, 1G, 2G
77 }
78
79 /* Reserve VideoCore memory */
80 alloc_ram(vcram_start, vcram_size, 0);
81 as_add_containing(vcram_start, vcram_start + vcram_size - 1, AS_ATTR_RAM, "vcram", "ram");
82
83 /* Add a below1G tag for legacy peripherals */
84 if (ramsize > GIG(1UL)) {
85 as_add_containing(BCM2712_DRAM_BASE, vcram_start - 1, AS_ATTR_RAM, "below1G", "ram");
86 }
87
88 if (debug_flag > 3) {
89 kprintf("vcram start-size: %P-%P, memory size: %l\n", vcram_start, vcram_size, ramsize);
90 }
91}
