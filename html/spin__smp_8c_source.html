27unsigned
28spin_smp_num_cpu(void) {
29 // The "yield" is to give any other cores a chance to run the
30 // code in spin_smp_init.S.
31 asm("yield");
32 return spin_num_cpu;
33}

36int
37spin_smp_start(unsigned cpu, void (*start)(void)) {
38 /*
39 * Secondary cores will be spinning in spin_smp_init.S watching for
40 * spin_start_addr to indicate this core should wake up.
41 */
42#if defined(__aarch64__)
43 spin_start_addr = (uintptr_t)start | ((uintptr_t)cpu << 56);
44#else
45 extern unsigned _btext[];
46 spin_start_addr = ((uintptr_t)start - (uintptr_t)_btext)
47 | ((uintptr_t)cpu << 24);
48#endif
49 __asm__ __volatile__(
50 "dsb sy\n"
51 "sev\n"
52 : : : "memory"
53 );
54
55 // Wait for other processor to notice
56 while(spin_start_addr != 0) {
57 asm("yield");
58 }
59
60 return 1;
61}
