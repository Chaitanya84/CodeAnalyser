362void gic_v3_set_paddr_range(const paddr_t gicd, const paddr_t gicr, size_t gicr_size, const paddr_t gits)
363{
364 ASSERT(base_addresses_initialized == 0);
365
366 gicd_paddr_base = gicd;
367 gicr_paddr_base = gicr;
368 /*
369 * Now set the GITS paddr
370 * Don't clobber previous calls to gic_v3_its_set_paddr()
371 */
372 if (gits != NULL_PADDR) gic_v3_its_set_paddr(0, gits);
373
374 base_addresses_initialized = 1; /* don't move this */
375
376 /* initialize the GIC implementation and version information */
377 gic_init_version_info();
378
379 /* add syspage asinfo entries for GICD and GICR */
380 const unsigned off = as_find(AS_NULL_OFF, "memory", NULL);
381 as_add64(gicd_paddr_base, gicd_paddr_base + 0xFFFFu, AS_ATTR_DEV, "gicd", off);
382
383 /*
384 * there are as many GICRs/PEs as there are real processors
385 * Don't assume a 1:1 CPU to PE correlation
386 */
387 gicr_offset_shift_val = 17;
388 if (ARM_GIC_ARCH_VERSION_IS(gic_get_arch_version()) >= ARM_GIC_ARCH_VERSION_4_0) ++gicr_offset_shift_val;
389
390 const unsigned max_cpu = board_smp_num_cpu();
391 if (gicr_size == 0) {
392 gicr_size = (1u << gicr_offset_shift_val) * max_cpu;
393 gicr_index_limit = max_cpu;
394 } else {
395 gicr_index_limit = gicr_size >> gicr_offset_shift_val;
396 }
397 as_add64(gicr_paddr_base, gicr_paddr_base + gicr_size - 1, AS_ATTR_DEV, "gicr", off);
398}

619unsigned gic_v3_num_spis(void)
620{
621 ASSERT(base_addresses_initialized != 0);
622
623 const _Uint32t gicd_typer = in32(gicd_paddr_base + ARM_GICD_TYPER);
624 const _Uint32t ITLines = ((gicd_typer >> 0) & ARM_GICD_TYPER_ITLN);
625 /*
626 * ((ITLines + 1) * 32) - 1 represents the highest SPI INTID.
627 * The first 32 INTIDs are not SPIs, therefore the number of SPIs is
628 * less the above calculation by 32. INTIDs >= 1020 are reserved so
629 * clip at that value - 1.
630 */
631 if(((ITLines + 1U) * 32U) > (ARM_GIC_SPI_MAX_IRQ + 1U))
632 {
633 return (ARM_GIC_SPI_MAX_IRQ + 1U) - 32U;
634 }
635 return ITLines * 32U;
636}

642uint32_t gic_v3_num_espis(void)
643{
644 uint32_t ret = 0;
645 ASSERT(base_addresses_initialized != 0U);
646
647 const uint32_t gicd_typer = in32(gicd_paddr_base + ARM_GICD_TYPER);
648
649 if((gicd_typer & GICD_TYPER_ESPI) != 0U) {
650 ret = (32U * (((gicd_typer & GICD_TYPER_ESPI_RANGE) >> GICD_TYPER_ESPI_RANGE_S) + 1U));
651 }
652
653 return ret;
654}

661unsigned gic_v3_num_lpis(void)
662{
663 ASSERT(base_addresses_initialized != 0);
664
665 const _Uint32t gicd_typer = in32(gicd_paddr_base + ARM_GICD_TYPER);
666 unsigned nlpi = (gicd_typer >> 11) & 0x1f; /* try num_LPIs bits */
667
668 if ((gicd_typer & GICD_TYPER_LPIS) == 0) nlpi = 0; /* not supported */
669 /* if num_LPI bits is 0, get from IDbits */
670 else if (nlpi == 0) nlpi = ((_Uint32t)1 << (((gicd_typer >> 19) & 0x1f) + 1)) - ARM_GIC_LPI_MIN_IRQ;
671 /* otherwise 'nlpi is actually num_LPI bits so convert to want we want */
672 else nlpi = (_Uint32t)1 << (nlpi + 1);
673
674 return nlpi;
675}

685void gic_v3_set_intr_trig_mode(const unsigned irq, const unsigned edge)
686{
687 const uint32_t num_spis = gic_v3_num_spis();
688 const uint32_t num_espis = gic_v3_num_espis();
689 ASSERT(base_addresses_initialized != 0);
690
691 if ((irq >= ARM_GIC_SPI_MIN_IRQ) && (irq < (num_spis + ARM_GIC_SPI_MIN_IRQ)))
692 {
693 const _Uint32t mask = 0x2 << ((irq % 16) * 2);
694 const _Uint32t off = (irq / 16) * 4;
695
696 const _Uint32t cur_val = in32(gicd_paddr_base + ARM_GICD_ICFGRn + off);
697 _Uint32t val = cur_val;
698
699 if (edge != 0) val |= mask;
700 else val &= ~mask;
701
702 out32(gicd_paddr_base + ARM_GICD_ICFGRn + off, val);
703 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
704
705 if (debug_flag > 4)
706 {
707 kprintf("%s(%u, %s) @ ICFG<%u> [0x%x]: 0x%x -> 0x%x\n", __func__,
708 irq, (edge != 0) ? "edge" : "level",
709 irq / 16, gicd_paddr_base + ARM_GICD_ICFGRn + off, cur_val, val);
710 }
711 }
712 else if ((irq >= ARM_GIC_ESPI_MIN_IRQ) && (irq < (num_espis + ARM_GIC_ESPI_MIN_IRQ)))
713 {
714 const _Uint32t mask = 0x2 << ((irq % 16) * 2);
715 const _Uint32t off = ((irq - ARM_GIC_ESPI_MIN_IRQ) / 16) * 4;
716
717 const _Uint32t cur_val = in32(gicd_paddr_base + ARM_GICD_ICFGRnE + off);
718 _Uint32t val = cur_val;
719
720 if (edge != 0) val |= mask;
721 else val &= ~mask;
722
723 out32(gicd_paddr_base + ARM_GICD_ICFGRnE + off, val);
724 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
725
726 if (debug_flag > 4)
727 {
728 kprintf("%s(%u, %s) @ ICFG<%u>E [0x%x]: 0x%x -> 0x%x\n", __func__,
729 irq, (edge != 0) ? "edge" : "level",
730 irq / 16, gicd_paddr_base + ARM_GICD_ICFGRnE + off, cur_val, val);
731 }
732 }
733 else
734 {
735 kprintf("%s(%u, %s) invalid\n", __func__, irq, (edge != 0) ? "edge" : "level");
736 }
737}

791int gic_v3_spi_add_entry(struct startup_intrinfo * const spi_entry)
792{
793 ASSERT(base_addresses_initialized != 0);
794 ASSERT(spi_intrinfo.num_entries < NUM_ELTS(spi_intrinfo.co));
795 /* do range checking for normal (v3) */
796 ASSERT((spi_entry->vector_base >= ARM_GIC_SPI_MIN_IRQ) && (spi_entry->vector_base <= ARM_GIC_SPI_MAX_IRQ));
797 ASSERT((spi_entry->vector_base + spi_entry->num_vectors) <= (gic_v3_num_spis() + ARM_GIC_SPI_MIN_IRQ));
798
799 int r = 0;
800 const unsigned new_start = spi_entry->vector_base;
801 const unsigned new_last = new_start + spi_entry->num_vectors - 1;
802
803 if (new_last >= (gic_v3_num_spis() + ARM_GIC_SPI_MIN_IRQ)) r = -1;
804 else
805 {
806 if ((spi_entry->flags & INTR_FLAG_MSI) != 0)
807 {
808 /* make sure Message based SPIs are supported */
809 const _Uint32t gicd_typer = in32(gicd_paddr_base + ARM_GICD_TYPER);
810 if ((gicd_typer & GICD_TYPER_MBIS) == 0)
811 {
812 if (debug_flag > 3) kprintf("Message based SPIs are not supported\n");
813 r = -1;
814 }
815 }
816 if (r == 0)
817 {
818 /*
819 * check for an overlap with existing entries
820 * IMPORTANT
821 * we'll also check to make sure that the entry being added has vectors
822 * numerically larger than those previously added. There is an as yet
823 * to be determined issue if the entries are not added in this way. While
824 * the vectors are all in the syspage and get allocated to the requester
825 * properly, no interrupts will occur for the out of order entries. This
826 * appears to be a procnto problem because the correct bits are getting
827 * masked/unmasked and it is only the order of adding entries, and not
828 * anything to do with the vectors in the entries that makes a difference.
829 * The patched callouts also look fine
830 */
831 unsigned i;
832 for (i=0; i<spi_intrinfo.num_entries; i++)
833 {
834 const struct startup_intrinfo * const cur_spi_entry = spi_intrinfo.co[i].entry;
835 ASSERT(cur_spi_entry != NULL);
836 const unsigned cur_start = cur_spi_entry->vector_base;
837 const unsigned cur_last = cur_start + cur_spi_entry->num_vectors - 1;
838
839 if ((new_start >= cur_start) && (new_start <= cur_last))
840 {
841 if (debug_flag > 3)
842 {
843 kprintf("start overlap: new_start: %u, cur_start: %u, cur_last: %u\n",
844 new_start, cur_start, cur_last);
845 }
846 break;
847 }
848 else if ((new_last >= cur_start) && (new_last <= cur_last))
849 {
850 if (debug_flag > 3)
851 {
852 kprintf("end overlap: new_last: %u, cur_start: %u, cur_last: %u\n",
853 new_last, cur_start, cur_last);
854 }
855 break;
856 }
857 /*
858 * check for an out-of-order entry as described above. Note that we
859 * don't necessarily care that the entries are contiguous, just that
860 * they are in increasing order
861 */
862 else if (new_start <= cur_last)
863 {
864 if (debug_flag > 3)
865 {
866 kprintf("out-of-order: new_last: %u, cur_last: %u\n",
867 new_last, cur_last);
868 }
869 break;
870 }
871 }
872 if (i != spi_intrinfo.num_entries) r = -1;
873 else
874 {
875 spi_entry->cascade_vector = _NTO_INTR_SPARE;
876 spi_entry->cpu_intr_base = 0;
877 spi_entry->cpu_intr_stride = 0;
878 spi_entry->id.genflags = 0;
879 spi_entry->id.size = 0;
880 spi_entry->id.rtn = &interrupt_id_gic_v3_spi;
881 spi_entry->eoi.genflags = 0;
882 spi_entry->eoi.size = 0;
883 spi_entry->eoi.rtn = &interrupt_eoi_gic_v3_spi_sr;
884 spi_entry->mask = &interrupt_mask_gic_v3_spi;
885 spi_entry->unmask = &interrupt_unmask_gic_v3_spi;
886 /* don't touch the config or local_stride fields. The patch_data will
887 * be established in gic_v3_initialize() */
888
889 spi_intrinfo.co[spi_intrinfo.num_entries].entry = spi_entry;
890 ++spi_intrinfo.num_entries;
891 }
892 }
893 }
894 if (debug_flag > 2)
895 {
896 kprintf("Add SPI entry %u for vectors %u -> %u [SPI max: %u], %s\n",
897 spi_intrinfo.num_entries - ((r == 0) ? 1 : 0),
898 spi_entry->vector_base, spi_entry->vector_base + spi_entry->num_vectors - 1U,
899 gic_v3_num_spis(), (r == 0) ? "Ok" : "Failed");
900 }
901 return r;

1268void gic_v3_initialize(void)
1269{
1270 ASSERT(base_addresses_initialized != 0);
1271
1272 /*
1273 * if 'use_memory_mapped_callouts' is not set, make sure that we don't
1274 * need them and correct as necessary
1275 */
1276 if ((use_memory_mapped_callouts == 0) && ((aa64_sr_rd64(id_aa64pfr0_el1) & ((_Uint64t)0xF << 24)) == 0))
1277 {
1278 kprintf("%s(), setting 'use_memory_mapped_callouts'\n", __func__);
1279 use_memory_mapped_callouts = 1;
1280 }
1281 /*
1282 * See comments in gic_v3_use_mm_reg_callouts() for why this ASSERT is here
1283 * If you hit this, you either called gic_v3_use_mm_reg_callouts() with a
1284 * bogus GICC paddr or you didn't call gic_v3_use_mm_reg_callouts() and you
1285 * need to
1286 */
1287 ASSERT((use_memory_mapped_callouts == 0) || (gicc_paddr_base != NULL_PADDR));
1288
1289 /*
1290 * Initialize the GIC cpu map with invalid values
1291 * Note that the number of elements in this 'gic_map' is PROCESSORS_MAX
1292 * as specified in cpu_init_syspage_memory() and so we should probably be
1293 * initializing that many values. But since this field is not an array that
1294 * we can use NUM_ELTS() on and a size variable does not exist, we'll use the
1295 * real number of processors on this platform as returned by board_smp_num_cpu().
1296 * so that at least as many entries are initialized as there are real processors.
1297 * Previously the code was using 'lsp.syspage.p->num_cpu' which can be arbitrarily
1298 * made smaller with the -P command line option
1299 */
1300 const unsigned max_cpu = board_smp_num_cpu();
1301 struct aarch64_gic_map_entry * const gic_map = lsp.cpu.aarch64_gic_map.p;
1302 for (unsigned i=0; i<max_cpu; i++)
1303 {
1304 gic_map->gic_cpu[i] = ~0ul;
1305 }
1306
1307 /* Disable distributor */
1308 out32(gicd_paddr_base + ARM_GICD_CTLR, 0);
1309 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
1310
1311 /* get the number of SPI interrupt lines and calculate the max INTID */
1312 const _Uint32t gicd_typer = in32(gicd_paddr_base + ARM_GICD_TYPER);
1313 const unsigned itlines = ((gicd_typer & ARM_GICD_TYPER_ITLN) + 1U);
1314 const unsigned itn = ((itlines * 32U) > (ARM_GIC_SPI_MAX_IRQ + 1U)) ? (ARM_GIC_SPI_MAX_IRQ + 1U) : (itlines * 32U);
1315
1316 if (debug_flag > 2) kprintf("GICv3: gicd_typer 0x%x\n", gicd_typer);
1317 if (debug_flag > 1) kprintf("GICv3: itlines: %u, max INTID: %d\n", itlines, itn - 1);
1318
1319 /*
1320 * Disable all interrupts and clear pending state
1321 * SGIs and PPIs depend on the security state. We're currently skipping them
1322 * until gic_v3_gicc_init() hence the starting 'reg_idx' of 1
1323 * There are 32 INTIDs per register
1324 */
1325 for (unsigned reg_idx=1; reg_idx<itlines; reg_idx++)
1326 {
1327 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1328 out32(gicd_paddr_base + ARM_GICD_ICENABLERn + reg_offset, 0xFFFFFFFF);
1329 out32(gicd_paddr_base + ARM_GICD_ICPENDRn + reg_offset, 0xFFFFFFFF);
1330 /* All interrupts are in Group1 */
1331 out32(gicd_paddr_base + ARM_GICD_IGROUPRn + reg_offset, 0xFFFFFFFF);
1332 /* SPI interrupts are in Group1NS */
1333 out32(gicd_paddr_base + ARM_GICD_IGRPMODRn + reg_offset, 0x00000000);
1334 }
1335
1336 /*
1337 * Set default priority of all SPI interrupts to 0xA0. For SGIs and PPIs,
1338 * GICR_IPRIORITYR is used and will be initialized in gic_v3_gicc_init()
1339 * hence the starting 'reg_idx' of 8.
1340 */
1341 for (unsigned reg_idx=8; reg_idx<(8*itlines); reg_idx++)
1342 {
1343 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1344 out32(gicd_paddr_base + ARM_GICD_IPRIORITYn + reg_offset, 0xA0A0A0A0);
1345 }
1346
1347 /*
1348 * Default all interrupts as level triggered. Modifications will be done as
1349 * required for MSIs
1350 * SGIs and PPIs depend on the security state. We're currently skipping them
1351 * hence the starting 'reg_idx' of 2
1352 * There are 16 INTIDs per register
1353 */
1354 for (unsigned reg_idx=2; reg_idx<(itn/16); reg_idx++)
1355 {
1356 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1357 out32(gicd_paddr_base + ARM_GICD_ICFGRn + reg_offset, 0);
1358 }
1359 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
1360
1361 /* get the number of ESPI interrupt lines and INTIDs */
1362 const unsigned espilines = ((gicd_typer & GICD_TYPER_ESPI) == 0) ? 0 : (((gicd_typer & GICD_TYPER_ESPI_RANGE) >> GICD_TYPER_ESPI_RANGE_S) + 1U);
1363 const unsigned espin = (32U * espilines);
1364
1365 if ((debug_flag > 1) && (espin > 0U))
1366 {
1367 kprintf("GICv3: espilines: %u, max INTID: %d\n", espilines, espin + ARM_GIC_ESPI_MIN_IRQ - 1U);
1368 }
1369
1370 /*
1371 * Disable all ESPI interrupts and clear pending state
1372 * There are 32 INTIDs per register
1373 */
1374 for (unsigned reg_idx=0; reg_idx<espilines; reg_idx++)
1375 {
1376 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1377 out32(gicd_paddr_base + ARM_GICD_ICENABLERnE + reg_offset, 0xFFFFFFFF);
1378 out32(gicd_paddr_base + ARM_GICD_ICPENDRnE + reg_offset, 0xFFFFFFFF);
1379 /* ESPI interrupts are in Group1NS */
1380 out32(gicd_paddr_base + ARM_GICD_IGRPMODRnE + reg_offset, 0x00000000);
1381 }
1382
1383 /*
1384 * Set default priority of all ESPI interrupts to 0xA0.
1385 */
1386 for (unsigned reg_idx=0; reg_idx<(8U*espilines); reg_idx++)
1387 {
1388 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1389 out32(gicd_paddr_base + ARM_GICD_IPRIORITYnE + reg_offset, 0xA0A0A0A0);
1390 }
1391
1392 /*
1393 * Default all interrupts as level triggered. Modifications will be done as
1394 * required for MSIs
1395 * There are 16 INTIDs per register
1396 */
1397 for (unsigned reg_idx=0; reg_idx<(espin/16U); reg_idx++)
1398 {
1399 const unsigned reg_offset = reg_idx * sizeof(_Uint32t); /* 32 bit register */
1400 out32(gicd_paddr_base + ARM_GICD_ICFGRnE + reg_offset, 0);
1401 }
1402 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
1403
1404 /* Enable GICD */
1405 if (gicd_typer & GICD_TYPER_SecurityExtn)
1406 {
1407 /*
1408 * GIC implementation supports two security states.
1409 * This must be done in two separate writes because ARE_NS being already
1410 * set changes whether we should be using ENABLEGRP1A or ENABLEGRP1
1411 */
1412 out32(gicd_paddr_base + ARM_GICD_CTLR, GICD_CTLR_ARE_NS);
1413 dsb();
1414 out32(gicd_paddr_base + ARM_GICD_CTLR, GICD_CTLR_ARE_NS | GICD_CTLR_ENABLEGRP1A_NS);
1415 }
1416 else
1417 {
1418 /*
1419 * GIC implementation supports a single security state
1420 * GICD_CTLR.DS is RAO/WI in this case
1421 */
1422 out32(gicd_paddr_base + ARM_GICD_CTLR, GICD_CTLR_ARE_1s | GICD_CTLR_ENABLEGRP1_1s);
1423 }
1424 wait_for_rwp(gicd_paddr_base + ARM_GICD_CTLR, 0x80000000);
1425
1426 // Only get the AffX fields from MPIDR, clear Interrupt Routing Mode.
1427 const _Uint64t affinity = aa64_sr_rd64(mpidr_el1) & 0xff00fffffful;
1428
1429 /*
1430 * Route all SPI INTIDs to the current CPU, assumed to be CPU0
1431 * Must be done after ARE is set
1432 * There is one register per INTID
1433 */
1434 for (unsigned reg_idx=ARM_GIC_SPI_MIN_IRQ; reg_idx<itn; reg_idx++)
1435 {
1436 const unsigned reg_offset = reg_idx * sizeof(_Uint64t); /* 64 bit register */
1437 out64(gicd_paddr_base + ARM_GICD_IROUTERn + reg_offset, affinity);
1438 }
1439
1440 /*
1441 * Route all ESPI INTIDs to the current CPU, assumed to be CPU0
1442 * Must be done after ARE is set
1443 * There is one register per INTID
1444 */
1445 for (unsigned reg_idx=0; reg_idx<espin; reg_idx++)
1446 {
1447 const unsigned reg_offset = reg_idx * sizeof(_Uint64t); /* 64 bit register */
1448 out64(gicd_paddr_base + ARM_GICD_IROUTERnE + reg_offset, affinity);
1449 dsb();
1450 }
1451
1452 /* map memory needed in the callouts */
1453 const _Uintptrt gicd_vaddr_base = callout_io_map(0x10000, gicd_paddr_base);
1454 if (debug_flag > 1) kprintf("GICD @ %L mapped to %L\n", gicd_paddr_base, gicd_vaddr_base);
1455
1456 const size_t gicr_size = (1u << gicr_offset_shift_val) * gicr_index_limit;
1457 const _Uintptrt gicr_vaddr_base = callout_io_map(gicr_size, gicr_paddr_base);
1458 if (debug_flag > 1) kprintf("GICR @ %L mapped to %L\n", gicr_paddr_base, gicr_vaddr_base);
1459 /* IMPORTANT: 'gic_gicr_vaddr' is still used for the IPI callout so need to set it here */
1460 gic_gicr_vaddr = gicr_vaddr_base;
1461
1462 _Uintptrt gicc_vaddr_base = 0;
1463 if (use_memory_mapped_callouts != 0)
1464 {
1465 gicc_vaddr_base = callout_io_map(ARM_GICC_SIZE, gicc_paddr_base);
1466 if (debug_flag > 1) kprintf("GICC @ %L mapped to %L\n", gicc_paddr_base, gicc_vaddr_base);
1467 }

1582 if (lpi_params.num_lpis > 0)
1583 {
1584 if (debug_flag > 1) kprintf("GICv3: %d LPI interrupts\n", lpi_params.num_lpis);
1585
1586 ASSERT(lpi_params.num_lpis <= gic_v3_num_lpis()); /* implementation defined maximum */

1589 ASSERT(lpi_params.cfg.num_tbls == 1);

1591 {

1593 struct _lpi_tbl_params_s * const lpi_cfg_tbl_params = &lpi_params.cfg.tbl_params[tbl_num];
1594

1597 {

1599 lpi_cfg_tbl_params->paddr = alloc_ram(NULL_PADDR, lpi_params.num_lpis, ARM_GIC_ESPI_MIN_IRQ);
1600 }
1601 /* make sure the address is valid. We're done if its not */
1602 ASSERT(lpi_cfg_tbl_params->paddr != NULL_PADDR);
1603 ASSERT((lpi_cfg_tbl_params->paddr & (_Uint64t)0xFFFFFFFFFF000) == lpi_cfg_tbl_params->paddr);
1604
1605 /* set priority, enabled bit off */
1606 memset(MAKE_1TO1_PTR(lpi_cfg_tbl_params->paddr), 0xa2, lpi_params.num_lpis);
1607
1608 DCACHE_FLUSH_FOR_GIC_ITS(MAKE_1TO1_PTR(lpi_cfg_tbl_params->paddr), lpi_params.num_lpis);
1609
1610 /* map the LPI config table for use in the callouts */
1611 lpi_cfg_tbl_params->vaddr = (_Uintptrt)callout_memory_map(lpi_params.num_lpis,
1612 lpi_cfg_tbl_params->paddr, lpi_cfg_tbl_params->mmap_flags);
1613 if (debug_flag > 0)
1614 {
1615 kprintf("LPI config table #%u @ %L, callout vaddr: %L\n",
1616 tbl_num + 1, lpi_cfg_tbl_params->paddr, lpi_cfg_tbl_params->vaddr);
1617 }

1620 * Note that in some implementations, the shareability bits are non-modifiable
1621 * and if they are 0b00 or 0b11, then cacheability should be set to
1622 * non-cacheable so we check whether shareability is settable and adjust
1623 * the parameters accordingly
1624 */
1625 unsigned cfg_tbl_share_bits = (in64(gicr_paddr_base + ARM_GICR_PROPBASER) >> 10) & 0x3;
1626 const unsigned flipped_cur_share_bits = ~cfg_tbl_share_bits & 0x3;
1627 out64(gicr_paddr_base + ARM_GICR_PROPBASER, (_Uint64t)flipped_cur_share_bits << 10);
1628 cfg_tbl_share_bits = (in64(gicr_paddr_base + ARM_GICR_PROPBASER) >> 10) & 0x3;
1629 const unsigned cfg_tbl_share_bits_settable = (cfg_tbl_share_bits == flipped_cur_share_bits);
1630

1632 {
1633 if (debug_flag > 1) kprintf("PROPBASER shareability bits non-settable, use non-cached attributes\n");
1634

1636 lpi_cfg_tbl_params->mem.cacheability.inner = arm_gic_tbl_cacheability_NONE;
1637 lpi_cfg_tbl_params->mem.cacheability.outer = arm_gic_tbl_cacheability_OUTER_IS_NORMAL_INNER;
1638 }
1639
1640 /*
1641 * build the PROPBASER register value starting with the LPI config
1642 * table physical address.
1643 */
1644 lpi_cfg_tbl_params->propbaser = lpi_cfg_tbl_params->paddr;
1645 /* set inner and outer cacheability */
1646 lpi_cfg_tbl_params->propbaser |= ((_Uint64t)lpi_cfg_tbl_params->mem.cacheability.inner << 7);
1647 lpi_cfg_tbl_params->propbaser |= ((_Uint64t)lpi_cfg_tbl_params->mem.cacheability.outer << 56);
1648 /* set shareability */
1649 lpi_cfg_tbl_params->propbaser |= ((_Uint64t)lpi_cfg_tbl_params->mem.shareability << 10);
1650
1651 unsigned bits = 14; /* minimum when LPIs are supported */
1652 while (((_Uint64t)1 << bits) < (lpi_params.num_lpis + ARM_GIC_LPI_MIN_IRQ)) {
1653 ++bits;
1654 }
1655 lpi_cfg_tbl_params->propbaser |= (bits - 1);
1656
1657#ifndef SET_PROPBASER_IN_GICC_INIT
1658 /*
1659 * program the PROPBASER value into the redistributor. Here is where
1660 * changes will be required in order to support multiple LPI config tables.
1661 * Because we are running this code as CPU 0, we program the PE associated
1662 * with that CPUs PROPBASER with the LPI table parameters. In gic_v3_gicc_init()
1663 * we (currently) use this same value for all other redistributors. The API
1664 * that would to be added to allow for multiple LPI config tables will also need
1665 * to be able assign the tables to specific redistributors and gic_v3_gicc_init()
1666 * updated to use that information to program the other PE redistributors
1667 */
1668 out64(gicr_paddr_base + ARM_GICR_PROPBASER, lpi_cfg_tbl_params->propbaser);
1669
1670 if (debug_flag > 1) kprintf("Setting PROPBASER to %L for CPU0\n", in64(gicr_paddr_base + ARM_GICR_PROPBASER));
1671#endif
1672 }

1674 const unsigned uses_ITS = gic_v3_lpi_uses_its();
1675
1676 if (debug_flag > 1) kprintf("Implementation supports %s physical LPIs\n", uses_ITS ? "ITS based" : "Direct");
1677
1678 const unsigned its_num = 0;
1679 if (uses_ITS != 0) gic_v3_its_initialize(its_num);
1680
1681 for (unsigned i=0; i<lpi_intrinfo.num_entries; i++)
1682 {
1683 struct startup_intrinfo * const intrinfo = lpi_intrinfo.co[i].entry;
1684
1685 if (use_memory_mapped_callouts != 0)
1686 {
1687 intrinfo->eoi.rtn = &interrupt_eoi_gic_v3_lpi_mm;

1698 lpi_intrinfo.co[i].patch_data.lpi_its.lpi_cfgtbl_vaddr = lpi_cfg_params->vaddr;
1699 lpi_intrinfo.co[i].patch_data.lpi_its.cmd_q_vaddr = gic_v3_its_cmd_q_vaddr(its_num);

1715 }
1716 else
1717 {

1721 const struct _lpi_tbl_params_s * const lpi_cfg_params = select_lpi_cfg_table(0);
1722
1723 lpi_intrinfo.co[i].patch_data.lpi_direct.gicr_base_vaddr = gicr_vaddr_base;

1728 }

1730 }

1746static void gic_v3_gicc_init(const unsigned cpu)
1747{
1748 paddr_t gic_core_gicr = gicr_paddr_base;
1749
1750 /* Get core affinity */
1751 _Uint64t const mpidr = aa64_sr_rd64(mpidr_el1);
1752 _Uint64t const cpu_aff = (_Uint32t)((mpidr & 0xFFFFFF) | ((mpidr >> 8) & 0xFF000000));
1753
1754 unsigned gicr_index = 0;
1755 do {
1756 /* Get GICR affinity bits (offset + 4, [32 - 63] bits of TYPER register) */
1757 _Uint32t const gicr_aff = in32(gic_core_gicr + ARM_GICR_TYPER + 4);
1758 if (cpu_aff == gicr_aff) {
1759 break;
1760 }
1761 gic_core_gicr += (1 << gicr_offset_shift_val);
1762 ++gicr_index;
1763
1764 ASSERT(gicr_index < gicr_index_limit);
1765
1766 } while (1);
1767 // When there is more than one redistributor region, the top half will indicate it.
1768 lsp.cpu.aarch64_gicr_map.p->gicr_idx[cpu] = gicr_index;
1769
1770 /* Make sure ProcessorSleep bit is off in WAKER? */
1771 _Uint32t gicr_waker = in32(gic_core_gicr + ARM_GICR_WAKER);
1772 if ((gicr_waker & (0x1 << 1)) != 0)
1773 {
1774 kprintf("** CPU %d PE is not awake\n", cpu);
1775 int loop = 1000;
1776 while ((gicr_waker & (0x1 << 2)) == 0)
1777 {
1778 gicr_waker = in32(gic_core_gicr + ARM_GICR_WAKER);
1779 ASSERT(--loop > 0);
1780 }
1781 out32(gic_core_gicr + ARM_GICR_WAKER, gicr_waker & ~(0x1 << 1));
1782 }
1783
1784 /* Configure LPIs */
1785 if (lpi_params.num_lpis != 0)
1786 {
1787 /*
1788 * each redistributor will have its own pending table. This will either
1789 * be allocated now, or will have been preallocated and specified in
1790 * 'lpi_params' (no API to do this yet)
1791 */
1792 /* set up a pointer to the appropriate LPI pending table parameters */
1793 const unsigned params_idx = (lpi_params.pending.num_tbls > 1) ? cpu : 0;
1794 ASSERT(params_idx < lpi_params.pending.num_tbls);
1795 struct _lpi_tbl_params_s * const lpi_pend_tbl_params = &lpi_params.pending.tbl_params[params_idx];
1796 /* set up the 'pendtbl_paddr' to either point to local storage or the specific pending table parameters */
1797 paddr_t _pendtbl_paddr = NULL_PADDR;
1798 paddr_t * const pendtbl_paddr_p = (lpi_params.pending.num_tbls > 1) ? &lpi_pend_tbl_params->paddr : &_pendtbl_paddr;
1799 size_t dcache_flush_len = 0;
1800
1801 /* allocate as required */
1802 if (*pendtbl_paddr_p == NULL_PADDR)
1803 {
1804 /*
1805 * Unlike the LPI config table, the LPI pending table is sized to
1806 * hold all INTIDs including the 8192 non LPI interrupts and each
1807 * entry holds the pending status for 8 INTIDs.
1808 * Note that the number of INTIDs supported by a redistributor can
1809 * be less than that indicated by the distributor
1810 */
1811 const size_t tbl_size = ROUND((ARM_GIC_LPI_MIN_IRQ + lpi_params.num_lpis) / 8, 1024);
1812 if (debug_flag > 1)
1813 {
1814 kprintf("LPI Pend table, num_lpis: %u, min required size: %u, actual size: %u\n",
1815 lpi_params.num_lpis, (ARM_GIC_LPI_MIN_IRQ + lpi_params.num_lpis) / 8, tbl_size);
1816 }
1817 *pendtbl_paddr_p = calloc_ram(tbl_size, 0x10000); /* 64KB aligned */
1818 dcache_flush_len = tbl_size;
1819 }
1820
1821 if (debug_flag > 1) kprintf("LPI pending table @ %L for CPU%d\n", *pendtbl_paddr_p, cpu);
1822 /* make sure the address is valid */
1823 ASSERT(*pendtbl_paddr_p != NULL_PADDR);
1824 ASSERT((*pendtbl_paddr_p & (_Uint64t)0xFFFFFFFFF0000) == *pendtbl_paddr_p);
1825
1826 /*
1827 * Note that in some implementations, the shareability bits are non-modifiable
1828 * and if they are 0b00 or 0b11, then cacheability should be set to
1829 * non-cacheable so we check whether shareability is settable and adjust
1830 * the parameters accordingly
1831 */
1832 unsigned pending_tbl_share_bits = (in64(gic_core_gicr + ARM_GICR_PENDBASER) >> 10) & 0x3;
1833 const unsigned flipped_cur_share_bits = ~pending_tbl_share_bits & 0x3;

1843 lpi_pend_tbl_params->mem.cacheability.inner = arm_gic_tbl_cacheability_NONE;
1844 lpi_pend_tbl_params->mem.cacheability.outer = arm_gic_tbl_cacheability_OUTER_IS_NORMAL_INNER;
1845 }

1847 * build the PENDBASER register value starting with the LPI pending
1848 * table physical address.
1849 */
1850 _Uint64t pendbaser = *pendtbl_paddr_p;
1851 /* set inner and outer cacheability */
1852 pendbaser |= ((_Uint64t)lpi_pend_tbl_params->mem.cacheability.inner << 7);
1853 pendbaser |= ((_Uint64t)lpi_pend_tbl_params->mem.cacheability.outer << 56);
1854 /* set shareability */
1855 pendbaser |= ((_Uint64t)lpi_pend_tbl_params->mem.shareability << 10);

1859 out64(gic_core_gicr + ARM_GICR_PENDBASER, pendbaser | (1ul << 62));
1860
1861 if (debug_flag > 1)
1862 {
1863 kprintf("Set PENDBASER to %L for CPU%d\n",
1864 in64(gic_core_gicr + ARM_GICR_PENDBASER), cpu);
1865 }

1867#ifndef SET_PROPBASER_IN_GICC_INIT
1868 if (cpu > 0)
1869#endif
1870 {
1871 const struct _lpi_tbl_params_s * const lpi_cfg_params = select_lpi_cfg_table(cpu);
1872 out64(gic_core_gicr + ARM_GICR_PROPBASER, lpi_cfg_params->propbaser);
1873
1874 if (debug_flag > 1)
1875 {
1876 kprintf("Set PROPBASER to %L for CPU%d\n",
1877 in64(gic_core_gicr + ARM_GICR_PROPBASER), cpu);
1878 }
1879 }
1880 /* invalidate */
1881 const unsigned uses_ITS = gic_v3_lpi_uses_its();
1882 if (uses_ITS == 0) out64(gic_core_gicr + ARM_GICR_INVALLR, 0);
1883 else gic_v3_its_mapc(0, cpu, gic_core_gicr); /* only 1 ITS instance supported */
1884
1885 /* Enable LPI's */
1886 out32(gic_core_gicr + ARM_GICR_CTLR, 1);
1887 kprintf("Enable LPIs in GICR_CTLR @ %P for CPU%d\n", gic_core_gicr + ARM_GICR_CTLR, cpu);
1888 }
1889
1890 /* SGI base address */
1891 gic_core_gicr += ARM_GICR_SGI_BASE_OFFSET;
1892 if (debug_flag > 1) {
1893 kprintf("cpu%d: Core GICR SGI address: 0x%x\n", cpu, gic_core_gicr);
1894 }
1895
1896 /* SGIs and PPs are in Group1NS (GICD_CTLR.DS = 0) */
1897 out32(gic_core_gicr + ARM_GICR_IGROUPR0, 0xFFFFFFFF);
1898 out32(gic_core_gicr + ARM_GICR_IGRPMODR0, 0x00000000);
1899
1900 /* Set SGI/PPI priority (all 0xA0, except for SGI0 which is 0xC0) */
1901 out32(gic_core_gicr + ARM_GICR_IPRIORITYR0, 0xA0A0A0C0);
1902 unsigned i;
1903 for(i = 1; i < 8; ++i) {
1904 out32(gic_core_gicr + ARM_GICR_IPRIORITYR0 + (i*4), 0xA0A0A0A0);
1905 }
1906
1907 /* Clear/disable all SGI/PPI interrupts */
1908 out32(gic_core_gicr + ARM_GICR_ICPENDR0, 0xFFFFFFFF);
1909 out32(gic_core_gicr + ARM_GICR_ICENABLER0, 0xFFFFFFFF);
1910
1911 /* Enable SGI 0 interrupt (used for IPI's) */
1912 out32(gic_core_gicr + ARM_GICR_ISENABLER0, 0x0000001);
1913
1914 wait_for_rwp(gic_core_gicr - ARM_GICR_SGI_BASE_OFFSET + ARM_GICR_CTLR, 0x08);

1916 /* Initialize CPU GIC interface part */
1917 struct aarch64_gic_map_entry *gic_map = lsp.cpu.aarch64_gic_map.p;
1918 if (gic_sendipi == &sendipi_gic_v3_mm)
1919 {
1920 /* using memory mapped interface */
1921 // XXX assumes cpu is the same as gicr_index
1922 gic_map->gic_cpu[cpu] = ((1u << gicr_offset_shift_val)*cpu)
1923 + ARM_GICR_SGI_BASE_OFFSET+ARM_GICR_ISPENDR0;
1924
1925 /* set priority mask to allow all interrupts */
1926 out32(gicc_paddr_base + ARM_GICC_PMR, 0xF0);
1927 /* set binary point value */
1928 out32(gicc_paddr_base + ARM_GICC_BPR, 0);
1929 /* enable group 1 interrupts, disable IRQ/FIQ bypass signals */
1930 out32(gicc_paddr_base + ARM_GICC_CTLR, 0x61);
1931 }
1932 else
1933 {
1934 /* Create value for ARM_GICC_SGI1R_EL1 register used in "sendipi" callout function */
1935 _Uint64t const aff0 = cpu_aff & 0xfful;
1936 _Uint64t const aff1 = (cpu_aff >> 8) & 0xfful;
1937 _Uint64t const aff2 = (cpu_aff >> 16) & 0xfful;
1938 _Uint64t const aff3 = (cpu_aff >> 24) & 0xfful;
1939
1940 _Uint64t icc_sgi1r = 1ul << (aff0 & 0x0ful);
1941 icc_sgi1r |= (aff1 << 16);
1942 icc_sgi1r |= (aff2 << 32);
1943 icc_sgi1r |= (aff3 << 48);
1944
1945 if (aff0 > 0x0ful) {
1946 _Uint64t const ctlr = aa64_sr_rd64(S3_0_C12_C12_4); // ARM_GICC_CTLR_EL1

1952 icc_sgi1r |= (aff0 & 0xf0ul) << (44-4);

1958 kprintf("WARNING: cpu%d: aff0: %d and ICC_CTLR_EL1.RSS = 0\n", cpu, aff0);
1959 }
1960 }
1961

1963
1964 /* Enable the system register interface (icc_sre_el1) */
1965 aa64_sr_wr32(S3_0_C12_C12_5, 7);
1966 isb();
1967
1968 /* Set priority mask to allow all interrupts and enable cpu interface Set ARM_GICC_PMR = 0x0F */
1969 aa64_sr_wr32(S3_0_C4_C6_0, 0xf0);
1970
1971 /* Set binary point value, ARM_GICC_BPR1_EL1 */
1972 aa64_sr_wr32(S3_0_C12_C12_3, 0);

1975 aa64_sr_wr32(S3_0_C12_C12_4, 0);
1976
1977 /* Enable signaling of each interrupt group, ARM_GICC_IGRPEN1_ELx = 1 */
1978 aa64_sr_wr32(S3_0_C12_C12_7, 1);
1979 }
1980
1981 if (debug_flag > 1) {
1982 kprintf("cpu%d: GICv3 cpu interface 0x%l\n", cpu, gic_map->gic_cpu[cpu]);
1983 }
