33static char *
34find_next_arg(char *str, char ** const end, unsigned terminate)
35{
36 while (*str == ' ') {
37 ++str;
38 }
39 if (*str == '\0') {
40 *end = str;
41 return str;
42 }
43
44 char *next = str;
45 while ((*next != ' ') && (*next != '\0')) {
46 ++next;
47 }
48
49 if (*next != '\0') {
50 if (terminate) {
51 *next = '\0';
52 }
53 ++next;
54 }
55 *end = next;
56 return str;
57}

82 int len;

84 if (prop == NULL) {

96 if (*cmdline == '\0') return;
97 if (!strcmp(cmdline, "--")) break;
