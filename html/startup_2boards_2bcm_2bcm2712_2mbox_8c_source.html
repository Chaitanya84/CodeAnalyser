147static uint32_t mbox_send_message(void) {
148 uint32_t rdata;
149
150 // wait for ready
151 for (;;) {
152 rdata = *((volatile uint32_t *) (MBOX_REG_BASE + MBOX0_STATUS));
153 if (!(rdata & MBOX_STATUS_FULL)) {
154 break;
155 }
156 }
157
158 // send
159 rdata = (MBOX_BUFFER_ADDR & ~0xf) | MBOX_SEND_CHANNEL;
160 *((volatile uint32_t *) (MBOX_REG_BASE + MBOX1_WRITE)) = rdata;
161
162 // wait for tx done
163 for (;;) {
164 rdata = *((volatile uint32_t *) (MBOX_REG_BASE + MBOX0_STATUS));
165 if (!(rdata & MBOX_STATUS_FULL)) {
166 break;
167 }
168 }
169
170 for (;;) {
171 // wait for response
172 rdata = *((volatile uint32_t *) (MBOX_REG_BASE + MBOX0_STATUS));
173 if (rdata & MBOX_STATUS_EMPTY) {
174 continue;
175 }
176
177 // make sure it's the channel we're waiting for
178 rdata = *((volatile uint32_t *) (MBOX_REG_BASE + MBOX0_READ));
179 if ((rdata & 0xf) == MBOX_SEND_CHANNEL) {
180 return rdata & ~0xf;
181 }
182 }
183
184 return 0;
185}

267char *mbox_get_cmdline(void) {
268 volatile struct {
269 struct mbox_msg_header header;
270 struct mbox_msg_tag tag;
271 struct {
272 char line[512 - 24];
273 } body;
274 uint32_t end;
275 } *msg = (volatile void *) MBOX_BUFFER_ADDR;
276
277 MSG_INIT(msg, MBOX_TAG_GET_CMDLINE);
278 mbox_send_message();
279
280 unsigned len = msg->tag.code & ~MBOX_TAG_RESPONSE;
281 if (len >= sizeof msg->body.line) {
282 return 0;
283 }
284 msg->body.line[len] = '\0';
285 return (char *) msg->body.line;
286}
