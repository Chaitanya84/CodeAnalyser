26/*
27 ===============================================================================
28 arm_gic_version_t
29
30 This type is used to encode major/minor version information for both the GIC
31 architecture as well as the implementation into a _Uint16t compatible type
32 We'll keep both a major and minor to handle the nuances of minor revisions
33
34*/
35typedef struct __attribute__((packed,aligned(2)))
36{
37 /* a UINT8_MAX value is invalid and indicates no information */
38 _Uint8t major;
39 _Uint8t minor;
40} arm_gic_version_t;
41
42/*
43 ===============================================================================
44 ARM_GIC_ARCH_VERSION_IS
45
46 These macros are provided for easy architecture checks within code and can be
47 used as follows
48
49 const arm_gic_version_t arch_ver = gic_get_arch_version();
50 if (ARM_GIC_ARCH_VERSION_IS(arch_ver) == ARM_GIC_ARCH_VERSION_3_1)
51 {
52 }
53
54*/
55#define __agav_major_shift (offsetof(arm_gic_version_t, major) * 8)
56#define __agav_minor_shift (offsetof(arm_gic_version_t, minor) * 8)
57
58static inline _Uint16t _arm_gic_version_is(const arm_gic_version_t version)
59{
60 return *((_Uint16t *)&version);
61}
62#define ARM_GIC_ARCH_VERSION_IS(_v_) _arm_gic_version_is((_v_))
63#define ARM_GIC_ARCH_VERSION_1_0 ((_Uint16t)(((_Uint8t)1 << __agav_major_shift) | ((_Uint8t)0 << __agav_minor_shift)))
64#define ARM_GIC_ARCH_VERSION_2_0 ((_Uint16t)(((_Uint8t)2 << __agav_major_shift) | ((_Uint8t)0 << __agav_minor_shift)))
65#define ARM_GIC_ARCH_VERSION_3_0 ((_Uint16t)(((_Uint8t)3 << __agav_major_shift) | ((_Uint8t)0 << __agav_minor_shift)))
66#define ARM_GIC_ARCH_VERSION_3_1 ((_Uint16t)(((_Uint8t)3 << __agav_major_shift) | ((_Uint8t)1 << __agav_minor_shift)))
67#define ARM_GIC_ARCH_VERSION_4_0 ((_Uint16t)(((_Uint8t)4 << __agav_major_shift) | ((_Uint8t)0 << __agav_minor_shift)))

69/*
70 ===============================================================================
71 arm_gic_id_t
72
73 This type is used to encode package identification and version information for
74 the GIC implementation into a _Uint64t compatible type.
75 It effectively contains the IIDR and PIDR2 information in a form that does not
76 require additional masking and shifting and allows for easier differentiation
77 of architecture minor versions
78
79*/
80typedef struct __attribute__((packed,aligned(8)))
81{
82 arm_gic_version_t arch_version;
83 struct
84 {
85 arm_gic_version_t version;
86 _Uint16t company;
87 _Uint8t id;
88 _Uint8t resv;
89 } implementation;
90} arm_gic_id_t;
91
92/*
93 ===============================================================================
94 GIC_IMPL_ID_IS
95
96 These macros are provided for easy GIC implementation identifier checks within
97 code and can be used as follows
98
99 const arm_gic_id_t gic_id = gic_get_id();
100 if (gic_id == ARM_GIC_500_r0p1)
101 {
102 }
103
104*/
105#define __gii_archver_shift (offsetof(arm_gic_id_t, arch_version) * 8)
106#define __gii_implver_shift (offsetof(arm_gic_id_t, implementation.version) * 8)
107#define __gii_implco_shift (offsetof(arm_gic_id_t, implementation.company) * 8)
108#define __gii_implid_shift (offsetof(arm_gic_id_t, implementation.id) * 8)
109
110static inline _Uint64t _gic_impl_id_is(const arm_gic_id_t impl_id)
111{
112 return *((_Uint64t *)&impl_id);
113}
114#define GIC_IMPL_ID_IS(_v_) _gic_impl_id_is((_v_))

618 arm_gic_lpi_params_t
619
620 This structure defines the parameters associated with GIC LPIs
621
622*/
623typedef struct
624{
625 _Uint32t num_lpis; /* 0 - implementation defined, max 4294967295 */
626 /*
627 * Normally there is one LPI configuration table for all redistributors
628 * and an LPI pending table per redistributor. However the spec does not
629 * preclude multiple LPI configuration tables and in theory the attributes
630 * of each of the config and pending tables "could" be different. While
631 * we don't expect this to be the case under normal circumstances, we'll
632 * accommodate it. The default attributes for the config and pending tables
633 * will be held in the 'cfg.dflt_tbl_params' and 'pending.dflt_tbl_params'
634 * fields respectively and the 'cfg.tbl_params' and 'pending.tbl_params'
635 * pointers will be initialized to point to the corresponding one. The
636 * 'num_tbls' fields will be set to 1.
637 * If it is decided to have different attributes for either the config or
638 * the pending tables, then the 'cfg.tbl_params' and/or 'pending.tbl_params'
639 * fields will point to suitably allocated memory to hold as many
640 * 'struct _lpi_tbl_params_s' structures as required and the 'respective
641 * num_tbls' fields updated accordingly.
642 * In this case the 'cfg.dflt_tbl_params' and/or 'pending.dflt_tbl_params'
643 * fields will not be used other than to perhaps initialize the structures
644 * in the allocated memory
645 */
646 struct _lpi_tbl_s
647 {
648 unsigned num_tbls;
649 struct _lpi_tbl_params_s
650 {
651 paddr_t paddr;
652 _Uintptrt vaddr;
653 _Uint64t propbaser; /* value of the GIRC PROPBASER register */
654 int mmap_flags;
655 gic_tbl_params_t mem;
656 } dflt_tbl_params;
657 struct _lpi_tbl_params_s *tbl_params;
658 } cfg;
659 struct _lpi_tbl_s pending;

661} arm_gic_lpi_params_t;
