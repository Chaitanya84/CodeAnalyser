34#define CPUID_MASK 0xff00fff0u

36struct aarch64_cpuid *
37aarch64_cpuid_find(void) {
38 unsigned const cpuid = aa64_sr_rd32(midr_el1);
39
40 struct aarch64_cpuid *id;
41 struct aarch64_cpuid **idp = aarch64_cpuid;
42 for(;;) {
43 id = *idp;
44 if(id == NULL) break;
45 if(id->midr == (cpuid & CPUID_MASK)) break;
46 ++idp;
47 }
48 return id;
49}
