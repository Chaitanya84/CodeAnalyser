53static int sd_parse_cid( sdio_dev_t * const dev, sdio_cid_t *cid, uint32_t * const rsp )
54{
55 int idx;
56
57 memset( cid, 0, sizeof( sdio_cid_t ) );
58
59 cid->mid = sdio_extract_bits( rsp, 128, 120, 8 );
60 cid->oid = sdio_extract_bits( rsp, 128, 104, 16 );
61
62 for( idx = 0; idx < 5; idx++ ) {
63 cid->pnm[idx] = (char)sdio_extract_bits( rsp, 128, 96 - idx * 8, 8 );
64 }
65
66 cid->psn = sdio_extract_bits( rsp, 128, 24, 32 );
67 cid->prv = sdio_extract_bits( rsp, 128, 56, 8 );
68 cid->month = sdio_extract_bits( rsp, 128, 8, 4 );
69 cid->year = sdio_extract_bits( rsp, 128, 12, 8 ) + 2000;
70
71 return( EOK );
72}

74static int sd_parse_csd( sdio_dev_t *dev, sdio_csd_t *csd, uint32_t * const rsp )
75{
76 uint32_t blksz;
77 uint32_t csize;
78 uint32_t csizem;
79 static const uint32_t sd_tran_speed_fu[] = { 10000, 100000, 1000000, 10000000, 0, 0, 0, 0 };
80 static const uint32_t sd_tran_speed_mf[] = { 0, 10, 12, 13, 15, 20, 25, 30,
81 35, 40, 45, 50, 55, 60, 70, 80 };
82
83 memset( csd, 0, sizeof( sdio_csd_t ) );
84 csd->csd_structure = (uint8_t)sdio_extract_bits( rsp, 128, 126, 2 );
85 csd->taac = (uint8_t)sdio_extract_bits( rsp, 128, 112, 8 );
86 csd->nsac = (uint8_t)sdio_extract_bits( rsp, 128, 104, 8 );
87 csd->tran_speed = (uint8_t)sdio_extract_bits( rsp, 128, 96, 8 );
88 csd->ccc = (uint16_t)sdio_extract_bits( rsp, 128, 84, 12 );
89 csd->read_bl_len = (uint8_t)sdio_extract_bits( rsp, 128, 80, 4 );
90 csd->read_bl_partial = (uint8_t)sdio_extract_bits( rsp, 128, 79, 1 );
91 csd->write_blk_misalign = (uint8_t)sdio_extract_bits( rsp, 128, 78, 1 );
92 csd->read_blk_misalign = (uint8_t)sdio_extract_bits( rsp, 128, 77, 1 );
93 csd->dsr_imp = (uint8_t)sdio_extract_bits( rsp, 128, 76, 1 );
94
95 switch( csd->csd_structure ) {
96 case CSD_STRUCT_VER_10: // Standard Capacity
97 csd->c_size = sdio_extract_bits( rsp, 128, 62, 12 );
98 csd->vdd_r_curr_min = (uint8_t)sdio_extract_bits( rsp, 128, 59, 3 );
99 csd->vdd_r_curr_max = (uint8_t)sdio_extract_bits( rsp, 128, 56, 3 );
100 csd->vdd_w_curr_min = (uint8_t)sdio_extract_bits( rsp, 128, 53, 3 );
101 csd->vdd_w_curr_max = (uint8_t)sdio_extract_bits( rsp, 128, 50, 3 );
102 csd->c_size_mult = (uint8_t)sdio_extract_bits( rsp, 128, 47, 3 );
103 blksz = 1 << csd->read_bl_len;
104 csize = csd->c_size + 1;
105 csizem = 1 << ( csd->c_size_mult + 2 );
106 break;
107
108 case CSD_STRUCT_VER_20: // High Capacity / Extended Capacity
109 csd->c_size = sdio_extract_bits( rsp, 128, 48, 22 );
110 blksz = SDIO_DFLT_BLKSZ;
111 csize = csd->c_size + 1;
112 csizem = 1024;
113 dev->caps |= DEV_CAP_HC;
114 break;
115
116 case CSD_STRUCT_VER_30: // Ultra Capacity
117 csd->c_size = sdio_extract_bits( rsp, 128, 48, 28 );
118#ifndef SDIO_UC_SUPPORT
119 // limit size to 22 bits until we get a card to test
120 if( csd->c_size > CSD_VER_20_CSIZE_MAX ) {
121 csd->c_size = CSD_VER_20_CSIZE_MAX;
122 }
123#endif
124 blksz = SDIO_DFLT_BLKSZ;
125 csize = csd->c_size + 1;
126 csizem = 1024;
127 dev->caps |= DEV_CAP_HC | DEV_CAP_UC;
128 break;
129
130 default:
131 return( EINVAL );
132 }
133
134 csd->erase_blk_en = (uint8_t)sdio_extract_bits( rsp, 128, 46, 1 );
135 csd->sector_size = (uint8_t)sdio_extract_bits( rsp, 128, 39, 7 );
136 csd->wp_grp_size = (uint8_t)sdio_extract_bits( rsp, 128, 32, 7 );
137 csd->wp_grp_enable = (uint8_t)sdio_extract_bits( rsp, 128, 31, 1 );
138 csd->r2w_factor = (uint8_t)sdio_extract_bits( rsp, 128, 26, 3 );
139 csd->write_bl_len = (uint8_t)sdio_extract_bits( rsp, 128, 22, 4 );
140 csd->write_bl_partial = (uint8_t)sdio_extract_bits( rsp, 128, 21, 1 );
141// csd->file_format_grp = (uint8_t)sdio_extract_bits( rsp, 128, 15, 1 );
142// csd->copy = (uint8_t)sdio_extract_bits( rsp, 128, 14, 1 );
143 csd->write_protect = (uint8_t)sdio_extract_bits( rsp, 128, 12, 2 );
144// csd->file_format = (uint8_t)sdio_extract_bits( rsp, 128, 10, 2 );
145
146 // force to 512 byte block
147 if( ( blksz > SDIO_DFLT_BLKSZ ) && ( ( blksz % SDIO_DFLT_BLKSZ ) == 0 ) ) {
148 csize = csize * ( blksz / SDIO_DFLT_BLKSZ );
149 blksz = SDIO_DFLT_BLKSZ;
150 }
151
152 csd->blksz = blksz;
153 csd->sectors = csize * csizem;
154
155 csd->dtr_max = sd_tran_speed_fu[csd->tran_speed & 0x7] *
156 sd_tran_speed_mf[(csd->tran_speed >> 3 ) & 0xf];
157
158 return( EOK );
159}

188static int sd_rw_extr( sdio_dev_t * const dev, const int dir, const int mio, const int fno, const int mw, const uint32_t addr, const int len_msk, uint8_t * const extr )
189{
190 sdio_cmd_t *cmd;
191 int cc;
192 int arg;
193 uint8_t *ebuf;
194 int status;
195 sdio_sge_t sge;
196
197 cmd = sdio_alloc_cmd( );
198 if( cmd == NULL ) {
199 return( ENOMEM );
200 }
201
202 ebuf = sdio_alloc( SD_EXTR_SIZE );
203 if( ebuf == NULL ) {
204 sdio_free_cmd( cmd );
205 return( ENOMEM );
206 }
207
208 memset( ebuf, 0, SD_EXTR_SIZE );
209
210 cc = ( dir & SCF_DIR_OUT ) ? SD_WRITE_EXTR_SINGLE : SD_READ_EXTR_SINGLE;
211
212#if 0
213[31]MIO 0:Memory, 1:I/O
214[30:27] FNO
215[26] MW
216[25:9] ADDR
217[8:0] LEN/MASK mw=1 8bits, mw=0 9bits
218#endif
219
220 arg = SD_EXTR_MIO( mio );
221 arg |= SD_EXTR_FNO( fno );
222 arg |= ( SD_EXTR_MWM( mw ) );
223 arg |= ( SD_EXTR_ADDR( addr ) );
224 arg |= ( SD_EXTR_LEN_MSK( len_msk, mw ) );
225
226 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, cc, arg );
227 sge.sg_count = SD_EXTR_SIZE;
228 sge.sg_address = SDIO_DATA_PTR_P( ebuf );
229 sdio_setup_cmd_io( cmd, dir, 1, SD_EXTR_SIZE, &sge, 1, NULL );
230
231 if( ( dir & SCF_DIR_OUT ) ) {
232 memcpy( ebuf, extr, mw ? 1 : len_msk );
233 }
234
235 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
236 if( status == EOK ) {
237 if( ( dir & SCF_DIR_IN ) ) {
238 memcpy( extr, ebuf, mw ? 1 : len_msk );
239 }
240 }
241 else {
242 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: Error %s %s extention register (%d) - fno %d, addr 0x%x, %s 0x%x",
243 __FUNCTION__, ( dir & SCF_DIR_OUT ) ? "writting" : "reading", mio ? "io" : "memory", status, fno, addr,
244 mw ? "mask" : "len", len_msk );
245
246 }
247
248 sdio_free( ebuf, SD_EXTR_SIZE );
249
250 sdio_free_cmd( cmd );
251
252 return( status );
253}

265static int sdio_send_relative_addr( sdio_hc_t *hc, uint32_t *rca )
266{
267 sdio_dev_t *dev;
268 sdio_cmd_t *cmd;
269 int status;
270 int retries;
271
272 cmd = sdio_alloc_cmd( );
273 if( cmd == NULL ) {
274 return( ENOMEM );
275 }
276
277 dev = &hc->device;
278 *rca = 0;
279 retries = SD_SEND_RELATIVE_ADDR_RETRIES;
280
281 do {
282 sdio_setup_cmd( cmd, SCF_CTYPE_BCR | SCF_RSP_R6, SD_SEND_RELATIVE_ADDR, 0 );
283 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, SDIO_CMD_RETRIES );
284 if( status != EOK ) {
285 break;
286 }
287
288 *rca = cmd->rsp[0] >> 16;
289 if( *rca != 0 ) { // wait for non-zero RCA
290 break;
291 }
292 } while( retries-- );
293
294 if( *rca == 0 ) { // validate RCA
295 status = EIO;
296 }
297
298 sdio_free_cmd( cmd );
299
300 return( status );
301}

341int sd_cache( sdio_dev_t *dev, const int op, uint32_t timeout )
342{
343 uint8_t rval;
344 int status;
345
346 switch( op ) {
347 case SDIO_CACHE_DISABLE:
348 status = sd_extr_wr( dev, dev->perf_enh_fcn_rsa, PERF_ENH_CACHE, 0 );
349 if( status == EOK ) {
350 dev->flags &= ~DEV_FLAG_WCE;
351 }
352 break;
353
354 case SDIO_CACHE_ENABLE:
355 status = sd_extr_wr( dev, dev->perf_enh_fcn_rsa, PERF_ENH_CACHE, PERF_ENH_CACHE_EN );
356 if( status == EOK ) {
357 dev->flags |= DEV_FLAG_WCE;
358 }
359 break;
360
361 case SDIO_CACHE_FLUSH:
362 if( ( dev->flags & DEV_FLAG_WCE ) ) {
363 if( ( status = sd_extr_wr( dev, dev->perf_enh_fcn_rsa, PERF_ENH_CACHE_FLUSH, PERF_ENH_CACHE_FLUSH_START ) ) ) {
364 return( status );
365 }
366
367 // wait up to 1 second
368 for( timeout = 1000; timeout; timeout-- ) {
369 status = sd_extr_rd( dev, dev->perf_enh_fcn_rsa, PERF_ENH_CACHE_FLUSH, &rval );
370 if( status == EOK ) {
371 if( ( rval & PERF_ENH_CACHE_FLUSH_START ) == 0 ) {
372 status = EOK;
373 break;
374 }
375 delay( 1 );
376 }
377 }
378
379 if( !timeout ) {
380 status = ETIMEDOUT;
381 }
382 }
383 else {
384 status = EOK;
385 }
386 break;
387
388 default:
389 status = EINVAL;
390 break;
391 }
392
393 return( status );
394}

415int sd_switch( sdio_dev_t * const dev, const int mode, const int grp, const uint8_t val, uint8_t * const switch_status )
416{
417 sdio_cmd_t *cmd;
418 uint32_t arg;
419 uint8_t *sbuf;
420 int status;
421 uint16_t sw_status;
422 sdio_sge_t sge;
423
424 cmd = sdio_alloc_cmd( );
425 if( cmd == NULL ) {
426 return( ENOMEM );
427 }
428
429 sbuf = sdio_alloc( SD_SF_STATUS_SIZE );
430 if( sbuf == NULL ) {
431 sdio_free_cmd( cmd );
432 return( ENOMEM );
433 }
434
435 arg = ( mode << 31 ) | 0x00ffffffU;
436 arg &= ~( 0xf << ( grp * 4 ) );
437 arg |= ( val << ( grp * 4 ) );
438
439 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, SD_SWITCH_FUNC, arg );
440 sge.sg_count = SD_SF_STATUS_SIZE;
441 sge.sg_address = SDIO_DATA_PTR_P( sbuf );
442 sdio_setup_cmd_io( cmd, SCF_DIR_IN, 1, SD_SF_STATUS_SIZE, &sge, 1, NULL );
443
444 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
445 if( status == EOK ) {
446 memcpy( switch_status, sbuf, SD_SF_STATUS_SIZE );
447
448 if( mode == SD_SF_MODE_SET ) {
449 // check function busy status
450 sw_status = switch_status[29 - ( grp * 2 )] | ( switch_status[28 - ( grp * 2 )] << 8 );
451 if( sw_status & ( 1 << val ) ) {
452 status = EBUSY;
453 }
454 }
455 }
456 else {
457 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: Error Switching %d", __FUNCTION__, status );
458 }
459
460 sdio_free( sbuf, SD_SF_STATUS_SIZE );
461 sdio_free_cmd( cmd );
462
463 return( status );
464}

595 addr = ( dev->caps & DEV_CAP_HC ) ? lba : ( lba * dev->csd.blksz );
596 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1, SD_ERASE_WR_BLK_START, addr );
597 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
598 if( status != EOK ) {
599 break;
600 }
601
602 timeout = sd_erase_timeout( dev, flgs, nlba );
603 lba += nlba;
604 nlba -= nlba;
605 lba--;
606
607 addr = ( dev->caps & DEV_CAP_HC ) ? lba : ( lba * dev->csd.blksz );
608 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1, SD_ERASE_WR_BLK_END, addr );
609 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
610 if( status != EOK ) {
611 break;
612 }
613
614 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1B | SCF_WAIT_DRDY, SD_ERASE, flgs );
615 status = _sdio_send_cmd( dev, cmd, NULL, timeout, 0 );
616 if( status != EOK ) {
617 break;
618 }
619 }
620
621 sdio_free_cmd( cmd );
622
623 return( status );
624}

711static int sd_read_switch( sdio_dev_t *dev )
712{
713 sdio_hc_t *hc;
714 sd_switch_cap_t *swcaps;
715 int status;
716 uint8_t ss[SD_SF_STATUS_SIZE];
717
718 hc = dev->hc;
719 swcaps = &dev->swcaps;
720
721 if( !( dev->csd.ccc & CCC_SWITCH ) ) {
722 return( EOK );
723 }
724
725 // get supported bus speeds
726 if( ( status = sd_switch( dev, SD_SF_MODE_CHECK, SD_SF_GRP_BUS_SPD, SD_SF_CUR_FCN, ss ) ) ) {
727 return( status );
728 }
729
730 if( ( ss[13] & 0x2 ) ) {
731 swcaps->dtr_max_hs = 50000000;
732
733 if( ( hc->caps & HC_CAP_HS ) ) {
734 dev->caps |= DEV_CAP_HS;
735 }
736 }
737
738 if( dev->scr.sd_spec3 ) {
739 swcaps->bus_mode = ss[13] & SD_BUS_MODE_MSK;
740 swcaps->drv_type = ss[9] & SD_DRV_TYPE_MSK;
741 swcaps->curr_limit = ss[7] & SD_CURR_LIMIT_MSK;
742
743 if( ( swcaps->bus_mode & SD_BUS_MODE_UHS ) ) {
744 dev->caps |= DEV_CAP_UHS;
745 }
746 }
747
748 swcaps->cmd_sys = ss[11];
749
750 if( ( ss[11] & SD_CMD_SYS_EC ) || ( ss[11] & SD_CMD_SYS_ASSD ) ) {
751 dev->caps |= DEV_CAP_ASSD;
752 }
753
754 return( status );
755}

757static int sd_read_status( sdio_dev_t *dev )
758{
759 int idx;
760 int status;
761 uint32_t au;
762 uint32_t e_size;
763 uint32_t e_offset;
764 uint32_t e_timeout;
765 uint32_t sds[SD_STATUS_SIZE / 4];
766
767
768 if( !( dev->csd.ccc & CCC_APP_SPEC ) ) {
769 return( EOK );
770 }
771
772 if( ( status = sd_app_sd_status( dev, sds ) ) ) {
773 return( status );
774 }
775
776 for( idx = 0; idx < 16; idx++ ) {
777 sds[idx] = ENDIAN_BE32( sds[idx] );
778 }
779
780 au = sdio_extract_bits( sds, 512, 428, 4 );
781 if( au ) {
782 dev->sds.au_size = 1 << ( au + 4 );
783
784 e_size = sdio_extract_bits( sds, 512, 408, 16 );
785 e_offset = sdio_extract_bits( sds, 512, 400, 2 );
786 e_timeout = sdio_extract_bits( sds, 512, 402, 6 );
787 if( e_size && e_timeout ) {
788 dev->sds.erase_offset = e_offset * 1000;
789 dev->sds.erase_timeout = ( e_timeout * 1000 ) / e_size;
790 }
791 }
792
793 au = sdio_extract_bits( sds, 512, 392, 4 );
794 if( au ) {
795// dev->sds.uhs_au_size = 1 << ( au + 4 );
796 }
797
798 dev->sds.perf_class = sdio_extract_bits( sds, 512, 336, 4 );
799 dev->sds.perf_enh = sdio_extract_bits( sds, 512, 328, 8 );
800 dev->sds.speed_class = sdio_extract_bits( sds, 512, 440, 8 );
801 dev->sds.uhs_speed_grade = sdio_extract_bits( sds, 512, 396, 4 );
802 return( status );
803}

805__attribute__((used))
806static int sd_parse_extended_fcn( sdio_dev_t *dev )
807{
808 sd_fcn_ext_hdr *feh;
809 sd_fcn_ext *fext;
810 uint32_t fcn;
811 uint8_t rval;
812 int status;
813 uint8_t efb[SD_FCN_PG_SZ * 2] = { 0 }; // 512 * 2
814
815 feh = (sd_fcn_ext_hdr *)efb;
816 fext = (sd_fcn_ext *)(feh + 1);
817
818 if( !( dev->scr.cmd_support & SCR_CMD48_49_SUP ) ) {
819 return( ENOTSUP );
820 }
821
822 // read general information page 0
823 if( ( status = sd_rw_extr( dev, SCF_DIR_IN, SD_EXTR_MEM, 0, 0, 0, SD_FCN_PG_SZ, efb ) ) ) {
824 return( status );
825 }
826
827 // read general information page 1 (optional no error check)
828 sd_rw_extr( dev, SCF_DIR_IN, SD_EXTR_MEM, 0, 0, SD_FCN_PG_SZ, SD_FCN_PG_SZ, efb + SD_FCN_PG_SZ );
829
830 for( fcn = 1; fcn <= feh->number_extensions; fcn++ ) {
831 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, dev->hc->cfg.verbosity, 0, "%s: fcn %d, sfc %d, fcc %d, fmc %d, fmn %s, pfc %d, fn %s, pne %d, nrs %d, rsa 0x%x, fno %d, saddr 0x%x",
832 __FUNCTION__, fcn, fext->sfc, fext->fcc, fext->fmc, fext->fmn, fext->pfc, fext->fn, fext->pne, fext->nrs,
833 fext->rsa[0], SD_FCN_EXT_FNO( fext->rsa[0] ), SD_FCN_EXT_SADDR( fext->rsa[0] ) );
834 switch( fext->sfc ) {
835 case SFC_PWR_MGNT:
836 dev->pwr_mgnt_fcn_rsa = fext->rsa[0];
837 status = sd_extr_rd( dev, dev->pwr_mgnt_fcn_rsa, PWR_MGNT_STATUS, &rval );
838 if( status == EOK ) {
839 if( ( rval & PWR_MGNT_STATUS_POFS ) ) {
840 dev->caps |= DEV_CAP_PWROFF_NOTIFY;
841 }
842 }
843 break;
844 case SFC_PERF_ENH:
845 dev->perf_enh_fcn_rsa = fext->rsa[0];
846 status = sd_extr_rd( dev, dev->perf_enh_fcn_rsa, PERF_ENH_CACHE_SUP, &rval );
847 if( status == EOK ) {
848 if( ( rval & PERF_ENH_CACHE_SUPPORTED ) ) {
849 dev->caps |= DEV_CAP_CACHE;
850 }
851 }
852 break;
853 default:
854 break;
855 }
856 fext = (sd_fcn_ext *)&efb[fext->pne];
857
858 // range check
859 if( ( (uintptr_t)fext + sizeof( sd_fcn_ext ) ) > (uintptr_t)( efb + sizeof( efb ) ) ) {
860 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, dev->hc->cfg.verbosity, 0, "%s: Range limit", __FUNCTION__ );
861 break;
862 }
863 }
864
865 return( status );
866}

1377 return( status );
1378 }

1381 return( ENXIO );
1382 }
1383
1384 memcpy( dev->raw_cid, cid, sizeof( cid ) );
1385

1389 }
