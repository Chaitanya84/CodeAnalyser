26void uefi_print_str(char *str)
27{
28 if ((efi_system_table != NULL) &&
29 (efi_system_table->ConOut != NULL) &&
30 (efi_system_table->ConOut->OutputString != NULL))
31 {
32 const unsigned len = strlen(str);
33 unsigned carriage_ret = 0;
34 unsigned i;
35
36 for (i=0; i<len; i++) {
37 if (str[i] == '\n') ++carriage_ret;
38 }
39 _Char16t *uefi_str = alloca((2 * len) + ((1 + carriage_ret) * sizeof(*uefi_str)));
40
41 if (uefi_str != NULL)
42 {
43 unsigned j;
44 for (i=j=0; i<len; i++, j++) {
45 uefi_str[j] = str[i];
46 if (str[i] == '\n') uefi_str[++j] = '\r';
47 }
48 uefi_str[j] = '\0';
49
50 efi_system_table->ConOut->OutputString(efi_system_table->ConOut, uefi_str);
51 }
52 }
53}

86/*
87 * The suspend/resume are matched and should only be called by the boot processor
88 *
89 * They are required because the AP's cannot seem to use the boot services so in
90 * order to defer exiting boot services as long as possible so we get as much
91 * output as possible, we suspend/resume around start_aps()
92*/
93static EFI_SIMPLE_TEXT_OUTPUT_PROTOCOL *cur_ConOut = NULL;
94static unsigned suspend_cnt = 0;
95
96static void _uefi_io_suspend(void)
97{
98 if (suspend_cnt++ == 0)
99 {
100 uefi_io_flush();
101 cur_ConOut = efi_system_table->ConOut;
102 efi_system_table->ConOut = NULL;
103 }
104}
105static void _uefi_io_resume(void)
106{
107 if (suspend_cnt == 1)
108 {
109 efi_system_table->ConOut = cur_ConOut;
110 }
111 --suspend_cnt;
112}
