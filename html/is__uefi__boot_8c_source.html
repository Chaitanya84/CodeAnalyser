21/* return true (!0) or false (0) depending on whether this was a UEFI boot */
22unsigned is_uefi_boot(void)
23{
24 static unsigned uefi = 0x55555555; /* a value that a boolean will never be */
25
26 /* Assign the standard arguments for an UEFI application */
27 efi_image_handle = (void*)boot_regs[0];
28 efi_system_table = (void*)boot_regs[1];
29
30 if (uefi == 0x55555555)
31 {
32 uefi = 0;
33
34 do {
35 /* Check validity of the UEFI service pointers */
36 if (efi_system_table == NULL) {
37 break;
38 }
39 if (efi_system_table->BootServices == NULL) {
40 break;
41 }
42 if (efi_system_table->RuntimeServices == NULL) {
43 break;
44 }
45
46 /* Verify EFI_SYSTEM_TABLE signature */
47 if (efi_system_table->Hdr.Signature != EFI_SYSTEM_TABLE_SIGNATURE) {
48 break;
49 }
50
51 /* Verify EFI_BOOT_SERVICES signature */
52 if (efi_system_table->BootServices->Hdr.Signature != EFI_BOOT_SERVICES_SIGNATURE) {
53 break;
54 }
55
56 /* Verify EFI_RUNTIME_SERVICES signature */
57 if (efi_system_table->RuntimeServices->Hdr.Signature != EFI_RUNTIME_SERVICES_SIGNATURE) {
58 break;
59 }
60 uefi = 1;
61
62 } while (0);
63
64 if (uefi == 0) {
65 efi_image_handle = NULL;
66 efi_system_table = NULL;
67 }
68 }
69
70 return uefi;
71}
