32static void
33get_freq(unsigned long *fp) {
34 unsigned long freq;
35 char *end;
36
37 freq = strtoul(optarg, &end, 10);
38 if(optarg != end) {
39 switch(*end) {
40 case 'm': case 'M':
41 freq *= 1000;
42 /* fall through */
43 case 'k': case 'K':
44 freq *= 1000;
45 /* fall through */
46 case 'h': case 'H':
47 ++end;
48 break;
49 default:
50 if(freq < 1000) freq *= 1000000;
51 break;
52 }
53 *fp = freq;
54 optarg = end;
55 }
56 if(*optarg == ',') ++optarg;
57}

59void
60handle_common_option(int opt) {
61 unsigned num;
62 char *cp;
63 unsigned long hundred_loop_time, overhead;
64 struct reserved_ram *next;
65 paddr_t resmem_addr;
66 size_t resmem_size;
67
68 if(!cpu_handle_common_option(opt)) {
69 switch(opt) {
70 case 'C':
71 secure_system = 1;
72 break;
73 case 'c':
74 // E.g.:
75 // -c big:0xf0,little:0x0f
76 for (;;) {
77 char * clusdata = strchr(optarg, ':');
78 if (clusdata == NULL) break;
79
80 *clusdata = 0;
81 uint64_t const cpumask = strtoul(clusdata+1, &clusdata, 0);
82 if (!cpu_cluster_add(optarg, cpumask)) break;
83 if (*clusdata != ',') break;
84 optarg = clusdata + 1;
85 }
86 break;
87 case 'D':
88 /* kprintf output channel */
89 debug_opt[0] = optarg;
90 break;
91 case 'f':
92 /* cpu frequency, clock cycles frequency, timer chip frequency, time needed to program hw timer */
93 get_freq(&cpu_freq);
94 get_freq(&cycles_freq);
95 get_freq(&timer_freq);
96 char *end;
97 timer_prog_time = strtoul(optarg, &end, 10);
98 if (end == optarg) {
99 timer_prog_time = 0;
100 }
101 break;
102 case 'K':
103 /* kdebug remote debug protocol channel */
104 debug_opt[1] = optarg;
105 break;
106 case 'N':
107 add_typed_string(_CS_HOSTNAME, optarg);
108 break;
109 case 'R':
110 next = ws_alloc(sizeof(struct reserved_ram));
111 if(next == NULL) {
112 crash("No memory for reserved_ram structure.\n");
113 }
114 next->next = reserved_ram_list;
115 reserved_ram_list = next;
116
117 reserved_ram_list->low_region = 0;
118 reserved_ram_list->align = 0;
119 reserved_ram_list->name = NULL;
120 reserved_ram_list->size = getsize(optarg, &cp);
121 if(*cp == ',') {
122 reserved_ram_list->align = getsize(cp + 1, &cp);
123 }
124 if(*cp == ',') {
125 reserved_ram_list->name = cp+1;
126 }
127 if(reserved_ram_list->name != NULL) {
128 cp = strchr(reserved_ram_list->name, ',');
129 if(cp != NULL) {
130 reserved_ram_list->low_region = (unsigned)strtoul(cp+1, NULL, 10);
131 *cp = '\0';
132 }
133 }
134 break;
135 case 'r':
136 // reserve user-specified memory from startup
137 resmem_addr = getsize(optarg, &cp);
138 if(*cp == ',') {
139 resmem_size = getsize(cp + 1, &cp);
140 avoid_ram(resmem_addr, resmem_size);
141 }
142 // else: ignore the option - a size is required
143 break;
144 case 'S':
145 //This gets handled later so that debug code isn't brought in all
146 //the time.
147 break;
148 case 'T':
149 misc_flags |= MISC_FLAG_SUPPRESS_BOOTTIME;
150 break;
151 case 'P':
152 num = strtoul(optarg, &optarg, 10);
153 if(num > 0) max_cpus = num;
154 break;
155 case 'v':
156 debug_flag++;
157 break;
158 case 'F':
159 case 'A':
160 case 'M':
161 case 'j':
162 case 'Z':
163 // All handled later, see init_system_private...
164 break;
165 case 'I':
166 // Fastboot/Restore IFS
167 rifs_flag |= RIFS_FLAG_ENABLE;
168
169 num = strtoul(optarg, &optarg, 10);
170 if(num) {
171 rifs_flag |= RIFS_FLAG_CKSUM;
172 }
173 break;
174 case 'i':
175 // Fastboot/Secondary IFS
176 rifs_flag |= RIFS_FLAG_IFS2;
177 ifs2_size = strtoul(optarg, &optarg, 0);
178
179 // Valid flags are: 'R', 'RK', or 'K' (equivalent to 'RK')
180 if((cp = strchr(optarg, ',')) != NULL) {
181 if(*(cp + 1) == 'R') {
182 rifs_flag |= RIFS_FLAG_IFS2_RESTORE;
183 cp++;
184 }
185 if(*(cp + 1) == 'K') {
186 rifs_flag |= RIFS_FLAG_IFS2_RESTORE;
187 rifs_flag |= RIFS_FLAG_IFS2_CKSUM;
188 cp++;
189 }
190 }
191 if((cp != NULL) && ((cp = strchr(cp, ',')) != NULL)) {
192 rifs_flag |= RIFS_FLAG_IFS2_SRC;
193 ifs2_paddr_src = strtoul(cp + 1, &cp, 0);
194 }
195 if((cp != NULL) && ((cp = strchr(cp, ',')) != NULL)) {
196 rifs_flag |= RIFS_FLAG_IFS2_DST;
197 // Get destination value and make sure it is at a 4K page boundary
198 ifs2_paddr_dst = 0xFFFFF000 & strtoul(cp + 1, &cp, 0);
199 }
200 break;
201 case 'o':
202 hundred_loop_time = (unsigned long) strtoul(optarg, &cp, 0);
203 if(*cp == ',') {
204 overhead = (unsigned long) strtoul(cp + 1, &cp, 0);
205 hwi_add_nanospin(hundred_loop_time, overhead);
206 }
207 break;
208 case 'H':
209 in_hvc = 1;
210 break;
211 }
212 }

215void
216set_debug(unsigned channel, const struct debug_device *dev, const char *options) {
217 unsigned base;
218 unsigned i;
219
220 dev->init(channel, options, dev->defaults[channel]);
221 if(channel == 0) set_print_char(dev->put);
222 base = offsetof(struct callout_entry, debug);
223 base += channel * sizeof(struct debug_callout);
224 for(i = 0; i < NUM_ELTS(dev->callouts); ++i) {
225 const struct callout_rtn *rtn = dev->callouts[i];
226
227 if(rtn != NULL) {
228 add_callout(base + i*sizeof(callout_fp_t), rtn);
229 }
230 }

233static void
234select_one(const struct debug_device *dev, unsigned size, unsigned channel) {
235 unsigned len;
236 const struct debug_device *selected;
237 const struct debug_device *end;
238 char *arg;
239
240 end = (void *)((uint8_t *) dev + size);
241 arg = debug_opt[channel];
242 if(arg == NULL) {
243 for( ;; ) {
244 if(dev >= end) return;
245 if(dev->defaults[channel] != NULL) break;
246 ++dev;
247 }
248 set_debug(channel, dev, "");
249 return;
250 }
251 selected = NULL;
252 for(;;) {
253 const char *defaults;
254
255 if(dev >= end) break;
256 defaults = dev->defaults[channel];
257 if(defaults != NULL) {
258 char ch;
259
260 len = strlen(dev->name);
261 ch = arg[len];
262 if((memcmp(arg, dev->name, len) == 0)
263 &&(ch==' ' || ch=='\t' || ch=='\0' || ch=='.')) {
264 selected = dev;
265 arg += len;
266 if(*arg == '.') ++arg;
267 break;
268 }
269 if((selected == NULL) && (defaults[0] != '\0')) {
270 //If we don't match any of the names, the first device
271 //in the list with actual options the default device.
272 selected = dev;
273 }
274 }
275 ++dev;
276 }
277 if(selected != NULL) set_debug(channel, selected, arg);
278}
