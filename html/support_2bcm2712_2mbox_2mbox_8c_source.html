90static int mbox_msg(const uint32_t tag, void *buf, const uint32_t bufsize) {
91 struct {
92 uint32_t hdrsize, hdrcode;
93#define MBOX_PROCESS_REQUEST 0
94 uint32_t tag, tagsize, tagcode;
95#define MBOX_TAG_REQUEST (0U << 31)
96#define MBOX_TAG_RESPONSE (1U << 31)
97#define MBOX_TAG_NULL 0
98 uint8_t buf[bufsize];
99 uint32_t tagend;
100 } *msg;
101 static void *out = NULL;
102
103 static uint32_t physaddr;
104
105 typedef struct {
106 uint32_t rdwr, _x04, _x08, _x0c;
107 uint32_t poll, send, status, config;
108#define MBOX_STATUS_FULL (0x80000000U)
109#define MBOX_STATUS_EMPTY (0x40000000U)
110#define MBOX_SEND_CHANNEL (8U)
111#define MBOX_SEND_CHANNEL_MASK (0xfU)
112 } mbox_t;
113 static volatile mbox_t *mbox = NULL; // [0]:fromvc [1]:tovc
114
115 if (out == NULL) {
116 int fd;
117 off_t offset;
118 fd = posix_typed_mem_open("/sysram&below1G", O_RDWR, POSIX_TYPED_MEM_ALLOCATE_CONTIG);
119 if (fd < 0) {
120 (void)printf("Failed to open typed memory(%s) \n", strerror( errno ));
121 exit(EXIT_FAILURE);
122 }
123 out = mmap(NULL, 0x1000, PROT_READ | PROT_WRITE | PROT_NOCACHE, MAP_SHARED, fd, 0);
124 if (out == MAP_FAILED) {
125 (void)printf("Failed to allocate buffer!\n");
126 (void)close(fd);
127 exit(EXIT_FAILURE);
128 }
129
130 if (mem_offset(out, NOFD, 1, &offset, NULL) != EOK) {
131 (void)munmap(out, 0x1000);
132 (void)close(fd);
133 (void)printf("Failed to obtain physical address\n");
134 exit(EXIT_FAILURE);
135 }
136
137 physaddr = (uint32_t)offset;
138 (void)close(fd);
139 }
140
141 msg = out;
142 msg->hdrsize = (uint32_t)sizeof(*msg);
143 msg->hdrcode = MBOX_PROCESS_REQUEST;
144 msg->tag = tag;
145 msg->tagsize = bufsize;
146 msg->tagcode = MBOX_TAG_REQUEST;
147 (void)memcpy((void*)msg->buf, buf, bufsize);
148 msg->tagend = MBOX_TAG_NULL;
149
150#define MBOX_REGS (0x107c013880UL)
151 int wait_cnt = 0;
152 if (mbox == NULL) {
153 mbox = mmap_device_memory(NULL, (sizeof(mbox_t)*2U), PROT_NOCACHE|PROT_READ|PROT_WRITE, 0, MBOX_REGS);
154 if (mbox == MAP_FAILED) {
155 (void)printf("Failed to map mbox registers\n");
156 exit (EXIT_FAILURE);
157 }
158 }
159 if (mbox != NULL) {
160 // check for space on tovc channel to send with timeout
161 for (;;) {
162 if ((mbox[1].status & MBOX_STATUS_FULL) == 0U) {
163 break;
164 }
165 (void)usleep(100);
166 if (++wait_cnt > 5000) {
167 (void)printf("Check for tovc channel full timeout\n");
168 return -1;
169 }
170 }
171
172 // send
173 mbox[1].rdwr = (physaddr & ~MBOX_SEND_CHANNEL_MASK) | MBOX_SEND_CHANNEL;
174
175 // wait for tx done
176 for (;;) {
177 if ((mbox[1].status & MBOX_STATUS_FULL) == 0U) {
178 break;
179 }
180 (void)usleep(100);
181 if (++wait_cnt > 5000) {
182 (void)printf("Wait for tx done timeout\n");
183 return -1;
184 }
185 }
186
187 for (;;) {
188 wait_cnt = 0;
189 // wait for response on fromvc channel to receive with timeout
190 for (;;) {
191 if ((mbox[0].status & MBOX_STATUS_EMPTY) == 0U) {
192 break;
193 }
194 (void)usleep(100);
195 if (++wait_cnt > 5000) {
196 (void)printf("Wait for response timeout\n");
197 return -2;
198 }
199 }
200
201 // is it the channel we're waiting for?
202 const uint32_t rd = mbox[0].rdwr;
203 if ((rd & MBOX_SEND_CHANNEL_MASK) == MBOX_SEND_CHANNEL) {
204 const uint32_t tagcode = msg->tagcode;
205 uint32_t len = tagcode & ~MBOX_TAG_RESPONSE;
206
207 if (tag != MBOX_TAG_GET_FIRMWAREHASH) {
208 if (len > 0U) {
209 if (len > bufsize) {
210 // passed in bufsize is not big enough to hold return response
211 len = bufsize;
212 }
213 (void)memcpy(buf, (void*)msg->buf, len);
214 }
215 return (int)len;
216 }
217 else {
218 /* firmwware hash should return 20 bytes but return code indicates 8 only */
219 (void)memcpy(buf, (void*)msg->buf, bufsize);
220 return bufsize;
221 }
222 }
223 }
224 /*NOTREACHED*/
225 }
226 return 0;
227}

247static void fmt_x(const void *const buf, const uint32_t len) {
248 /* mac address returns 6 byte, need to round up to get last two octets */
249 const uint32_t length = len + (sizeof(uint32_t) - 1U);
250 for (unsigned i = 0; i < (length / sizeof(uint32_t)); i++) {
251 (void)printf("%08x ", ((uint32_t *) buf)[i]);
252 }
253 (void)printf("\n");
254}
