128// The first byte of the CDB is the operation code. The top 3 bits
129// are the group code field and the bottom 5 bits are the command code
130// field.
131// Group 0 - six-byte commands
132// Group 1 - ten-byte commands
133// Group 2 - ten-byte commands
134// Group 3 - reserved
135// Group 4 - reserved
136// Group 5 - twelve-byte commands
137// Group 6 - vendor specific
138// Group 7 - vendor specific
139#define SC_GROUP( _cmd ) ( (_cmd) >> 5 )
