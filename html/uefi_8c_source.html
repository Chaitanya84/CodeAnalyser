40void *
41uefi_find_config_tbl(const EFI_GUID *const uid) {
42 EFI_CONFIGURATION_TABLE *tbl;
43 unsigned i;
44
45 tbl = efi_system_table->ConfigurationTable;
46 for(i = 0; i < efi_system_table->NumberOfTableEntries; ++i) {
47 if(memcmp(&tbl->VendorGuid, uid, sizeof(*uid)) == 0) {
48 return tbl->VendorTable;
49 }
50 ++tbl;
51 }
52 return NULL;
53}

58static void _uefi_exit_boot_services(void)
59{
60 ASSERT(uefi_init_cnt > 0);
61
62 if (--uefi_init_cnt == 0)
63 {
64 const EFI_BOOT_SERVICES *const bs = efi_system_table->BootServices;
65 _Uintn map_key;
66 _Uintn desc_size;
67 uint32_t desc_version;
68 _Uintn map_size = 0;
69 EFI_MEMORY_DESCRIPTOR *desc = NULL;
70 EFI_STATUS status = 0;
71 unsigned uefi_exit_retry_cnt = 0;
72
73 if (debug_flag > 0) kprintf("%s, exiting\n", __FUNCTION__);
74 uefi_io_flush();
75
76 do {
77 /* Get the UEFI memory map size */
78 status = bs->GetMemoryMap(&map_size, desc, &map_key, &desc_size, &desc_version);
79 if (status != EFI_BUFFER_TOO_SMALL) {
80 crash("Failed to query the UEFI memory map with error %l\n", status);
81 }
82
83 do {
84 /* Allocating the memory creates at least one new descriptor */
85 map_size += desc_size;
86 status = bs->AllocatePool(EfiLoaderData, map_size, (void **)&desc);
87 if ((desc == NULL) || (status != EFI_SUCCESS)) {
88 crash("Fatal error while allocating the UEFI memory map buffer\n");
89 }
90
91 /* Get the UEFI memory map */
92 status = bs->GetMemoryMap(&map_size, desc, &map_key, &desc_size, &desc_version);
93 if (status != EFI_SUCCESS) {
94 bs->FreePool(desc);
95 desc = NULL;
96 }
97 } while (status == EFI_BUFFER_TOO_SMALL);
98
99 if (status != EFI_SUCCESS) {
100 crash("Failed to get the UEFI memory map with error %l\n", status);
101 }
102
103 status = bs->ExitBootServices(efi_image_handle, map_key);
104
105 if ((status == EFI_INVALID_PARAMETER) && (uefi_exit_retry_cnt <= UEFI_EXIT_RETRY_MAX)) {
106 uefi_exit_retry_cnt++;
107 kprintf("retry uefi exit: %u\n", uefi_exit_retry_cnt);
108 bs->FreePool(desc);
109 desc = NULL;
110 map_size=0;
111 continue;
112 }
113 else if (status != EFI_SUCCESS) {
114 kprintf("exit status: %u\n", status);
115 crash("Could not exit boot services\n");
116 }
117 else { /* EFI_SUCCESS */
118 break;
119 }
120 } while (uefi_exit_retry_cnt <= UEFI_EXIT_RETRY_MAX);
121
122 efi_system_table->ConOut = NULL; /* no more output */
123 efi_system_table->ConIn = NULL; /* no more input */
124 efi_boot_services = NULL; /* no more EFI_BOOT_SERVICES */
125 uefi_exit_boot_services_f = NULL;
126 }
127 else if (debug_flag > 0) kprintf("%s, deferred, cnt = %u\n", __FUNCTION__, uefi_init_cnt);
128
129 /*
130 * QNX procnto expects QNX startup to disable the interrupts.
131 * Failing to disable the interrupts may result in a crash during
132 * the early initialization of the kernel.
133 */
134 InterruptDisable();
135}
