36static void fdt_get_mac_addr(uint8_t* const mac)
37{
38 static char *bootargs_str = NULL;
39
40 int const chosen = fdt_path_offset(fdt, "/chosen");
41 if (chosen <= 0) {
42 kprintf("Did not find chosen in FDT\n");
43 return;
44 }
45
46 int len;
47 const struct fdt_property *const prop = fdt_get_property(fdt, chosen, "bootargs", &len);
48 if (prop == NULL) {
49 kprintf("Did not find chosen bootargs in FDT\n");
50 return;
51 }
52 bootargs_str = ws_alloc(len);
53 if (bootargs_str == NULL) {
54 return;
55 }
56 memcpy(bootargs_str, prop->data, (size_t)len);
57 if (debug_flag > 4) {
58 kprintf("fdt: bootargs %s\n", bootargs_str);
59 }
60 for (;;) {
61 char * const arg = find_next_arg(bootargs_str, &bootargs_str, 0);
62 char *macstr;
63 if (*arg == '\0') return;
64 macstr=strstr(arg, "smsc95xx.macaddr=");
65 if (macstr != NULL) {
66 macstr=macstr+(sizeof("smsc95xx.macaddr=")-1);
67 int i;
68 for (i=0; i<6; i++) {
69 mac[i] = (uint8_t)strtoul(macstr, NULL, 16);
70 if (macstr[2] == ':') {
71 macstr += 3;
72 }
73 else break;
74 }
75 break;
76 }
77 }
78}

80static uint32_t fdt_get_board_revision(void)
81{
82 const int node_offset = fdt_path_offset(fdt, "/system");
83 int prop_len;
84 struct fdt_property const * const prop = fdt_get_property(fdt, node_offset, "linux,revision", &prop_len);
85 if (prop == NULL) return 0;
86
87 fdt32_t const * const data = (fdt32_t const *)prop->data;
88 return fdt32_to_cpu(data[0]);
89}

104uint32_t get_board_revision(void)
105{
106 static uint32_t revision = 0;
107
108 if (revision == 0) {
109 revision = mbox_get_board_revision();
110 if (revision == 0) {
111 revision = fdt_get_board_revision();
112 if (revision == 0) {
113 /* hard code to 00c04170 - 4GB target */
114 /* 0xd04170 - 8GB target */
115 revision = 0x00c04170;
116 kprintf("Use default board revision\n");
117 }
118 if (debug_flag > 1) {
119 kprintf("FDT: board revision 0x%x\n", revision);
120 }
121 }
122 else {
123 if (debug_flag > 1) {
124 kprintf("mbox: board revision 0x%x\n", revision);
125 }
126 }
127 }
128 return revision;
129}

131void init_hwinfo(void)
132{
133 const unsigned hwi_bus_internal = 0;
134 /* Add ENET */
135 {
136 unsigned i, hwi_off;
137 uint8_t mac[6] = { 0 };
138 hwiattr_enet_t attr = HWIATTR_ENET_T_INITIALIZER;
139 hwiattr_common_t common_attr = HWIATTR_COMMON_INITIALIZER;
140 HWIATTR_ENET_SET_NUM_IRQ(&attr, 1);
141
142 uint32_t irqs[] = {
143 BCM2712_RP1_ENET_IRQ
144 };
145
146 HWIATTR_SET_NUM_IRQ(&common_attr, NUM_ELTS(irqs));
147
148 /* Create DWC0 */
149 HWIATTR_ENET_SET_LOCATION(&attr, BCM2712_RP1_ENET_BASE, BCM2712_RP1_ENET_SIZE, 0, hwi_find_as(BCM2712_RP1_ENET_BASE, 1));
150 hwi_off = hwidev_add_enet(BCM2712_HWI_CGEM, &attr, hwi_bus_internal);
151 hwitag_add_common(hwi_off, &attr);
152 ASSERT(hwi_find_unit(hwi_off) == 0);
153
154 /* Add IRQ number */
155 for(i = 0; i < NUM_ELTS(irqs); i++) {
156 hwitag_set_ivec(hwi_off, i, irqs[i]);
157 }
158
159 get_enet_mac_address(mac);
160 hwitag_add_nicaddr(hwi_off, mac, sizeof(mac));
161 }
162
163 /* add the WATCHDOG device */
164 {
165 unsigned hwi_off;
166 hwiattr_timer_t attr = HWIATTR_TIMER_T_INITIALIZER;
167 const struct hwi_inputclk clksrc_kick = {.clk = 500, .div = 1};
168 HWIATTR_TIMER_SET_NUM_CLK(&attr, 1);
169 HWIATTR_TIMER_SET_LOCATION(&attr, BCM2712_PM_BASE, BCM2712_PM_SIZE, 0, hwi_find_as(BCM2712_PM_BASE, 1));
170 hwi_off = hwidev_add_timer("wdog", &attr, HWI_NULL_OFF);
171 ASSERT(hwi_off != HWI_NULL_OFF);
172 hwitag_set_inputclk(hwi_off, 0, (struct hwi_inputclk *)&clksrc_kick);
173
174 hwi_off = hwidev_add("wdt,options", 0, HWI_NULL_OFF);
175 hwitag_add_regname(hwi_off, "write_width", 32);
176 hwitag_add_regname(hwi_off, "enable_width", 32);
177 }
178}

196{
197 const uintptr_t base = (uintptr_t) PCIE_EXT_MIP_INTC_ADDR;
198
199 /* set all MSI-Xs edge-triggered, masked in for the host, masked out for the VPU */
200 out32(base + MIP_INTC_MASKL_HOST, 0);

202 out32(base + MIP_INTC_MASKL_VPU, 0xffffffff);
203 out32(base + MIP_INTC_MASKH_VPU, 0xffffffff);
204 out32(base + MIP_INTC_CFGL_HOST, 0xffffffff);
205 out32(base + MIP_INTC_CFGH_HOST, 0xffffffff);
