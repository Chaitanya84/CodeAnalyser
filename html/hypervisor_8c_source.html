32hyp_flags_t
33arch_hypervisor_validate_flags(hyp_flags_t const option_flags) {
34 if (option_flags == HYP_FLAG_INVALID) {
35 crash("Unrecognized hypervisor option flag\n");
36 }
37
38 int const current_el = (aa64_sr_rd64(CurrentEL) >> 2);
39 if (current_el == 1) {
40 // There is nothing we can do unless we are already at EL2
41 if (option_flags & HYP_FLAG_ENABLED) {
42 crash("Hypervisor support requested but CPU has no EL2 support\n");
43 }
44 return HYP_FLAG_DISABLED;
45 }
46
47 if (option_flags & HYP_FLAG_ENABLED) {
48 int const el2_host_supported = (aa64_sr_rd64(id_aa64mmfr1_el1) & 0x0f00);
49 int const el2_host_requested = (option_flags & HYP_FLAG_EL2_HOST);
50 int const el1_host_requested = (option_flags & HYP_FLAG_EL1_HOST);
51
52 if (el2_host_requested && !el2_host_supported) {
53 crash("Hypervisor EL2 Host requested but CPU does not support this\n");
54 }
55 if (el1_host_requested || !el2_host_supported) {
56 return (HYP_FLAG_ENABLED | HYP_FLAG_EL1_HOST);
57 }
58 return (HYP_FLAG_ENABLED | HYP_FLAG_EL2_HOST);
59 }
60
61 return HYP_FLAG_DISABLED;
62}

75void
76arch_hypervisor_init(unsigned const cpunum, hyp_flags_t const hyp_flags) {
77 int const current_el = (aa64_sr_rd64(CurrentEL) >> 2);
78
79 switch(hyp_flags) {
80 case HYP_FLAG_DISABLED:
81 if (debug_flag && (cpunum == 0)) {
82 kprintf("Hypervisor support disabled\n");
83 }
84 if (current_el != 1) {
85 drop_to_el1();
86 }
87 break;
88 case (HYP_FLAG_ENABLED | HYP_FLAG_EL2_HOST):
89 if (debug_flag && (cpunum == 0)) {
90 kprintf("Enabling EL2 host hypervisor support (VHE)\n");
91 }
92 hyp_enable_el2_host();
93 break;
94 case (HYP_FLAG_ENABLED | HYP_FLAG_EL1_HOST):
95 if (cpunum == 0) {
96 if (debug_flag) {
97 kprintf("Enabling EL1 host hypervisor support\n");
98 }
99
100 // allocate memory for an EL2 exception vector table
101 transfer_vector_table = alloc_ram(NULL_PADDR, 0x800, 0x800);
102 if (transfer_vector_table == NULL_PADDR) {
103 crash("No memory for EL2 exception table.\n");
104 }
105 as_add(transfer_vector_table, transfer_vector_table+0x7ff, AS_ATTR_RAM,
106 "hypervisor_vector", as_find(AS_NULL_OFF, "memory", NULL));
107 }
108 hyp_enable_el1_host(transfer_vector_table);
109 drop_to_el1();
110 break;
111 default:
112 crash("Unknown required hypervisor mode\n");
113 break;
114 }
115}
