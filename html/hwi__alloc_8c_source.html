56void *
57hwi_alloc_tag(const char *name, unsigned size, unsigned align) {
58 hwi_tag *tag;
59 unsigned name_off;
60
61 align_tag(align);
62 name_off = add_string(name);
63 unsigned need = size + sizeof(struct hwi_prefix);
64 unsigned spare = lsp.hwinfo.size - offset;
65 if(need > spare) {
66 // grow by chunk of 128 bytes.
67 grow_syspage_section(&lsp.hwinfo, ROUND(need - spare, 128));
68 }
69 uint8_t *p = (uint8_t *)lsp.hwinfo.p->data;
70 tag = (hwi_tag *)(p + last_item);
71 tag->item.itemsize += size / 4;
72 tag = (hwi_tag *)(p + offset);
73 tag->prefix.size = size / 4;
74 tag->prefix.name = name_off;
75 offset += size;
76 return(tag);
77}

79void *
80hwi_alloc_item(const char *name, unsigned size, unsigned align,
81 const char *itemname, unsigned owner) {
82 hwi_tag *tag;
83 unsigned saved_in;
84
85 //Do add_string() up front so that the tag pointer doesn't move due
86 //to shifting section sizes.
87 saved_in = add_string(itemname);
88 align_tag(align);
89 last_item = offset;
90 tag = hwi_alloc_tag(name, size, align);
91 tag->item.itemname = saved_in;
92 tag->item.owner = owner;
93 if(owner != HWI_NULL_OFF) {
94 hwi_tag *owner_tag;
95
96 owner_tag = hwi_off2tag(owner);
97 owner_tag->item.kids++;
98 }
99 return(tag);
100}
