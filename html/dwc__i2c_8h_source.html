140/*
141 * SOFT RESET (offset 0x204)
142 *
143 * Bits |Access |Default |Description
144 * |Type | |
145 * -------+-------+--------+---------------------------------------------------------------------------------------------------------------
146 * 31:3 |na |30'h0 |Reserved
147 * -------+-------+--------+---------------------------------------------------------------------------------------------------------------
148 * 2 |RW |1'h0 |iDMA Software Reset Control
149 * | | | 0 – IP is in reset (Reset Asserted)
150 * | | | 1 – IP is NOT at reset (Reset Released)
151 * -------+-------+--------+---------------------------------------------------------------------------------------------------------------
152 * 1:0 |RW |2'h0 |reset_i2c (reset_i2c)
153 * | | |I2C Host Controller reset.
154 * | | |Used to reset the I2C Host Controller by SW control. All I2C Configuration State and Operational State will be
155 * | | |forced to the Default state.
156 * | | |There is no timing requirement (SW can assert and de-assert in back to back transactions)
157 * | | |This reset does NOT impact the LPSS cluster level settings by BIOS, the PCI configuration header information,
158 * | | | DMA channel configuration and interrupt assignment/mapping/etc.
159 * | | |Driver should re-initialize registers related to Driver context following an I2C host controller reset.
160 * | | | 00 = I2C Host Controller is in reset (Reset Asserted)
161 * | | | 01 = Reserved
162 * | | | 10 = Reserved
163 * | | | 11 = I2C Host Controller is NOT at reset (Reset Released)
164 */
165#define DW_IC_SOFT_RESET_MASK 0x03U
166#define DW_IC_SOFT_RESET_ASSERTED 0x00U
167#define DW_IC_SOFT_RESET_NOT_ASSERTED 0x03U
