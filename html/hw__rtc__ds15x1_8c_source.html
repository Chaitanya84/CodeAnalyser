28/*
29 * Clock setup for DS1501/1511 part.
30 * addresses that we use (all fields are BCD)
31 *
32 * 0 seconds
33 * 1 minutes
34 * 2 hours
35 * 3 day of week (we set this up but do not use it)
36 * 4 day
37 * 5 month
38 * 6 year (0-99)
39 * 7 century
40 * f control b register
41 */
42
43#define TRANSFER_ENABLE (1 << 7)
44
45unsigned long
46rtc_time_ds15x1(paddr_t base, unsigned reg_shift, int mmap, int cent_reg) {
47 struct tm tm;
48 int centry;
49
50 /*
51 * Tell Neutrino what kind of chip for 'rtc' utility
52 */
53 hwi_add_rtc("ds15x1", base, reg_shift, 16, mmap, 0x07);
54
55 chip_access(base, reg_shift, mmap, 16);
56
57 /*
58 * Set TE bit
59 */
60 chip_write8(0xf, chip_read8(0xf) | TRANSFER_ENABLE);
61
62 do {
63 tm.tm_sec = chip_read8(0);
64 tm.tm_min = chip_read8(1);
65 tm.tm_hour = chip_read8(2);
66 tm.tm_mday = chip_read8(4);
67 tm.tm_mon = chip_read8(5);
68 tm.tm_year = chip_read8(6);
69 centry = chip_read8(7);
70
71 /*
72 * Loop while time inconsistent
73 */
74 } while(tm.tm_sec != chip_read8(0));
75
76 chip_done();
77
78 tm.tm_mon &= 0x1F;
79
80 tm.tm_sec = bcd2bin(tm.tm_sec);
81 tm.tm_min = bcd2bin(tm.tm_min);
82 tm.tm_hour = bcd2bin(tm.tm_hour);
83 tm.tm_mday = bcd2bin(tm.tm_mday);
84 tm.tm_mon = bcd2bin(tm.tm_mon);
85 tm.tm_year = bcd2bin(tm.tm_year);
86
87 if (centry == 0x20)
88 tm.tm_year += 100;
89
90 tm.tm_mon -= 1;
91
92 return(calc_time_t(&tm));
93}
