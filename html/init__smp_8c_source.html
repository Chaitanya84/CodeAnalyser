39static void
40transfer_aps(void) {
41 unsigned i;
42 volatile uint32_t *pending;
43
44 pending = &SYSPAGE_ENTRY(smp)->pending;
45
46 for(i = 1; i < lsp.syspage.p->num_cpu; ++i) {
47 //Get one AP into the syspage spin callout
48 *pending = 1;
49 mem_barrier();
50#ifdef __MIPS__
51 setcp0_count(0);
52#endif
53 syspage_available = i;
54 do {
55 mem_barrier(); // give the bus a break.
56 } while(*pending != 0);
57 }
58}

61static void
62start_aps(void) {
63 int i;
64 unsigned count;
65
66 /* the AP's cannot seem to use the boot services so suspend UEFI based output */
67 uefi_io_suspend();
68
69 for(i = 1; i < lsp.syspage.p->num_cpu; ++i) {
70 cpu_starting = i + 1;
71
72 if(board_smp_start(i, smp_start)) {
73 count = 0;
74 do {
75 if(++count == 0) ap_fail(i);
76 } while(cpu_starting != 0);
77 } else {
78 ap_fail(i);
79 }
80 }
81 uefi_io_resume();
82
83 smp_hook_rtn = transfer_aps;
84}

88callout_output_smp(int sizing) {

101void
102init_smp() {
103 struct smp_entry *smp;
104 unsigned num_cpus;
105
106 if(max_cpus > 1) {
107 num_cpus = board_smp_num_cpu();
108 if(num_cpus > max_cpus) num_cpus = max_cpus;
109 lsp.syspage.p->num_cpu = num_cpus;
110
111 if(num_cpus > 1) {
112 smp = set_syspage_section(&lsp.smp, sizeof(*lsp.smp.p));
113
114 smp_spin_vaddr = (void (*)(void))&smp_spin;
115
116 board_smp_init(smp, num_cpus);
117
118 smp_hook_rtn = start_aps;
119
120 //Hook into the sizing/writing list
121 smp_prev_output_rtn = callout_output_rtn;
122 callout_output_rtn = (output_callout_t *)callout_output_smp;
123 }
124 }
