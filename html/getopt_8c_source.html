27int getopt (int argc, char * const argv[], char const * opts) {
28 static char *cp;
29 static int last_optind;
30 int opt;
31 static char const * opts_str;
32
33 if (opts) {
34 opts_str = opts; // save the option list
35 } else {
36 opts = opts_str; // use the option list that saved in the previous call
37 }
38
39 if (optind == 0) {
40 optind = 1;
41 last_optind = 0;
42 }
43 if (optind >= argc) {
44 return -1;
45 }
46 if (last_optind != optind) {
47 cp = argv [optind];
48 last_optind = optind;
49 if (*cp++ != '-' || *cp == 0) {
50 return -1;
51 }
52 }
53 if (*cp == '-') {
54 ++optind;
55 return -1;
56 }
57 for (; *opts; ++opts) {
58 if (*opts == ':') continue;
59 if (*cp == *opts) break;
60 }
61 optarg = cp;
62 ++optind;
63 opt = *opts++;
64 if(opt) {
65 if (*++cp) {
66 if (*opts == ':') {
67 optarg = cp;
68 } else {
69 --optind;
70 }
71 } else {
72 if (*opts == ':') {
73 if (optind < argc) {
74 optarg = argv [optind];
75 optind++;
76 } else {
77 opt = '?';
78 }
79 }
80 }
81 } else {
82 opt = '?';
83 }
84 return opt;
85}
