23// NOTE: This file is shared between pidin and the startup library.
24// Ordinarily, we'd put it in a library that both code references, but
25// that would complicate building startups in a manner that we don't
26// want. Instead, a copy of this file is placed in both locations
27// and a distinct print_sysp.h file in both directories provide the
28// 'glue' to hook up this code with the particularities of the environment.
29// If you make any changes to this file, you should also copy the new
30// version to the other location (after first checking that no-one's
31// modified the other copy and forgotten to update this location).
32// Normally this won't be a problem since the file will only change in
33// response to system page layout changes and the copy in startup will
34// be modified first and then copied to the pidin tree.

36// This file is in:

47static void
48print_typed_strings(void) {
49 struct typed_strings_entry *string = _SYSPAGE_ENTRY(PSP_SYSPAGE, typed_strings);
50 void *type_ptr;
51 unsigned type;
52 unsigned i;
53
54 i = 0;
55 for( ;; ) {
56 type_ptr = &string->data[i];
57 type = (*(uint32_t *)type_ptr);
58 if(type == _CS_NONE) break;
59 i += sizeof(uint32_t);
60 kprintf(" off:%d type:%d string:'%s'\n", i-sizeof(uint32_t), type, &string->data[i]);
61 i += strlen(&string->data[i]) + 1;
62 i = (i + (sizeof(uint32_t)-1)) & ~(sizeof(uint32_t)-1);
63 }
64}

67static void
68print_strings(void) {
69 char *p = _SYSPAGE_ENTRY(PSP_SYSPAGE,strings)->data;
70 char *start = p;
71 unsigned off;
72 unsigned len;
73 char buff[MAX_BUFF_SIZE];
74
75 kprintf(" ");
76 off = 1;
77 while(*p != '\0') {
78 PSP_SNPRINTF(buff, MAX_BUFF_SIZE, " [%d]'%s'", (unsigned)(p - start), p);
79 buff[MAX_BUFF_SIZE - 1] = '\0';
80 len = (unsigned)strlen(buff);
81 if((off + len) >= MAX_BUFF_SIZE) {
82 kprintf("\n ");
83 off = 1;
84 }
85 kprintf("%s", buff);
86 off += len;
87 p += strlen(p) + 1;
88 }
89 kprintf("\n");
90}

168static void
169print_hwinfo(void) {
170 hwi_tag *tag = (hwi_tag *)_SYSPAGE_ENTRY(PSP_SYSPAGE,hwinfo);
171 void *base;
172 void *next;
173 char *name;
174
175 while(tag->prefix.size != 0) {
176 next = (hwi_tag *)((uint32_t *)tag +
177 tag->prefix.size);
178 base = (void *)(&tag->prefix + 1);
179 name = get_string(tag->prefix.name);
180 kprintf(" %d) size:%d tag:%d(%s)",
181 hwi_tag2off(tag),
182 tag->prefix.size,
183 tag->prefix.name, name);
184 if(*name >= 'A' && *name <= 'Z') {
185 base = (void *) (&tag->item + 1);
186 kprintf(" isize:%d, iname:%d(%s), owner:%d, kids:%d",
187 tag->item.itemsize,
188 tag->item.itemname,
189 get_string(tag->item.itemname),
190 tag->item.owner, tag->item.kids);
191 }
192 if(base != next) {
193 kprintf("\n ");
194 while(base < next) {
195 uint8_t *p = base;
196
197 kprintf(" %b", *p);
198 base = p + 1;
199 }
200 }
201 kprintf("\n");
202 tag = next;
203 }
204}

427void
428print_syspage_enable(const char *name) {
429 unsigned i;
430 unsigned on_bit;
431 unsigned off_mask;
432
433 if(*name == '~') {
434 ++name;
435 on_bit = EXPLICIT_DISABLE;
436 off_mask = ~EXPLICIT_ENABLE;
437 } else {
438 on_bit = EXPLICIT_ENABLE;
439 off_mask = ~EXPLICIT_DISABLE;
440 }
441 for(i = 0; i < NUM_ELTS(sp_section); ++i) {
442 if(strcmp(sp_section[i].name, name) == 0) {
443 sp_section[i].flags &= off_mask;
444 sp_section[i].flags |= on_bit;
445 }
446 if(on_bit & EXPLICIT_ENABLE) {
447 // If we have an explict enable, we mark all the sections
448 // as implicitly disabled so that we don't print them out
449 // unless we end up with an explict enablement of the entry
450 sp_section[i].flags |= IMPLICIT_DISABLE;
451 }
452 }

455void
456print_syspage_sections(void) {
457 unsigned i;
458 unsigned flags;
459 unsigned type;
460
461 kprintf("Header size=0x%x, Total Size=0x%x, Version=%d.%d, #Cpu=%d, Type=%d\n",
462 PSP_SYSPAGE->size,
463 PSP_SYSPAGE->total_size,
464 PSP_SYSPAGE->version.major,
465 PSP_SYSPAGE->version.minor,
466 PSP_SYSPAGE->num_cpu,
467 PSP_SYSPAGE->type);
468
469 for(i = 0; i < NUM_ELTS(sp_section); ++i) {
470 flags = sp_section[i].flags;
471 if(flags & EXPLICIT_ENABLE) {
472 flags &= ~IMPLICIT_DISABLE;
473 }
474 type = flags & SYSPAGE_TYPE_MASK;
475 if((type != 0) && (type != (PSP_SYSPAGE->type + 1))) {
476 // Not a section on this system page
477 flags |= EXPLICIT_DISABLE;
478 }
479 if(sp_section[i].loc > PSP_SYSPAGE->size) {
480 // Section isn't present on this system page
481 flags |= EXPLICIT_DISABLE;
482 }
483 if(!(flags & (EXPLICIT_DISABLE|IMPLICIT_DISABLE))) {
484 kprintf("Section:%s ", sp_section[i].name);
485 if(sp_section[i].flags & INFO_SECTION) {
486 syspage_array_info *info;
487
488 info = (void *)((uint8_t *)PSP_SYSPAGE + sp_section[i].loc);
489 kprintf("offset:0x%x size:0x%x", info->entry_off, info->entry_size);
490 if(flags & ARRAY_SECTION) {
491 kprintf(" elsize:0x%x", info->element_size);
492 }
493 kprintf("\n");
494 if(info->entry_size > 0 && PSP_VERBOSE(2)) {
495 sp_section[i].print();
496 }
497 } else {
498 kprintf("offset:0x%x\n", sp_section[i].loc);
499 if(PSP_VERBOSE(2)) {
500 sp_section[i].print();
501 }
502 }
503 }
504 }
