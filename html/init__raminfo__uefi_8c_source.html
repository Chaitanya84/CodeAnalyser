25static _Uintn
26do_raminfo(void) {
27 const EFI_BOOT_SERVICES *const bs = efi_system_table->BootServices;
28 _Uintn map_size = 0;
29 EFI_MEMORY_DESCRIPTOR *desc = NULL;
30 EFI_STATUS status = 0;
31 _Uintn map_key;
32 _Uintn desc_size;
33 uint32_t desc_version;
34
35 // Get the UEFI memory map size
36 status = bs->GetMemoryMap(&map_size, desc, &map_key, &desc_size, &desc_version);
37 if (status != EFI_BUFFER_TOO_SMALL) {
38 crash("Failed to query the UEFI memory map with error %l\n", status);
39 }
40
41 do {
42 // Allocating the memory creates at least one new descriptor
43 map_size += desc_size;
44 status = bs->AllocatePool(EfiLoaderData, map_size, (void **)&desc);
45 if ((desc == NULL) || (status != EFI_SUCCESS)) {
46 crash("Fatal error while allocating the UEFI memory map buffer\n");
47 }
48
49 // Get the UEFI memory map
50 status = bs->GetMemoryMap(&map_size, desc, &map_key, &desc_size, &desc_version);
51 if (status != EFI_SUCCESS) {
52 bs->FreePool(desc);
53 desc = NULL;
54 }
55 } while (status == EFI_BUFFER_TOO_SMALL);
56 if (status != EFI_SUCCESS) {
57 crash("Failed to get the UEFI memory map with error %l\n", status);
58 }
59
60 paddr_t starting_paddr = NULL_PADDR;
61 paddr_t ending_paddr = NULL_PADDR;
62 EFI_MEMORY_DESCRIPTOR *avoid_list = NULL;
63 EFI_MEMORY_DESCRIPTOR **owner = &avoid_list;
64
65 // Tell the startup library about all the ram
66 while(map_size != 0) {
67 // Make sure the start address fits in a paddr_t (think 32bit systems)
68 if((paddr_t)desc->PhysicalStart == desc->PhysicalStart) {
69 uint64_t nbytes = desc->NumberOfPages * 0x1000u;
70 paddr64_t const end_paddr = desc->PhysicalStart + nbytes;
71 if((paddr_t)end_paddr != end_paddr) {
72 // have to shrink size
73 nbytes = (~(paddr_t)0 + 1) - desc->PhysicalStart;
74 desc->NumberOfPages = nbytes / 0x1000u;
75 }
76 switch(desc->Type) {
77 case EfiReservedMemoryType:
78 case EfiUnusableMemory:
79 case EfiMemoryMappedIO:
80 case EfiMemoryMappedIOPortSpace:
81 if (debug_flag > 3) {
82 kprintf("memory type: %d PhysicalStart: %L size: %L (%L)\n",
83 desc->Type, desc->PhysicalStart, nbytes, desc->NumberOfPages);
84 }
85 break;
86 case EfiACPIReclaimMemory:
87 case EfiACPIMemoryNVS:
88 case EfiPalCode:
89 case EfiLoaderCode:
90 case EfiLoaderData:
91 case EfiBootServicesCode:
92 case EfiBootServicesData:
93 case EfiRuntimeServicesCode:
94 case EfiRuntimeServicesData:
95 if((PADDR_T)desc->PhysicalStart == desc->PhysicalStart) {
96 // Maintain the list in a low to high paddr order
97 // so that we're more likely to mark the unusable
98 // regions as such before alloc_ram() hands them
99 // out to somebody else.
100 *owner = desc;
101 owner = (void *)&desc->VirtualStart;
102 *owner = NULL;
103 }
104 /* FALLTHROUGH */
105 case EfiConventionalMemory:
106 if(desc->PhysicalStart != ending_paddr) {
107 if(starting_paddr != NULL_PADDR) {
108 add_ram(starting_paddr, ending_paddr - starting_paddr);
109 if (debug_flag > 2) {
110 kprintf("add_ram: starting_paddr: %L size: %L (%L)\n",
111 starting_paddr, (ending_paddr - starting_paddr), (ending_paddr - starting_paddr)/0x1000);
112 }
113 }
114 starting_paddr = desc->PhysicalStart;
115 ending_paddr = starting_paddr;
116 }
117 ending_paddr += nbytes;
118 if (debug_flag > 2) {
119 kprintf("memory type: %d PhysicalStart: %L size:%L (%L)\n",
120 desc->Type, desc->PhysicalStart, nbytes, desc->NumberOfPages);
121 }
122 break;
123 default:
124 crash("Unexpected memory type: %d\n", desc->Type);
125 }
126 }
127 desc = (void *)((uintptr_t)desc + desc_size);
128 map_size -= desc_size;
129 }
130 if(starting_paddr != ending_paddr) {
131 add_ram(starting_paddr, ending_paddr - starting_paddr);
132 if (debug_flag > 1) {
133 kprintf("add_ram: start: %L size: %L\n", starting_paddr, ending_paddr - starting_paddr);
134 }
135 }
136 //Tell the library about all the areas we need to avoid
137 while(avoid_list != NULL) {
138 size_t const nbytes = avoid_list->NumberOfPages*0x1000u;
139 switch(avoid_list->Type) {
140 case EfiACPIReclaimMemory:
141 case EfiACPIMemoryNVS:
142 case EfiPalCode:
143 case EfiRuntimeServicesCode:
144 case EfiRuntimeServicesData:
145 // This memory we can't use at all
146 alloc_ram(avoid_list->PhysicalStart, nbytes, 0);
147 if (debug_flag > 2) {
148 kprintf("alloc_ram: Type: %d PhysicalStart: %L size: %L (%L)\n",
149 avoid_list->Type, avoid_list->PhysicalStart, nbytes, nbytes/0x1000);
150 }
151 break;
152 case EfiLoaderCode:
153 case EfiLoaderData:
154 case EfiBootServicesCode:
155 case EfiBootServicesData:
156 // This memory, we can use after startup is done
157 avoid_ram(avoid_list->PhysicalStart, nbytes);
158 if (debug_flag > 2) {
159 kprintf("avoid_ram: Type: %d PhysicalStart: %L size: %L (%L)\n",
160 avoid_list->Type, avoid_list->PhysicalStart, nbytes, nbytes/0x1000);
161 }
162 break;
163 }
164 avoid_list = (void *)(uintptr_t)avoid_list->VirtualStart;
165 }
166 return map_key;
167}

169int
170init_raminfo_uefi(void) {
171
172 if(efi_system_table == NULL) {
173 return 0;
174 }
175
176 do_raminfo();
177
178 uefi_exit_boot_services();
179
180 return 1;
181}
