42/* Enable watchdog timer */
43void bcm2712_wdt_enable(const uint32_t timeout_value)
44{
45 const uintptr_t base = (uintptr_t) BCM2712_PM_BASE;
46 uint32_t val;
47 uint64_t ticks;
48
49 if (timeout_value <= 1000U) {
50 val = WDT_DEFAULT_TICKS;
51 } else {
52 ticks = timeout_value * PM_WDOG_MS_TO_TICKS;
53 if (ticks > WDT_MAX_TICKS) {
54 ticks = WDT_MAX_TICKS;
55 }
56 val = (uint32_t) ticks;
57 }
58
59 /* Disable watch dog */
60 out32(base + BCM2712_PM_RSTC, PM_PASSWORD | PM_RSTC_STOP);
61
62 /* Reload the timer value */
63 out32(base + BCM2712_PM_WDOG, PM_PASSWORD | val);
64
65 if (debug_flag > 3) {
66 kprintf("wdt: Reload the timer value %x = %x\n", base + BCM2712_PM_WDOG, PM_PASSWORD | val);
67 kprintf("wdt: read %x = %x\n", base + BCM2712_PM_RSTC, in32(base + BCM2712_PM_RSTC));
68 }
69 /* Enable the wdog timer */
70 val = in32(base + BCM2712_PM_RSTC) & PM_RSTC_WRCFG_CLR;
71 val |= (PM_PASSWORD | PM_RSTC_WRCFG_FULL_RESET);
72
73 if (debug_flag > 3) {
74 kprintf("wdt: Enable the wdog timer %x = %x\n", base + BCM2712_PM_RSTC, val);
75 }
76
77 out32(base + BCM2712_PM_RSTC, val);
78}
