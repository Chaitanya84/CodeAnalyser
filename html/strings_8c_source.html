69add_typed_string(int type_index, const char *name) {
70 char *p = lsp.typed_strings.p->data;
71 unsigned i;
72 unsigned len;
73 unsigned spare;
74 unsigned need;
75
76 del_typed_string(type_index);
77 i = find_typed_string(_CS_NONE);
78 len = strlen(name) + 1;
79 len = ROUND(len, sizeof(uint32_t));
80 need = len + 2*sizeof(uint32_t);
81 spare = lsp.typed_strings.size - i;
82 if(spare < need) {
83 grow_syspage_section(&lsp.typed_strings, need - spare);
84 }
85 *(uint32_t *)&p[i] = type_index;
86 strlcpy(&p[i+sizeof(uint32_t)], name, lsp.typed_strings.size - i);
87 return(i);
88}

90unsigned
91add_string(const char *name) {
92 char *p = lsp.strings.p->data;
93 char *str;
94 unsigned i;
95 unsigned len;
96 unsigned spare;
97
98 i = 0;
99 for( ;; ) {
100 str = &p[i];
101 if(*str == '\0') break;
102 if(strcmp(name, str) == 0) return(i);
103 i += strlen(str)+1;
104 }
105 len = strlen(name) + 2;
106 spare = lsp.strings.size - i;
107 if(len > spare) {
108 // grow by chunk of 128 bytes.
109 grow_syspage_section(&lsp.strings, ROUND(len - spare, 128));
110 }
111 strlcpy(str, name, lsp.strings.size - i);
112 return(i);
113}
