50void
51hypervisor_init(unsigned const cpunum) {
52 hyp_flags_t selected_flags;
53
54 selected_flags = arch_hypervisor_validate_flags(option_flags);
55
56 // Cpu 0 determines the required flags all others must support
57 if (cpunum == 0) {
58 required_flags = selected_flags;
59 } else {
60 if (selected_flags != required_flags) {
61 crash("Hypervisor setup: cpu %d, does not match cpu 0 capability");
62 }
63 }
64
65 arch_hypervisor_init(cpunum, required_flags);
66}
