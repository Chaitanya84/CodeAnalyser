42// We keep the maximum size we're allowed to grow things in
43// the lsp.syspage->p.size field - we'll set it to it's proper value
44// once we're done, that'll also disallow any further growing since
45// it's going to be less than the total_size.
46
47void *
48grow_syspage_section(void *p, unsigned add) {
49 local_section *sect = p;
50 local_section *s2;
51 uint8_t *bottom;
52 unsigned new_size;
53 unsigned max_size;
54 unsigned len;
55 uint8_t *new;
56
57 if(add > 0) {
58 add = ROUND(add, sizeof(uint64_t));
59 new_size = lsp.syspage.p->total_size + add;
60 max_size = lsp.syspage.p->size;
61 if(new_size > max_size) {
62 crash("Syspage memory request of %d exceeds maximum of %d bytes.\n",
63 new_size, max_size);
64 }
65 bottom = (uint8_t *)sect->p + sect->size;
66 for(s2 = SECTION_START; s2 < SECTION_END; ++s2) {
67 if((s2 != sect) && ((void *)s2->p >= (void *)bottom)) {
68 s2->p = (void *)((uint8_t *)s2->p + add);
69 }
70 }
71 new = bottom + add;
72 len = lsp.syspage.p->total_size - PTR_DIFF(bottom, lsp.syspage.p);
73 memmove(new, bottom, len);
74 callout_reloc_data(bottom, len, (ptrdiff_t)add);
75 memset(bottom, 0, add);
76 sect->size += add;
77 lsp.syspage.p->total_size = new_size;
78 }
79 return(sect->p);
80}

92void
93init_syspage_memory(void *base, unsigned max_size) {
94 local_section *sect;
95 unsigned spsize = ROUND(sizeof(*lsp.syspage.p),sizeof(uint64_t));
96
97 for(sect = SECTION_START; sect < SECTION_END; ++sect) {
98 sect->p = (void *)((uint8_t *)base + spsize);
99 sect->size = 0;
100 }
101 lsp.syspage.p = base;
102 lsp.syspage.size = spsize;
103 memset(base, 0, spsize);
104 fixed_size = spsize;
105 lsp.syspage.p->version.major = SYSPAGE_VERSION_MAJOR;
106 lsp.syspage.p->version.minor = SYSPAGE_VERSION_MINOR;
107 lsp.syspage.p->total_size = spsize;
108 lsp.syspage.p->size = max_size;
109 lsp.syspage.p->asinfo.element_size = sizeof(struct asinfo_entry);
110 lsp.syspage.p->cpuinfo.element_size = sizeof(struct cpuinfo_entry);
111 lsp.syspage.p->cacheattr.element_size = sizeof(struct cacheattr_entry);
112 lsp.syspage.p->intrinfo.element_size = sizeof(struct intrinfo_entry);
113 set_syspage_section(&lsp.callout, sizeof(*lsp.callout.p));
114 grow_syspage_section(&lsp.typed_strings, sizeof(uint32_t));
115 grow_syspage_section(&lsp.strings, sizeof(char));
116 grow_syspage_section(&lsp.hwinfo, sizeof(struct hwi_prefix));
117 lsp.syspage.p->cluster.element_size = sizeof(struct cluster_entry);
118
119 lsp.syspage.p->num_cpu = 1;
120
121 cpu_init_syspage_memory();
122
123 hwi_default();
124}

126void
127reloc_syspage_memory(void *base, unsigned max_size) {
128 local_section *sect;
129 ptrdiff_t diff;
130
131 memmove(base, lsp.syspage.p, lsp.syspage.p->total_size);
132 diff = PTR_DIFF(base, lsp.syspage.p);
133 callout_reloc_data(lsp.syspage.p, lsp.syspage.p->total_size, diff);
134 for(sect = SECTION_START; sect < SECTION_END; ++sect) {
135 sect->p = (void *)((uint8_t *)sect->p + diff);
136 }
137 lsp.syspage.p->size = max_size;
138}

140void
141alloc_syspage_memory() {
142 PADDR_T cpupage_paddr;
143 unsigned i;
144 struct cpupage_entry cpu;
145
146 struct syspage_entry *sp = lsp.syspage.p;
147
148 //figure out size of callouts and it's rw_size.
149 unsigned rwsize;
150 callouts_size = output_callouts(&rwsize);
151
152 //Figure out total size required for the system page. It is then followed by
153 //callout rw data, callout code, cpupage_entry for each cpu.
154 //Allow for four more asinfo_entry's in case the syspage allocation(s)
155 //splits the address range(s).
156 // in order to make callout code to be readonly, we need to make sure the callout code
157 // is page aligned, so we need ROUNDPG before and after callout code.
158 unsigned spsize = ROUNDPG(sp->total_size
159 + rwsize
160 + 4*sizeof(struct asinfo_entry));
161 spsize += ROUNDPG(callouts_size); // this make sure cpupage_entries are in a separate page
162
163 _syspage_ptr = cpu_alloc_syspage_memory(&cpupage_paddr, &syspage_paddr, spsize);
164
165 #define INIT_ENTRY(field) \
166 sp->field.entry_size = lsp.field.size; \
167 sp->field.entry_off = PTR_DIFF(lsp.field.p, sp)
168
169 sp->size = sizeof(*lsp.syspage.p); // disallows further growing
170 sp->type = CPU_SYSPAGE_TYPE;
171
172 INIT_ENTRY(system_private);
173 INIT_ENTRY(asinfo);
174 INIT_ENTRY(hwinfo);
175 INIT_ENTRY(cpuinfo);
176 INIT_ENTRY(cacheattr);
177 INIT_ENTRY(qtime);
178 INIT_ENTRY(callout);
179 INIT_ENTRY(typed_strings);
180 INIT_ENTRY(strings);
181 INIT_ENTRY(intrinfo);
182 INIT_ENTRY(smp);
183 INIT_ENTRY(hypinfo);
184 INIT_ENTRY(cluster);
185
186 memset(&cpu, 0, sizeof(cpu));
187 for(i = 0; i < sp->num_cpu; ++i) {
188 struct cpuinfo_entry *cp;
189
190 cp = startup_memory_map(sizeof(cpu), cpupage_paddr, PROT_READ|PROT_WRITE);
191 cpu.cpu = i;
192 memmove(cp, &cpu, sizeof(cpu));
193 startup_memory_unmap(cp);
194 cpupage_paddr += lsp.system_private.p->cpupage_spacing;
195 }
196
197 //write the callouts to the syspage.
198 output_callouts(NULL);
199
200 //make the callouts read only execute.
201 uintptr_t callout_off = ROUNDPG(sp->total_size + rwsize);
202 cpu_syspage_remap((uintptr_t)lsp.system_private.p->kern_syspageptr+callout_off, syspage_paddr+callout_off, callouts_size, PF_R|PF_X);
203}
