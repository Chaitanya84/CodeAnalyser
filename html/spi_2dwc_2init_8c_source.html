44static uint32_t dw_spi_get_fifo_len(const dw_spi_t *const spi)
45{
46 uint32_t thr_lev_hold, left, right, i;
47
48 thr_lev_hold = dw_read32(spi, DW_SPI_TXFTLR);
49
50 if (thr_lev_hold != 0) {
51 left = thr_lev_hold;
52 } else {
53 left = DW_SPI_MIN_FIFO_LENGTH;
54 }
55 right = DW_SPI_MAX_FIFO_LENGTH + 1;
56
57 for (i = left; i <= right; i++) {
58 dw_write32(spi, DW_SPI_TXFTLR, i);
59 if (dw_read32(spi, DW_SPI_TXFTLR) != i) {
60 break;
61 }
62 }
63 dw_write32(spi, DW_SPI_TXFTLR, thr_lev_hold); /* restore old fifo threshold */
64
65 return (i);
66}

101static int dw_process_args(dw_spi_t *spi, char *options)
102{
103 char *value;
104 int opt = 0;
105 const char *c;
106 char *dw_opts[] = {
107 [OPTION_CS_MAX] = "cs_max", /* Chip select max */
108 [OPTION_SSTE] = "sste", /* Slave select Toggle enable */
109 [OPTION_LOOPBACK] = "loopback", /* Loopback interface for testing */
110#ifdef DW_DMA
111 [OPTION_RX_CHANNEL] = "dma_rx_chan", /* Receive channel ID for DMA */
112 [OPTION_TX_CHANNEL] = "dma_tx_chan", /* Transmit channel ID for DMA */
113#endif
114 [NUM_OPTIONS] = NULL
115 };
116
117 if (options == NULL) {
118 spi_slogf(_SLOG_WARNING, "%s: no board specific args passed", __func__);
119 return EOK;
120 }
121
122 while ((options != NULL) && (*options != '\0')) {
123 c = options;
124 opt = getsubopt(&options, dw_opts, &value);
125 if (opt == -1) {
126 spi_slogf(_SLOG_ERROR, "%s: unsupported SPI device driver args: %s", __func__, c);
127 return EINVAL;
128 }
129
130 switch (opt) {
131 case OPTION_CS_MAX:
132 if (value) {
133 spi->cs_max = (uint32_t) strtoul(value, NULL, 0);
134
135 if ((spi->cs_max < 1) || (spi->cs_max > 32)) {
136 spi_slogf(_SLOG_ERROR, "%s: Invalid cs_max value: %u", __func__,
137 spi->cs_max);
138 return EINVAL;
139 }
140 spi_slogf(_SLOG_DEBUG2, "%s: cs_max = %u", __func__,
141 spi->cs_max);
142 } else {
143 spi_slogf(_SLOG_ERROR, "%s: no cs_max value provided",
144 __func__);
145 return EINVAL;
146 }
147 break;
148 case OPTION_SSTE:
149 spi->sste = true;
150 spi_slogf(_SLOG_DEBUG2, "%s: SSTE is enabled", __func__);
151 break;
152 case OPTION_LOOPBACK:
153 spi->loopback = true; /* Enable loopback mode, used for testing */
154 break;
155#ifdef DW_DMA
156 case OPTION_RX_CHANNEL:
157 spi->dma.rx_channel = (uint32_t)strtoul(value, NULL, 0);
158 break;
159 case OPTION_TX_CHANNEL:
160 spi->dma.tx_channel = (uint32_t)strtoul(value, NULL, 0);
161 break;
162#endif
163 default:
164 spi_slogf(_SLOG_ERROR, "%s: wrong option", __func__);
165 return EINVAL;
166 }
167 }
168 return EOK;
169}

177static int map_in_device_registers(dw_spi_t *spi)
178{
179 const void *const hw_addr = mmap_device_memory(NULL, spi->map_size, PROT_READ|PROT_WRITE|PROT_NOCACHE, MAP_SHARED, spi->pbase);
180 if (hw_addr == MAP_FAILED) {
181 spi_slogf(_SLOG_ERROR, "%s: mmap_device_memory failed pbase:%x, map size: %d", __func__, spi->pbase, spi->map_size);
182 return ENOMEM;
183 }
184
185 spi->vbase = (uintptr_t)hw_addr;
186
187 return EOK;
188}

196static int dw_init_device(dw_spi_t *const spi)
197{
198 int status;
199 spi_bus_t *const bus = spi->bus;
200 spi_dev_t *spi_dev = bus->devlist;
201
202 /*
203 * Initial device configuration with defaults from config file
204 */
205 while (spi_dev != NULL) {
206 status = dw_cfg(spi, spi_dev);
207 if (status != EOK) {
208 spi_slogf(_SLOG_ERROR, "%s: dw_cfg failed", __func__);
209 return status;
210 }
211 spi_dev = spi_dev->next;
212 }
213
214 dw_read32(spi, DW_SPI_ICR);
215 dw_write32(spi, DW_SPI_IMR, DW_SPI_INT_MASK);
216 dw_write32(spi, DW_SPI_CTRLR1, 0);
217 dw_spi_deselect_slave(spi);
218
219 /* get fifo size */
220 spi->fifo_len = dw_spi_get_fifo_len(spi);
221
222#ifdef DW_DMA
223 if (bus->funcs->dma_xfer != NULL) {
224 /* check valid dma threshold */
225 if (spi->bus->dma_thld > spi->fifo_len) {
226 spi_slogf(_SLOG_WARNING, "%s: SPI DMA threshold %d is greater than fifo len %d on bus.", __func__, spi->bus->dma_thld, spi->fifo_len);
227 spi->bus->dma_thld = (uint8_t)(spi->fifo_len / SPI_FIFO_LEN_DIV);
228 spi_slogf(_SLOG_INFO, "%s: SPI DMA threshold will be configured as %d on bus.", __func__, spi->bus->dma_thld);
229 }
230
231 /* Init DMA */
232 status = dw_spi_init_dma(spi);
233 if (status != EOK) {
234 spi_slogf(_SLOG_ERROR, "%s: dw_spi_init_dma failed", __func__);
235 }
236 }
237#endif
238
239 /* Attach SPI interrupt */
240 status = dw_attach_intr(spi);
241 if (status != EOK) {
242 spi_slogf(_SLOG_ERROR, "%s: dw_attach_intr failed", __func__);
243 }
244
245 return status;
246}

254int spi_init(spi_bus_t *bus)
255{
256 dw_spi_t *spi = NULL;
257 int status = EOK;
258
259 if (bus == NULL) {
260 spi_slogf(_SLOG_ERROR, "%s: no SPI bus structure !", __func__);
261 return EINVAL;
262 }
263
264 spi = calloc(1, sizeof(dw_spi_t));
265 if (spi == NULL) {
266 spi_slogf(_SLOG_ERROR, "%s: failed to allocate memory !", __func__);
267 return ENOMEM;
268 }
269
270 /* Save spi_ctrl to driver structure */
271 spi->spi_ctrl = bus->spi_ctrl;
272 spi->bus = bus;
273
274 /* Get other SPI driver functions */
275 bus->funcs->spi_fini = dw_fini;
276 bus->funcs->drvinfo = dw_drvinfo;
277 bus->funcs->devinfo = dw_devinfo;
278 bus->funcs->setcfg = dw_setcfg;
279 bus->funcs->xfer = dw_xfer;
280 bus->funcs->dma_xfer = NULL;
281 bus->funcs->dma_allocbuf = NULL;
282 bus->funcs->dma_freebuf = NULL;
283
284 /* Set default value */
285 spi->iid = -1;
286 spi->vbase = (uintptr_t)MAP_FAILED;
287
288 /* Set the default value of the SPI device */
289 spi->loopback = false;
290 spi->pbase = bus->pbase;
291 spi->map_size = SPI0_SIZE;
292 spi->cs_max = SPI_CS_MAX_DEF;
293 spi->sste = false;
294#ifdef DW_DMA
295 spi->dma.rx_channel = CHANNEL_ID_MAX;
296 spi->dma.tx_channel = CHANNEL_ID_MAX;
297#endif
298
299 /* Process args, override the defaults */
300 if (bus->bs != NULL) {
301 status = dw_process_args(spi, bus->bs);
302 if (status != EOK) {
303 spi_slogf(_SLOG_ERROR, "%s: dw_process_args failed", __func__);
304 dw_fini(spi);
305 return status;
306 }
307 }
308
309#ifdef DW_DMA
310 if ((spi->dma.tx_channel != CHANNEL_ID_MAX) && (spi->dma.rx_channel != CHANNEL_ID_MAX)) {
311 bus->funcs->dma_xfer = dw_spi_dmaxfer;
312 bus->funcs->dma_allocbuf = dw_spi_dma_allocbuf;
313 bus->funcs->dma_freebuf = dw_spi_dma_freebuf;
314 }
315#endif
316
317 /* Check the device interrupt number */
318 if (spi->bus->irq <= 0) {
319 spi_slogf(_SLOG_ERROR, "%s: Invalid IRQ: %x", __func__, spi->bus->irq);
320 return EINVAL;
321 }
322
323 /*map SPI controller base address */
324 status = map_in_device_registers(spi);
325 if (status != EOK) {
326 spi_slogf(_SLOG_ERROR, "%s: map_in_device_registers failed", __func__);
327 dw_fini(spi);
328 return status;
329 }
330
331 /* Disable spi to set controls */
332 dw_spi_enable(spi, DW_SPI_DISABLE);
333
334 /* Init SPI device */
335 status = dw_init_device(spi);
336 if (status != EOK) {
337 dw_fini(spi);
338 return status;
339 }
340 spi_slogf(_SLOG_DEBUG2, "%s: dw_init_device succeeded", __func__);
341
342 /*
343 * Create SPI chip select devices
344 */
345 status = spi_create_devs(bus->devlist);
346 if (status != EOK) {
347 spi_slogf(_SLOG_ERROR, "%s: Disabling SPI device", __func__, bus->devlist->devinfo.name);
348 dw_fini(spi);
349 return status;
350 }
351
352 /* Save driver structure to drvhdl */
353 bus->drvhdl = spi;
354
355 return status;
356}
