50int board_smp_start(const unsigned cpu, void (*start)(void))
51{
52#if 1
53 return psci_smp_start(cpu, start);
54#else
55 typedef void (*start_t)(void);
56 volatile start_t *cpu_start = start;
57 uint64_t status;
58
59 status = psci_call(PSCI_CPU_ON, (uint64_t)(cpu << 8), (uint64_t)start, 0);
60 if (status != PSCI_SUCCESS) {
61 crash("%s: psci_call PSCI_CPU_ON cpu: %lu failed! return: %L. \n", __func__, cpu, status);
62 }
63
64 if (debug_flag) {
65 kprintf("%s: psci_call PSCI_CPU_ON cpu: %lu ok\n", __func__, cpu);
66 }
67 return 1;
68#endif
69}
