32/*
33 * CP15 Auxiliary control register
34 *
35 * Accessed by mcr/mrc p15, 0, Rd, c1, c0, 1
36 */
37#define MPCORE_CP15_AUX_SMP (1 << 5) /* enable SMP mode */
38#define MPCORE_CP15_AUX_EXCL (1 << 4) /* L1/L2 caches are exclusive */
39#define MPCORE_CP15_AUX_F (1 << 3) /* enable instruction folding */
40#define MPCORE_CP15_AUX_SB (1 << 2) /* enable static prediction */
41#define MPCORE_CP15_AUX_DB (1 << 1) /* enable dynamic prediction */
42#define MPCORE_CP15_AUX_RS (1 << 0) /* enable return stack */
