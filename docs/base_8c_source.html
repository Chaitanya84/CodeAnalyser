119// return position of MSB
120int fls( int val )
121{
122 int idx;
123
124 idx = 32;
125 while( val ) {
126 if( ( val & 0x80000000u ) ) return( idx );
127 val <<= 1;
128 idx--;
129 }
130 return( 0 );
131}

248static char *sdio_strchr( char *string, char token )
249{
250 char *cp;
251 int escaped;
252
253 escaped = 0;
254 for( cp = string; *cp != '\0'; ++cp ) {
255 if( escaped ) {
256 escaped = 0;
257 --cp;
258 memmove( cp, cp + 1, strlen( cp ) );
259 }
260 else if( *cp == '\\' ) {
261 escaped = !0;
262 }
263 else if( *cp == token ) {
264 return( cp );
265 }
266 else {
267 // nothing
268 }
269 }
270 return( NULL );
271}

273static int sdio_parse_tuple( char *string, char separator, char **entry, ... )
274{
275 va_list args;
276 char *cp;
277 int nargs;
278
279 nargs = 0;
280 va_start( args, entry );
281 while( ( string != NULL ) || ( entry != NULL ) ) {
282 cp = (string != NULL) ? sdio_strchr( string, separator ) : NULL;
283 if( entry != NULL ) {
284 *entry = string;
285 ++nargs;
286 entry = va_arg( args, char ** );
287 }
288 if( cp != NULL ) {
289 *cp++ = '\0';
290 }
291 string = cp;
292 }
293 va_end( args );
294 return( nargs );
295}

306int sdio_hc_getsubopt( char **optionp, char * const *tokens, char **valuep )
307{
308 char *p;
309 char *opt;
310 int len;
311 int index;
312 const char *token;
313
314 *valuep = NULL;
315 opt = *optionp;
316
317 for( p = opt, len = 0; *p && ( *p != ':' ); p++, len++ ) {
318 if( *p == '=' ) {
319 for( *valuep = ++p; *p && ( *p != ':' ); p++ ) {
320 // Nothing to do
321 }
322 break;
323 }
324 }
325
326 if( *p ) {
327 *p++ = '\0';
328 }
329
330 *optionp = p;
331
332 for( index = 0; ( token = *tokens++ ); index++ ) {
333 if( ( *token == *opt ) && (len > 0) &&!strncmp( token, opt, len ) && ( token[len] == '\0' ) ) {
334 return( index );
335 }
336 }
337
338 *valuep = opt;
339
340 return( -1 );
341}

352int sdio_create_thread( pthread_t *tid, pthread_attr_t *aattr, void *(*func)(void *), void * const arg, const int priority, uint32_t *tstate, char * const name )
353{
354 pthread_t _tid;
355 pthread_attr_t *pattr;
356 pthread_attr_t attr;
357 struct sched_param param = {0};
358 int status;
359 int ret = EOK;
360 if( tid == NULL ) {
361 tid = &_tid;
362 }
363
364 if( tstate ) {
365 *tstate = SDIO_TSTATE_CREATING;
366 }
367
368 pattr = aattr;
369 if( aattr == NULL ) {
370 pattr = &attr;
371 ret = pthread_attr_init( &attr );
372 if (ret != EOK) {
373 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_init failed %s", __func__, strerror( ret ) );
374 return (-1);
375 }
376 ret = pthread_attr_setschedpolicy( &attr, SCHED_RR );
377 if (ret != EOK) {
378 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_setschedpolicy failed %s", __func__, strerror( ret ) );
379 return (-1);
380 }
381 param.sched_priority = priority;
382 ret = pthread_attr_setschedparam( &attr, &param );
383 if (ret != EOK) {
384 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_setschedparam failed %s", __func__, strerror( ret ) );
385 return (-1);
386 }
387 ret = pthread_attr_setinheritsched( &attr, PTHREAD_EXPLICIT_SCHED );
388 if (ret != EOK) {
389 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_setinheritsched failed %s", __func__, strerror( ret ) );
390 return (-1);
391 }
392 ret = pthread_attr_setstacksize( &attr, SDIO_STACK_SIZE );
393 if (ret != EOK) {
394 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_setstacksize failed %s", __func__, strerror( ret ) );
395 return (-1);
396 }
397 }
398
399 status = pthread_create( tid, pattr, func, arg );
400 if (status != EOK) {
401 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_create failed %s", __func__, strerror( status ) );
402 return (-1);
403 }
404
405 if( aattr == NULL ) {
406 ret = pthread_attr_destroy( &attr );
407 if (ret != EOK) {
408 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_attr_destroy failed %d", __func__, ret);
409 return (-1);
410 }
411
412 }
413 if( status ) {
414 return( status );
415 }
416
417 if( name ) {
418#if ( _NTO_VERSION >= 632 )
419 if( pthread_setname_np( *tid, name ) ) {
420// sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_setname_np %s", __FUNCTION__, strerror( status ) );
421 }
422#endif
423 }
424
425 if( tstate ) {
426 pthread_sleepon_lock( );
427 while( !( *tstate ) ) {
428 pthread_sleepon_wait( tstate );
429 }
430 pthread_sleepon_unlock( );
431 status = (*tstate == SDIO_TSTATE_INITIALIZED) ? EOK : EIO;
432 }
433
434 return( status );
435}

437uint32_t sdio_extract_bits( uint32_t * const data, const int bits, const int start, const int size )
438{
439 int idx;
440 int shift;
441 uint32_t value;
442
443 idx = ( bits / 32 ) - ( start / 32 ) - 1;
444
445 shift = start & 31;
446 value = data[idx] >> shift;
447 if( size + shift > 32 ) {
448 value |= data[idx - 1] << ( 32 - shift );
449 }
450
451 return( value & ( ( 1llu << size ) - 1 ) );
452}

473int sdio_sg_nxt( sdio_hc_t *hc, uint8_t **addr, uint32_t *len, const uint32_t blksz )
474{
475 sdio_wspc_t *wspc;
476
477 wspc = &hc->wspc;
478
479 if( !wspc->nsg && !wspc->sgc ) { // sge's complete
480 return( SDIO_TRUE );
481 }
482
483 if( !wspc->sgc ) { // advance to next sg entry
484 wspc->sge++;
485 wspc->nsg--;
486 wspc->sgc = wspc->sge->sg_count;
487 wspc->sga = SDIO_DATA_PTR_V( wspc->sge->sg_address );
488 }
489
490 *addr = wspc->sga;
491 *len = min( wspc->sgc, blksz );
492 wspc->sga += *len;
493 wspc->sgc -= *len;
494
495 return( SDIO_FALSE );
496}

607int _sdio_pwrmgnt( sdio_dev_t * const dev, uint32_t pm )
608{
609 sdio_hc_t *hc;
610 int status;
611
612 hc = dev->hc;
613 status = EOK;
614
615 if( ( hc->entry.pm == NULL ) || ( pm == hc->pm_state ) || hc->wspc.cmd ) {
616 return( EOK );
617 }
618
619 switch( pm ) {
620 case PM_IDLE:
621// sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 3, "%s: idle", __FUNCTION__ );
622
623 if( hc->pm_state == PM_ACTIVE ) {
624 hc->entry.pm( hc, pm );
625 }
626 else {
627 return( EOK );
628 }
629 break;
630
631 case PM_ACTIVE:
632// sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 3, "%s (path %d): active", __FUNCTION__, hc->path );
633
634 hc->entry.pm( hc, pm );
635 if( hc->pm_state == PM_SUSPEND ) {
636 hc->pm_state = pm;
637 _sdio_reset( dev );
638
639 // start change detect timer
640 if( !( hc->caps & HC_CAP_CD_INTR ) && !( hc->caps & HC_CAP_SLOT_TYPE_EMBEDDED ) ) {
641 sdio_timer_settime( sdio_ctrl.cd_timerid, SDIO_CD_INTERVAL, 0, SDIO_TRUE );
642 }
643 }
644 else if( ( hc->pm_state == PM_SLEEP ) && ( hc->caps & HC_CAP_SLEEP ) ) {
645 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 3, "%s (path %d): sleep to active", __FUNCTION__, hc->path );
646 hc->pm_state = pm;
647 if( ( dev->caps & DEV_CAP_SLEEP ) ) {
648 status = mmc_sleep_awake( dev, 0 );
649 if( status == EOK ) {
650 status = sdio_select_card( dev, dev->rca );
651 }
652 if( status ) {
653 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: awake fail", __FUNCTION__ );
654 status = _sdio_reset( dev );
655 }
656 }
657 else {
658 if( ( dev->flags & DEV_FLAG_PRESENT ) ) {
659 status = _sdio_reset( dev );
660 }
661 }
662 }
663 else {
664 // nothing
665 }
666
667 if( hc->tuning_count ) { // start tuning timer
668 if( sdio_timer_settime( hc->tuning_timerid, hc->tuning_count, 0, SDIO_FALSE ) ) {
669 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: sdio_timer_settime (tuning start)", __FUNCTION__ );
670 }
671 }
672 break;
673
674 case PM_SLEEP:
675// sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 3, "%s: sleep", __FUNCTION__ );
676
677 if( hc->pm_state == PM_IDLE ) { // wakeup when idle to deselect
678 hc->entry.pm( hc, PM_ACTIVE );
679 }
680
681 if( ( hc->caps & HC_CAP_SLEEP ) ) {
682 if( ( dev->caps & DEV_CAP_SLEEP ) ) {
683 if( status == EOK ) {
684 status = sdio_select_card( dev, 0 );
685 if( status == EOK ) {
686// Should we wait for STANDBY state before putting the device to sleep?
687// if( ( status = _sdio_wait_card_status( dev, NULL, CDS_CUR_STATE_MSK, CDS_CUR_STATE_STANDBY, SDIO_TIME_DEFAULT ) ) == EOK ) {
688 if( ( dev->dtype == DEV_TYPE_MMC ) ) {
689 status = mmc_sleep_awake( dev, 1 );
690 }
691// }
692 }
693
694 if( status ) {
695 _sdio_reset( dev );
696 }
697 }
698 }
699
700 if( status ) {
701 pm = PM_IDLE; // set state back to idle
702 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: sleep fail", __FUNCTION__ );
703 }
704 else {
705 if( hc->tuning_count ) { // stop tuning timer
706 if( sdio_timer_settime( hc->tuning_timerid, 0, 0, SDIO_FALSE ) ) {
707 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: sdio_timer_settime (tuning stop)", __FUNCTION__ );
708 }
709 }
710 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 3, "%s (path %d): sleep hc cnt %"PRId64"", __FUNCTION__, hc->path, ++hc->pm_sleep_cnt );
711 }
712 hc->entry.pm( hc, pm );
713 }
714 break;
715
716 case PM_SUSPEND:
717 if( hc->tuning_count ) { // stop tuning timer
718 if( sdio_timer_settime( hc->tuning_timerid, 0, 0, SDIO_FALSE ) ) {
719 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: sdio_timer_settime (tuning stop)", __FUNCTION__ );
720 }
721 }
722 // stop change detect timer
723 if( !( hc->caps & HC_CAP_CD_INTR ) && !( hc->caps & HC_CAP_SLOT_TYPE_EMBEDDED ) ) {
724 sdio_timer_settime( sdio_ctrl.cd_timerid, 0, 0, SDIO_FALSE );
725 }
726
727 hc->entry.pm( hc, pm );
728 break;
729
730 default:
731 break;
732 }
733
734 hc->pm_state = pm;
735
736 return( status );
737}

1029int _sdio_wait_card_status( sdio_dev_t * const dev, uint32_t *rsp, const uint32_t mask, const uint32_t val, uint32_t msec )
1030{
1031 sdio_hc_t *hc;
1032 int status;
1033 uint32_t resp[4];
1034
1035 hc = dev->hc;
1036 status = EOK;
1037 rsp = rsp ? rsp : resp;
1038
1039 for( msec = max( msec, 1 ); msec; msec-- ) {
1040 status = _sdio_send_status( dev, rsp, SDIO_FALSE );
1041 if( status != EOK ) {
1042 break;
1043 }
1044
1045 if( hc->cfg.verbosity > 3 ) {
1046 sdio_rsp( dev, rsp );
1047 }
1048
1049 if( ( rsp[0] & CDS_ERROR_MSK ) ) {
1050 status = EIO;
1051 break;
1052 }
1053
1054 if( ( rsp[0] & mask ) == val ) {
1055 status = EOK;
1056 break;
1057 }
1058
1059 delay( 1 );
1060 }
1061
1062 if( !msec ) {
1063 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: mask %x, val %x, card status %x", __FUNCTION__, mask, val, rsp[0] );
1064 sdio_rsp( dev, rsp );
1065 status = ETIMEDOUT;
1066 }
1067
1068 return( status );
1069}

1085int _sdio_lock_unlock( sdio_dev_t *dev, const int op, uint8_t * const pwd, const uint32_t pwd_len )
1086{
1087 struct sdio_cmd *cmd;
1088 sdio_sge_t sge;
1089 uint8_t *ld;
1090 uint32_t len;
1091 int status;
1092 int timeout;
1093 uint32_t rsp[4];
1094
1095 len = pwd_len + 2; // +2 for op field, and pwd_len field
1096
1097 if( ( len & 0x1 ) ) {
1098 len++;
1099 }
1100
1101 timeout = ( op & SD_LU_ERASE ) ? sdio_erase_timeout( dev, SD_ERASE_NORM, dev->csd.sectors ): SDIO_TIME_DEFAULT;
1102
1103 cmd = sdio_alloc_cmd( );
1104 if( cmd == NULL ) {
1105 return( ENOMEM );
1106 }
1107
1108 ld = sdio_alloc( len );
1109 if( ld == NULL ) {
1110 sdio_free_cmd( cmd );
1111 return( ENOMEM );
1112 }
1113
1114 ld[0] = op;
1115 ld[1] = pwd_len;
1116 if( pwd_len ) {
1117 memcpy( &ld[2], pwd, pwd_len );
1118 }
1119
1120 status = _sdio_set_block_length( dev, len );
1121 if( status == EOK ) {
1122 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, MMC_LOCK_UNLOCK, 0 );
1123 sge.sg_count = len;
1124 sge.sg_address = SDIO_DATA_PTR_P( ld );
1125 sdio_setup_cmd_io( cmd, SCF_DIR_OUT, 1, len, &sge, 1, NULL );
1126
1127 status = _sdio_send_cmd( dev, cmd, NULL, timeout, SDIO_CMD_RETRIES );
1128 _sdio_set_block_length( dev, SDIO_DFLT_BLKSZ );
1129 }
1130
1131 if( status == EOK ) {
1132 status = _sdio_send_status( dev, rsp, SDIO_FALSE );
1133 if( status == EOK ) {
1134 if( ( rsp[0] & CDS_CARD_IS_LOCKED ) ) {
1135 atomic_set( &dev->flags, DEV_FLAG_LOCKED );
1136 }
1137
1138 if( ( dev->flags & DEV_FLAG_LOCKED ) && !( rsp[0] & CDS_CARD_IS_LOCKED ) ) {
1139 atomic_clr( &dev->flags, DEV_FLAG_LOCKED );
1140 }
1141 }
1142 }
1143
1144 sdio_free( ld, len );
1145 sdio_free_cmd( cmd );
1146
1147 return( status );
1148}

1197int _sdio_reset( sdio_dev_t *dev )
1198{
1199 sdio_hc_t *hc;
1200 int retry;
1201 int status;
1202
1203 hc = dev->hc;
1204 status = EOK;
1205 retry = SDIO_RESET_RETRIES;
1206
1207 if( ( hc->flags & HC_FLAG_RST ) ) {
1208 return( EBUSY );
1209 }
1210
1211 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 1, "%s (path %d): ", __FUNCTION__, hc->path );
1212
1213 hc->flags |= HC_FLAG_RST;
1214 hc->flags &= ~HC_FLAG_SKIP_PWRUP;
1215 dev->rca = 0;
1216 dev->pactive = 0;
1217 dev->flags &= ~DEV_FLAG_WCE;
1218
1219 do {
1220 sdio_power( hc, SDIO_PWR_OFF );
1221 sdio_power( hc, SDIO_PWR_ON );
1222
1223 switch( dev->dtype ) {
1224 case DEV_TYPE_MMC:
1225 status = mmc_init_device( hc, dev->ocr, SDIO_TRUE );
1226 break;
1227
1228 case DEV_TYPE_SD:
1229 status = sd_init_device( hc, dev->ocr, SDIO_TRUE );
1230 break;
1231
1232 default:
1233 status = ENXIO;
1234 break;
1235 }
1236
1237 if( status == ENXIO ) { // device has been removed
1238 sdio_hc_event( hc, HC_EV_CD );
1239 break;
1240 }
1241
1242 } while( --retry && ( status != EOK ) );
1243
1244 hc->flags &= ~HC_FLAG_RST;
1245
1246 return( status );
1247}

1273int sdio_wait_cmd( sdio_hc_t *hc, struct sdio_cmd * const cmd, const uint64_t tms )
1274{
1275 uint64_t ct;
1276 int status;
1277 struct timespec abstime;
1278
1279 status = EOK;
1280 int ret;
1281 ct = _syspage_time( CLOCK_MONOTONIC );
1282 nsec2timespec( &abstime, ct + SDIO_TIMEOUT_MS_TO_NS( tms ) );
1283
1284
1285 ret = pthread_mutex_lock( &hc->mutex );
1286 if (ret != EOK) {
1287 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, 0, 1, "abort due to lock mutex err: %d", ret);
1288 abort();
1289 }
1290 while( cmd->status == CS_CMD_INPROG ) {
1291 status = pthread_cond_timedwait( &hc->cond, &hc->mutex, &abstime );
1292 if( status == ETIMEDOUT ) {
1293 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: TIMEOUT %"PRId64"ms (errno %d) CMD %d, flgs 0x%x, arg 0x%x, blks %d, blksz %d",
1294 __FUNCTION__, tms, status, cmd->opcode, cmd->flags, cmd->arg, cmd->blks, cmd->blksz );
1295 break;
1296 }
1297 }
1298 (void)pthread_mutex_unlock( &hc->mutex );
1299
1300 return( status );
1301}

1381int _sdio_send_cmd( sdio_dev_t * const dev, struct sdio_cmd *cmd,
1382 void (*func)( struct sdio_device *, sdio_cmd_t *, void *),
1383 const uint32_t timeout, int retries )
1384{
1385 sdio_hc_t *hc;
1386 uint64_t acmd_caps;
1387 int status;
1388
1389 hc = dev->hc;
1390 acmd_caps = 0;
1391
1392 status = _sdio_pwrmgnt( dev, PM_ACTIVE );
1393 if( status != EOK ) {
1394 return( status );
1395 }
1396
1397 if( ( cmd->flags & SCF_SBC_RLW ) ) {
1398 // Reliable write is enabled via bit 31 in set block count,
1399 // so temporarily disable HC ACMD12/ACMD23 features.
1400 acmd_caps = hc->caps & ( HC_CAP_ACMD23 | HC_CAP_ACMD12 );
1401 hc->caps &= ~( HC_CAP_ACMD23 | HC_CAP_ACMD12 );
1402 }
1403
1404 do {
1405 cmd->status = CS_CMD_INPROG;
1406
1407 if( ( cmd->flags & SCF_APP_CMD ) ) {
1408 status = sd_app_cmd( dev );
1409 if( status != EOK ) {
1410 break;
1411 }
1412 }
1413
1414 if( ( cmd->flags & SCF_SBC ) && !( hc->caps & HC_CAP_ACMD23 ) ) {
1415 status = _sdio_set_block_count( dev, cmd->blks, ( cmd->flags & SCF_SBC_RLW ) ? SBC_RL_WRITE : 0 );
1416 if( status != EOK ) {
1417 break;
1418 }
1419 }
1420
1421#ifdef SDIO_UC_SUPPORT
1422 if( ( cmd->flags & SCF_SUA ) ) {
1423 if( ( status = sdio_set_upper_addr( dev, cmd->earg ) ) ) {
1424 break;
1425 }
1426 }
1427#endif
1428
1429 status = sdio_issue_cmd( dev, cmd, timeout );
1430 if( status != EOK ) {
1431 // calling function should handle error recovery
1432 break;
1433 }
1434
1435 switch( cmd->status ) {
1436 case CS_CMD_CMP:
1437 status = EOK;
1438 break;
1439
1440 case CS_CMD_TO_ERR:
1441 status = ETIMEDOUT;
1442 break;
1443
1444 case CS_CARD_REMOVED:
1445 return( ENXIO );
1446
1447 case CS_DATA_CRC_ERR:
1448 if( ( dev->flags & DEV_FLAG_PRESENT ) && !( hc->flags & HC_FLAG_RST )
1449 && ( dev->flags & ( DEV_FLAG_UHS | DEV_FLAG_HS200 | DEV_FLAG_HS400 ) ) ) {
1450 atomic_set( &hc->flags, HC_FLAG_TUNE );
1451 }
1452 status = EIO;
1453 break;
1454
1455 default:
1456 status = EIO;
1457 break;
1458 }
1459
1460 if( status == EOK ) {
1461 if( ( cmd->flags & SCF_WAIT_DRDY ) && !( hc->caps & HC_CAP_BSY ) ) {
1462 status = _sdio_wait_card_status( dev, NULL, CDS_READY_FOR_DATA | CDS_CUR_STATE_MSK, CDS_READY_FOR_DATA | CDS_CUR_STATE_TRAN, timeout );
1463 if( status != EOK ) {
1464 break;
1465 }
1466 }
1467 break;
1468 }
1469
1470 if( ( cmd->rsp[0] & CDS_ILLEGAL_COMMAND ) ) {
1471 break;
1472 }
1473
1474 } while( retries-- );
1475
1476 if( acmd_caps ) { // Restore ACMD12/ACMD23 feature flags
1477 hc->caps |= acmd_caps;
1478 }
1479
1480 return( status );
1481}

1694{

1709 abort();

1715 atomic_clr( &dev->flags, ( DEV_FLAG_MEDIA_CHANGE | DEV_FLAG_INVALID_CARD | DEV_FLAG_WRITE_PROTECT ) );

1718 atomic_clr( &dev->flags, DEV_FLAG_PRESENT );

1721 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 1, "abort due to lock mutex err: %d", ret);

1726 for( device = TAILQ_FIRST( &sdio_ctrl.dlist ); device != NULL; device = TAILQ_NEXT( device, dlink ) ) {
1727 if( ( device->instance.path == hc->path ) && ( device->instance.generation == hc->generation ) ) {
1728 break;
1729 }
1730 }
1731
1732 if( device ) {
1733 // we can't use _sdio_synchronize to set the
1734 // pending flag since we already hold the mutex
1735 atomic_set( &device->flags, DEV_FLAG_RMV_PENDING );
1736
1737 while( device->usage ) {
1738 ret = pthread_cond_wait( &sdio_ctrl.cd_cond, &sdio_ctrl.mutex );
1739 if (ret != EOK) {
1740 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 0, 0, "%s: pthread_cond_wait failed %s", __func__, strerror( ret ) );
1741 break;
1742 }
1743 }
1744 }
1745
1746 (void)pthread_mutex_unlock( &sdio_ctrl.mutex );
1747
1748 if( device && ( funcs->removal != NULL ) ) {
1749 funcs->removal( (void *)&sdio_ctrl, &device->instance );
1750 }
1751

1851 struct sigevent event;
1852
1853 sc = hdl;
1854 init = SDIO_FAILURE;
1855
1856 sc->cd_chid = ChannelCreate( _NTO_CHF_PRIVATE | _NTO_CHF_DISCONNECT | _NTO_CHF_UNBLOCK );

1886 sdio_cd( hc );
1887 }
1888 }
1889 break;
1890
1891 case SDIO_EV_CD:

1895

1897// sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: unknown pulse rid %d, type %x, subtype %, pulse %x, value %x, scoid %x", __FUNCTION__, rid, pulse.type, pulse.subtype, pulse.code, pulse.value, pulse.scoid );

1899 }
1900
1901 }
1902
