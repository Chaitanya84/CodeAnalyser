28is_elf64(paddr_t addr)
29{
30 Elf64_Ehdr *p, hdr, *hp=&hdr;
31
32 p = startup_memory_map(sizeof(hdr), addr, PROT_READ);
33 hdr = *p;
34 startup_memory_unmap(p);
35
36 if (memcmp(hp->e_ident, ELFMAG, SELFMAG)
37 || hp->e_ident [EI_DATA] != ELFDATANATIVE
38 || hp->e_ident[EI_CLASS] != ELFCLASS64
39 || hp->e_phnum == 0
40 || hp->e_phentsize != sizeof(Elf64_Phdr)) {
41 return 0;
42 }
43 return 1;
44}
