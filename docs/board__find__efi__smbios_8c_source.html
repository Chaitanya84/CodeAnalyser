30static void dump_hex(_Uint8t *data, _Uint8t len)
31{
32 _Uint8t i;
33 for (i=0; i<len; i++) {
34 if ((i % 16) == 0) {
35 kprintf("0x%x: ", i);
36 }
37 kprintf("%b ", data[i]);
38 if ((i % 16) == 15) {
39 kprintf("\n");
40 }
41 }
42 kprintf("\n");
43}

97static void dump_smbios_data(smbios_structure_t *smbios_data_p)
98{
99 _Uint8t *data;
100 _Uint8t len = 0;
101
102 if (debug_flag > 4) {
103 kprintf("Hdr: %u-%u-%u Buf: %L\n",
104 smbios_data_p->Hdr->Type, smbios_data_p->Hdr->Length, smbios_data_p->Hdr->Handle, smbios_data_p->buf);
105
106 /* get smbios structure string section length */
107 data = smbios_data_p->buf + smbios_data_p->Hdr->Length;
108 while (data[0] || data[1]) {
109 data++;
110 len++;
111 }
112 len += 2 + smbios_data_p->Hdr->Length;
113
114 data = smbios_data_p->buf;
115 dump_hex(data, len);
116 }
117}

150const char *get_smbios_string(smbios_header_t *hdr, smbios_string_idx_t idx)
151{
152 char *data = (char *)hdr;
153
154 if ((hdr != NULL) && (idx > 0)) {
155 data += hdr->Length;
156 while (idx > 1) {
157 data += strlen(data);
158 data++;
159 idx--;
160 }
161 return data;
162 }
163 return "";
164}

166paddr_t board_find_efi_smbios_table(void)
167{
168 unsigned i;
169 void *cfg_tbl;
170
171 if (smbios_table == NULL_PADDR) {
172 for (i = 0; i < NUM_ELTS(smbios_guids); ++i) {
173 cfg_tbl = uefi_find_config_tbl(&smbios_guids[i]);
174 if (cfg_tbl != NULL) {
175 if (i==0) {
176 smbios_table_3_t *table = (smbios_table_3_t *) cfg_tbl;
177 use_smbios_3 = 1;
178 dump_smbios_3_table(table);
179 as_add((paddr_t)table, (paddr_t)table+(table->TableMaximumSize)-1, AS_ATTR_RAM,
180 "smbios3", as_find(AS_NULL_OFF, "memory", NULL));
181 }
182 else {
183 smbios_table_t *table = (smbios_table_t *) cfg_tbl;
184 dump_smbios_table(table);
185 as_add((paddr_t)table, (paddr_t)table+(table->TableLength)-1, AS_ATTR_RAM,
186 "smbios", as_find(AS_NULL_OFF, "memory", NULL));
187 }
188 smbios_table = (paddr_t)cfg_tbl;
189 break;
190 }
191 }
192 }
193 return smbios_table;
194}

196paddr_t board_find_efi_smbios_by_type(_Uint8t type)
197{
198 paddr_t table;
199 smbios_structure_t smbios_data;
200 smbios_structure_t smbios_end_data;
201 _Uint8t *data_p;
202
203 table = board_find_efi_smbios_table();
204 if (table == NULL_PADDR) {
205 kprintf("Could not find smbios table\n");
206 return NULL_PADDR;
207 }
208
209 if (use_smbios_3 == 0) {
210 smbios_data.buf = (_Uint8t *)(_Uint64t)((smbios_table_t *)table)->TableAddress;
211 smbios_end_data.buf = smbios_data.buf + ((smbios_table_t *)table)->TableLength;
212 }
213 else {
214 smbios_data.buf = (_Uint8t *)(_Uint64t)((smbios_table_3_t *)table)->TableAddress;
215 smbios_end_data.buf = smbios_data.buf + ((smbios_table_3_t *)table)->TableMaximumSize;
216 }
217
218 while (1) {
219
220 if (smbios_data.Hdr->Type == type) {
221 if(debug_flag > 2) {
222 kprintf("Find smbios type %u data at %L\n", type, smbios_data.buf);
223 }
224 dump_smbios_data(&smbios_data);
225 if (type == 1) {
226 dump_smbios_data_type_1((smbios_table_type_1_t *)smbios_data.buf);
227 }
228 if (type == 4) {
229 dump_smbios_data_type_4((smbios_table_type_4_t *)smbios_data.buf);
230 }
231 return (paddr_t)smbios_data.buf;
232 }
233
234 if (smbios_data.Hdr->Type == 127) {
235 /* reach to the last type */
236 if(debug_flag > 2) {
237 kprintf("Find smbios end type data\n");
238 }
239 break;
240 }
241 /* point to the next smbios structure */
242 data_p = smbios_data.buf + smbios_data.Hdr->Length;
243
244 /* skip string section, smbios structure ends with two consecutive 0s */
245 while (data_p[0] || data_p[1]) {
246 data_p++;
247 }
248 data_p += 2;
249 smbios_data.buf = data_p;
250 if (smbios_data.buf >= smbios_end_data.buf)
251 break;
252 }
253 if (debug_flag > 2) {
254 kprintf("Could not find type %u smbios data\n", type);
255 }
256 return NULL_PADDR;
257}
