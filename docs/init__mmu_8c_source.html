29 * Index MAIR ARMv8 meaning ARMv7 meaning
30 * ----- ---- ------------- -------------

34 * 3 0xee normal-WBnWA normal-WBnWA

90/*
91 * Initialise MMU structures
92 */
93void
94init_mmu(void)
95{
96 unsigned long mmfr0;
97 paddr_t start;
98 paddr_t end;
99
100 /*
101 * Set TTBR1 bit range for two slots, one for the kernel and one for the
102 * message passing window.
103 */
104 if (__PAGESIZE > 0x1000) { //64K page
105 /*
106 * 64K page size.
107 * T1SZ, bits [21:16], The region size is 2^(64-T1SZ) bytes, 42 bits per
108 * slot, for a total of 43 bits to accommodate the two slots.
109 */
110 tcr_el1 &= ~(0x3f |(0x3ull << 30) | (0x3ull << 14));
111 tcr_el1 |= AARCH64_TCR_EL1_T0SZ(42)
112 | AARCH64_TCR_EL1_T1SZ(43)
113 | AARCH64_TCR_EL1_TG1_64K
114 | AARCH64_TCR_EL1_TG0_64K;
115 } else {
116 /*
117 * default 4K page size.
118 * T1SZ, bits [21:16], The region size is 2^(64-T1SZ) bytes, 39 bits per
119 * slot, for a total of 40 bits to accommodate the two slots.
120 */
121 tcr_el1 |= AARCH64_TCR_EL1_T1SZ(40);
122 }
123
124 /*
125 * Set TCR_EL1 ASID/PA sizes and shareability
126 */
127 __asm__ __volatile__("mrs %0, id_aa64mmfr0_el1" : "=r"(mmfr0));
128 if (MMFR0_ASIDBITS(mmfr0)) {
129 tcr_el1 |= AARCH64_TCR_EL1_AS;
130 }
131 tcr_el1 |= AARCH64_TCR_EL1_IPS(MMFR0_PARANGE(mmfr0));
132
133 /*
134 * Because the cpu reads/writes and page table walk mechanism
135 * are separate "observers" in the ARMv8 memory architecture,
136 * we need to always mark page tables as inner-shareable,
137 * even if there's only one cpu.
138 * Mark page tables as inner-shareable
139 */
140 tcr_el1 |= AARCH64_TCR_EL1_SH1_ISH | AARCH64_TCR_EL1_SH0_ISH;
141
142 if (debug_flag) {
143 static int par[] = { 32, 36, 40, 42, 44, 48, 52 };
144
145 kprintf("MMU: %d-bit ASID %d-bit PA TCR_EL1=%v\n",
146 MMFR0_ASIDBITS(mmfr0) ? 16 : 8,
147 par[MMFR0_PARANGE(mmfr0)],
148 tcr_el1);
149 }
150
151 /*
152 * Set up initial page tables etc.
153 */
154 aarch64_map_init();
155
156 /*
157 * Create identity mapping for startup code/data
158 */
159 start = TRUNCPG(full_image_paddr);
160 end = ROUNDPG(full_image_paddr + shdr->startup_size);
161 aarch64_map(start, start, end-start, PROT_READ|PROT_WRITE|PROT_EXEC);
162}
