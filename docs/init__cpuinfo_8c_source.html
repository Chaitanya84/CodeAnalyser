46/*
47 * Read cache size id register for this level and cache type
48 */
49static inline void
50cache_sizes(unsigned csel, unsigned *nsets, unsigned *assoc, unsigned *lsize)
51{
52 aa64_sr_wr32(csselr_el1, csel);
53 isb();
54 unsigned const ccsidr = aa64_sr_rd32(ccsidr_el1);
55 *nsets = ((ccsidr >> 13) & 0x7fff) + 1;
56 *assoc = ((ccsidr >> 3) & 0x3ff) + 1;
57 *lsize = 1 << ((ccsidr & 7) + 4);
58}

60static void
61init_cache(struct cpuinfo_entry *cpu, unsigned cpunum)
62{
63 unsigned clidr;
64 unsigned nsets;
65 unsigned lsize;
66 unsigned assoc;
67
68 /*
69 * Cache maintenance instructions operate on all levels so we only
70 * add cacheattr information for the L1 cache(s).
71 */
72 clidr = aa64_sr_rd32(clidr_el1);
73 if (clidr & 4) {
74 cache_sizes(0, &nsets, &assoc, &lsize);
75 cpu->data_cache = add_cache_ways(cpu->data_cache,
76 CACHE_FLAG_UNIFIED | cache_snooped,
77 lsize, nsets * assoc, assoc,
78 &cache_armv8_dcache);
79 } else {
80 if (clidr & 1) {
81 cache_sizes(1, &nsets, &assoc, &lsize);
82 cpu->ins_cache = add_cache_ways(cpu->ins_cache,
83 CACHE_FLAG_INSTR,
84 lsize, nsets * assoc, assoc,
85 &cache_armv8_icache);
86 }
87 if (clidr & 2) {
88 cache_sizes(0, &nsets, &assoc, &lsize);
89 cpu->data_cache = add_cache_ways(cpu->data_cache,
90 CACHE_FLAG_DATA | cache_snooped,
91 lsize, nsets * assoc, assoc,
92 &cache_armv8_dcache);
93 }
94 }
95
96 if (debug_flag) {
97 int i;
98 unsigned ctype;
99 unsigned ctr;
100 static const char *const l1ip[] = {
101 "unknown", "AIVIVT", "VIPT", "PIPT"
102 };
103
104 ctr = aa64_sr_rd32(dczid_el0);
105 if ((ctr & (1u << 4)) == 0) {
106 kprintf("cpu%d: DCZID_EL0=0x%x (%d bytes)\n",
107 cpunum, ctr, 1 << ((ctr & 0xf) + 2));
108 }
109
110 ctr = aa64_sr_rd32(ctr_el0);
111 kprintf("cpu%d: CWG=%d ERG=%d Dminline=%d Iminline=%d %s\n",
112 cpunum,
113 (ctr >> 24) & 0xf,
114 (ctr >> 20) & 0xf,
115 (ctr >> 16) & 0xf,
116 ctr & 0xf,
117 l1ip[(ctr >> 14) & 3]);
118
119 kprintf("cpu%d: CLIDR=%w LoUU=%d LoC=%d LoUIS=%d\n",
120 cpunum,
121 clidr,
122 (clidr >> 27) & 7,
123 (clidr >> 24) & 7,
124 (clidr >> 21) & 7);
125
126 for (i = 0, ctype = clidr; i < 7; i++, ctype >>= 3) {
127 unsigned t = (ctype & 7);
128
129 if (t == 0 || t > 4) {
130 /*
131 * Cache not present or has a bogus value
132 */
133 break;
134 }
135
136 if (t & 4) {
137 cache_sizes((i << 1), &nsets, &assoc, &lsize);
138 kprintf("cpu%d: L%d Unified %dK linesz=%d set/way=%d/%d\n",
139 cpunum, i+1,
140 lsize * nsets * assoc / 1024,
141 lsize, nsets, assoc);
142 } else {
143 /*
144 * Might have separate I/D caches at each level
145 */
146 if (t & 1) {
147 cache_sizes((i << 1) | 1, &nsets, &assoc, &lsize);
148 kprintf("cpu%d: L%d Icache %dK linesz=%d set/way=%d/%d\n",
149 cpunum, i+1,
150 lsize * nsets * assoc / 1024,
151 lsize, nsets, assoc);
152 }
153 if (t & 2) {
154 cache_sizes((i << 1), &nsets, &assoc, &lsize);
155 kprintf("cpu%d: L%d Dcache %dK linesz=%d set/way=%d/%d\n",
156 cpunum, i+1,
157 lsize * nsets * assoc / 1024,
158 lsize, nsets, assoc);
159 }
160 }
161 }
162 }

165void
166init_one_cpuinfo(unsigned cpunum)
167{
168 struct cpuinfo_entry *cpu;
169 unsigned cpuid;
170 unsigned long mpidr;
171 const char *name;
172 unsigned long pfr;
173 unsigned long isar0;
174
175
176 cpu = &lsp.cpuinfo.p[cpunum];
177
178 /*
179 * Get the CPUID and processor name
180 */
181 cpuid = aa64_sr_rd32(midr_el1);
182 cpu->cpu = cpuid;
183 struct aarch64_cpuid *const id = aarch64_cpuid_find();
184 name = id ? id->name : "Unknown";
185 cpu->name = add_string(name);
186
187 mpidr = aa64_sr_rd64(mpidr_el1);
188 cpu->smp_hwcoreid = mpidr;
189
190 if (cpu_freq != 0) {
191 cpu->speed = cpu_freq / 1000000;
192 } else {
193 cpu->speed = aarch64_cpuspeed();
194 }
195
196 if (debug_flag) {
197 kprintf("cpu%d: MPIDR=%v\n", cpunum, mpidr);
198 if (CPUID_IS_ARM(cpuid)) {
199 kprintf("cpu%d: MIDR=%w %s r%dp%d\n",
200 cpunum, cpuid, name,
201 ARM_VARIANT(cpuid), ARM_REVISION(cpuid));
202 } else {
203 kprintf("cpu%d: MIDR=%w %s\n", cpunum, cpuid, name);
204 }
205 }
206
207 cpu->flags = 0;
208 if (shdr->flags1 & STARTUP_HDR_FLAGS1_VIRTUAL) {
209 cpu->flags |= CPU_FLAG_MMU;
210 }
211 if (lsp.syspage.p->num_cpu > 1) {
212 cpu->flags |= AARCH64_CPU_FLAG_SMP;
213 }
214
215 /*
216 * Detect if we have FP/SIMD.
217 * Both share the same register file so set CPU_FLAG_FPU.
218 */
219 pfr = aa64_sr_rd64(id_aa64pfr0_el1);
220 if (AA64PFR0_SIMD(pfr) || AA64PFR0_FP(pfr)) {
221 cpu->flags |= CPU_FLAG_FPU;
222 if (AA64PFR0_SIMD(pfr)) {
223 cpu->flags |= AARCH64_CPU_FLAG_SIMD;
224 }
225 }
226
227 /*
228 * Detect if we have SVE and the user enables it.
229 */
230 if(sve_requested != 0) {
231 pfr = aa64_sr_rd64(id_aa64pfr0_el1);
232 if (AA64PFR0_SVE(pfr)) {
233 cpu->flags |= AARCH64_CPU_FLAG_SVE;
234 } else {
235 kprintf("WARN: SVE requested, but not supported by CPU\n");
236 }
237 }
238
239 /*
240 * Enable EL0 access to CNTVCT_EL0 for ClockCycles()
241 */
242 aa64_sr_wr32(cntkctl_el1, 1 << 1);
243
244 /*
245 * Set legacy ARMv7 flags for Aarch32 binary compatibility
246 */
247 if (cpu->flags & AARCH64_CPU_FLAG_SMP) {
248 cpu->flags |= AARCH32_CPU_FLAG_V7_MP;
249 }
250 if (cpu->flags & CPU_FLAG_FPU) {
251 cpu->flags |= AARCH32_CPU_FLAG_VFP_D32;
252 }
253 cpu->flags |= AARCH32_CPU_FLAG_V6|AARCH32_CPU_FLAG_V7|AARCH32_CPU_FLAG_IDIV;
254
255 /*
256 * Enable Spectre V2 fix
257 */
258 if (spectre_v2_active > 0) {
259 if ((id != NULL) && ((id->midr == cpuid_a57.midr) || (id->midr == cpuid_a72.midr))) {
260 cpu->flags |= AARCH64_CPU_SPECTRE_V2_MMU;
261 } else {
262 cpu->flags |= AARCH64_CPU_SPECTRE_V2_FIX;
263 }
264 }
265
266 /*
267 * Enable SSBS support
268 */
269 if (ssbs_active > 0) {
270 cpu->flags |= AARCH64_CPU_SSBS;
271 }
272
273
274 /* Detect if we have LSE atomic instructions */
275 isar0 = aa64_sr_rd64(id_aa64isar0_el1);
276 if (AA64ISAR0_LSE(isar0)) {
277 cpu->flags |= AARCH64_CPU_FLAG_LSE;
278 }
279 if (AA64ISAR0_AES(isar0)) {
280 cpu->flags |= AARCH64_CPU_FLAG_AES;
281 }
282 if (AA64ISAR0_PMULL(isar0)) {
283 cpu->flags |= AARCH64_CPU_FLAG_PMULL | AARCH64_CPU_FLAG_AES;
284 }
285 if (AA64ISAR0_SHA1(isar0)) {
286 cpu->flags |= AARCH64_CPU_FLAG_SHA1;
287 }
288 if (AA64ISAR0_SHA256(isar0)) {
289 cpu->flags |= AARCH64_CPU_FLAG_SHA256;
290 }
291 if (AA64ISAR0_SHA512(isar0)) {
292 cpu->flags |= AARCH64_CPU_FLAG_SHA512 | AARCH64_CPU_FLAG_SHA256;
293 }
294 if (AA64ISAR0_SHA3(isar0)) {
295 cpu->flags |= AARCH64_CPU_FLAG_SHA3;
296 }
297 if (AA64ISAR0_RNDR(isar0)) {
298 cpu->flags |= AARCH64_CPU_FLAG_RNDR;
299 }
300#ifdef AARCH64_CPU_PAUTH
301 /* Detect if we have pointer authentication instructions */
302 if(pauth_active != 0) {
303 unsigned long isar1 = aa64_sr_rd64(id_aa64isar1_el1) & (0xfful << 4);
304 if (isar1 != 0) {
305 cpu->flags |= AARCH64_CPU_PAUTH;
306 }
307 }
308#endif
309
310 unsigned ctr = aa64_sr_rd32(ctr_el0);
311 if ((ctr & (1U << 29)) != 0U) {
312 cpu->flags |= AARCH64_CPU_FLAG_ICACHE_COHERENT;
313 }
314
315 /*
316 * Set up cache callouts
317 */
318 init_cache(cpu, cpunum);
319
320 /*
321 * Set up GIC CPU interface for this cpu
322 */
323 if (gic_cpu_init) {
324 gic_cpu_init(cpunum);
325 }
326
327 /*
328 * Perform any cpu-specific cpu information setup
329 */
330 if (id && id->cpuinfo) {
331 id->cpuinfo(cpunum, cpu);
332 }
333
334 /*
335 * zero CPACR_EL1, which will disable exception for EVT, FPU, and SVE.
336 */
337 aa64_sr_wr32(cpacr_el1, 0);
338
339 /* Dump ID registers */
340 uint64_t * const dump_space =
341 &lsp.cpu.aarch64_idreg_dump.p->idreg[cpunum*AARCH64_IDDUMP_ENTRIES];

435void
436init_cpuinfo()
437{
438 struct cpuinfo_entry *cpu;
439 unsigned num;
440 unsigned i;
441
442 num = lsp.syspage.p->num_cpu;
443
444 cpu = set_syspage_section(&lsp.cpuinfo, sizeof(*lsp.cpuinfo.p) * num);
445 for (i = 0; i < num; i++) {
446 cpu[i].ins_cache = system_icache_idx;
447 cpu[i].data_cache = system_dcache_idx;
448 }
449
450 // 6 bits worth of ID reg space, minus C0, so 56 entries
451 set_syspage_section(&lsp.cpu.aarch64_idreg_dump,
452 (unsigned)(num * sizeof(_Uint64t) * AARCH64_IDDUMP_ENTRIES));
453
454 init_one_cpuinfo(0);
455}
