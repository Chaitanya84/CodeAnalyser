56static void
57ws_init(void) {
58 // We can be fairly sure there's some ram following :-).
59 ws_early_next = ROUND(full_ram_paddr + shdr->ram_size, sizeof(uint64_t));
60
61 avoid_list = MAKE_1TO1_PTR(ws_early_next);
62 avoid_list[0].start = ws_early_next;
63 ws_early_next += ROUND(MAX_AVOID_LIST_SIZE, sizeof(uint64_t));
64 avoid_list[0].end = ws_early_next;
65 avoid_list[1].end = 0;
66 avoid_list_size = sizeof(avoid_list[0])*2;
67
68 ram_list = ws_alloc(MAX_RAM_LIST_SIZE);
69 ram_list->size = 0;
70 ram_list_size = sizeof(struct ram_entry);
71
72 //Avoid using memory covered by startup
73 avoid_ram(full_ram_paddr, shdr->startup_size);
74
75#ifdef CPU_MEM_AVOID_START
76 //Avoid using memory that might have something interesting
77 avoid_ram(CPU_MEM_AVOID_START, CPU_MEM_AVOID_SIZE);
78#endif
79}

82static int
83add_mem(paddr_t start, paddr_t size) {
84 paddr64_t e, s;
85 struct ram_entry *r;
86 struct ram_entry *ram = ram_list;
87
88//kprintf( "add_mem(%P,%P)\n", start, size );
89 if(size > 0) {
90 for(r = ram; r->size != 0; r++) {
91 if(r->addr > start) break;
92 }
93 if(ram_list_size > (MAX_RAM_LIST_SIZE - sizeof(*r))) {
94 crash("RAM entry array overflow\n");
95 }
96 memmove(&r[1], r, ram_list_size - PTR_DIFF(r, ram));
97 ram_list_size += sizeof(*r);
98 r->addr = start;
99 r->size = size;
100 for(r = ram+1; r->size != 0; r++) {
101 e = r[-1].addr + r[-1].size;
102 s = r->addr;
103 if(s <= e) {
104 if((s + r->size) > e) e = s + r->size;
105 memmove(r, &r[1], ram_list_size - PTR_DIFF(&r[1], ram));
106 ram_list_size -= sizeof(*r);
107 --r;
108 r->size = e - r->addr;
109 }
110 }
111 }
112 return 0;
113}

115static int
116fix_size(paddr64_t *sp, paddr64_t *ep, paddr_t as, paddr_t ae) {
117 paddr64_t s = *sp;
118 paddr64_t e = *ep;
119
120 if((s >= ae) || (e <= as)) {
121 // Candidate mem has no overlap with the avoid range. All is good.
122 return 1;
123 }
124 if(s >= as) {
125 // candidate mem start within avoid range
126 if(e <= ae) {
127 // Candidate mem wholly contained in avoid range. All is lost.
128 return 0;
129 }
130 // Restrict mem start to be after the end of the avoid range
131 *sp = ae;
132 } else if(e < ae) {
133 // end of candidate mem is within avoid range
134 // Restrict the mem end to be the start of the avoid range (should be -1?)
135 *ep = as;
136 } else if((as-s) > (e-ae)) {
137 // avoid range is in the middle of the candidate mem
138 // and the larger piece is at the beginning
139 // Restrict the mem end to be the start of the avoid range (should be -1?)
140 *ep = as;
141 } else {
142 // avoid range is in the middle of the candidate mem
143 // and the larger piece is at the end
144 // Restrict the mem start to be after the end of the avoid range
145 *sp = ae;
146 }
147 return 1;
148}

150paddr_t
151find_top_ram_aligned_limit(paddr_t size, unsigned align, unsigned pagesize, paddr_t limit) {
152 struct ram_entry *r;
153 struct avoid_entry *a;
154 paddr64_t s;
155 paddr64_t e;
156 paddr_t chk;
157 paddr_t top = 0;
158
159 if (limit < size) {
160 return NULL_PADDR;
161 }
162
163 size = ROUND(size, pagesize);
164 for(r = ram_list; r->size != 0; r++) {
165 s = r->addr;
166 e = s + r->size;
167 for(a = avoid_list; a->end != 0; ++a) {
168 if(!fix_size(&s, &e, a->start, a->end)) goto next_one;
169 }
170 // Find an address at the end of the ram block. If align is given,
171 // truncate to the alignment value.
172 if ( (e-s) >= size ) {
173 chk = align ? TRUNC(e-size, align) : e-size;
174 if (chk >= s) {
175 if(chk > top) {
176 const paddr_t ptop = top;
177 top = chk;
178 if ((top + size) >= limit) {
179 top = limit - size;
180 if (top < s) {
181 top = ptop;
182 }
183 break;
184 }
185 }
186 }
187 }
188next_one: ;
189 }
190 if(top == 0) top = NULL_PADDR;
191 return top;
192}

203paddr_t
204find_ram_in_range(paddr_t range_start, paddr_t range_end, size_t size, unsigned align, unsigned colour, unsigned mask) {
205 struct ram_entry *ram = ram_list;
206 struct ram_entry *r;
207 paddr64_t s;
208 paddr64_t e;
209 paddr_t addr;
210 struct avoid_entry *a;
211
212 for(r = ram; r->size != 0; r++) {
213 int avoided=0;
214 s = r->addr;
215 e = s + r->size;
216// kprintf("%s: Candidate s:0x%L e:0x%L size:0x%L rqsize:%x\n", __FUNCTION__, s, e, r->size, size);
217 for(a = avoid_list; a->end != 0; ++a) {
218 if(!fix_size(&s, &e, a->start, a->end)) {
219 avoided=1;
220 break;
221 }
222 }
223 if (avoided) {
224 // No point in continuing to check this one
225 continue;
226 }
227
228 addr = ROUND(s, align);
229
230 if((addr & mask) != colour) {
231 unsigned incr;
232
233 incr = ((mask - 1) | mask) + 1;
234 addr = ((addr + incr) & ~mask) | colour;
235 }
236
237 if (range_start) {
238 if (addr < (paddr_t) range_start) {
239 addr = range_start;
240 }
241 }
242 if (range_end) {
243 if (e > (paddr_t) range_end) {
244 e = range_end;
245 }
246 }
247
248 if(addr >= e) {
249 // Adjusted out of contention by alignment or ranges
250 continue;
251 }
252
253// kprintf("%s: Adjusted addr:0x%L e:0x%L size:0x%L\n", __FUNCTION__, addr, e, e-addr);
254 if(size <= (e-addr)) {
255 // This one is big enough
256 return addr;
257 }
258 }
259 return NULL_PADDR;
260}

262PADDR_T
263find_ram(size_t size, unsigned align, unsigned colour, unsigned mask) {
264 paddr_t result;
265
266 // we need to restrict this to PADDR_T range
267 result = find_ram_in_range(0, (PADDR_T)~(paddr_t)0, size, align, colour, mask);
268 if (result == NULL_PADDR) {
269 return NULL_PADDR_STARTUP;
270 }
271 return (PADDR_T) result;
272}

274paddr_t
275alloc_ram(paddr_t addr, paddr_t size, unsigned align) {
276 struct ram_entry *ram = ram_list;
277 struct ram_entry *r;
278 paddr64_t s, e, end;
279
280//kprintf("alloc_ram(%P,%P,%x) => ", addr, size, align);
281 if(addr == NULL_PADDR) {
282 addr = find_ram(size, align, 0, 0);
283 if(addr == NULL_PADDR_STARTUP) {
284//kprintf("NULL_PADDR\n");
285 return NULL_PADDR;
286 }
287 if(secure_system) {
288 // On a secure system, we want to zero everything
289 memset(MAKE_1TO1_PTR((paddr32_t)addr), 0, size);
290 }
291 }
292//kprintf("%P\n", addr);
293 end = addr + size;
294 for(r = ram; r->size != 0; r++) {
295 s = r->addr;
296 e = s + r->size;
297 if((addr <= s && end > s)
298 || (addr >= s && addr < e)
299 || (end >= s && end < e)) {
300 if(end > e) e=end;
301 if(addr < s) s=addr;
302 if(addr == s && end == e) {
303 memmove(r, &r[1], ram_list_size - PTR_DIFF(&r[1], ram));
304 ram_list_size -= sizeof(*r);
305 --r;
306 } else if(addr == s) {
307 r->size = e - end;
308 r->addr = end;
309 } else {
310 r->size = addr - s;
311 if(e != end) {
312 add_mem(end, e - end);
313 }
314 }
315 }
316 }
317
318 //
319 // If add_sysram() has been called, we need to adjust the start/end
320 // fields
321 //
322
323 if(sysram_added) {
324 unsigned off;
325 struct asinfo_entry *as;
326
327 --end;
328 off = AS_NULL_OFF;
329 for( ;; ) {
330 off = as_find_containing(off, addr, end, "sysram");
331 if(off == AS_NULL_OFF) break;
332 as = as_off2info(off);
333 s = as->start;
334 e = as->end;
335 if(end > e) e=end;
336 if(addr < s) s=addr;
337 if(addr == s && end == e) {
338 crash("Can't remove whole sysram entry.\n");
339 } else if(addr == s) {
340 as->start = end + 1;
341 } else {
342 as->end = addr - 1;
343 if(e != end) {
344 as_add(end+1, e, AS_ATTR_RAM, "sysram", as->owner);
345 }
346 }
347 }
348 }
349 return addr;
350}

363void
364add_ram(paddr_t start, paddr_t size) {
365 struct watch_entry *curr;
366 unsigned mem;
367
368 if(lsp.asinfo.size == 0) {
369 mem = as_default();
370 as_add(0, GIG(4)-1, 0, "below4G", mem);
371 init_asinfo(mem);
372 }
373 as_add_containing(start, start + size - 1, AS_ATTR_RAM, "ram", "memory");
374
375 add_mem(start, size);
376
377 //Avoid using memory covered by image
378 alloc_ram(full_ram_paddr+shdr->startup_size,
379 shdr->ram_size-shdr->startup_size, 1);
380
381 //Avoid using memory reserved for the CPU (exception table)
382 #if defined(CPU_MEM_RESERVE_SIZE) && (CPU_MEM_RESERVE_SIZE > 0)
383 alloc_ram(CPU_MEM_RESERVE_START, CPU_MEM_RESERVE_SIZE, 1);
384 #endif
385
386 //Get more space for the temporary system page we're building.
387 if(temp_syspage == NULL_PADDR_STARTUP) {
388 temp_syspage = find_ram(TEMP_SYSPAGE_SIZE, __PAGESIZE, 0, 0);
389 if(temp_syspage != NULL_PADDR_STARTUP) {
390 avoid_ram(temp_syspage, TEMP_SYSPAGE_SIZE);
391 reloc_syspage_memory(MAKE_1TO1_PTR(temp_syspage), TEMP_SYSPAGE_SIZE);
392 }
393 }
394
395 for(curr = watch_list; curr != NULL; curr = curr->next) {
396 curr->watcher(start, size);
397 }
398}

400void
401avoid_ram(PADDR_T start, size_t size) {
402 struct avoid_entry *avoid;
403
404 if(avoid_list == NULL) {
405 ws_init();
406 }
407 avoid = (void *)((uint8_t *)avoid_list + avoid_list_size);
408 if((avoid_list_size >= (sizeof(struct avoid_entry))) && (avoid[-2].end == start)) {
409 avoid[-2].end += size;
410 } else {
411 avoid[-1].start = start;
412 avoid[-1].end = start + size;
413 avoid[0].end = 0;
414 avoid_list_size += sizeof(*avoid);
415 }
416 if(avoid_list_size > MAX_AVOID_LIST_SIZE) {
417 crash("Avoid RAM entry array overflow\n");
418 }
419}

421//
422// Add all the "sysram" entries to the asinfo section. Note that we haven't
423// allocated the system page storage yet, so we're going to have adjust
424// the start/end fields when that happens.
425//
426void
427add_sysram() {
428 struct ram_entry *ram;
429 struct watch_entry *curr;
430
431 // Tell all the watchers that there won't be any more RAM added
432 for(curr = watch_list; curr != NULL; curr = curr->next) {
433 curr->watcher(0, 0);
434 }
435
436 for(ram = ram_list; ram->size != 0; ++ram) {
437 as_add_containing(ram->addr, ram->addr+ram->size-1,
438 AS_ATTR_RAM, "sysram", "ram");
439 }
440 sysram_added = 1;
441}

444void *
445ws_alloc(size_t size) {
446 PADDR_T paddr;
447
448 if(avoid_list == NULL) {
449 ws_init();
450 }
451
452 size = ROUND(size, sizeof(uint64_t));
453 if(ram_list_size <= sizeof(struct ram_entry)) {
454 // no add_ram()'s have been done yet.
455 paddr = ws_early_next;
456 ws_early_next += size;
457 } else {
458 // we've been told where memory is.
459 paddr = find_ram(size, sizeof(uint64_t), 0, 0);
460 if(paddr == NULL_PADDR_STARTUP) {
461 return(NULL);
462 }
463 }
464 avoid_ram(paddr, size);
465 return(MAKE_1TO1_PTR(paddr));
466}

480/*
481 * reserve_ram:
482 * Alloc some ram and create a "reserved" asinfo for it.
483 *
484 * Parameters:
485 * size Size in bytes of the ram to allocate.
486 * align Alignment requirement for the ram or 0 for no alignment.
487 * pagesize System page size. Block will be rounded up to this size.
488 *
489 * Return Value:
490 * none
491 */
492void
493reserve_ram(unsigned size, unsigned align, const char * name, unsigned pagesize, const unsigned low_region) {
494 paddr_t reserved = NULL_PADDR;
495
496 if (size == 0) {
497 return;
498 }
499
500 if(low_region == 0) {
501 reserved = find_top_ram_aligned(size, align ? align : pagesize, pagesize);
502 if(reserved == NULL_PADDR) {
503 crash("Could not reserve 0x%x bytes from top of memory.\n", size);
504 }
505 }
506
507 reserved = alloc_ram(reserved, size, align ? align : pagesize);
508 if(reserved == NULL_PADDR) {
509 crash("Could not reserve 0x%x bytes of memory.\n", size);
510 }
511
512 if((name == NULL) || (name[0] == '\0')) {
513 name = "reserved";
514 }
515 as_add_containing(reserved, (reserved + size - 1), AS_ATTR_RAM, name, "ram");
516}
