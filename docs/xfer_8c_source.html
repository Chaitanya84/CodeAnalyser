46int32_t dw_spi_select_slave(const dw_spi_t *const spi, const int slv)
47{
48 /* check if spi busy */
49 if (dw_spi_busy(spi)) {
50 return -1;
51 }
52
53 dw_write32(spi, DW_SPI_SER, (1UL << slv));
54 return 0;
55}

66int dw_prepare_for_transfer(dw_spi_t *spi, const spi_cfg_t *const cfg, const uint32_t tnbytes, const uint32_t rnbytes)
67{
68 static const uint32_t dscales[] = {0, 0, 1, 2, 2};
69
70 /* Data bits per word */
71 const uint32_t nbits = cfg->mode & SPI_MODE_WORD_WIDTH_MASK;
72 if ((nbits <= SPI_MIN_WORD_WIDTH) || (nbits > SPI_MAX_WORD_WIDTH)) {
73 return EINVAL;
74 }
75
76 /* Spi must be disablled to change configuration */
77 dw_spi_enable(spi, DW_SPI_DISABLE);
78
79 /* Data bytes per word */
80 spi->dlen = (nbits + DW_NBIT_ROUNDER) / 8;
81 spi->dscale = dscales[spi->dlen];
82 if (spi->dlen == 3) {
83 spi->dlen = 4;
84 }
85
86 /* Transation data length - word */
87 const uint32_t nbytes = max(tnbytes, rnbytes);
88 spi->xlen = nbytes >> spi->dscale;
89 if ((spi->xlen <= 0) || (nbytes != (spi->xlen << spi->dscale))) {
90 return EINVAL;
91 }
92
93 /* Check clock rate */
94 if (cfg->clock_rate == 0) {
95 spi_slogf(_SLOG_ERROR, "%s: invalid clock rate %u", __func__, cfg->clock_rate);
96 return EINVAL;
97 }
98
99 spi->cr0 = (DW_SPI_CTRLR0_FRF_MOTO_SPI << DW_SPI_CTRLR0_FRF_OFFSET) | (DW_SPI_CTRLR0_TMOD_TR << DW_SPI_CTRLR0_TMOD_OFFSET);
100
101 /* set nbits value */
102 spi->cr0 |= DW_PSSI_CTRLR0_DFS_CONFIG(spi->dlen);
103
104 /* Update cr0 value */
105 if ((cfg->mode & SPI_MODE_CPOL_1) != 0) {
106 spi->cr0 |= DW_SPI_CTRLR0_SCPOL; /* Clock Polarity */
107 }
108
109 if ((cfg->mode & SPI_MODE_CPHA_1) != 0) {
110 spi->cr0 |= DW_SPI_CTRLR0_SCPHA; /* Clock Phase */
111 }
112
113 if (spi->loopback) {
114 spi->cr0 |= DW_PSSI_CTRLR0_SRL; /* Loop-Back Mode */
115 }
116
117 if (spi->sste) {
118 spi->cr0 |= DW_SPI_SSTE; /* Slave select Toggle Enable */
119 }
120
121 const uint32_t version = dw_read32(spi, DW_SPI_VERSION);
122 if (version >= DW_HSSI_102A) {
123 spi->cr0 |= (1 << DW_SPI_CTRLR0_IS_MST_OFFSET); /* enable as master */
124 }
125
126 dw_write32(spi, DW_SPI_BAUDR, (uint32_t)(spi->bus->input_clk / cfg->clock_rate));
127 dw_write32(spi, DW_SPI_CTRLR0, spi->cr0);
128 dw_write32(spi, DW_SPI_CTRLR1, 0);
129
130 /* Calculate the transaction time, minimal 1 us */
131 spi->xtime_us = max(1, (uint64_t)nbits * 1000 * 1000 * spi->xlen / cfg->clock_rate);
132
133 /* Reset RX and TX length */
134 spi->rlen = 0;
135 spi->tlen = 0;
136
137 return EOK;
138}

150int dw_xfer(void *const hdl, spi_dev_t *const spi_dev, uint8_t *const buf, const uint32_t tnbytes, const uint32_t rnbytes)
151{
152 dw_spi_t *spi = hdl;
153 const spi_cfg_t *const cfg = &spi_dev->devinfo.cfg;
154 int status = EOK;
155
156#ifdef DW_DMA
157 spi->dma_active = false;
158#endif
159
160 /* Module busy */
161 if (dw_spi_busy(spi) == true) {
162 spi_slogf(_SLOG_ERROR, "%s: Controller is busy", __func__);
163 return EBUSY;
164 }
165
166 spi->pbuf = buf;
167
168 /* Configure SPI device */
169 status = dw_prepare_for_transfer(spi, cfg, tnbytes, rnbytes);
170 if (status != EOK) {
171 spi_slogf(_SLOG_ERROR, "%s: dw_prepare_for_transfer failed", __func__);
172 return status;
173 }
174
175 /* Disable DMA transaction on SPI when non-DMA and DMA co-exits */
176 dw_write32(spi, DW_SPI_DMACR, 0);
177
178 /* Enable FIFO Interrupts and SPI */
179 dw_device_reset(spi);
180
181 /* Prefill transmit FIFO */
182 spi_slogf(_SLOG_DEBUG1, "%s: Starting prefill transmit FIFO", __func__);
183
184 const uint32_t len = min(spi->xlen, spi->fifo_len / SPI_FIFO_LEN_DIV);
185 dw_write_fifo(spi, len);
186 spi->tlen += len;
187 const uint32_t thresh = len;
188
189 if (spi->tlen < spi->xlen) {
190 dw_write32(spi, DW_SPI_IMR, DW_SPI_INT_MASK);
191 /* When the number of receive FIFO entries is greater than
192 or equal to this value + 1, the receive FIFO full interrupt
193 is triggered.*/
194 dw_write32(spi, DW_SPI_RXFTLR, (thresh - 1U));
195 /* When the number of transmit FIFO entries is less than or
196 equal to this value, the transmit FIFO empty interrupt is
197 triggered. */
198 dw_write32(spi, DW_SPI_TXFTLR, thresh);
199 } else {
200 dw_write32(spi, DW_SPI_IMR, DW_SPI_INT_RXUI
201 | DW_SPI_INT_RXOI | DW_SPI_INT_RXFI);
202 dw_write32(spi, DW_SPI_RXFTLR, (thresh - 1U));
203 }
204
205 dw_spi_select_slave(spi, spi_dev->devinfo.devno);
206
207 /* Wait for exchange to finish */
208 status = dw_wait(spi);
209 if (status != EOK) {
210 spi_slogf(_SLOG_ERROR, "%s: xfer timeout: xlen = %u, tlen = %u, rlen = %u xtime = %lu us",
211 __func__, spi->xlen, spi->tlen, spi->rlen, spi->xtime_us);
212 }
213
214 dw_spi_enable(spi, DW_SPI_DISABLE);
215
216 return status;
217}
