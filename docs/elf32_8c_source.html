29static Elf32_Phdr *
30elf32_phdr(Elf32_Ehdr *hdr, paddr32_t addr)
31{
32 Elf32_Ehdr *p;
33
34 p = startup_memory_map(sizeof(*hdr), addr, PROT_READ);
35 *hdr = *p;
36 startup_memory_unmap(p);
37
38 if (memcmp(hdr->e_ident, ELFMAG, SELFMAG)
39 || hdr->e_ident [EI_DATA] != ELFDATANATIVE
40 || hdr->e_machine != EM_NATIVE
41 || hdr->e_phnum == 0
42 || hdr->e_phentsize != sizeof(Elf32_Phdr))
43 return(NULL);
44
45 return startup_memory_map(hdr->e_phnum * sizeof(Elf32_Phdr),
46 addr + hdr->e_phoff, PROT_READ|PROT_WRITE);
47}

52static void
53load_elf32mmu(paddr32_t addr, Elf32_Ehdr *hdr, Elf32_Phdr *phdr)
54{
55 int i;
56
57 for(i = 0; i < hdr->e_phnum; ++i, ++phdr) {
58 switch(phdr->p_type) {
59 case PT_LOAD:
60 {
61 size_t memsz = phdr->p_memsz + (phdr->p_vaddr & PAGEBITS);
62 size_t filesz = phdr->p_filesz + (phdr->p_vaddr & PAGEBITS);
63 uintptr_t vaddr = phdr->p_vaddr & ~PAGEBITS;
64 paddr32_t paddr = phdr->p_paddr & ~PAGEBITS;
65 paddr32_t daddr;
66
67 if((boot_vaddr_base == 0u) || (boot_vaddr_base > vaddr)){
68 boot_vaddr_base = vaddr;
69 }
70 if(boot_vaddr_end < (vaddr + memsz)){
71 boot_vaddr_end = vaddr + memsz;
72 }
73
74 if((phdr->p_flags & PF_W) == 0 && memsz == filesz) {
75 filesz = memsz = ROUNDPG(filesz);
76 }
77 if(phdr->p_paddr != 0) {
78 if (phdr->p_memsz == phdr->p_filesz) {
79 // mkifs has padded out the data/bss section
80 filesz = memsz = ROUNDPG(filesz);
81 }
82 elf_map(vaddr, paddr, filesz & ~PAGEBITS, phdr->p_flags);
83 memsz = ROUNDPG(memsz - (filesz & ~PAGEBITS));
84 if(memsz) {
85 daddr = calloc_ram(memsz, __PAGESIZE);
86 copy_memory(daddr, (paddr + (filesz & ~PAGEBITS)), filesz & PAGEBITS);
87 elf_map(vaddr + (filesz & ~PAGEBITS), daddr, memsz, phdr->p_flags);
88 }
89 } else {
90#ifndef BOOTSTRAPS_RUN_ONE_TO_ONE
91 /*
92 * We need to do different things depending on whether the bootstrap
93 * executables a run in a physical <-> virtual one to one mapping
94 * area (MIPS, SH, PPC) or if they use the normal virtual address
95 * mapping gear (X86, ARM). Basically, is startup or procnto
96 * enabling the MMU on the CPU.
97 */
98 #error BOOTSTRAPS_RUN_ONE_TO_ONE must be defined
99#endif
100#if BOOTSTRAPS_RUN_ONE_TO_ONE
101 daddr = alloc_ram(vaddr - shdr->paddr_bias, ROUNDPG(memsz), __PAGESIZE);
102 if(daddr != vaddr - shdr->paddr_bias) {
103 crash("Error: can not allocate RAM for proc in XIP.\n");
104 }
105 memmove((void *)phdr->p_vaddr,
106 MAKE_1TO1_PTR(addr + phdr->p_offset),
107 phdr->p_filesz);
108 memset((void *)(phdr->p_vaddr + phdr->p_filesz), 0,
109 (phdr->p_memsz - phdr->p_filesz));
110#else
111 daddr = calloc_ram(ROUNDPG(memsz), __PAGESIZE);
112 copy_memory(daddr, (addr + phdr->p_offset) - (phdr->p_vaddr & PAGEBITS), filesz);
113 elf_map(vaddr, daddr, memsz, phdr->p_flags);
114#endif
115 }
116 }
117 break;
118 }
119 }
120}

123uintptr_t
124load_elf32(paddr32_t addr)
125{
126 Elf32_Ehdr hdr;
127 Elf32_Phdr *phdr;
128 Elf32_Phdr *phdr_start;
129 int i;
130
131 if((phdr_start = elf32_phdr(&hdr, addr)) == NULL) {
132 return(~0L);
133 }
134
135 if(!(shdr->flags1 & STARTUP_HDR_FLAGS1_VIRTUAL)) {
136 phdr = phdr_start;
137 for(i = 0; i < hdr.e_phnum; ++i, ++phdr) {
138 switch(phdr->p_type) {
139 case PT_LOAD:
140 if(MAKE_1TO1_PTR(phdr->p_paddr) != (void *)(uintptr_t)phdr->p_vaddr) {
141 alloc_ram(phdr->p_vaddr - shdr->paddr_bias, phdr->p_memsz, 1);
142 memmove((void *)(uintptr_t)phdr->p_vaddr,
143 MAKE_1TO1_PTR(addr + phdr->p_offset),
144 phdr->p_filesz);
145 memset((void *)(uintptr_t)(phdr->p_vaddr+phdr->p_filesz), 0,
146 phdr->p_memsz - phdr->p_filesz);
147 }
148 break;
149 }
150 }
151#ifdef __X86__
152 } else if(load_elf32mmu_4m(addr, &hdr, phdr_start)) {
153 // Nothing to do
154#endif
155 } else {
156 load_elf32mmu(addr, &hdr, phdr_start);
157 }
158
159 startup_memory_unmap(phdr_start);
160
161 return hdr.e_entry;
162}
