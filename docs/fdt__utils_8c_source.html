25/*
26 * FDT "/chosen" has:
27 * stdout-path = "serial10:115200n8" for default debug console
28 * FDT "/aliases" has:
29 * serial10 = "/soc/serial@7d001000"
30 * FDT node "serial@7d001000" has:
31 * "interrupts = <0x00 0x79 0x04>" for revision 1.0 targets
32 * "interrupts = <0x00 0x78 0x04>" for revision 1.1 targets
33 */
34
35struct fdt_info
36{
37 void *base;
38 uint64_t size;
39};
40
41static int32_t map_fdt(struct asinfo_entry const * const as, char const * const name, void * const arg)
42{
43 int32_t ret = 0;
44 struct fdt_info * const fdt = (struct fdt_info *) arg;
45
46 (void)name;
47
48 fdt->size = (as->end - as->start + 1UL);
49 fdt->base = mmap64(NULL, (size_t)fdt->size, PROT_READ, MAP_SHARED | MAP_PHYS, NOFD, (off_t)as->start);
50 if (fdt->base == (void *)MAP_FAILED) {
51 fdt->base = NULL;
52 ret = 1;
53 }
54 else {
55 if (fdt_check_header(fdt->base) != 0) {
56 (void)munmap(fdt->base, (size_t)fdt->size);
57 fdt->base = NULL;
58 (void)slogf (_SLOG_SETCODE (_SLOGC_CHAR, 0), _SLOG_ERROR, "FDT header check failed");
59 ret = 1;
60 }
61 }
62 return ret;
63}

65/*
66 * Retrieve debug console interrupt number from FDT at runtime
67 */
68int32_t fdt_debug_console_irq_fix(DEV_PL011 *dev)
69{
70 struct fdt_info fdt = { .base = NULL, .size = 0UL };
71 int32_t ret = 0;
72
73 // Map the FDT.
74 if (walk_asinfo("fdt", map_fdt, (void *)&fdt) != 0) {
75 ret = -1;
76 }
77 else {
78 if (fdt.base == NULL) {
79 ret = -2;
80 }
81 else {
82 const int32_t chosen_offset = fdt_path_offset(fdt.base, "/chosen");
83 if (chosen_offset < 0) {
84 ret = -3;
85 }
86 else {
87 const char *stdout_path_str = fdt_getprop(fdt.base, chosen_offset, "stdout-path", NULL);
88 if (stdout_path_str == NULL) {
89 ret = -4;
90 }
91 else {
92 int32_t namelen = 0;
93 int32_t node_offset = 0;
94 int32_t prop_len = 0;
95 uint32_t irq = 0;
96 /* find default console serial from "stdout-path = "serial10:115200n8" */
97 const char *p = strchr(stdout_path_str, ':');
98 namelen = (p != NULL) ? (int32_t)(p - stdout_path_str) : (int32_t)strlen(stdout_path_str);
99 node_offset = fdt_path_offset_namelen(fdt.base, stdout_path_str, namelen);
100 if (node_offset < 0) {
101 ret = -5;
102 }
103 else {
104 struct fdt_property const * const fdt_prop = fdt_get_property(fdt.base, node_offset, "interrupts", &prop_len);
105 if (fdt_prop == NULL) {
106 ret = -6;
107 }
108 else {
109 if (prop_len != (int32_t)(3U * sizeof(fdt32_t))) {
110 ret = -7;
111 }
112 else {
113 // Extract the second cell of interrupts.
114 const fdt32_t * const data = (fdt32_t const *)fdt_prop->data;
115 irq = fdt32_to_cpu(data[1]) + 32U; /* FDT does not include interrupt 0~31 */
116 if (dev->intr != irq) {
117 (void)slogf (_SLOG_SETCODE (_SLOGC_CHAR, 0), _SLOG_NOTICE, "Override Serial console IRQ from %u to %u", dev->intr, irq);
118 dev->intr = irq;
119 }
120 }
121 }
122 }
123 }
124 }
125 (void)munmap(fdt.base, fdt.size);
126 }
127 }
128 return ret;
129}
