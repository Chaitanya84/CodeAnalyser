38struct syspage_entry *
39cpu_alloc_syspage_memory(PADDR_T *cpupagep, PADDR_T *syspagep, unsigned spsize)
40{
41 struct syspage_entry *sp = lsp.syspage.p;
42 struct system_private_entry *private;
43 unsigned size;
44 unsigned cpsize;
45 paddr_t syspage_paddr;
46
47 #define SP_OFFSET(field) PTR_DIFF(lsp.cpu.field.p, sp)
48 #define INIT_ENTRY(_cpu, _field) \
49 sp->un._cpu._field.entry_size = lsp.cpu._cpu##_##_field.size; \
50 sp->un._cpu._field.entry_off = SP_OFFSET(_cpu##_##_field)
51
52 spsize = ROUND(spsize, syspage_roundup);
53 cpsize = sp->num_cpu * sizeof(struct cpupage_entry);
54
55 /*
56 * Allocate the system page (and cpupage entries) and save it away.
57 * The system page must be 4K aligned in a virtual system
58 */
59 size = spsize + cpsize;
60
61 syspage_paddr = calloc_ram(size, lsp.system_private.p->pagesize);
62 if (syspage_paddr == NULL_PADDR) {
63 crash("could not allocate 0x%x bytes for syspage/cpupage\n", size);
64 }
65
66 private = lsp.system_private.p;
67
68 private->kern_syspageptr = TOPTR(aarch64_map(~0L, syspage_paddr, size, PROT_READ|PROT_WRITE|PROT_EXEC));
69 private->kern_cpupageptr = TOPTR((uintptr_t)private->kern_syspageptr + spsize);
70
71 private->user_syspageptr = TOPTR(aarch64_map(~0L, syspage_paddr, size, PROT_READ|PROT_EXEC|PROT_USER));
72 private->user_cpupageptr = TOPTR((uintptr_t)private->user_syspageptr + spsize);
73
74 private->cpupage_spacing = sizeof(struct cpupage_entry);
75
76 *syspagep = syspage_paddr;
77 *cpupagep = syspage_paddr + spsize;
78
79 /*
80 * Set up aarch64 specific syspage section
81 */
82 sp->un.aarch64.L0_vaddr = L0_vaddr;
83 INIT_ENTRY(aarch64, gic_map);
84 INIT_ENTRY(aarch64, gicr_map);
85 INIT_ENTRY(aarch64, idreg_dump);
86
87 return((void *)syspage_paddr);
88}
