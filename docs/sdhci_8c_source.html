84static int sdhci_waitmask( sdio_hc_t * const hc, const uint32_t reg, const uint32_t mask, const uint32_t val, const uint32_t usec )
85{
86 sdhci_hc_t const *sdhc;
87 uintptr_t base;
88 uint32_t cnt;
89 int stat;
90 uint32_t rval;
91 uint32_t iter;
92
93 sdhc = hc->cs_hdl;
94 base = sdhc->base;
95 stat = ETIMEDOUT;
96 rval = 0;
97
98 // fast poll for 1ms - 1us intervals
99 for( cnt = min( usec, 1000 ); cnt; cnt-- ) {
100 rval = sdhci_in32( base + reg );
101 if( ( rval & mask ) == val ) {
102 stat = EOK;
103 break;
104 }
105 nanospin_ns( 1000L );
106 }
107
108 if( ( usec > 1000 ) && stat ) {
109 iter = usec / 1000U;
110 for( cnt = iter; cnt; cnt-- ) {
111 rval = sdhci_in32( base + reg );
112 if( ( rval & mask ) == val ) {
113 stat = EOK;
114 break;
115 }
116 delay( 1 );
117 }
118 }
119
120#ifdef SDHCI_DEBUG
121 if( !cnt ) {
122 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: reg %x, mask %x, val %x, rval %x", __func__, reg, mask, val, sdhci_in32( base + reg ) );
123 }
124#endif
125
126 return( stat );
127}

129int sdhci_reset( sdio_hc_t * const hc, const uint32_t rst )
130{
131 sdhci_hc_t const *sdhc;
132 uintptr_t base;
133 uint32_t sctl;
134 int status;
135
136#ifdef SDHCI_DEBUG
137 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: rst %x", __FUNCTION__, rst );
138#endif
139
140 sdhc = hc->cs_hdl;
141 base = sdhc->base;
142
143 sctl = sdhci_in32( base + SDHCI_SYSCTL );
144
145 /* wait up to 100 ms for reset to complete */
146 sdhci_out32( base + SDHCI_SYSCTL, sctl | rst );
147#ifdef SDIO_HC_SC1810X
148 /*
149 * Clear reset bits: On Socionext SC1810A EVB board,
150 * the Reset register bits are unable to get cleared automatically.
151 * Need to manually clear them after issuing reset.
152 */
153 delay( 1 );
154 sdhci_out32( base + SDHCI_SYSCTL, sctl );
155#endif
156 status = sdhci_waitmask( hc, SDHCI_SYSCTL, rst, 0, 100000 );
157
158 if( status ) {
159 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: timeout", __FUNCTION__ );
160 }
161
162#ifdef SDIO_HC_TEGRA
163 tegra_reset_all(hc, rst);
164#endif
165
166 if( hc->flags & SF_ICE_PERSIST ) {
167 if( rst & SDHCI_SYSCTL_SRA ) {
168 /* For SDHCI cadence ip that need a 1ms delay, to ensure SD Card has been reset properly */
169 delay( 1 );
170 sdhci_out32( base + SDHCI_SYSCTL, SDHCI_SYSCTL_ICE);
171 if( sdhci_waitmask( hc, SDHCI_SYSCTL, SDHCI_SYSCTL_ICS, SDHCI_SYSCTL_ICS, 20000 ) != EOK ) {
172 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: wait ics timeout......", __func__ );
173 }
174 }
175 }
176
177 return( status );
178}

180static int sdhci_pio_xfer( sdio_hc_t * const hc, sdio_cmd_t const * const cmd )
181{
182 sdhci_hc_t const *sdhc;
183 uintptr_t base;
184 int msk;
185 uint32_t len;
186 uint32_t blksz;
187 uint8_t *addr;
188
189 sdhc = hc->cs_hdl;
190 base = sdhc->base;
191 msk = ( cmd->flags & SCF_DIR_IN ) ? SDHCI_PSTATE_BRE : SDHCI_PSTATE_BWE;
192
193 while( ( sdhci_in32( base + SDHCI_PSTATE ) & msk ) ) {
194 blksz = cmd->blksz;
195 while( blksz ) {
196 if( sdio_sg_nxt( hc, &addr, &len, blksz ) ) {
197 break;
198 }
199 blksz -= len;
200 len /= 4;
201 if( ( cmd->flags & SCF_DIR_IN ) ) {
202 sdhci_in32s( addr, len, base + SDHCI_DATA );
203 }
204 else {
205 sdhci_out32s( addr, len, base + SDHCI_DATA );
206 }
207 }
208 }
209 return( EOK );
210}

212static int sdhci_intr_event( sdio_hc_t * const hc )
213{
214 sdhci_hc_t const *sdhc;
215 sdio_cmd_t *cmd;
216 uint32_t sts;
217 uint32_t cs;
218 uint32_t idx;
219 uint32_t rsp;
220 uint8_t rbyte;
221 uintptr_t base;
222
223 sdhc = hc->cs_hdl;
224 base = sdhc->base;
225 cs = CS_CMD_INPROG;
226
227#ifdef SDHCI_DEBUG
228// sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: is 0x%x, ie 0x%x", __FUNCTION__, sdhci_in32( base + SDHCI_IS ), sdhci_in32( base + SDHCI_IE ) );
229// sdhci_reg_dump( hc, __FUNCTION__, __LINE__ );
230#endif
231
232#if !defined(__ARM__) && !defined(__aarch64__)
233 if( !( sdhci_in16( base + SDHCI_SLOT_IS ) & ( 1 << hc->slot ) ) ) {
234 return( EOK );
235 }
236#endif
237
238 sts = sdhci_in32( base + SDHCI_IS );
239
240 if( !sts ) { // shared interrupt
241 return( EOK );
242 }
243
244 sts &= sdhci_in32( base + SDHCI_IE ) | SDHCI_INTR_ERRI;
245 sdhci_out32( base + SDHCI_IS, sts );
246
247 if( ( sts & ( SDHCI_INTR_CINS | SDHCI_INTR_CREM ) ) ) {
248 sdio_hc_event( hc, HC_EV_CD );
249 }
250
251 if( ( sts & ( SDCHI_INTR_RETUNE ) ) ) {
252 sdio_hc_event( hc, HC_EV_TUNE );
253 }
254
255 cmd = hc->wspc.cmd;
256 if( cmd == NULL ) {
257 return( EOK );
258 }
259
260 if( ( sts & SDHCI_INTR_ERRI ) ) { // Check of errors
261 if( sts & SDHCI_INTR_DTO ) {
262 cs = CS_DATA_TO_ERR;
263 }
264 if( sts & SDHCI_INTR_DCRC ) {
265 cs = CS_DATA_CRC_ERR;
266 }
267 if( sts & SDHCI_INTR_DEB ) {
268 cs = CS_DATA_END_ERR;
269 }
270 if( sts & SDHCI_INTR_CTO ) {
271 cs = CS_CMD_TO_ERR;
272 }
273 if( sts & SDHCI_INTR_CCRC ) {
274 cs = CS_CMD_CRC_ERR;
275 }
276 if( sts & SDHCI_INTR_CEB ) {
277 cs = CS_CMD_END_ERR;
278 }
279 if( sts & SDHCI_INTR_CIE ) {
280 cs = CS_CMD_IDX_ERR;
281 }
282 if( sts & SDHCI_INTR_ADMAE ) {
283 cs = CS_DATA_TO_ERR;
284 }
285 if( sts & SDHCI_INTR_ACE ) {
286// sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: is 0x%x, ac12 0x%x", __FUNCTION__, sdhci_in32( base + SDHCI_IS ), sdhci_in32( base + SDHCI_AC12 ) );
287 cs = CS_DATA_TO_ERR;
288 }
289 if( !cs ) {
290 cs = CS_CMD_CMP_ERR;
291 }
292 sdhci_reset( hc, SDHCI_SYSCTL_SRD );
293 sdhci_reset( hc, SDHCI_SYSCTL_SRC );
294 }
295 else {
296 if( ( sts & SDHCI_INTR_CC ) ) {
297 if( ( cmd->flags & SCF_RSP_136 ) ) {
298 // CRC is not included in the response reg, left
299 // shift 8 bits to match the 128 CID/CSD structure
300 rbyte = 0;
301 for( idx = 0; idx < 4; idx++ ) {
302 rsp = sdhci_in32( base + SDHCI_RESP0 + idx * 4 );
303 cmd->rsp[3 - idx] = (rsp << 8) + rbyte;
304 rbyte = rsp >> 24;
305 }
306 }
307 else if( ( cmd->flags & SCF_RSP_PRESENT ) ) {
308 cmd->rsp[0] = sdhci_in32( base + SDHCI_RESP0 );
309 }
310 else {
311 // nothing
312 }
313 cs = CS_CMD_CMP;
314 }
315
316 if( ( sts & SDHCI_INTR_TC ) ) {
317 cs = CS_CMD_CMP;
318 cmd->rsp[0] = sdhci_in32( base + SDHCI_RESP0 );
319 }
320 else if( ( sts & SDHCI_INTR_DMA ) ) { // restart on dma boundary
321 sdhci_out32( base + SDHCI_SDMA_ARG2, sdhci_in32( base + SDHCI_SDMA_ARG2 ) );
322 }
323 else {
324 // nothing
325 }
326
327 if( ( sts & ( SDHCI_INTR_BWR | SDHCI_INTR_BRR ) ) ) {
328 if( ( cmd->opcode == SD_SEND_TUNING_BLOCK ) ||
329 ( cmd->opcode == MMC_SEND_TUNING_BLOCK ) ) {
330 cs = CS_CMD_CMP;
331 }
332 else {
333 cs = sdhci_pio_xfer( hc, cmd );
334 }
335 }
336 }
337
338 if( ( sts & SDHCI_INTR_CREM ) ) {
339 cs = CS_CARD_REMOVED;
340 }
341
342 if( cs != CS_CMD_INPROG ) {
343 sdio_cmd_cmplt( hc, cmd, cs );
344 }
345
346 return( EOK );
347}

349static int sdhci_event( sdio_hc_t * const hc, sdio_event_t * const ev )
350{
351 int status;
352
353 switch( ev->code ) {
354 case HC_EV_INTR:
355 status = sdhci_intr_event( hc );
356 InterruptUnmask( hc->cfg.irq[0], hc->hc_iid );
357 break;
358
359 default:
360#ifdef BS_EVENT
361 status = bs_event( hc, ev );
362#else
363 status = ENOTSUP;
364#endif
365 break;
366 }
367
368 return( status );
369}

371static int sdhci_adma_setup( sdio_hc_t *hc, sdio_cmd_t * const cmd )
372{
373 sdhci_hc_t *sdhc;
374 sdhci_adma64_t *adma;
375 sdio_sge_t *sgp;
376 uint32_t sgc;
377 uint32_t sgi;
378 uint32_t acnt;
379 uint32_t alen;
380 uint32_t sg_count;
381 uint32_t desc_sz;
382 paddr64_t paddr;
383 sdio_hc_cfg_t *cfg;
384
385 cfg = &hc->cfg;
386 sdhc = hc->cs_hdl;
387 adma = sdhc->adma;
388
389#ifdef SDHCI_DEBUG
390 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: ", __FUNCTION__ );
391#endif
392
393 sgc = cmd->sgc;
394 sgp = cmd->sgl;
395
396 if( !( cmd->flags & SCF_DATA_PHYS ) ) {
397 sdio_vtop_sg( sgp, sdhc->sgl, sgc, cmd->mhdl );
398 sgp = sdhc->sgl;
399 }
400
401 desc_sz = sizeof( sdhci_adma32_t );
402 if( ( sdhc->flags & SF_USE_ADMA64 ) ) {
403 desc_sz = SDHCI_ADMA2_64_DESC_SZ( sdhc );
404 }
405
406 acnt = 0;
407 for( sgi = 0; sgi < sgc; sgi++ ) {
408 paddr = sgp->sg_address + cfg->bmstr_xlat;
409 sg_count = sgp->sg_count;
410 while( sg_count ) {
411 alen = min( sg_count, SDHCI_ADMA2_MAX_XFER );
412 adma->attr = SDHCI_ADMA2_VALID | SDHCI_ADMA2_TRAN;
413 adma->addr_lo = (uint32_t)paddr;
414 adma->len = alen;
415 if( ( sdhc->flags & SF_USE_ADMA64 ) ) {
416 adma->addr_hi = paddr >> 32;
417 if( ( sdhc->flags & SF_V4_MODE ) ) {
418 ((sdhci_v4_adma64_t *)adma)->reserved = 0; /* SBZ */
419 }
420 }
421 sg_count -= alen;
422 paddr += alen;
423 adma = (sdhci_adma64_t *)( (uintptr_t)adma + desc_sz );
424 if( ++acnt > ADMA_DESC_MAX ) {
425 return( ENOTSUP );
426 }
427 }
428 sgp++;
429 }
430
431 adma = (sdhci_adma64_t *)( (uintptr_t)adma - desc_sz );
432 adma->attr |= SDHCI_ADMA2_END;
433
434 __cpu_membarrier();
435
436 sdhci_out32( sdhc->base + SDHCI_ADMA_ADDRL, (uint32_t)(sdhc->admap + cfg->bmstr_xlat) );
437 if( ( sdhc->flags & SF_USE_ADMA64 ) ) {
438 sdhci_out32( sdhc->base + SDHCI_ADMA_ADDRH, (uint32_t)((sdhc->admap + cfg->bmstr_xlat) >> 32) );
439 }
440
441 return( EOK );
442}

469static int sdhci_xfer_setup( sdio_hc_t * const hc, sdio_cmd_t * const cmd,
470 uint32_t *command, uint32_t *imask )
471{
472 sdhci_hc_t const *sdhc;
473 uintptr_t base;
474 uint32_t hctl;
475 uint16_t hctl2;
476 int status;
477
478 sdhc = hc->cs_hdl;
479 status = EOK;
480 base = sdhc->base;
481 hctl = sdhci_in32( base + SDHCI_HCTL ) & ~SDHCI_HCTL_DMA_MSK;
482 hctl2 = sdhci_in16( base + SDHCI_HCTL2 );
483 *command |= ( cmd->flags & SCF_DIR_IN ) ? ( SDHCI_CMD_DP | SDHCI_CMD_DDIR ) : SDHCI_CMD_DP;
484 *imask |= SDHCI_INTR_TC | SDHCI_INTR_DTO | SDHCI_INTR_DCRC | SDHCI_INTR_DEB;
485
486 status = sdio_sg_start( hc, cmd->sgl, cmd->sgc );
487
488 if( cmd->sgc && ( hc->caps & HC_CAP_DMA ) ) {
489 if( ( sdhc->flags & SF_USE_ADMA ) ) {
490 status = sdhci_adma_setup( hc, cmd );
491 if( status == EOK ) {
492 if( sdhc->flags & SF_V4_MODE ) {
493 hctl |= SDHCI_HCTL_ADMA2;
494 hctl2 |= SDHCI_HCTL2_V4_MODE;
495 hctl2 &= ~SDHCI_HCTL2_64BIT_ADDR;
496 if( sdhc->flags & SF_USE_ADMA64 ) {
497 hctl2 |= SDHCI_HCTL2_64BIT_ADDR;
498 }
499 } else {
500 hctl |= ( sdhc->flags & SF_USE_ADMA64 ) ? SDHCI_HCTL_ADMA64 : SDHCI_HCTL_ADMA32;
501 }
502 *command |= SDHCI_CMD_DE;
503 }
504 }
505 else {
506 status = sdhci_sdma_setup( hc, cmd );
507 if( status == EOK ) {
508 hctl |= SDHCI_HCTL_SDMA;
509 *command |= SDHCI_CMD_DE;
510 }
511 }
512 }
513
514 if( status || !( hc->caps & HC_CAP_DMA ) ) { // use PIO
515 if( ( cmd->flags & SCF_DATA_PHYS ) ) {
516 return( status );
517 }
518 status = EOK;
519 *imask |= ( cmd->flags & SCF_DIR_IN ) ? SDHCI_INTR_BRR : SDHCI_INTR_BWR;
520 }
521
522 if( cmd->blks > 1 ) {
523 *command |= SDHCI_CMD_MBS | SDHCI_CMD_BCE;
524 if( ( hc->caps & HC_CAP_ACMD23 ) && ( cmd->flags & SCF_SBC ) ) {
525 *command |= SDHCI_CMD_ACMD23;
526 sdhci_out32( base + SDHCI_SDMA_ARG2, cmd->blks );
527 }
528 else if( ( hc->caps & HC_CAP_ACMD12 ) ) {
529 *command |= SDHCI_CMD_ACMD12;
530 }
531 else {
532 // nothing
533 }
534 }
535
536 sdhci_out32( base + SDHCI_HCTL, hctl );
537
538 if( ( sdhc->flags & SF_V4_MODE ) ) {
539 sdhci_out16( base + SDHCI_HCTL2, hctl2 );
540 sdhci_out32( base + SDHCI_BLK, cmd->blksz );
541 sdhci_out32( base + SDHCI_32BIT_BLK_CNT, cmd->blks );
542 }
543 else {
544 sdhci_out32( base + SDHCI_BLK, cmd->blksz | SDHCI_BLK_SDMA_BOUNDARY_512K |
545 ( cmd->blks << SDHCI_BLK_BLKCNT_SHIFT ) );
546 }
547
548 return( status );
549}

551static int sdhci_cmd( sdio_hc_t * const hc, sdio_cmd_t * const cmd )
552{
553 sdhci_hc_t const *sdhc;
554 uintptr_t base;
555 uint32_t pmask;
556 uint32_t imask;
557 int status;
558 uint32_t command;
559
560 sdhc = hc->cs_hdl;
561 base = sdhc->base;
562 imask = SDHCI_INTR_DFLTS;
563
564#ifndef BS_GPIO_INS_RMV
565 if( !( hc->caps & HC_CAP_SLOT_TYPE_EMBEDDED ) ) {
566 imask |= SDHCI_INTR_CINS | SDHCI_INTR_CREM;
567 }
568#endif
569
570 pmask = SDHCI_PSTATE_CMDI;
571 command = cmd->opcode << 24;
572
573 sdhci_out32( base + SDHCI_IS, SDHCI_INTR_CLR_MSK ); // Clear Status
574
575 if( cmd->opcode == MMC_STOP_TRANSMISSION ) {
576 command |= SDHCI_CMD_TYPE_CMD12;
577 }
578
579 if( ( cmd->flags & SCF_DATA_MSK ) ) {
580 pmask |= SDHCI_PSTATE_DATI;
581 status = sdhci_xfer_setup( hc, cmd, &command, &imask );
582 if( status != EOK ) {
583 return( status );
584 }
585 }
586 else {
587 imask |= SDHCI_INTR_CC; // Enable command complete intr
588 }
589
590 if( ( cmd->flags & SCF_RSP_PRESENT ) ) {
591 if( ( cmd->flags & SCF_RSP_136 ) ) {
592 command |= SDHCI_CMD_RSP_TYPE_136;
593 }
594 else if( ( cmd->flags & SCF_RSP_BUSY ) ) { // Response with busy check
595 command |= SDHCI_CMD_RSP_TYPE_48b;
596 }
597 else {
598 command |= SDHCI_CMD_RSP_TYPE_48;
599 }
600
601 if( ( cmd->flags & SCF_RSP_OPCODE ) ) { // Index check
602 command |= SDHCI_CMD_CICE;
603 }
604
605 if( ( cmd->flags & SCF_RSP_CRC ) ) { // CRC check
606 command |= SDHCI_CMD_CCCE;
607 }
608 }
609
610 status = sdhci_waitmask( hc, SDHCI_PSTATE, pmask, 0, 100000 );
611 if( status ) {
612 return( status );
613 }
614
615 sdhci_out32( base + SDHCI_IE, imask );
616 sdhci_out32( base + SDHCI_ARG, cmd->arg );
617 sdhci_out32( base + SDHCI_CMD, command );
618
619 return( EOK );
620}

628{
629 sdhci_hc_t const *sdhc;
630 uintptr_t base;
631 uint32_t hctl;
632
633#ifdef SDHCI_DEBUG
634 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: vdd 0x%x", __FUNCTION__, vdd );
635#endif
636
637 sdhc = hc->cs_hdl;
638 base = sdhc->base;
639
640 // disable interrupts
641 sdhci_out32( base + SDHCI_IE, 0 );
642

644 sdhci_reset( hc, SDHCI_SYSCTL_SRC );

646 sdhci_reset( hc, SDHCI_SYSCTL_SRA );

650 if( vdd ) {
651 hctl = sdhci_in32( base + SDHCI_HCTL ) & ~SDHCI_HCTL_SDVS_MSK;
652 switch( vdd ) {
653 case OCR_VDD_17_195:
654 hctl |= SDHCI_HCTL_SDVS1V8;
655 break;
656
657 case OCR_VDD_30_31:
658 case OCR_VDD_29_30:
659 hctl |= SDHCI_HCTL_SDVS3V0;
660 break;
661
662 case OCR_VDD_33_34:
663 case OCR_VDD_32_33:
664 hctl |= SDHCI_HCTL_SDVS3V3;
665 break;
666
667 default:
668 return( EINVAL );
669 }
670
671 sdhci_out32( base + SDHCI_HCTL, hctl );
672 sdhci_out32( base + SDHCI_HCTL, hctl | SDHCI_HCTL_SDBP );
673 sdhci_waitmask( hc, SDHCI_HCTL, SDHCI_HCTL_SDBP, SDHCI_HCTL_SDBP, 10000 );
674// delay( 10 );
675 }
676 else {
677 sdhci_out32( base + SDHCI_HCTL, 0 );
678 }
679
680#ifndef BS_GPIO_INS_RMV
681 if( !( hc->caps & HC_CAP_SLOT_TYPE_EMBEDDED ) ) {
682 sdhci_out32( base + SDHCI_IE, SDHCI_INTR_CINS | SDHCI_INTR_CREM );
683 }
684#endif
685
686 hc->vdd = vdd;

688 return( EOK );

738static int sdhci_clk( sdio_hc_t *hc, const uint32_t clk )
739{
740 sdhci_hc_t const *sdhc;
741 uintptr_t base;
742 uint32_t sctl;
743 uint32_t clkd;
744
745#ifdef SDHCI_DEBUG
746 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: clk %d", __FUNCTION__, clk );
747#endif
748
749 sdhc = hc->cs_hdl;
750 base = sdhc->base;
751 clkd = 0;
752 sctl = sdhci_in32( base + SDHCI_SYSCTL );
753
754 if ( hc->flags & SF_ICE_PERSIST ) {
755 /* stop clock except for ICE */
756 sctl &= ~(SDHCI_SYSCTL_CEN | SDHCI_SYSCTL_DTO_MSK |
757 SDHCI_SYSCTL_CGS_PROG | SDHCI_SYSCTL_SDCLKFS_MSK);
758 } else {
759 // stop clock
760 sctl &= ~(SDHCI_SYSCTL_ICE | SDHCI_SYSCTL_CEN | SDHCI_SYSCTL_DTO_MSK |
761 SDHCI_SYSCTL_CGS_PROG | SDHCI_SYSCTL_SDCLKFS_MSK);
762 }
763
764 sctl |= SDHCI_SYSCTL_DTO_MAX;
765 sdhci_out32( base + SDHCI_SYSCTL, sctl | SDHCI_SYSCTL_SRC | SDHCI_SYSCTL_SRD );
766
767 if( hc->version >= SDHCI_SPEC_VER_3 ) {
768 if( sdhc->clk_mul ) {
769 if( !( sdhci_in16( base + SDHCI_HCTL2 ) & SDHCI_HCTL2_PRESET_VAL ) ) {
770 for( clkd = 1; clkd < 1024; clkd++ ) {
771 if( ( hc->clk_max / clkd ) <= clk ) {
772 break;
773 }
774 }
775 sctl |= SDHCI_SYSCTL_CGS_PROG;
776 clkd--;
777 }
778 }
779 else {
780 // 10 bit Divided Clock Mode
781 for( clkd = 2; clkd < SDHCI_CLK_MAX_DIV_SPEC_VER_3; clkd += 2 ) {
782 if( ( hc->clk_max / clkd ) <= clk ) {
783 break;
784 }
785 }
786 if( hc->clk_max <= clk ) {
787 clkd = 1;
788 }
789 clkd >>= 1;
790 }
791 }
792 else {
793 // 8 bit Divided Clock Mode
794 for( clkd = 1; clkd < SDHCI_CLK_MAX_DIV_SPEC_VER_2; clkd *= 2 ) {
795 if( ( hc->clk_max / clkd ) <= clk ) {
796 break;
797 }
798 }
799 clkd >>= 1;
800 }
801
802 // enable internal clock
803 sctl |= SDHCI_SYSCTL_CLK( clkd ) | SDHCI_SYSCTL_ICE;
804 sdhci_out32( base + SDHCI_SYSCTL, sctl );
805
806 // wait for clock to stabilize
807 sdhci_waitmask( hc, SDHCI_SYSCTL, SDHCI_SYSCTL_ICS, SDHCI_SYSCTL_ICS, 10000 );
808
809 // enable clock to the card
810 sdhci_out32( base + SDHCI_SYSCTL, sctl | SDHCI_SYSCTL_CEN );
811
812 delay(1);
813
814 hc->clk = clk;
815
816 // On some boards, the clocks may not actually have stabilized as we think they have
817 usleep(1000);
818 return( EOK );
819}

821static int sdhci_timing( sdio_hc_t *hc, const uint32_t timing )
822{
823 sdhci_hc_t const *sdhc;
824 uintptr_t base;
825 uint32_t hctl;
826 uint16_t hctl2;
827
828#ifdef SDHCI_DEBUG
829 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: timing %d", __FUNCTION__, timing );
830#endif
831
832 sdhc = hc->cs_hdl;
833 base = sdhc->base;
834 hctl = sdhci_in32( base + SDHCI_HCTL ) & ~SDHCI_HCTL_HSE;
835 hctl2 = 0;
836
837 if( hc->version >= SDHCI_SPEC_VER_3 ) {
838 hctl2 = sdhci_in16( base + SDHCI_HCTL2 ) & ~SDHCI_HCTL2_MODE_MSK;
839 }
840
841 switch( timing ) {
842 case TIMING_HS400:
843 hctl |= SDHCI_HCTL_HSE;
844 hctl2 |= SDHCI_HCTL2_MODE_HS400;
845 break;
846
847 case TIMING_HS200:
848 hctl |= SDHCI_HCTL_HSE;
849 hctl2 |= SDHCI_HCTL2_MODE_HS200;
850 break;
851
852 case TIMING_SDR104:
853 hctl |= SDHCI_HCTL_HSE;
854 hctl2 |= SDHCI_HCTL2_MODE_SDR104;
855 break;
856
857 case TIMING_SDR50:
858 hctl |= SDHCI_HCTL_HSE;
859 hctl2 |= SDHCI_HCTL2_MODE_SDR50;
860 break;
861
862 case TIMING_SDR25:
863 hctl |= SDHCI_HCTL_HSE;
864 hctl2 |= SDHCI_HCTL2_MODE_SDR25;
865 break;
866
867 case TIMING_SDR12:
868 hctl |= SDHCI_HCTL_HSE;
869 break;
870
871 case TIMING_DDR50:
872 hctl |= SDHCI_HCTL_HSE;
873 hctl2 |= SDHCI_HCTL2_MODE_DDR50;
874 break;
875
876 case TIMING_HS:
877 hctl |= SDHCI_HCTL_HSE;
878 break;
879
880 case TIMING_LS:
881 break;
882
883 default:
884 break;
885 }
886
887 if( hc->version >= SDHCI_SPEC_VER_3 ) {
888 // check for preset value enable and disble clock
889 if( ( hctl2 & SDHCI_HCTL2_PRESET_VAL ) ) {
890 sdhci_out32( base + SDHCI_SYSCTL, sdhci_in32( base + SDHCI_SYSCTL ) & ~SDHCI_SYSCTL_CEN );
891 }
892
893 sdhci_out32( base + SDHCI_HCTL, hctl );
894 sdhci_out16( base + SDHCI_HCTL2, hctl2 );
895
896 if( ( hctl2 & SDHCI_HCTL2_PRESET_VAL ) ) {
897 sdhci_clk( hc, hc->clk );
898 }
899 }
900 else {
901 sdhci_out32( base + SDHCI_HCTL, hctl );
902 }
903
904 hc->timing = timing;
905
906 return( EOK );
907}

909static int sdhci_signal_voltage( sdio_hc_t *hc, const uint32_t signal_voltage )
910{
911 sdhci_hc_t const *sdhc;
912 uintptr_t base;
913 uint32_t sctl;
914
915#ifdef SDHCI_DEBUG
916 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: signal_voltage %d", __FUNCTION__, signal_voltage );
917#endif
918
919 sdhc = hc->cs_hdl;
920 base = sdhc->base;
921
922 if( hc->version < SDHCI_SPEC_VER_3 ) {
923 return( EOK );
924 }
925
926 if( hc->signal_voltage == signal_voltage ) {
927 if( signal_voltage == SIGNAL_VOLTAGE_3_3 ) {
928 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 1, "%s: voltage already 3.3V", __FUNCTION__ );
929 }
930 else if( signal_voltage == SIGNAL_VOLTAGE_1_8 ) {
931 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 1, "%s: voltage already 1.8V", __FUNCTION__ );
932 }
933 else {
934 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 1, "%s: voltage already: %u, return EOK", __FUNCTION__, signal_voltage );
935 }
936 return( EOK );
937 }
938
939 if( signal_voltage == SIGNAL_VOLTAGE_3_3 ) {
940 sdhci_out16( base + SDHCI_HCTL2, sdhci_in16( base + SDHCI_HCTL2 ) & ~SDHCI_HCTL2_SIG_1_8V );
941 delay( 5 ); // wait for 3.3V regulator to stabilize
942 if( ( sdhci_in16( base + SDHCI_HCTL2 ) & SDHCI_HCTL2_SIG_1_8V ) ) {
943 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: switch 3.3V failure", __FUNCTION__ );
944 return( EIO );
945 }
946 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 1, "%s: switch 3.3V OK", __FUNCTION__ );
947 }
948 else if( signal_voltage == SIGNAL_VOLTAGE_1_8 ) {
949 // disable clock
950 sctl = sdhci_in32( base + SDHCI_SYSCTL );
951 sdhci_out32( base + SDHCI_SYSCTL, sctl & ~SDHCI_SYSCTL_CEN );
952
953 // Check whether DAT[3:0] is 0000
954 // The specification has a check for DAT[3:0] == 0000, but doesn't work on some platforms
955// if( !( sdhci_in32( base + SDHCI_PSTATE ) & SDHCI_PSTATE_DLSL_MSK ) || (hc->flags & HC_FLAG_DEV_MMC) ) {
956 sdhci_out16( base + SDHCI_HCTL2, sdhci_in16( base + SDHCI_HCTL2 ) | SDHCI_HCTL2_SIG_1_8V );
957
958 delay( 5 );
959
960 if( ( sdhci_in16( base + SDHCI_HCTL2 ) & SDHCI_HCTL2_SIG_1_8V ) ) {
961 sdhci_out32( base + SDHCI_SYSCTL, sctl ); // re-enable clock
962
963 delay( 1 );
964#ifdef SDIO_HC_TEGRA
965 /* DAT[3:0] level doesn't change on Tegra SOC */
966 hc->signal_voltage = signal_voltage;
967 return( EOK );
968#endif
969 // If DAT[3:0] level is 1111b, then the switch succeeded
970 if( ( ( sdhci_in32( base + SDHCI_PSTATE ) & SDHCI_PSTATE_DLSL_MSK ) == SDHCI_PSTATE_DLSL_MSK ) || (hc->flags & HC_FLAG_DEV_MMC) ) {
971 hc->signal_voltage = signal_voltage;
972 sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 1, "%s: switch 1.8V OK", __FUNCTION__ );
973 return( EOK );
974 }
975 }
976// }
977
978 // voltage switch failed, cycle power
979 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: switch 1.8V failure", __FUNCTION__ );
980
981 sdhci_out32( base + SDHCI_HCTL, sdhci_in32( base + SDHCI_HCTL ) & ~SDHCI_HCTL_SDBP );
982 delay( 1 );
983 sdhci_out32( base + SDHCI_HCTL, sdhci_in32( base + SDHCI_HCTL ) | SDHCI_HCTL_SDBP );
984 return( EIO );
985 }
986 else {
987 // nothing
988 }
989
990 hc->signal_voltage = signal_voltage;
991
992 return( EOK );
993}

1039static int sdhci_tune( sdio_hc_t *hc, const uint32_t op )
1040{
1041 sdhci_hc_t const *sdhc;
1042 struct sdio_cmd *cmd;
1043 uintptr_t base;
1044 uint16_t hctl2;
1045 int tlc;
1046 uint8_t *rbuf;
1047 uint32_t tlen;
1048 int status;
1049
1050#ifdef SDHCI_DEBUG
1051 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: ", __FUNCTION__ );
1052#endif
1053
1054 sdhc = hc->cs_hdl;
1055 base = sdhc->base;
1056 tlen = ( hc->bus_width == BUS_WIDTH_8 ) ? 128U : 64U;
1057 rbuf = alloca(tlen);
1058
1059 if( hc->version < SDHCI_SPEC_VER_3 ) {
1060 return( EOK );
1061 }
1062
1063 hctl2 = sdhci_in16( base + SDHCI_HCTL2 );
1064
1065 // return if not HS200 or SDR104, and not SDR50 that requires tuning
1066 if( ( SDHCI_HCTL2_MODE( hctl2 ) != SDHCI_HCTL2_MODE_SDR104 ) &&
1067 ( SDHCI_HCTL2_MODE( hctl2 ) != SDHCI_HCTL2_MODE_HS200 ) &&
1068 ( ( SDHCI_HCTL2_MODE( hctl2 ) == SDHCI_HCTL2_MODE_SDR50 ) &&
1069 !( sdhc->flags & SF_TUNE_SDR50 ) ) ) {
1070 return( EOK );
1071 }
1072
1073 cmd = sdio_alloc_cmd( );
1074 if( cmd == NULL ) {
1075 return( ENOMEM );
1076 }
1077
1078 hctl2 |= SDHCI_HCTL2_EXEC_TUNING;
1079
1080 sdhci_out16( base + SDHCI_HCTL2, hctl2 );
1081
1082 for( tlc = SDHCI_TUNING_RETRIES; tlc; tlc-- ) {
1083 cmd->status = CS_CMD_INPROG;
1084 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, op, 0 );
1085 sdio_setup_cmd_io( cmd, SCF_DIR_IN, 1, tlen, NULL, 0, NULL );
1086
1087 status = sdio_issue_cmd( &hc->device, cmd, SDHCI_TUNING_TIMEOUT );
1088 if( status ) {
1089 break;
1090 }
1091
1092 sdhci_in32s( rbuf, tlen >> 2, base + SDHCI_DATA );
1093
1094 hctl2 = sdhci_in16( base + SDHCI_HCTL2 );
1095 if( !( hctl2 & SDHCI_HCTL2_EXEC_TUNING ) ) {
1096 break;
1097 }
1098 }
1099
1100 if( status || ( hctl2 & SDHCI_HCTL2_EXEC_TUNING ) ) {
1101 hctl2 &= ~( SDHCI_HCTL2_TUNED_CLK | SDHCI_HCTL2_EXEC_TUNING );
1102 sdhci_out16( base + SDHCI_HCTL2, hctl2 );
1103 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: tuning failure (%d 0x%x)", __FUNCTION__, status, hctl2 );
1104 status = EIO;
1105 }
1106
1107 sdio_free_cmd( cmd );
1108
1109 return( status );
1110}
