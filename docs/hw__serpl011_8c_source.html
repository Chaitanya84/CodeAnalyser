31static void
32parse_line(unsigned channel, const char *line, unsigned *baud, unsigned *clk)
33{
34 /*
35 * Get device base address and register stride
36 */
37 if (*line != '.' && *line != '\0') {
38 dbg_device[channel].base = strtopaddr(line, (char **)&line, 16);
39 if (*line == '^') {
40 dbg_device[channel].shift = strtoul(line+1, (char **)&line, 0);
41 }
42 }
43
44 /*
45 * Get baud rate
46 */
47 if (*line == '.') {
48 ++line;
49 }
50 if (*line != '.' && *line != '\0') {
51 *baud = strtoul(line, (char **)&line, 0);
52 }
53
54 /*
55 * Get clock rate
56 */
57 if (*line == '.') {
58 ++line;
59 }
60 if (*line != '.' && *line != '\0') {
61 *clk = strtoul(line, (char **)&line, 0);
62 }
63}

65/*
66 * Initialise one of the serial ports
67 */
68void
69init_pl011(unsigned channel, const char *init, const char *defaults)
70{
71 paddr_t base;
72 unsigned baud;
73 unsigned clk;
74 unsigned ibrd, fbrd;
75
76 /*
77 * Default clock and baud rate
78 */
79 clk = 14745600;
80 baud = 57600;
81
82 parse_line(channel, defaults, &baud, &clk);
83 parse_line(channel, init, &baud, &clk);
84 base = dbg_device[channel].base;
85
86 if (baud != 0) {
87 clk /= 16;
88
89 ibrd = clk / baud;
90 fbrd = (((clk % baud) * 64) + ((baud + 1) >> 1)) / baud;
91
92 while (in32(base + PL011_FR) & PL011_FR_BUSY)
93 ;
94 out32(base + PL011_CR, 0); /* Disable UART */
95
96 out32(base + PL011_ICR, 0x7FF); /* Clear all status */
97 out32(base + PL011_IBRD, ibrd);
98 out32(base + PL011_FBRD, fbrd);
99 out32(base + PL011_LCR_H, PL011_LCR_H_WLEN8 | PL011_LCR_H_FEN);
100 }
101
102 out32(base + PL011_CR, PL011_CR_UARTEN|PL011_CR_TXE|PL011_CR_RXE);
103}

105/*
106 * Send a character
107 */
108void
109put_pl011(int c)
110{
111 paddr_t base = dbg_device[0].base;
112
113 while ((in32(base + PL011_FR) & PL011_FR_TXFF) != 0)
114 ;
115 out32(base + PL011_DR, c & 0xff);
116}
