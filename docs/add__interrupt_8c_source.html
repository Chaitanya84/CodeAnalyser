43static output_callout_t *
44callout_output_intr(int sizing) {
45 unsigned i;
46 unsigned nelts;
47
48 nelts = lsp.intrinfo.size / sizeof(*lsp.intrinfo.p);
49 for(i = 0; i < nelts; ++i) {
50 uint8_t *base = (uint8_t *)&lsp.intrinfo.p[i];
51 unsigned j;
52
53 for(j = 0; j < NUM_ELTS(offsets); ++j) {
54 void (**rtn)(void) = (void (**)(void))(base + offsets[j]);
55
56 callout_output_one(sizing, rtn);
57 }
58 }
59 return(intr_prev_output_rtn);
60}

63struct intrinfo_entry *
64add_interrupt(const struct startup_intrinfo *startup_intr) {
65 struct intrinfo_entry *intr;
66 unsigned i;
67
68 if(intr_prev_output_rtn == NULL) {
69 //Hook into the sizing/writing list
70 intr_prev_output_rtn = callout_output_rtn;
71 callout_output_rtn = (output_callout_t *)callout_output_intr;
72 }
73
74 intr = grow_syspage_section(&lsp.intrinfo, sizeof(*intr));
75 //Point at newly allocated entry
76 intr = (void *)((uint8_t *)intr + lsp.intrinfo.size - sizeof(*intr));
77 intr->vector_base = startup_intr->vector_base;
78 intr->num_vectors = startup_intr->num_vectors;
79 intr->cascade_vector = startup_intr->cascade_vector;
80 intr->cpu_intr_base = startup_intr->cpu_intr_base;
81 intr->cpu_intr_stride = startup_intr->cpu_intr_stride;
82 intr->local_stride = startup_intr->local_stride;
83 intr->flags = startup_intr->flags;
84 intr->id.genflags = startup_intr->id.genflags;
85 intr->id.rtn = (void *)startup_intr->id.rtn;
86 if(startup_intr->id.rtn != NULL) {
87 intr->id.size = startup_intr->id.rtn->rtn_size;
88 }
89 intr->eoi.genflags = startup_intr->eoi.genflags;
90 intr->eoi.rtn = (void *)startup_intr->eoi.rtn;
91 if(startup_intr->eoi.rtn != NULL) {
92 intr->eoi.size = startup_intr->eoi.rtn->rtn_size;
93 }
94 intr->mask = (void *)startup_intr->mask;
95 intr->unmask = (void *)startup_intr->unmask;
96 intr->config = (void *)startup_intr->config;
97
98 for(i = 0; i < NUM_ELTS(offsets); ++i) {
99 void (**rtn)(void) = (void (**)(void))((uintptr_t)intr + offsets[i]);
100
101 callout_register_data(rtn, startup_intr->patch_data);
102 }
103 return(intr);
104}

106void
107add_interrupt_array(const struct startup_intrinfo *intrs, unsigned size) {
108 unsigned i;
109 unsigned incr;
110
111 incr = sizeof(*intrs);
112 //
113 //Backwards compatability stuff - can be removed once
114 //all the startups have been recompiled with the new startup_intrinfo
115 //structure
116 //
117 if((size % sizeof(*intrs)) != 0) {
118 incr -= sizeof(void *);
119 }
120 for(i = size / incr; i > 0; --i) {
121 add_interrupt(intrs);
122 intrs = (void *)((uintptr_t)intrs + incr);
123 }
124}
