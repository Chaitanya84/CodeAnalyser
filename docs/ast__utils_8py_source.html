53def iter_nodes(root) -> Iterator:
54 """Depth-first iteration over every node (named and anonymous)."""
55 stack = [root]
56 while stack:
57 node = stack.pop()
58 yield node
59 children = node.children
60 for i in range(len(children) - 1, -1, -1):
61 stack.append(children[i])
62

64def count_nodes(root) -> int:
65 return sum(1 for _ in iter_nodes(root))

68def collect_syntax_errors(root) -> List[Dict]:
69 """Recursively find ERROR and MISSING nodes in the AST.
70
71 A file is cleanly parsed only when both lists are empty. Malformed
72 regions are reported with locations but never raise.
73 """
74 errors: List[Dict] = []
75 for node in iter_nodes(root):
76 is_error = node.type == "ERROR" or node.is_error
77 is_missing = node.is_missing
78 if not (is_error or is_missing):
79 continue
80 if len(errors) >= MAX_REPORTED_ERRORS:
81 continue
82 errors.append(
83 {
84 "type": "ERROR" if is_error else "MISSING",
85 "node_type": node.type,
86 "start_line": start_line(node),
87 "start_column": start_column(node),
88 "end_line": end_line(node),
89 "end_column": end_column(node),
90 }
91 )
92 return errors
