38def load_graph(json_path: Path) -> nx.DiGraph:
39 data = json.loads(Path(json_path).read_text(encoding="utf-8"))
40 graph = nx.DiGraph()
41 for node in data.get("nodes", []):
42 graph.add_node(node["id"], **node)
43 for link in data.get("links", []):
44 graph.add_edge(link["source"], link["target"], type=link["type"])
45 log.info(
46 "visualizer loaded %d nodes and %d edges",
47 graph.number_of_nodes(),
48 graph.number_of_edges(),
49 )
50 return graph

66def _edge_trace(graph: nx.DiGraph, pos: Dict[str, List[float]], edge_type: str) -> go.Scatter3d:
67 xs: List = []
68 ys: List = []
69 zs: List = []
70 for source, target, data in graph.edges(data=True):
71 if data.get("type") != edge_type:
72 continue
73 if source not in pos or target not in pos:
74 continue
75 x0, y0, z0 = pos[source]
76 x1, y1, z1 = pos[target]
77 xs += [x0, x1, None]
78 ys += [y0, y1, None]
79 zs += [z0, z1, None]
80 style = EDGE_STYLES[edge_type]
81 return go.Scatter3d(
82 x=xs, y=ys, z=zs,
83 mode="lines",
84 name=f"{edge_type} (edges)",
85 line=dict(color=style["color"], width=style["width"]),
86 hoverinfo="skip",
87 legendgroup=edge_type,
88 )

91def _node_trace(graph: nx.DiGraph, pos: Dict[str, List[float]], node_type: str) -> go.Scatter3d:
92 xs, ys, zs, texts, custom = [], [], [], [], []
93 for node_id, data in graph.nodes(data=True):
94 if data.get("type") != node_type or node_id not in pos:
95 continue
96 x, y, z = pos[node_id]
97 xs.append(x)
98 ys.append(y)
99 zs.append(z)
100 kind = data.get("kind") or node_type
101 texts.append(
102 f"<b>{data.get('qualified_name', data.get('name', node_id))}</b>"
103 f"<br>Type: {node_type} ({kind})"
104 f"<br>File: {data.get('file', '?')}"
105 f"<br>Line: {data.get('line', '?')}"
106 )
107 custom.append(node_id)
108 label = {"function": "functions", "class": "classes", "variable": "variables"}.get(
109 node_type, node_type + "s"
110 )
111 return go.Scatter3d(
112 x=xs, y=ys, z=zs,
113 mode="markers",
114 name=label,
115 marker=dict(
116 size=NODE_SIZES.get(node_type, 5),
117 color=NODE_COLORS.get(node_type, "#ffffff"),
118 opacity=0.9,
119 line=dict(width=0.5, color="#0d1117"),
120 ),
121 text=texts,
122 hoverinfo="text",
123 customdata=custom,
124 legendgroup=node_type,
125 )

128def build_figure(graph: nx.DiGraph) -> go.Figure:
129 if graph.number_of_nodes() == 0:
130 return _empty_figure("No graph entities were found in the analyzed source tree.")
131
132 # nx.spring_layout requires scipy (sparse eigen-solver for initial
133 # placement); scipy is declared in requirements.txt.
134 pos = nx.spring_layout(graph, dim=3, seed=LAYOUT_SEED)
135
136 traces: List[go.Scatter3d] = []
137 for edge_type in ("calls", "inherits", "defines"):
138 traces.append(_edge_trace(graph, pos, edge_type))
139 for node_type in ("function", "class", "variable"):
140 traces.append(_node_trace(graph, pos, node_type))
141
142 axis = dict(
143 showbackground=False, showticklabels=False, showgrid=False,
144 zeroline=False, title="", showspikes=False,
145 )
146 layout = go.Layout(
147 title=dict(
148 text=(
149 f"Code Graph — {graph.number_of_nodes()} nodes, "
150 f"{graph.number_of_edges()} edges"
151 ),
152 font=dict(color="#c9d1d9", size=18),
153 ),
154 paper_bgcolor=BACKGROUND,
155 plot_bgcolor=BACKGROUND,
156 scene=dict(xaxis=axis, yaxis=axis, zaxis=axis, bgcolor=BACKGROUND),
157 legend=dict(
158 font=dict(color="#c9d1d9"),
159 bgcolor="rgba(22, 27, 34, 0.8)",
160 bordercolor="#30363d",
161 ),
162 margin=dict(l=0, r=0, t=40, b=0),
163 hoverlabel=dict(bgcolor="#161b22", font=dict(color="#c9d1d9")),
164 )
165 return go.Figure(data=traces, layout=layout)

168def render(json_path: Path, html_path: Path) -> None:
169 graph = load_graph(json_path)
170 figure = build_figure(graph)
171 html_path = Path(html_path)
172 html_path.parent.mkdir(parents=True, exist_ok=True)
173 figure.write_html(str(html_path), include_plotlyjs=True)
174 log.info("wrote %s", html_path)
