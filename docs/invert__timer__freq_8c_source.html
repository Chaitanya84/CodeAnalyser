28//
29// Convert the timer frequency (given in Hz) into a timer period and
30// store the result into the qtime structure with the highest possible
31// precision. This code is system independent and should not have to
32// be changed by end users.
33//
34void
35invert_timer_freq(struct qtime_entry *qtime, unsigned freq) {
36 int scale;
37 unsigned period;
38 unsigned rem;
39 unsigned div;
40 unsigned quot;
41
42 if(freq == 0) {
43 crash("Timer frequency is zero\n");
44 }
45 quot = 1;
46 scale = 0;
47 period = 0;
48
49 /* to ensure (quot = rem * 10) doesn't overflow ... */
50 if (freq > (UINT_MAX/10)) {
51 --scale;
52 freq /= 10;
53 }
54
55 for(;;) {
56 div = quot / freq;
57 period = period * 10 + div;
58 if(period >= ((UINT_MAX/10) - 10)) break;
59 rem = quot % freq;
60 if(rem == 0) break;
61 --scale;
62 quot = rem * 10;
63 }
64 qtime->timer_period = period;
65 qtime->timer_scale = scale;
66}
