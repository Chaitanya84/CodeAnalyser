64// Parse Card Identification Data
65static int mmc_parse_cid( sdio_dev_t * const dev, sdio_cid_t *cid, uint32_t * const rsp )
66{
67 int idx;
68 int status;
69
70 memset( cid, 0, sizeof( sdio_cid_t ) );
71
72 status = EOK;
73
74 switch( dev->csd.spec_vers ) {
75 case CSD_SPEC_VER_0:
76 case CSD_SPEC_VER_1:
77 cid->mid = sdio_extract_bits( rsp, 128, 104, 24 );
78 for( idx = 0; idx < 7; idx++ ) {
79 cid->pnm[idx] = (char)sdio_extract_bits( rsp, 128, 96 - idx * 8, 8 );
80 }
81 cid->prv = sdio_extract_bits( rsp, 128, 40, 8 );
82 cid->month = sdio_extract_bits( rsp, 128, 12, 4 );
83 cid->year = sdio_extract_bits( rsp, 128, 8, 4 ) + 1997;
84 break;
85
86 case CSD_SPEC_VER_2:
87 case CSD_SPEC_VER_3:
88 case CSD_SPEC_VER_4:
89 cid->mid = sdio_extract_bits( rsp, 128, 120, 8 );
90 cid->oid = sdio_extract_bits( rsp, 128, 104, 8 );
91 for( idx = 0; idx < 6; idx++ ) {
92 cid->pnm[idx] = (char)sdio_extract_bits( rsp, 128, 96 - idx * 8, 8 );
93 }
94
95 cid->prv = sdio_extract_bits( rsp, 128, 48, 8 );
96 cid->psn = sdio_extract_bits( rsp, 128, 16, 32 );
97 cid->month = sdio_extract_bits( rsp, 128, 12, 4 );
98 cid->year = sdio_extract_bits( rsp, 128, 8, 4 ) + 1997;
99 break;
100
101 default:
102 status = EINVAL;
103 break;
104 }
105
106 return( status );
107}

109// Parse Card Specfic Data
110static int mmc_parse_csd( sdio_dev_t * const dev, sdio_csd_t *csd, uint32_t * const rsp )
111{
112 uint32_t blksz;
113 uint32_t csize;
114 uint32_t csizem;
115 static const uint32_t mmc_tran_speed_fu[] = { 10000, 100000, 1000000, 10000000, 0, 0, 0, 0 };
116 static const uint32_t mmc_tran_speed_mf[] = { 0, 10, 12, 13, 15, 20, 25, 30,
117 35, 40, 45, 50, 55, 60, 70, 80 };
118
119 memset( csd, 0, sizeof( sdio_csd_t ) );
120 csd->csd_structure = (uint8_t)sdio_extract_bits( rsp, 128, 126, 2 );
121 csd->spec_vers = (uint8_t)sdio_extract_bits( rsp, 128, 122, 4 );
122 csd->taac = (uint8_t)sdio_extract_bits( rsp, 128, 112, 8 );
123 csd->nsac = (uint8_t)sdio_extract_bits( rsp, 128, 104, 8 );
124 csd->tran_speed = (uint8_t)sdio_extract_bits( rsp, 128, 96, 8 );
125 csd->ccc = (uint16_t)sdio_extract_bits( rsp, 128, 84, 12 );
126 csd->read_bl_len = (uint8_t)sdio_extract_bits( rsp, 128, 80, 4 );
127 csd->read_bl_partial = (uint8_t)sdio_extract_bits( rsp, 128, 79, 1 );
128 csd->write_blk_misalign = (uint8_t)sdio_extract_bits( rsp, 128, 78, 1 );
129 csd->read_blk_misalign = (uint8_t)sdio_extract_bits( rsp, 128, 77, 1 );
130 csd->dsr_imp = (uint8_t)sdio_extract_bits( rsp, 128, 76, 1 );
131 csd->c_size = sdio_extract_bits( rsp, 128, 62, 12 );
132 csd->vdd_r_curr_min = (uint8_t)sdio_extract_bits( rsp, 128, 59, 3 );
133 csd->vdd_r_curr_max = (uint8_t)sdio_extract_bits( rsp, 128, 56, 3 );
134 csd->vdd_w_curr_min = (uint8_t)sdio_extract_bits( rsp, 128, 53, 3 );
135 csd->vdd_w_curr_max = (uint8_t)sdio_extract_bits( rsp, 128, 50, 3 );
136 csd->c_size_mult = (uint8_t)sdio_extract_bits( rsp, 128, 47, 3 );
137 switch( csd->csd_structure ) {
138 case CSD_STRUCT_VER_10:
139 case CSD_STRUCT_VER_11:
140 csd->erase_grp_mult = 1;
141 csd->sector_size = (uint8_t)sdio_extract_bits( rsp, 128, 42, 5 );
142 csd->erase_grp_size = (uint8_t)sdio_extract_bits( rsp, 128, 37, 5 );
143 break;
144
145 case CSD_STRUCT_VER_12: // 4.1 - 4.3
146 case CSD_STRUCT_VER_EXT_CSD:
147 csd->erase_grp_mult = (uint8_t)sdio_extract_bits( rsp, 128, 37, 5 );
148 csd->erase_grp_size = (uint8_t)sdio_extract_bits( rsp, 128, 42, 5 );
149 break;
150
151 default:
152 break;
153 }
154
155 csd->wp_grp_size = (uint8_t)sdio_extract_bits( rsp, 128, 32, 5 );
156 csd->wp_grp_enable = (uint8_t)sdio_extract_bits( rsp, 128, 31, 1 );
157// csd->default_ecc = (uint8_t)sdio_extract_bits( rsp, 128, 29, 2 );
158 csd->r2w_factor = (uint8_t)sdio_extract_bits( rsp, 128, 26, 3 );
159 csd->write_bl_len = (uint8_t)sdio_extract_bits( rsp, 128, 22, 4 );
160 csd->write_bl_partial = (uint8_t)sdio_extract_bits( rsp, 128, 21, 1 );
161
162// csd->content_prot_app = (uint8_t)sdio_extract_bits( rsp, 128, 16, 1 );
163// csd->file_format_grp = (uint8_t)sdio_extract_bits( rsp, 128, 15, 1 );
164// csd->copy = (uint8_t)sdio_extract_bits( rsp, 128, 14, 1 );
165 csd->write_protect = (uint8_t)sdio_extract_bits( rsp, 128, 12, 2 );
166// csd->file_format = (uint8_t)sdio_extract_bits( rsp, 128, 10, 2 );
167// csd->ecc = (uint8_t)sdio_extract_bits( rsp, 128, 8, 2 );
168
169 blksz = 1 << csd->read_bl_len;
170 csize = csd->c_size + 1;
171 csizem = 1 << ( csd->c_size_mult + 2 );
172
173 // force to 512 byte block
174 if( ( blksz > SDIO_DFLT_BLKSZ ) && ( ( blksz % SDIO_DFLT_BLKSZ ) == 0 ) ) {
175 csize = csize * ( blksz / SDIO_DFLT_BLKSZ );
176 blksz = SDIO_DFLT_BLKSZ;
177 }
178
179 csd->blksz = blksz;
180 csd->sectors = csize * csizem;
181 csd->dtr_max = mmc_tran_speed_fu[csd->tran_speed & 0x7] *
182 mmc_tran_speed_mf[(csd->tran_speed >> 3 ) & 0xf];
183
184 return( EOK );
185}

187// Parse Extended Card Specfic Data
188static int mmc_parse_ext_csd( sdio_dev_t *dev, sdio_ecsd_t *ecsd, uint8_t *raw_ecsd )
189{
190 sdio_hc_t *hc;
191
192 memset( ecsd, 0, sizeof( sdio_ecsd_t ) );
193
194 hc = dev->hc;
195 dev->pactive = 0;
196 ecsd->ext_csd_rev = raw_ecsd[ECSD_REV];
197 ecsd->blksz = SDIO_DFLT_BLKSZ;
198
199 dev->caps |= DEV_CAP_CMD23;
200
201 if( ecsd->ext_csd_rev >= ECSD_REV_V4_2 ) {
202 ecsd->sectors = ( raw_ecsd[ECSD_SEC_CNT + 0] << 0 ) |
203 ( raw_ecsd[ECSD_SEC_CNT + 1] << 8 ) |
204 ( raw_ecsd[ECSD_SEC_CNT + 2] << 16 ) |
205 ( raw_ecsd[ECSD_SEC_CNT + 3] << 24 );
206 if( ecsd->sectors > ECSD_SEC_CNT_2GB ) {
207 dev->caps |= DEV_CAP_HC;
208 }
209 }
210
211 if( ecsd->ext_csd_rev >= ECSD_REV_V4_3 ) {
212 // boot partitions
213 ecsd->part_config = raw_ecsd[ECSD_PART_CONFIG] & ECSD_PC_ACCESS_MSK;
214 dev->pactive = ecsd->part_config;
215
216 if( raw_ecsd[ECSD_ACC_SIZE] ) {
217 ecsd->acc_size = 512 * ( 1 << ( raw_ecsd[ECSD_ACC_SIZE] - 1 ) );
218 }
219
220 // write reliable
221 dev->rel_wr_sec_c = raw_ecsd[ECSD_REL_WR_SEC_C];
222
223#ifdef MMC_REL_WR_LEGACY_SUP
224 if( ( raw_ecsd[ECSD_WR_REL_PARAM] & ECSD_WRP_EN_REL_WR ) || dev->rel_wr_sec_c ) {
225#else
226 if( ( raw_ecsd[ECSD_WR_REL_PARAM] & ECSD_WRP_EN_REL_WR ) ) {
227#endif
228 dev->caps |= DEV_CAP_WR_REL;
229
230 if( ( raw_ecsd[ECSD_WR_REL_PARAM] & ECSD_WRP_EN_REL_WR ) ) {
231 dev->caps |= DEV_CAP_WR_REL_ENH;
232 }
233 }
234
235 // sleep modes
236 dev->caps |= DEV_CAP_SLEEP;
237
238 if( raw_ecsd[ECSD_S_A_TIMEOUT] && (raw_ecsd[ECSD_S_A_TIMEOUT] <= 0x17) ) {
239 ecsd->s_a_timeout = 1 << raw_ecsd[ECSD_S_A_TIMEOUT];
240 }
241 }
242
243 if( ecsd->ext_csd_rev >= ECSD_REV_V4_4 ) {
244 // RPMB
245 // DDR dual data rate
246
247 // TRIM
248 if( raw_ecsd[ECSD_SEC_FEATURE_SUPPORT] & ECSD_SEC_GB_CL_EN ) {
249 dev->caps |= DEV_CAP_TRIM;
250 }
251
252 // SECURE
253 if( raw_ecsd[ECSD_SEC_FEATURE_SUPPORT] & ECSD_SEC_ER_EN ) {
254 dev->caps |= DEV_CAP_SECURE;
255 }
256 }
257
258 if( ecsd->ext_csd_rev >= ECSD_REV_V4_41 ) {
259 // GP partitions
260
261 // BKOPS
262 if( ( raw_ecsd[ECSD_BKOPS_SUPPORTED] & ECSD_BKOPS_SUP ) ) {
263 dev->caps |= DEV_CAP_BKOPS;
264 ecsd->bkops_en = raw_ecsd[ECSD_BKOPS_EN];
265 if( ( ecsd->bkops_en & ECSD_BKOPS_ENABLE ) ) {
266 dev->flags |= DEV_FLAG_BKOPS;
267 }
268 }
269
270 // HPI
271 if( ( raw_ecsd[ECSD_HPI_FEATURES] & EXT_HPI_FEATURES_SUP_CMD12 ) ) {
272 dev->caps |= DEV_CAP_HPI_CMD12;
273 }
274
275 // MDT handle 2012 roll over
276 if( dev->cid.year < MDT_YEAR_2010 ) {
277 dev->cid.year += ( MDT_YEAR_2013 - MDT_YEAR_1997 );
278 }
279 }
280
281 if( ecsd->ext_csd_rev >= ECSD_REV_V4_5 ) {
282 // HS200
283 // context management
284 // packed commands
285 // exception events
286 // cache
287 // dynamic capacity management
288 // large sector size
289 // power off notification
290
291 if( ( raw_ecsd[ECSD_USE_NATIVE_SECTOR] & ECSD_USE_NATIVE_SECTOR_EN ) ) {
292 switch( ( raw_ecsd[ECSD_NATIVE_SECTOR_SIZE] & ECSD_NATIVE_SECTOR_MSK ) ) {
293 case ECSD_NATIVE_SECTOR_512:
294 ecsd->blksz = SDIO_BLKSZ_512;
295 break;
296
297 case ECSD_NATIVE_SECTOR_4K:
298 ecsd->blksz = SDIO_BLKSZ_4K;
299 break;
300
301 default:
302 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: Invalid ECSD_NATIVE_SECTOR_SIZE 0x%x", __FUNCTION__, raw_ecsd[ECSD_NATIVE_SECTOR_SIZE] );
303 return( EINVAL );
304 }
305 }
306
307 if( raw_ecsd[ECSD_CACHE_SIZE + 0] || raw_ecsd[ECSD_CACHE_SIZE + 1] || raw_ecsd[ECSD_CACHE_SIZE + 2] || raw_ecsd[ECSD_CACHE_SIZE + 3] ) {
308 dev->caps |= DEV_CAP_CACHE;
309 }
310
311 ecsd->driver_strength = raw_ecsd[ECSD_DRIVER_STRENGTH];
312 if( !( ecsd->driver_strength & ( 1 << hc->drv_type ) ) ) {
313 hc->drv_type = 0; // fall back to default value
314 }
315
316 if( raw_ecsd[ECSD_SEC_FEATURE_SUPPORT] & ECSD_SEC_SANITIZE ) {
317 dev->caps |= DEV_CAP_SANITIZE;
318 }
319 dev->caps |= DEV_CAP_DISCARD; // DISCARD support
320 dev->caps |= DEV_CAP_PWROFF_NOTIFY; // Power off notification support
321 }
322
323 if( ecsd->ext_csd_rev >= ECSD_REV_V5 ) {
324 // HS400
325 }
326
327 if( ecsd->ext_csd_rev >= ECSD_REV_V5_1 ) {
328 // Enhanced strobe in HS400 mode - HS400ES
329 ecsd->strobe = raw_ecsd[ECSD_STROBE_SUPPORT];
330 if( ( hc->caps & HC_CAP_HS400ES ) && ( ecsd->strobe ) ) {
331 dev->caps |= DEV_CAP_HS400ES;
332 }
333 // Auto BKOPS
334 if( ( dev->caps & DEV_CAP_BKOPS ) ) {
335 dev->caps |= DEV_CAP_BKOPS_AUTO;
336 }
337 }
338
339 ecsd->card_type = raw_ecsd[ECSD_CARD_TYPE] & ECSD_CARD_TYPE_MSK;
340
341 if( ( hc->caps & ( HC_CAP_HS400 | HC_CAP_HS400ES ) ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS400 ) ) {
342 if( ( ( hc->caps & HC_CAP_SV_1_8V ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS400_1_8V ) ) ||
343 ( ( hc->caps & HC_CAP_SV_1_2V ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS400_1_2V ) ) ) {
344 dev->caps |= DEV_CAP_HS400 | DEV_CAP_HS;
345 ecsd->dtr_max_hs = DTR_MAX_HS400;
346 }
347 }
348
349 if( ( hc->caps & HC_CAP_HS200 ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS200 ) ) {
350 if( ( ( hc->caps & HC_CAP_SV_1_8V ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS200_1_8V ) ) ||
351 ( ( hc->caps & HC_CAP_SV_1_2V ) && ( ecsd->card_type & ECSD_CARD_TYPE_HS200_1_2V ) ) ) {
352 dev->caps |= DEV_CAP_HS200 | DEV_CAP_HS;
353 ecsd->dtr_max_hs = DTR_MAX_HS200;
354 }
355 }
356
357 if( !( dev->caps & DEV_CAP_HS ) && ( hc->caps & HC_CAP_HS ) && ( ecsd->card_type & ECSD_CARD_TYPE_52MHZ ) ) {
358 dev->caps |= DEV_CAP_HS;
359 ecsd->dtr_max_hs = DTR_MAX_HS52;
360 }
361
362 if( !( dev->caps & DEV_CAP_HS ) ) {
363 ecsd->dtr_max_hs = DTR_MAX_HS26;
364 }
365
366 if( ( hc->caps & HC_CAP_DDR50 ) && ( ecsd->card_type & ECSD_CARD_TYPE_DDR ) ) {
367 if( ( ( hc->caps & ( HC_CAP_SV_3_0V | HC_CAP_SV_1_8V ) ) && ( ecsd->card_type & ECSD_CARD_TYPE_DDR_1_8V ) ) ||
368 ( ( hc->caps & HC_CAP_SV_1_2V ) && ( ecsd->card_type & ECSD_CARD_TYPE_DDR_1_2V ) ) ) {
369 dev->caps |= DEV_CAP_DDR50;
370 }
371 }
372
373 if( raw_ecsd[ECSD_PARTITION_SWITCH_TIME] == 0 ) {
374 raw_ecsd[ECSD_PARTITION_SWITCH_TIME] = 100; // defualt to 1 second (10ms * 100)
375 }
376
377 if( raw_ecsd[ECSD_OUT_OF_INTERRUPT_TIME] == 0 ) {
378 raw_ecsd[ECSD_OUT_OF_INTERRUPT_TIME] = 100; // default to 1 second (10ms * 100)
379 }
380
381 ecsd->erase_grp_def = raw_ecsd[ECSD_ERASE_GRP_DEF];
382 ecsd->hc_erase_group_size = raw_ecsd[ECSD_ERASE_GRP_SIZE];
383 ecsd->hc_wp_grp_size = raw_ecsd[ECSD_HC_WP_GRP_SIZE];
384 ecsd->user_wp = raw_ecsd[ECSD_USER_WP];
385
386 return( EOK );
387}

421static int _mmc_switch( sdio_dev_t * const dev, uint32_t const cmdset, uint32_t const mode, uint32_t const index, uint32_t const value, uint32_t const timeout, uint32_t const wrdy )
422{
423 sdio_cmd_t *cmd;
424 int status;
425 uint32_t rsp;
426
427 cmd = sdio_alloc_cmd( );
428 if( cmd == NULL ) {
429 return( ENOMEM );
430 }
431
432 rsp = ( timeout ) ? SCF_RSP_R1B : SCF_RSP_R1;
433
434 sdio_setup_cmd( cmd, SCF_CTYPE_AC | rsp, MMC_SWITCH, ( mode << 24 ) | ( index << 16 ) | ( value << 8 ) | cmdset );
435
436 status = _sdio_send_cmd( dev, cmd, NULL, timeout, SDIO_CMD_RETRIES );
437 if( status == EOK ) {
438 // SanDisk errata. The parts can't handle
439 // commands within 100us after the switch.
440 delay( 1 );
441 if( wrdy ) {
442 status = _sdio_wait_card_status( dev, NULL, CDS_READY_FOR_DATA, CDS_READY_FOR_DATA, timeout );
443 }
444 }
445
446 sdio_free_cmd( cmd );
447
448 return( status );
449}

456// RPMB partition read write
457int mmc_rpmb_rw( sdio_dev_t * const dev, void * const pf, uint32_t const nf, uint32_t const flgs )
458{
459 sdio_hc_t *hc;
460 struct sdio_cmd *cmd;
461 void *buf;
462 sdio_sge_t sge;
463 uint32_t flags;
464 uint32_t sbcflg = 0;
465 int op;
466 uint64_t acmd_caps;
467 int status;
468
469 hc = dev->hc;
470
471 cmd = sdio_alloc_cmd( );
472 if( cmd == NULL ) {
473 return( ENOMEM );
474 }
475
476 buf = sdio_alloc( RPMB_FRAME_SIZE * nf );
477 if( buf == NULL ) {
478 sdio_free_cmd( cmd );
479 return( ENOMEM );
480 }
481
482 if( flgs == SCF_DIR_IN ) {
483 op = MMC_READ_MULTIPLE_BLOCK;
484 flags = SCF_CTYPE_ADTC | SCF_DIR_IN | SCF_RSP_R1;
485 }
486 else {
487 if( ( flgs & SCF_SBC_RLW ) ) {
488 sbcflg = SBC_RL_WRITE;
489 }
490 op = MMC_WRITE_MULTIPLE_BLOCK;
491 flags = SCF_CTYPE_ADTC | SCF_DIR_OUT | SCF_RSP_BUSY | SCF_RSP_R1;
492 memcpy( buf, pf, nf * RPMB_FRAME_SIZE );
493 }
494
495 status = _sdio_set_block_count( dev, nf, sbcflg );
496 if( status == EOK ) {
497 // We have to temporarily disable ACMD12/ACMD23, otherwise HC will try
498 // to use ACMD12/ACMD23 feature, which messes up the command sequence
499 acmd_caps = hc->caps & ( HC_CAP_ACMD23 | HC_CAP_ACMD12 );
500 hc->caps &= ~( HC_CAP_ACMD23 | HC_CAP_ACMD12 );
501
502 sge.sg_count = RPMB_FRAME_SIZE * nf;
503 sge.sg_address = (paddr_t)buf;
504 sdio_setup_cmd( cmd, flags, op, 0 );
505 sdio_setup_cmd_io( cmd, flags, nf, RPMB_FRAME_SIZE, &sge, 1, NULL );
506 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT * 5, 0 );
507
508 // Restore ACMD12/ACMD23 flag
509 hc->caps |= acmd_caps;
510
511 if( ( status == EOK ) && ( flgs == SCF_DIR_IN ) ) {
512 memcpy( pf, buf, nf * RPMB_FRAME_SIZE );
513 }
514 }
515 else {
516 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: set_block_count failed", __func__ );
517 }
518
519 sdio_free( buf, nf * RPMB_FRAME_SIZE );
520 sdio_free_cmd( cmd );
521
522 return( status );
523}

525int mmc_gen_man_cmd( sdio_dev_t * const dev, const uint8_t op, void * const buf, const uint32_t blklen, const uint32_t blkcnt, const uint32_t arg, const uint32_t flgs)
526{
527 sdio_hc_t *hc;
528 struct sdio_cmd *cmd;
529 sdio_sge_t sge;
530 void *dbuf;
531 int status;
532
533 hc = dev->hc;
534 dbuf = NULL;
535
536 if( blkcnt > 1 ) return ( ENOTSUP ); // We only support maximum one block data
537
538 cmd = sdio_alloc_cmd( );
539 if( cmd == NULL ) {
540 return( ENOMEM );
541 }
542
543 sdio_setup_cmd( cmd, flgs & ~SCF_DATA_MSK, op, arg );
544
545 if( ( flgs & SCF_CTYPE_ADTC ) ) {
546 if( blklen != SDIO_DFLT_BLKSZ ) {
547 status = _sdio_set_block_length( dev, blklen );
548 if( status != EOK ) {

550 sdio_free_cmd( cmd );

566 sge.sg_address = (paddr_t)dbuf;

569 }
570
571 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT * 5, 0 );
572 if( ( flgs & SCF_CTYPE_ADTC ) ) {
573 if( ( status == EOK ) && ( flgs & SCF_DIR_IN ) ) {
574 memcpy( buf, dbuf, blklen * blkcnt );

577 if( blklen != SDIO_DFLT_BLKSZ ) {
578 status = _sdio_set_block_length( dev, SDIO_DFLT_BLKSZ );
579 if( status != EOK ) {
580 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: sdio_set_block_length", __FUNCTION__ );
581 }
582 }
583 }
584
585 sdio_free_cmd( cmd );

688uint64_t mmc_erase_timeout( sdio_dev_t * const dev, const uint32_t etype, const uint64_t nlba )
689{
690 uint8_t *ecsd;
691 uint32_t tmult;
692 uint64_t timeout;
693 uint64_t erase_grp_size;
694
695 ecsd = dev->raw_ecsd;
696 tmult = MMC_ERASE_TIMEOUT;
697 erase_grp_size = mmc_erase_grp_size( dev ) / 512;
698
699 if( dev->ecsd.erase_grp_def ) {
700 switch( etype ) {
701 case MMC_ERASE_NORM:
702 tmult = MMC_ERASE_TIMEOUT * ecsd[ECSD_ERASE_MULT];
703 break;
704
705 case MMC_ERASE_TRIM:
706 case MMC_ERASE_DISCARD:
707 tmult = MMC_ERASE_TIMEOUT * ecsd[ECSD_TRIM_MULT];
708 break;
709
710 case MMC_ERASE_SECURE:
711 tmult = MMC_ERASE_TIMEOUT * ecsd[ECSD_ERASE_MULT] * ecsd[ECSD_SEC_ERASE_MULT];
712 break;
713
714 case MMC_ERASE_SECURE_TRIM:
715 case MMC_ERASE_SECURE_TRIM_PURGE:
716 tmult = MMC_ERASE_TIMEOUT * ecsd[ECSD_ERASE_MULT] * ecsd[ECSD_SEC_TRIM_MULT];
717 break;
718
719 default:
720 tmult = MMC_ERASE_TIMEOUT * ecsd[ECSD_ERASE_MULT];
721 break;
722 }
723 }
724
725 // +2 for potential TRIM/DISCARD aligment
726 timeout = tmult * ( ( nlba / erase_grp_size ) + 2 );
727
728 timeout = max( timeout, MMC_ERASE_TIMEOUT_MIN );
729
730 return( timeout );
731}

733int mmc_erase( sdio_dev_t * const dev, const uint32_t partition, const uint32_t flgs, uint64_t lba, uint32_t nlba )
734{
735 sdio_cmd_t *cmd;
736 int status;
737 uint64_t timeout;
738 uint64_t addr;
739
740 status = mmc_set_partition( dev, partition );
741 if( status != EOK ) {
742 return( status );
743 }
744
745 cmd = sdio_alloc_cmd( );
746 if( cmd == NULL ) {
747 return( ENOMEM );
748 }
749
750 while( nlba ) {
751 addr = ( dev->caps & DEV_CAP_HC ) ? lba : ( lba * dev->csd.blksz );
752 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1, MMC_TAG_ERASE_GROUP_START, addr );
753 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
754 if( status != EOK ) {
755 break;
756 }
757
758 timeout = mmc_erase_timeout( dev, flgs, nlba );
759 lba += nlba - 1;
760 nlba -= nlba;
761
762 addr = ( dev->caps & DEV_CAP_HC ) ? lba : ( lba * dev->csd.blksz );
763 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1, MMC_TAG_ERASE_GROUP_END, addr );
764 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
765 if( status != EOK ) {
766 break;
767 }
768
769 sdio_setup_cmd( cmd, SCF_CTYPE_AC | SCF_RSP_R1B | SCF_WAIT_DRDY, MMC_ERASE, flgs );
770 status = _sdio_send_cmd( dev, cmd, NULL, timeout, 0 );
771 if( status != EOK ) {
772 break;
773 }
774
775 status = _sdio_wait_card_status( dev, NULL, CDS_READY_FOR_DATA | CDS_CUR_STATE_MSK, CDS_READY_FOR_DATA | CDS_CUR_STATE_TRAN, timeout );
776 if( status != EOK ) {
777 break;
778 }
779 }
780
781 sdio_free_cmd( cmd );
782
783 return( status );
784}

883static int mmc_bus_test( sdio_dev_t * const dev, const uint32_t width )
884{
885 sdio_cmd_t *cmd;
886 sdio_sge_t sge;
887 int idx;
888 uint8_t *data;
889 int status;
890 uint8_t const *pattern;
891 static const uint8_t pattern8[8] = { 0x55, 0xaa, 0, 0, 0, 0, 0, 0 };
892 static const uint8_t pattern4[4] = { 0x5a, 0, 0, 0 };
893
894 cmd = sdio_alloc_cmd( );
895 if( cmd == NULL ) {
896 return( ENOMEM );
897 }
898
899 data = sdio_alloc( width );
900 if( data == NULL ) {
901 sdio_free_cmd( cmd );
902 return( ENOMEM );
903 }
904
905 pattern = ( width == 8 ) ? pattern8 : pattern4;
906 sge.sg_count = width;
907 sge.sg_address = SDIO_DATA_PTR_P( data );
908 memcpy( data, pattern, width );
909 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, MMC_BUSTEST_W, 0 );
910 sdio_setup_cmd_io( cmd, SCF_DIR_OUT, 1, width, &sge, 1, NULL );
911 _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
912
913 sdio_setup_cmd( cmd, SCF_CTYPE_ADTC | SCF_RSP_R1, MMC_BUSTEST_R, 0 );
914 sdio_setup_cmd_io( cmd, SCF_DIR_IN, 1, width, &sge, 1, NULL );
915 status = _sdio_send_cmd( dev, cmd, NULL, SDIO_TIME_DEFAULT, 0 );
916 if( status == EOK ) {
917 for( idx = 0; idx < width / 4; idx++ ) {
918 if( ( pattern[idx] ^ data[idx] ) != 0xff ) {
919 status = EIO;
920 break;
921 }
922 }
923 }
924
925 sdio_free( data, width );
926 sdio_free_cmd( cmd );
927 return( status );
928}

1387static int mmc_init_bus( sdio_hc_t *hc )
1388{
1389 sdio_dev_t *dev;
1390 uint32_t bus_width;
1391 int status;
1392
1393 dev = &hc->device;
1394
1395 bus_width = mmc_bus_width( dev, 0 );
1396
1397 // enable HS400ES if supported
1398 if( ( dev->caps & DEV_CAP_HS400ES ) && ( dev->ecsd.dtr_max_hs > DTR_MAX_HS52 ) && bus_width >= BUS_WIDTH_8 ) {
1399 status = mmc_init_hs400es( hc, bus_width );
1400 }
1401
1402 // enable HS400 if supported
1403 if( !( dev->flags & DEV_FLAG_HS400ES ) && ( dev->caps & DEV_CAP_HS400 ) && ( dev->ecsd.dtr_max_hs > DTR_MAX_HS52 ) && bus_width >= BUS_WIDTH_8 ) {
1404 status = mmc_init_hs400( hc, bus_width );
1405 }
1406
1407 // enable HS200 if supported
1408 if( !( dev->flags & ( DEV_FLAG_HS400 | DEV_FLAG_HS400ES ) ) && ( dev->caps & DEV_CAP_HS200 ) && ( dev->ecsd.dtr_max_hs > DTR_MAX_HS52 ) && bus_width >= BUS_WIDTH_4 ) {
1409 status = mmc_init_hs200( hc, bus_width );
1410 }
1411
1412 // enable HS if supported
1413 if( !( dev->flags & ( DEV_FLAG_HS200 | DEV_FLAG_HS400 | DEV_FLAG_HS400ES ) ) && ( dev->caps & DEV_CAP_HS ) ) {
1414 status = mmc_init_hs( hc );
1415 }
1416
1417 if( ( dev->flags & DEV_FLAG_HS ) ) {
1418 if( bus_width >= BUS_WIDTH_4 ) {
1419 if( ( dev->caps & DEV_CAP_DDR50 ) ) { // enable DDR50 if supported
1420 status = mmc_init_ddr( hc, bus_width );
1421 }
1422
1423 if( !( dev->flags & DEV_FLAG_DDR ) ) { // enable 4/8 bit mode
1424 status = mmc_switch( dev, MMC_SWITCH_CMDSET_DFLT, MMC_SWITCH_MODE_WRITE, ECSD_BUS_WIDTH, bus_width / 4, SDIO_TIME_DEFAULT );
1425 if( status != EOK ) {
1426 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: switch ECSD_BUS_WIDTH %x", __FUNCTION__, bus_width / 4 );
1427 }
1428 else {
1429 sdio_bus_width( hc, bus_width );
1430 }
1431 }
1432 }
1433 }
1434
1435 if( !( dev->flags & ( DEV_FLAG_HS400ES | DEV_FLAG_HS400 | DEV_FLAG_HS200 | DEV_FLAG_HS ) ) ) {
1436 // we either have a device where dev->csd.spec_vers < CSD_SPEC_VER_4
1437 // or all higher modes have failed, so set set max ls timing
1438 sdio_timing( hc, TIMING_LS );
1439 sdio_clock( hc, dev->csd.dtr_max );
1440 }
1441
1442 status = _sdio_set_block_length( dev, SDIO_DFLT_BLKSZ );
1443 if( status != EOK ) {
1444 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: sdio_set_block_length", __FUNCTION__ );
1445 return( status );
1446 }
1447
1448 return( status );
1449}
