14# Extensions analyzed by the tool. Everything else is skipped.

17# Files with these extensions use the C grammar; all others use C++.
18C_EXTENSIONS = {".c"}

20# Directories never entered. Extend freely.
21IGNORED_DIRECTORIES = {
22 ".git",
23 ".vscode",
24 ".idea",
25 "__pycache__",

30# Directory names equal to a prefix or starting with "<prefix>-" are ignored

32IGNORED_DIRECTORY_PREFIXES = ("build", "cmake-build")

34IGNORED_FILES = {"Makefile", "makefile", "CMakeLists.txt", ".gitignore"}
35IGNORED_SUFFIXES = {".o", ".a", ".so", ".dll", ".exe"}

38def is_ignored_dir(name: str) -> bool:
39 if name in IGNORED_DIRECTORIES:
40 return True
41 for prefix in IGNORED_DIRECTORY_PREFIXES:
42 if name == prefix or name.startswith(prefix + "-"):
43 return True
44 return False

47def is_ignored_file(name: str) -> bool:
48 if name in IGNORED_FILES:
49 return True

58def scan(root: Path) -> List[Path]:
59 """Recursively collect analyzable source files under *root*.
60
61 Returns a sorted list (deterministic ordering). Unreadable directories
62 are logged and skipped rather than aborting the scan.
63 """
64 root = root.resolve()
65 collected: List[Path] = []
66 stack = [root]
67 while stack:
68 current = stack.pop()
69 try:
70 entries = sorted(current.iterdir(), key=lambda p: p.name)
71 except OSError as exc:
72 log.warning("cannot read directory %s: %s", current, exc)
73 continue
74 for entry in entries:
75 try:
76 if entry.is_dir() and not entry.is_symlink():
77 if not is_ignored_dir(entry.name):
78 stack.append(entry)
79 elif entry.is_file():
80 if is_ignored_file(entry.name):
81 continue
82 if entry.suffix.lower() in SOURCE_EXTENSIONS:
83 collected.append(entry)
84 except OSError as exc:
85 log.warning("cannot stat %s: %s", entry, exc)
86 collected.sort(key=lambda p: normalize_relative(p, root))
87 log.info("scanner discovered %d source files under %s", len(collected), root)
88 return collected
