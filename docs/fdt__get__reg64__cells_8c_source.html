50int fdt_get_reg64_cells(const void *fdt, int node, int index, int num,
51 uint64_t *addr, uint64_t *size, int addr_cells, int size_cells) {
52 int len;
53 const fdt32_t *regprop;
54 int ent_cells;
55 int entlen;
56
57 // Find "reg" property in node
58 regprop = fdt_getprop(fdt, node, "reg", &len);
59 if(regprop == NULL) {
60 return len;
61 }
62
63 // Get the number of 32 bit cells for address and size
64 if(addr_cells == -1) {
65 // Finding parent requires scanning the complete device tree
66 const int parent = fdt_parent_offset(fdt, node);
67 if(parent < 0) {
68 return parent;
69 }
70 addr_cells = fdt_address_cells(fdt, parent);
71 size_cells = fdt_size_cells(fdt, parent);
72 }
73
74 // Validate the number of 32 bit cells for address and size
75 if((addr_cells < 1) || (addr_cells > 2)) {
76 return -FDT_ERR_BADNCELLS;
77 }
78 if((size_cells < 0) || (size_cells > 2)) {
79 return -FDT_ERR_BADNCELLS;
80 }
81 ent_cells = addr_cells + size_cells;
82
83 // Calculate number of entries
84 entlen = ent_cells * (int)(sizeof(*regprop));
85 if(len % entlen) {
86 return -FDT_ERR_BADOFFSET;
87 }
88 len /= entlen;
89 if((index < 0) || (index >= len)) {
90 return -FDT_ERR_NOTFOUND;
91 }
92 regprop += index * ent_cells;
93 len -= index;
94 if(!addr) {
95 return len;
96 }
97
98 if((size_cells == 0) != (size == 0)) {
99 return -FDT_ERR_BADNCELLS;
100 }
101
102 if(num > len) {
103 num = len;
104 }
105 while(num-- > 0) {
106 union {
107 fdt32_t cells[2];
108 fdt64_t val;
109 } temp;
110
111 // Get reg entries
112 switch(addr_cells) {
113 case 1:
114 *addr = fdt32_to_cpu(regprop[0]);
115 break;
116 case 2:
117 // Extract as two moves to avoid alignment problems
118 temp.cells[0] = regprop[0];
119 temp.cells[1] = regprop[1];
120 *addr = fdt64_to_cpu(temp.val);
121 break;
122 default:
123 // Never reached, addr_cells were checked above and can only be 1 or
124 // 2.
125 break;
126 }
127 regprop += addr_cells;
128 addr++;
129
130 if(size) {
131 switch(size_cells) {
132 case 0:
133 break;
134 case 1:
135 *size = fdt32_to_cpu(*((fdt32_t *)regprop));
136 break;
137 case 2:
138 // Extract as two moves to avoid alignment problems
139 temp.cells[0] = regprop[0];
140 temp.cells[1] = regprop[1];
141 *size = fdt64_to_cpu(temp.val);
142 break;
143 default:
144 // Never reached, size_cells were checked above and can only be
145 // 0, 1 or 2.
146 break;
147 }
148 regprop += size_cells;
149 size++;
150 }
151 }
152
153 return len;
154}
