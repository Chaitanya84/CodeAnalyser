132void dw_write_fifo(const dw_spi_t *const spi, const uint32_t len) {
133 switch (spi->dscale) {
134 case 0:
135 dw_write_fifo_uint8(spi, ((uint8_t*) spi->pbuf) + spi->tlen, len);
136 spi_slogf(_SLOG_DEBUG3, "%s: 8 bit %u", __func__, spi->dscale);
137 break;
138 case 1:
139 dw_write_fifo_uint16(spi, ((uint16_t*) spi->pbuf) + spi->tlen, len);
140 spi_slogf(_SLOG_DEBUG3, "%s: 16 bit %u", __func__, spi->dscale);
141 break;
142 case 2:
143 dw_write_fifo_uint32(spi, ((uint32_t*) spi->pbuf) + spi->tlen, len);
144 spi_slogf(_SLOG_DEBUG3, "%s: 32 bit %u", __func__, spi->dscale);
145 break;
146 default:
147 spi_slogf(_SLOG_ERROR, "%s: invalid dscale %u", __func__, spi->dscale);
148 break;
149 }

182static int dw_process_interrupts(dw_spi_t *spi) {
183 uint32_t len;
184
185 /* Get interrupt status */
186 const uint32_t value = dw_read32(spi, DW_SPI_SR);
187
188 // Check Interrupt valid
189 if ((value & DW_SPI_SR_TX_ERR) != 0) {
190 /* Disable SPI and mask interrupts */
191 dw_spi_enable(spi, DW_SPI_DISABLE);
192 dw_write32(spi, DW_SPI_IMR, 0);
193
194 spi_slogf(_SLOG_ERROR, "%s: Transmit error", __func__);
195 return SPI_INTR_ERR;
196 }
197 if ((value & DW_SPI_SR_DCOL) != 0) {
198 /* Disable SPI and mask interrupts */
199 dw_spi_enable(spi, DW_SPI_DISABLE);
200 dw_write32(spi, DW_SPI_IMR, 0);
201
202 spi_slogf(_SLOG_ERROR, "%s: Data collision error", __func__);
203 return SPI_INTR_ERR;
204 }
205
206 spi_slogf(_SLOG_DEBUG3, "%s: Starting interrupt routine read",
207 __func__);
208 len = dw_read32(spi, DW_SPI_RXFLR);
209 dw_read_fifo(spi, len);
210 spi->rlen += len;
211
212 if (spi->rlen == spi->xlen) {
213 /* Disable SPI and mask interrupts */
214 dw_spi_enable(spi, DW_SPI_DISABLE);
215 dw_write32(spi, DW_SPI_IMR, 0);
216
217 /* There's no more data to send, Exchange done */
218 return SPI_INTR_DONE;
219 }
220
221 if (spi->tlen < spi->xlen) {
222 //Get size of transmit
223 len = min(len, spi->xlen - spi->tlen);
224 spi_slogf(_SLOG_DEBUG3, "%s:Size of fifo: %u, Transmit transaction left: %u",
225 __func__, spi->fifo_len, spi->xlen - spi->tlen);
226
227 dw_write_fifo(spi, len);
228 spi->tlen += len;
229 }
230
231 if (spi->tlen >= spi->xlen) {
232 const uint32_t im = dw_read32(spi, DW_SPI_IMR);
233 dw_write32(spi, DW_SPI_IMR, im & ~DW_SPI_INT_TXEI & ~DW_SPI_INT_TXOI);
234 const uint32_t minlen = min(spi->xlen - spi->rlen, spi->fifo_len / SPI_FIFO_LEN_DIV);
235 const uint32_t thresh = minlen - 1U;
236 /* When the number of receive FIFO entries is greater than
237 or equal to this value + 1, the receive FIFO full interrupt
238 is triggered.*/
239 dw_write32(spi, DW_SPI_RXFTLR, thresh);
240 }
241
242
243 /* Transfer not complete */
244 /* Don't unblock xfer function until complete */
245 return SPI_INTR_CONTINUE;
246}

254int dw_wait(dw_spi_t *const spi)
255{
256 struct timespec timeout;
257 const uint64_t ns_timeout = spi->xtime_us * 50 * 1000UL; /* timeout in ns. 50 times for transation time */
258 spi_bus_t *const bus = spi->bus;
259 int status;
260#ifdef DW_DMA
261 dw_dma_t *const dma = &spi->dma;
262#endif
263
264 nsec2timespec(&timeout, clock_gettime_mon_ns() + ns_timeout);
265
266 spi_slogf(_SLOG_DEBUG2, "%s: Starting wait function", __func__);
267 while (1) {
268 /* The sem_timedwait_monotonic() function decrements the semaphore on successful completion */
269 status = sem_timedwait_monotonic(bus->sem, &timeout);
270
271 if (status == -1) {
272 spi_slogf(_SLOG_ERROR, "%s: sem_timedwait_monotonic failed: %s", __func__, strerror(errno));
273 status = errno;
274#ifdef DW_DMA
275 if (spi->dma_active) {
276 dma->dma_funcs.xfer_abort(dma->rx_ch_handle);
277 dma->dma_funcs.xfer_abort(dma->tx_ch_handle);
278 }
279#endif
280 break;
281 } else {
282#ifdef DW_DMA
283 if (spi->dma_active) {
284 spi_slogf(_SLOG_DEBUG2, "%s: DMA xfer is done", __func__);
285
286 spi->rlen = spi->xlen - dma->dma_funcs.bytes_left(dma->rx_ch_handle);
287
288 /* xfer_complete is used by some DMA lib to clear INT flags.
289 * It returns EOK on success, -1 on failure. Errors are logged in DMA lib.
290 * */
291 status = dma->dma_funcs.xfer_complete(dma->rx_ch_handle);
292 if (status == -1) {
293 spi_slogf(_SLOG_ERROR, "%s: RX xfer_complete with an error", __func__);
294 status = EIO;
295 break;
296 }
297
298 status = dma->dma_funcs.xfer_complete(dma->tx_ch_handle);
299 if (status == -1) {
300 spi_slogf(_SLOG_ERROR, "%s: TX xfer_complete with an error", __func__);
301 status = EIO;
302 break;
303 }
304 break;
305 }
306 else
307#endif
308 {
309 spi_slogf(_SLOG_DEBUG3, "%s: Processing interrupts", __func__);
310 /* Received SPI interrupt event */
311 status = dw_process_interrupts(spi);
312 InterruptUnmask(spi->bus->irq, spi->iid);
313
314 if (status == SPI_INTR_CONTINUE) {
315 spi_slogf(_SLOG_DEBUG3, "%s: SPI transfer continue ...", __func__);
316 } else if (status == SPI_INTR_DONE) {
317 /* Transfer is completed */
318 spi_slogf(_SLOG_DEBUG2, "%s: SPI transfer is completed", __func__);
319 status = EOK;
320 break;
321 } else {
322 /* Transfer failed */
323 spi_slogf(_SLOG_ERROR, "%s: SPI transfer failed", __func__);
324 status = EIO;
325 break;
326 }
327 }
328 }
329 }
330
331 return status;
332}
