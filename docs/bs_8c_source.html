33/*
34 * Raspberry PI5 regulator "sd_io_1v8_reg" is controlled by gpio "SD_IOVDD_SEL" pin 3 the gpio group defined by
35 * FDT gpio@7d517c00
36 * For SD 3.3v operation, the gpio state is set to 0, for 1.8v operation, the gpio state is set to 1
37 */
38#define BCM2712_AON_GPIO_BASE 0x107d517c00ULL
39#define BCM2712_AON_GPIO_SIZE 0x40UL
40#define BCM2712_AON_GPIO_NUM 38
41#define BCM2712_AON_GPIO_BANK_NUM 2
42#define BCM2712_AON_GPIO_BANK0_GPIO 17
43#define BCM2712_AON_GPIO_BANK1_GPIO 6
44
45#define BCM2712_GIO_DATA (0x04U)
46#define BCM2712_GIO_IODIR (0x08U)
47
48#define BCM2712_GIO_BANK_SIZE (0x20U)
49
50#define GPIO_SD_IOVDD_SEL_LINE (0x3U)
51#define GPIO_SD_IOVDD_SEL_3_3V (0x0U)
52#define GPIO_SD_IOVDD_SEL_1_8V (0x1U)
53

55 uint32_t gpio;
56 uint32_t state;

65static int32_t bcm2712_bs_args(const sdio_hc_t * const hc, char *options)
66{
67 char *value;

69 static char *opts[] = { NULL };
70 int32_t ret = EOK;
71
72 while (options && (*options != '\0')) {
73 opt = sdio_hc_getsubopt(&options, opts, &value);
74 if (opt == -1) {
75 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: invalid BS options %s", __func__, options);
76 ret = EINVAL;
77 break;

79 }
80 return ret;
81}

83static int32_t gpio_state(const sdio_hc_t * const hc, struct gpio_state_t *gpioinfo, const bool set)
84{
85 uint32_t bank, offset;
86 uintptr_t gpio_base;
87 uint32_t gpio_val;
88 int32_t ret = EOK;
89
90 if (gpioinfo == NULL) {
91 ret = -1;
92 }
93 else {
94 bank = gpioinfo->gpio / BCM2712_GIO_BANK_SIZE;
95 offset = gpioinfo->gpio % BCM2712_GIO_BANK_SIZE; // gpio within bank
96
97 gpio_base = (uintptr_t) mmap_device_memory(NULL, BCM2712_AON_GPIO_SIZE, PROT_NOCACHE|PROT_READ|PROT_WRITE, 0, BCM2712_AON_GPIO_BASE);
98 if (gpio_base == (uintptr_t) MAP_FAILED) {
99 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: mmap (GPIO) failed: %s\n", __func__, strerror (errno));
100 ret = -1;
101 }
102 else {
103 gpio_val = in32(gpio_base + (bank*BCM2712_GIO_BANK_SIZE) + BCM2712_GIO_DATA);
104 if (set == false) {
105 if ((gpio_val & (1UL << offset)) != 0UL) {
106 gpioinfo->state = 1;
107 }
108 else {
109 gpioinfo->state = 0;
110 }
111 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_INFO, hc->cfg.verbosity, 1, "%s: gpio: %u gpio_base[0x%x]: 0x%x offset: %u, state %u\n",
112 __func__, gpioinfo->gpio, BCM2712_GIO_DATA, gpio_val, offset, gpioinfo->state);
113 }
114 else {
115 gpio_val = (gpio_val & ~(1UL << offset)) | (gpioinfo->state << offset);
116 out32(gpio_base + (bank*BCM2712_GIO_BANK_SIZE) + BCM2712_GIO_DATA, gpio_val);
117 }
118 (void)munmap_device_memory((void *)gpio_base, BCM2712_AON_GPIO_SIZE);
119 }
120 }
121 return ret;
122}

157/*
158 * This function is a variant of sdhci_signal_voltage, it calls rpi5 specific gpio APIs
159 * to switch between 3.3v and 1.8v operation
160 */
161static int32_t bcm2712_signal_voltage(sdio_hc_t *hc, uint32_t signal_voltage)
162{
163 const struct bcm2712_ext_t * const ext = (struct bcm2712_ext_t *)hc->bs_hdl;
164 const int32_t gpio_sd_iovdd_sel_state = get_vdd_sd_io_state(hc);
165
166 if (signal_voltage == (uint32_t)SIGNAL_VOLTAGE_3_3) {
167 if (gpio_sd_iovdd_sel_state != GPIO_SD_IOVDD_SEL_3_3V) {
168 set_vdd_sd_io_state(hc, GPIO_SD_IOVDD_SEL_3_3V);
169 (void)delay(5);
170 // force ext->signal_voltage to switch to SIGNAL_VOLTAGE_3_3
171 hc->signal_voltage = 0;
172 }
173 }
174 else if (signal_voltage == (uint32_t)SIGNAL_VOLTAGE_1_8) {
175 if (gpio_sd_iovdd_sel_state != GPIO_SD_IOVDD_SEL_1_8V) {
176 set_vdd_sd_io_state(hc, GPIO_SD_IOVDD_SEL_1_8V);
177 (void)delay(5);
178 // force ext->signal_voltage to switch to SIGNAL_VOLTAGE_1_8
179 hc->signal_voltage = 0;
180 }
181 }
182 else {
183 // unknown gpio "SD_IOVDD_SEL" state, do nothing
184 }
185 return (ext->signal_voltage(hc, signal_voltage));
186}

201/*
202 * Pi5 FDT has a non-standard CFG region with offset 0x400 (size=0x200) of host controller
203 * The default map size of 4k of host controller should cover this CFG region
204 */
205static int32_t bcm2712_bs_cfg_init(const sdio_hc_t *hc)
206{
207 int32_t ret = EOK;
208 const sdhci_hc_t *sdhc = (sdhci_hc_t *)hc->cs_hdl;
209 if (sdhc == NULL) {
210 ret = EFAULT;
211 }
212 else {
213 const uintptr_t base = sdhc->base;
214 if ((base == (uintptr_t)MAP_FAILED) || (base == (uintptr_t)NULL)) {
215 ret = ENOMEM;
216 }
217 else {
218 uint32_t rd_val, wr_val;
219 /*
220 * Note:
221 * If driver command argument specifies size less that 0x600, than mmap_device_memory
222 * is needed.
223 */
224 rd_val = in32(base + SDIO_CFG_REG_OFFSET + SDIO_CFG_SD_PIN_SEL_REG);
225 wr_val = rd_val | SDIO_CFG_SD_PIN_SEL_SD;
226 out32(base + SDIO_CFG_REG_OFFSET + SDIO_CFG_SD_PIN_SEL_REG, wr_val);
227
228 /*
229 * Tunning support, select the delay line PHY as the clock source.
230 */
231 rd_val = in32(base + SDIO_CFG_REG_OFFSET + SDIO_CFG_MAX_50MHZ_MODE_REG);
232 wr_val = (rd_val & ~SDIO_CFG_MAX_50MHZ_MODE_ENABLE) | SDIO_CFG_MAX_50MHZ_MODE_STRAP_OVERRIDE;
233 out32(base + SDIO_CFG_REG_OFFSET + SDIO_CFG_MAX_50MHZ_MODE_REG, wr_val);
234 }
235 }
236 return ret;
237}

239static int32_t bcm2712_bs_init(sdio_hc_t *hc)
240{
241 static struct bcm2712_ext_t ext;
242 int32_t status = EOK;
243 const sdio_hc_cfg_t * const cfg = &hc->cfg;
244
245 hc->bs_hdl = (void *)&ext;
246
247 if (bcm2712_bs_args(hc, cfg->options) != EOK) {
248 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: command argument, %s",
249 __func__, strerror(status));
250 status = EINVAL;
251 }
252 else {
253 status = sdhci_init(hc);
254 if (status == EOK) {
255 status = bcm2712_bs_cfg_init(hc);
256 if (status == EOK) {
257 /* board specific handling of setting voltage and tuning */
258 ext.signal_voltage = hc->entry.signal_voltage;
259 hc->entry.signal_voltage = bcm2712_signal_voltage;
260 ext.sdhci_tune = hc->entry.tune;
261 hc->entry.tune = bcm2712_sdhci_tune;
262
263 if (!(hc->caps & HC_CAP_SLOT_TYPE_EMBEDDED)) {
264 /* Overwrite some of the capabilities that are set by sdhci_init() */
265 hc->caps &= ~HC_CAP_CD_INTR; /* HC doesn't support card detection interrupt */
266 }
267 (void)sdio_slogf( _SLOGC_SDIODI, _SLOG_INFO, 1, 0, "%s: controller version: 0x%x caps: %lx",
268 __func__, hc->version, hc->caps);
269 }
270 else {
271 (void)sdio_slogf(_SLOGC_SDIODI, _SLOG_ERROR, hc->cfg.verbosity, 0, "%s: configuration, %s",
272 __func__, strerror(status));
273 }
274 }
275 }
276 return status;
277}
