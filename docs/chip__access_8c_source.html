28static union {
29 uint8_t *p;
30 uintptr_t port;
31} chip_base;
32static unsigned chip_shift;
33static int chip_mmap;
34
35void
36chip_access(paddr_t base, unsigned reg_shift, unsigned mem_mapped, unsigned size) {
37 chip_shift = reg_shift;
38 chip_mmap = mem_mapped;
39 if(mem_mapped) {
40 chip_base.p = startup_memory_map(size << reg_shift, base,
41 PROT_READ|PROT_WRITE|PROT_NOCACHE);
42 } else {
43 chip_base.port = startup_io_map(size << reg_shift, base);
44 }
45}
46
47void
48chip_done() {
49 if(chip_mmap) {
50 startup_memory_unmap(chip_base.p);
51 } else {
52 startup_io_unmap(chip_base.port);
53 }
54}
