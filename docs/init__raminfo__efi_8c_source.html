49#if RAMINFO_EFI_DEBUG
50 const char *efi_types[] = {
51 [EfiReservedMemoryType] = "Reserved ",
52 [EfiLoaderCode] = "Loader Code ",
53 [EfiLoaderData] = "Loader Data ",
54 [EfiBootServicesCode] = "Boot Code ",
55 [EfiBootServicesData] = "Boot Data ",

58 [EfiConventionalMemory] = "Conventional ",
59 [EfiUnusableMemory] = "Unusable ",
60 [EfiACPIReclaimMemory] = "ACPIReclaim ",
61 [EfiACPIMemoryNVS] = "ACPIMemoryNVS",

63 [EfiMemoryMappedIOPortSpace] ="Port Space ",
64 };
65#endif
66
67 for (int pass = 0; pass < 3; ++pass) {
68 _Uint64t cur_start = ~0ul;
69 _Uint64t cur_size = 0;
70
71 for (desc = efi_walk_map(NULL); desc != NULL; desc = efi_walk_map(desc)) {
72#if EFI_RAMINFO_DEBUG
73 char attrs[] = "---------------";
74 if (desc->Attribute & EFI_MEMORY_UC) {
75 attrs[0] = 'U';
76 }
77 if (desc->Attribute & EFI_MEMORY_WC) {
78 attrs[1] = 'C';
79 }
80 if (desc->Attribute & EFI_MEMORY_WT) {
81 attrs[2] = 'T';
82 }
83 if (desc->Attribute & EFI_MEMORY_WB) {
84 attrs[3] = 'B';
85 }
86 if (desc->Attribute & EFI_MEMORY_UCE) {
87 attrs[4] = 'U';
88 }
89 if (desc->Attribute & EFI_MEMORY_WP) {
90 attrs[5] = 'W';
91 }
92 if (desc->Attribute & EFI_MEMORY_RP) {
93 attrs[6] = 'R';
94 }
95 if (desc->Attribute & EFI_MEMORY_XP) {
96 attrs[7] = 'X';
97 }
98 if (desc->Attribute & EFI_MEMORY_NV) {
99 attrs[8] = 'N';
100 }
101 if (desc->Attribute & EFI_MEMORY_MORE_RELIABLE) {
102 attrs[9] = 'M';
103 }
104 if (desc->Attribute & EFI_MEMORY_RO) {
105 attrs[10] = 'R';
106 }
107 if (desc->Attribute & EFI_MEMORY_SP) {
108 attrs[11] = 'S';
109 }
110 if (desc->Attribute & EFI_MEMORY_CPU_CRYPTO) {
111 attrs[12] = 'C';
112 }
113 if (desc->Attribute & EFI_MEMORY_RUNTIME) {
114 attrs[13] = 'R';
115 }
116 if (desc->Attribute & EFI_MEMORY_ISA_VALID) {
117 attrs[14] = 'I';
118 }
119 const char *type_str = "???";
120 if (desc->Type < EfiMaxMemoryType) {
121 type_str = efi_types[desc->Type];
122 }
123 kprintf("Type=%s PS=%l NOP=%l Attr=%s\r\n", type_str, desc->PhysicalStart, (_Uint64t)desc, attrs);
124#endif
125
126 _Uint64t const sz = 0x1000ul * desc->NumberOfPages;
127 switch (desc->Type) {
128 case EfiLoaderCode:
129 case EfiLoaderData:
130 case EfiBootServicesCode:
131 case EfiBootServicesData:
132 case EfiRuntimeServicesCode:
133 case EfiRuntimeServicesData:
134 case EfiACPIReclaimMemory:
135 if (pass == 0) {
136 avoid_ram(desc->PhysicalStart, sz);
137 }
138 if (pass == 2) {
139 alloc_ram(desc->PhysicalStart, sz, 0);
140 }
141 // FALLTHROUGH
142 case EfiConventionalMemory:
143 if (pass == 1) {
144 if (desc->PhysicalStart == (cur_start + cur_size)) {
145 cur_size += sz;
146 } else {
147 add_ram(cur_start, cur_size);
148 cur_start = desc->PhysicalStart;
149 cur_size = sz;
150 }
151 }
152 break;
153
154 default:
155 if (pass == 0) {
156 avoid_ram(desc->PhysicalStart, sz);
157 }
158 break;
159 }
160 }
161 if ((pass == 1) && (cur_size != 0)) {
162 add_ram(cur_start, cur_size);
163 }
164 }
165}
