27/*******************************************************************************
28* hwibus_add_i2c
29*
30* See sys/hwinfo.h for interface description
31*
32* Returns: the offset in the hwinfo section of the added bus or HWI_NULL_OFF
33* on error
34*/
35unsigned hwibus_add_i2c(unsigned parent_hwi_off, hwiattr_i2c_t *attr)
36{
37 unsigned hwi_off = hwibus_add(HWI_ITEM_BUS_I2C, parent_hwi_off);
38
39 if ((hwi_off != HWI_NULL_OFF) && (attr != NULL))
40 {
41 unsigned i;
42
43 hwitag_add_common(hwi_off, &attr->common);
44 for (i=0; i<attr->num_clks; i++)
45 hwitag_add_inputclk(hwi_off, 0, 1); // make divisor 1, just in case ;)
46 for (i=0; i<attr->num_busattr; i++)
47 hwitag_add_busattr(hwi_off, NULL); // create empty busattr tags
48 }
49 return hwi_off;
50}
