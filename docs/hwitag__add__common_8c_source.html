48/*******************************************************************************
49* hwitag_add_common
50*
51* add all of the common tags to device <hwi_off> and and set the tag fields as
52* specified in <attr>.
53*
54* Returns: -1 on error or the vector tag index that was set (base 0)
55*
56*/
57void hwitag_add_common(unsigned hwi_off, void *_attr)
58{
59 hwiattr_common_t *attr = (hwiattr_common_t *)_attr;
60
61 hwitag_add_location(hwi_off, attr->location.base, attr->location.len, attr->location.regshift, attr->location.addrspace);
62 ADD_IRQ_TAGS(hwi_off, attr->num_irq);
63 if (attr->dll != NULL) hwitag_add_dll(hwi_off, attr->dll);
64 if (attr->optstr != NULL) hwitag_add_optstr(hwi_off, attr->optstr);
65 ADD_ERRATA_TAGS(hwi_off, attr->num_errata);
66 ADD_DMA_TAGS(hwi_off, attr->num_dma);
67}
