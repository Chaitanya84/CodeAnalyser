64def run(source: Path, output: Path, html: Path) -> int:
65 root = source.resolve()
66 if not root.exists():
67 log.error("source directory does not exist: %s", source)
68 return 2
69 if not root.is_dir():
70 log.error("source path is not a directory: %s", source)
71 return 2
72
73 # ---- Pass 1: scan + parse -------------------------------------------
74 files = scan(root)
75 parser = SourceParser()
76 parsed_files = []
77 for path in files:
78 parsed_files.append(parser.parse_file(path, normalize_relative(path, root)))
79
80 # ---- Pass 2: entity extraction --------------------------------------
81 extractor = EntityExtractor()
82 for parsed in parsed_files:
83 try:
84 extractor.extract_file(parsed)
85 except Exception as exc: # one bad file must not kill the pipeline
86 log.error("extraction failed for %s: %s", parsed.relative_path, exc)
87
88 # ---- Pass 3: global symbol index ------------------------------------
89 symbols = SymbolTable()
90 symbols.add_all(extractor.nodes)
91 log.info("symbol index: %d unique entities", len(symbols.nodes))
92
93 # ---- Pass 4: relationship resolution --------------------------------
94 resolver = RelationshipResolver(symbols)
95 links = resolver.resolve_all(extractor.pending_calls, extractor.pending_inheritance)
96
97 # ---- Pass 5: validation ----------------------------------------------
98 nodes, links, validation_issues = validate_graph(
99 list(symbols.nodes.values()), links
100 )
101
102 # ---- statistics + diagnostics ---------------------------------------
103 files_with_errors = [p for p in parsed_files if p.has_syntax_error]
104 files_unreadable = [p for p in parsed_files if p.read_error]
105 statistics = {
106 "files_scanned": len(files),
107 "files_parsed": sum(1 for p in parsed_files if p.parsed_ok),
108 "files_with_syntax_errors": len(files_with_errors),
109 "files_unreadable": len(files_unreadable),
110 "ast_nodes": sum(p.total_nodes for p in parsed_files),
111 "nodes": len(nodes),
112 "links": len(links),
113 "functions": sum(1 for n in nodes if n.type == "function" and n.is_definition),
114 "function_declarations": sum(
115 1 for n in nodes if n.type == "function" and not n.is_definition
116 ),
117 "classes": sum(1 for n in nodes if n.type == "class" and n.kind == "class"),
118 "structs": sum(1 for n in nodes if n.type == "class" and n.kind == "struct"),
119 "variables": sum(1 for n in nodes if n.type == "variable"),
120 "calls": resolver.resolved_calls,
121 "inherits": resolver.resolved_inheritance,
122 "defines": resolver.resolved_defines,
123 "resolved_relationships": (
124 resolver.resolved_calls
125 + resolver.resolved_inheritance
126 + resolver.resolved_defines
127 ),
128 "ambiguous_calls": len(resolver.ambiguous_calls),
129 "unresolved_calls": len(resolver.unresolved_calls),
130 "unresolved_inheritance": len(resolver.unresolved_inheritance),
131 "duplicate_symbols": len(symbols.duplicate_ids),
132 "validation_issues": len(validation_issues),
133 }
134 diagnostics = {
135 "files": sorted(
136 (p.diagnostic() for p in parsed_files),
137 key=lambda d: d["file"],
138 ),
139 "duplicate_symbols": sorted(set(symbols.duplicate_ids)),
140 "unresolved_calls": resolver.unresolved_calls,
141 "ambiguous_calls": resolver.ambiguous_calls,
142 "unresolved_inheritance": resolver.unresolved_inheritance,
143 "ambiguous_defines": resolver.ambiguous_defines,
144 "validation_issues": validation_issues,
145 }
146
147 # ---- outputs ----------------------------------------------------------
148 payload = build_payload(str(root), nodes, links, statistics, diagnostics)
149 write_json(payload, output)
150 try:
151 render(output, html)
152 except Exception as exc:
153 log.error("visualization failed: %s", exc)
154 return 1
155
156 log.info(
157 "done: %d files, %d nodes, %d links (calls=%d inherits=%d defines=%d, "
158 "unresolved_calls=%d)",
159 statistics["files_scanned"],
160 statistics["nodes"],
161 statistics["links"],
162 statistics["calls"],
163 statistics["inherits"],
164 statistics["defines"],
165 statistics["unresolved_calls"],
166 )
167 return 0
