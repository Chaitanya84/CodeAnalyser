29 * alloc_qtime()
30 * Allocate the qtime section and initialize the boot time from the
31 * IPL info or an RTC chip. Also initializes some generic options.
32 */
33
34struct qtime_entry *
35alloc_qtime() {
36 if(lsp.qtime.size != 0) {
37 // Section already has been allocated & initialized
38 return lsp.qtime.p;
39 }
40
41 struct qtime_entry *qtime = set_syspage_section(&lsp.qtime, sizeof(*lsp.qtime.p));
42 const struct startup_info_time *time;
43
44 qtime->epoch = 1970;
45 if(!(misc_flags & MISC_FLAG_SUPPRESS_BOOTTIME)) {
46 time = find_startup_info(NULL, STARTUP_INFO_TIME);
47 qtime->boot_time = (time != NULL) ? time->time : rtc_time();
48 qtime->nsec_tod_adjust = qtime->boot_time * (uint64_t)1000000000;
49 }
50
51 return qtime;
52}
