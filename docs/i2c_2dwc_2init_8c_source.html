38int32_t dwc_i2c_enable(dwc_dev_t* const dev, const uint32_t enable)
39{
40 int32_t timeout = 1000;
41
42 do {
43 i2c_reg_write32(dev, DW_IC_ENABLE, enable);
44 if ((i2c_reg_read32(dev, DW_IC_ENABLE_STATUS) &
45 DW_IC_ENABLE_STATUS_ENABLE_MASK) == enable) {
46 return 0;
47 }
48
49 /*
50 * Wait 10 times the signaling period of the highest I2C
51 * transfer supported by the driver (for 400KHz this is 25us).
52 */
53 (void)usleep(25);
54 } while (timeout-- != 0);
55
56 logerr("%s: timeout in %sabling I2C adapter\n",
57 __func__, (enable != 0U) ? "en" : "dis");
58 return -ETIMEDOUT;
59}

61static uint32_t i2c_dw_scl_hcnt(const uint32_t ic_clk, const uint32_t tsymbol,
62 const uint32_t tf, const int32_t cond, const int32_t offset)
63{
64 /*
65 * DesignWare I2C core doesn't seem to have solid strategy to meet
66 * the tHD;STA timing spec. Configuring _HCNT based on tHIGH spec
67 * will result in violation of the tHD;STA spec.
68 */
69 if (cond != 0) {
70 /*
71 * Conditional expression:
72 *
73 * IC_[FS]S_SCL_HCNT + (1+4+3) >= IC_CLK * tHIGH
74 *
75 * This is based on the DW manuals, and represents an ideal
76 * configuration. The resulting I2C bus speed will be
77 * faster than any of the others.
78 *
79 * If your hardware is free from tHD;STA issue, try this one.
80 */
81 return (((ic_clk * tsymbol) + 500000U) / 1000000U)
82 - 8U + (uint32_t)offset;
83 } else {
84 /*
85 * Conditional expression:
86 *
87 * IC_[FS]S_SCL_HCNT + 3 >= IC_CLK * (tHD;STA + tf)
88 *
89 * This is just experimental rule; the tHD;STA period turned
90 * out to be proportinal to (_HCNT + 3). With this setting,
91 * we could meet both tHIGH and tHD;STA timing specs.
92 *
93 * If unsure, you'd better to take this alternative.
94 *
95 * The reason why we need to take into account "tf" here,
96 * is the same as described in i2c_dw_scl_lcnt().
97 */
98 return (((ic_clk * (tsymbol + tf)) + 500000U) / 1000000U)
99 - 3U + (uint32_t)offset;
100 }
101}

103static uint32_t i2c_dw_scl_lcnt(const uint32_t ic_clk, const uint32_t tlow,
104 const uint32_t tf, const int32_t offset)
105{
106 /*
107 * Conditional expression:
108 *
109 * IC_[FS]S_SCL_LCNT + 1 >= IC_CLK * (tlow + tf)
110 *
111 * DW I2C core starts counting the SCL CNTs for the LOW period
112 * of the SCL clock (tlow) as soon as it pulls the SCL line.
113 * In order to meet the tLOW timing spec, we need to take into
114 * account the fall time of SCL signal (tf). Default tf value
115 * should be 0.3 us, for safety.
116 */
117 return (((ic_clk * (tlow + tf)) + 500000U) / 1000000U)
118 - 1U + (uint32_t)offset;
119}

121int32_t dwc_i2c_bus_active(dwc_dev_t* const dev)
122{
123 /* Wait Bus Idle */
124 int32_t busy_retry = BUSY_RETRY;
125 int32_t ret;
126
127 ret = dwc_i2c_wait_bus_not_busy(dev);
128 while ((ret != 0) && (busy_retry >= 0)) {
129
130 logerr("Busy detected. retry=%d", busy_retry);
131 if (busy_retry-- == 0) {
132 dwc_i2c_reset(dev);
133 logfyi("One last try to check for busy...");
134 }
135 ret = dwc_i2c_wait_bus_not_busy(dev);
136 }
137
138 if (ret != 0) {
139 logerr("Can't get out of busy.");
140 return (int32_t)I2C_STATUS_BUSY;
141 }
142
143 if (busy_retry < BUSY_RETRY) {
144 logfyi("Successfully unstuck from busy.");
145 }
146
147 /* Disable the adapter */
148 if (0 != dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_DISABLE)) {
149 dwc_i2c_reset(dev);
150 return (int32_t)I2C_STATUS_BUSY;
151 }
152
153 /* Set slave address */
154 if (dev->slave_addr_fmt == I2C_ADDRFMT_7BIT) {
155 i2c_reg_write32(dev, DW_IC_TAR, dev->slave_addr);
156 } else {
157 i2c_reg_write32(dev, DW_IC_TAR,
158 dev->slave_addr | DW_IC_TAR_10BITADDR_MASTER);
159 }
160
161 /* set speed and master mode */
162 i2c_reg_write32(dev, DW_IC_CON, dev->master_cfg);
163
164 /* Enforce disabled interrupts (due to HW issues) */
165 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
166
167 /* Enable the adapter */
168 if (0 != dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_ENABLE)) {
169 dwc_i2c_reset(dev);
170 return (int32_t)I2C_STATUS_BUSY;
171 }
172
173 return 0;
174}

176static int32_t map_in_device_registers(dwc_dev_t *dev)
177{
178 const void* const hw_addr = mmap_device_memory(NULL, dev->map_size,
179 PROT_READ | PROT_WRITE | PROT_NOCACHE,
180 MAP_SHARED, dev->pbase);
181 if (hw_addr == MAP_FAILED) {
182 logerr( "mmap_device_memory failed");
183 return -1;
184 }
185
186 dev->vbase = (uintptr_t)hw_addr;
187
188 if (dev->verbose != 0U) {
189 logfyi("mmap_device_memory size %u @ %08x to %p\n",
190 (uint32_t)dev->map_size, (uint32_t)dev->pbase,
191 (void*)dev->vbase);
192 }
193
194 return 0;
195}

197static void dwc_i2c_set_fifo_threshold(dwc_dev_t *dev)
198{
199 uint32_t reg, tx_fifo_depth, rx_fifo_depth;
200
201 reg = i2c_reg_read32(dev, DW_IC_COMP_PARAM_1);
202 tx_fifo_depth = ((reg >> DW_IC_COMP_PARAM_1_TX_DEPTH_F) & 0xffU) + 1U;
203 rx_fifo_depth = ((reg >> DW_IC_COMP_PARAM_1_RX_DEPTH_F) & 0xffU) + 1U;
204
205 dev->fifo_depth = min(tx_fifo_depth, rx_fifo_depth);
206
207 i2c_reg_write32(dev, DW_IC_TX_TL, dev->fifo_depth / 2U);
208 i2c_reg_write32(dev, DW_IC_RX_TL, dev->fifo_depth / 2U);
209}

211void* dwc_i2c_init(int argc, char *argv[])
212{
213 dwc_dev_t *dev;
214 uint32_t reg;
215 uintptr_t iolevel;
216
217 iolevel = _NTO_IO_LEVEL_NONE | _NTO_TCTL_IO_LEVEL_INHERIT;
218 if (ThreadCtl(_NTO_TCTL_IO_LEVEL, (void *)iolevel) == -1) {
219 perror("ThreadCtl");
220 return NULL;
221 }
222
223 dev = calloc(1, sizeof(*dev)); // allocate device object
224 if (dev == NULL) {
225 return NULL;
226 }
227
228 if (dwc_i2c_parseopts(dev, argc, argv) == -1) {
229 goto fail_cleanup;
230 }
231
232 if (dwc_i2c_probe_device(dev) == -1) {
233 goto fail_cleanup;
234 }
235
236 if (map_in_device_registers(dev) != 0) {
237 goto fail_cleanup;
238 }
239
240 /* check I2C component type*/
241 reg = i2c_reg_read32(dev, DW_IC_COMP_TYPE);
242 if (reg != DW_IC_COMP_TYPE_VALUE) {
243 logerr("%s: Unknown I2C component type: 0x%08x",
244 __func__, reg);
245 goto fail_cleanup;
246 }
247
248 /* Attach interrupt */
249 struct sigevent intrevent;
250
251 SIGEV_INTR_INIT(&intrevent);
252
253 dev->iid = InterruptAttachEvent(dev->irq, &intrevent,
254 _NTO_INTR_FLAGS_TRK_MSK);
255 if (dev->iid == -1) {
256 logerr("%s: InterruptAttachEvent", __func__);
257 goto fail_cleanup;
258 }
259 logfyi("connected to IRQ %u", dev->irq);
260
261 /* Disable the I2C adapter */
262 (void)dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_DISABLE);
263
264 dwc_i2c_init_registers(dev);
265
266 return dev;
267
268fail_cleanup:
269 dwc_i2c_cleanup(dev);
270
271 return NULL;
272}

274void dwc_i2c_init_registers(dwc_dev_t *dev)
275{
276 uint32_t reg;
277
278 /* Get I2C input clock */
279 if (dev->clock_khz == 0U) {
280 dev->clock_khz = dwc_i2c_get_input_clock(dev);
281 }
282
283 logfyi("Set clock to %ld Khz", dev->clock_khz);
284
285#ifdef DWC_SUPPORT_FIXED_SCL
286 /* Set SCL timing parameters for standard-mode */
287 /* Intel Recommended setting; SF Case # 00168315 */
288 if (dev->fixed_scl != 0U) {
289 dev->ss_hcnt = FIXED_SS_HCNT;
290 dev->ss_lcnt = FIXED_SS_LCNT;
291 } else
292#endif
293 {
294 dev->ss_hcnt = i2c_dw_scl_hcnt(dev->clock_khz,
295 dev->std.high,
296 dev->std.fall,
297 0, // 0: DW default, 1: Ideal
298 0); // No offset
299 dev->ss_lcnt = i2c_dw_scl_lcnt(dev->clock_khz,
300 dev->std.low,
301 dev->std.fall,
302 0); // No offset
303 }
304
305 logfyi("DW_IC_SS_SCL_HCNT %08x ", dev->ss_hcnt);
306 logfyi("DW_IC_SS_SCL_LCNT %08x ", dev->ss_lcnt);
307
308 i2c_reg_write32(dev, DW_IC_SS_SCL_HCNT, dev->ss_hcnt);
309 i2c_reg_write32(dev, DW_IC_SS_SCL_LCNT, dev->ss_lcnt);
310
311#ifdef DWC_SUPPORT_FIXED_SCL
312 /* Set SCL timing parameters for fast-mode */
313 /* Intel Recommended setting; SF Case # 00168315 */
314 if (dev->fixed_scl != 0U) {
315 dev->fs_hcnt = FIXED_FS_HCNT;
316 dev->fs_lcnt = FIXED_FS_LCNT;
317 } else
318#endif
319 {
320 dev->fs_hcnt = i2c_dw_scl_hcnt(dev->clock_khz,
321 dev->fast.high,
322 dev->fast.fall,
323 0, // 0: DW default, 1: Ideal
324 0); // No offset
325 dev->fs_lcnt = i2c_dw_scl_lcnt(dev->clock_khz,
326 dev->fast.low,
327 dev->fast.fall,
328 0); // No offset
329 }
330
331 logfyi("DW_IC_FS_SCL_HCNT %08x ", dev->fs_hcnt);
332 logfyi("DW_IC_FS_SCL_LCNT %08x ", dev->fs_lcnt);
333
334 i2c_reg_write32(dev, DW_IC_FS_SCL_HCNT, dev->fs_hcnt);
335 i2c_reg_write32(dev, DW_IC_FS_SCL_LCNT, dev->fs_lcnt);
336
337 /* Configure SDA Hold Time */
338 reg = i2c_reg_read32(dev, DW_IC_COMP_VERSION);
339 if (reg >= DW_IC_SDA_HOLD_MIN_VERS) {
340#ifdef DWC_SUPPORT_FIXED_SCL
341 /* Intel Recommended setting; SF Case # 00168315 */
342 if (dev->fixed_scl != 0U) {
343 const uint32_t sda_hold_time = 28;
344 i2c_reg_write32(dev, DW_IC_SDA_HOLD, sda_hold_time);
345
346 logfyi("DW_IC_SDA_HOLD %08x ", sda_hold_time);
347 } else
348#endif
349 {
350 /*
351 * The I2C-bus specification requires at least 300 ns
352 * for the SDA hold time
353 */
354 const uint32_t sda_hold_time = max(dev->sda_hold_time,
355 ((dev->clock_khz * SDA_HOLD_TIME_VALUE_ns) + 500000U) / 1000000U);
356 if (sda_hold_time < dev->fs_lcnt) {
357 i2c_reg_write32(dev, DW_IC_SDA_HOLD, sda_hold_time);
358
359 logfyi("DW_IC_SDA_HOLD %08x ", sda_hold_time);
360 } else {
361 logerr("%s: SDA hold time %u too large for SCL_LCNT %u",
362 __func__, sda_hold_time, dev->fs_lcnt);
363 }
364 }
365 } else {
366 logerr("%s: Hardware too old to adjust SDA hold time!",
367 __func__);
368 }
369
370 /* Configure Tx/Rx FIFO threshold */
371 dwc_i2c_set_fifo_threshold(dev);
372
373 /* set default bus speed */
374 errno = EOK; /* CERT ERR30-C */
375 if ((dwc_i2c_set_bus_speed(dev, 100000, NULL) != 0) || (errno != EOK)) {
376 logerr("%s: Failed to set bus speed.", __func__);
377 }
378
379 /* Clear and Disable i2c Interrupt */
380 (void)i2c_reg_read32(dev, DW_IC_CLR_INTR);
381 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
382}
