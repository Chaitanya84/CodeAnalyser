1"""Relationship resolution pass.
2
3Runs strictly AFTER the global symbol index exists. Resolves pending call
4sites, inheritance specifiers and containment (defines) relationships into
5graph links, classifying each outcome as resolved / ambiguous / unresolved.
6
7Policy: prefer a correct omission over an incorrect edge. Ambiguous and
8unresolved relationships are recorded in diagnostics, never guessed.
9Self-edges (recursive calls) are valid and preserved.
10"""

64 def _resolve_defines(self) -> None:
65 """Link every entity whose scope is a known class to that class."""
66 for node in list(self.symbols.nodes.values()):
67 if not node.scope:
68 continue
69 classes = self.symbols.classes_by_qualified_name.get(node.scope, [])
70 if len(classes) == 1:
71 self._add_link(classes[0].id, node.id, "defines")
72 self.resolved_defines += 1
73 elif len(classes) > 1:
74 self._record(
75 self.ambiguous_defines,
76 {
77 "member": node.id,
78 "scope": node.scope,
79 "candidates": sorted(c.id for c in classes),
80 "file": node.file,
81 "line": node.line,
82 },
83 )

89 def _resolve_inheritance(self, pending: List[PendingInheritance]) -> None:
90 for record in pending:
91 base = record.base_raw.strip(":") # tolerate leading '::'
92 if not base:
93 continue
94 derived_segments = tuple(record.derived_qualified_name.split("::")[:-1])
95 target, ambiguous = self.symbols.lookup_class(base, derived_segments)
96 if target is not None:
97 self._add_link(record.derived_id, target.id, "inherits")
98 self.resolved_inheritance += 1
99 elif ambiguous:
100 self._record(
101 self.unresolved_inheritance,
102 {
103 "derived": record.derived_qualified_name,
104 "base": record.base_raw,
105 "reason": "ambiguous",
106 "file": record.file,
107 "line": record.line,
108 },
109 )
110 else:
111 self._record(
112 self.unresolved_inheritance,
113 {
114 "derived": record.derived_qualified_name,
115 "base": record.base_raw,
116 "reason": "base class not found",
117 "file": record.file,
118 "line": record.line,
119 },
120 )

126 def _resolve_calls(self, pending: List[PendingCall]) -> None:
127 for call in pending:
128 target, status = self._resolve_call(call)
129 if target is not None:
130 self._add_link(call.caller_id, target.id, "calls")
131 self.resolved_calls += 1
132 elif status == "ambiguous":
133 self._record(self.ambiguous_calls, self._call_diagnostic(call, status))
134 else:
135 self._record(self.unresolved_calls, self._call_diagnostic(call, status))

137 def _resolve_call(self, call: PendingCall) -> Tuple[Optional[CodeNode], str]:
138 if call.kind == "member":
139 return self._resolve_member_call(call)
140 if call.kind == "qualified":
141 return self._resolve_qualified_call(call)
142 if call.kind == "direct":
143 return self._resolve_direct_call(call)
144 return None, "unresolvable callee expression"

146 # -- direct (unqualified) calls ------------------------------------
147
148 def _resolve_direct_call(self, call: PendingCall) -> Tuple[Optional[CodeNode], str]:
149 """Resolution order: caller scope chain -> global -> unique by name."""
150 segments = list(call.scope_segments)
151 for i in range(len(segments), -1, -1):
152 prefix = "::".join(segments[:i])
153 candidate = f"{prefix}::{call.name}" if prefix else call.name
154 found = self.symbols.functions_by_qualified_name.get(candidate)
155 if found:
156 picked = self.symbols._prefer_definition(found)
157 return (picked, "resolved") if picked else (None, "ambiguous")
158 found = self.symbols.functions_by_name.get(call.name, [])
159 picked = self.symbols._prefer_definition(found) if found else None
160 if picked is not None:
161 return picked, "resolved"
162 if found:
163 return None, "ambiguous"
164 return None, "no matching function"

168 def _resolve_qualified_call(self, call: PendingCall) -> Tuple[Optional[CodeNode], str]:
169 raw = (call.raw or "").strip(":")
170 if not raw:
171 return None, "no matching function"
172 segments = list(call.scope_segments)
173 # try enclosing-namespace prefixes, longest first
174 for i in range(len(segments), -1, -1):
175 prefix = "::".join(segments[:i])
176 candidate = f"{prefix}::{raw}" if prefix else raw
177 found = self.symbols.functions_by_qualified_name.get(candidate)
178 if found:
179 picked = self.symbols._prefer_definition(found)
180 return (picked, "resolved") if picked else (None, "ambiguous")
181 # suffix fallback: a function whose qualified name ends with raw
182 matches = [
183 node
184 for node in self.symbols.functions_by_name.get(call.name, [])
185 if node.qualified_name == raw or node.qualified_name.endswith("::" + raw)
186 ]
187 picked = self.symbols._prefer_definition(matches) if matches else None
188 if picked is not None:
189 return picked, "resolved"
190 if matches:
191 return None, "ambiguous"
192 return None, "no matching function"

196 def _resolve_member_call(self, call: PendingCall) -> Tuple[Optional[CodeNode], str]:
197 if call.object_is_this:
198 if not call.class_scope:
199 return None, "'this' outside method context"
200 target, ambiguous = self.symbols.lookup_function_qname(
201 f"{call.class_scope}::{call.name}"
202 )
203 if target is not None:
204 return target, "resolved"
205 if ambiguous:
206 return None, "ambiguous"
207 return None, f"method not found on {call.class_scope}"
208
209 type_name: Optional[str] = None
210 if call.object_name:
211 type_name = call.local_types.get(call.object_name)
212 if type_name is None:
213 type_name = self.symbols.variable_type_name(
214 call.object_name, call.scope_segments
215 )
216 if not type_name:
217 return None, "receiver type unknown"
218
219 class_node, ambiguous = self.symbols.lookup_class(type_name, call.scope_segments)
220 if class_node is None:
221 return None, "ambiguous receiver type" if ambiguous else "receiver type unknown"
222
223 target, fn_ambiguous = self.symbols.lookup_function_qname(
224 f"{class_node.qualified_name}::{call.name}"
225 )
226 if target is not None:
227 return target, "resolved"
228 if fn_ambiguous:
229 return None, "ambiguous"
230 return None, f"method not found on {class_node.qualified_name}"
