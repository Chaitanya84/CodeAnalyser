33static char *
34find_next_arg(char *str, char ** const end, unsigned terminate)
35{
36 while (*str == ' ') {
37 ++str;
38 }
39 if (*str == '\0') {
40 *end = str;
41 return str;
42 }
43
44 char *next = str;
45 while ((*next != ' ') && (*next != '\0')) {
46 ++next;
47 }
48
49 if (*next != '\0') {
50 if (terminate) {
51 *next = '\0';
52 }
53 ++next;
54 }
55 *end = next;
56 return str;
57}

65void
66efi_tweak_cmdline(struct bootargs_entry * const bap)
67{
68 static char *next_arg = NULL;
69
70 if (next_arg == NULL) {
71 next_arg = efi_convert_command_line();
72 if (next_arg == NULL) {
73 return;
74 }
75 // Skip the first argument if it doesn't start with a '-'
76 // Firmware puts the name of the image as the first argument
77 char *str;
78 next_arg = find_next_arg(next_arg, &str, 0);
79 if (*next_arg != '-') {
80 next_arg = str;
81 }
82 }
83
84 for (;;) {
85 char * const cmdline = find_next_arg(next_arg, &next_arg, 1);
86 if (*cmdline == '\0') return;
87 if (!strcmp(cmdline, "--")) break;
88 bootstrap_arg_adjust(bap, NULL, cmdline);
89 }
90}
