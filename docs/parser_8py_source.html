24def get_language_name_for_file(path: Path) -> str:
25 """Select the grammar for a source file.
26
27 Only ``.c`` uses the C grammar; every other supported extension
28 (including all headers) uses the C++ grammar.
29 """
30 return "c" if path.suffix.lower() in C_EXTENSIONS else "cpp"

66class SourceParser:
67 """Loads both grammars once and parses files on demand."""

69 def __init__(self) -> None:
70 self._languages: Dict[str, Language] = {
71 "c": Language(tree_sitter_c.language()),

73 }

79 def parse_file(self, path: Path, relative_path: str) -> ParsedFile:
80 language = get_language_name_for_file(path)
81 result = ParsedFile(path=path, relative_path=relative_path, language=language)
82
83 try:
84 source = path.read_bytes()
85 except OSError as exc:
86 result.read_error = f"{type(exc).__name__}: {exc}"
87 log.warning("cannot read %s: %s", path, exc)
88 return result
89
90 result.source = source
91 try:
92 tree = self._parsers[language].parse(source)
93 except Exception as exc: # grammar/driver failure must not kill the run
94 result.read_error = f"parser failure: {type(exc).__name__}: {exc}"
95 log.warning("failed to parse %s: %s", path, exc)
96 return result
97
98 result.tree = tree
99 result.total_nodes = count_nodes(tree.root_node)
100 errors = collect_syntax_errors(tree.root_node)
101 result.errors = errors
102 result.error_nodes = sum(1 for e in errors if e["type"] == "ERROR")
103 result.missing_nodes = sum(1 for e in errors if e["type"] == "MISSING")
104 # counts above are capped by MAX_REPORTED_ERRORS; also check flags so a
105 # file with more errors than the reporting cap is still flagged.
106 result.has_syntax_error = bool(errors) or tree.root_node.has_error
107 if result.has_syntax_error:
108 log.debug(
109 "syntax errors in %s (errors=%d missing=%d)",
110 relative_path,
111 result.error_nodes,
112 result.missing_nodes,
113 )
114 return result
