1"""C/C++ grammar compatibility layer.
2
3All knowledge of concrete tree-sitter-c / tree-sitter-cpp node types and
4named fields lives here, so the rest of the pipeline never touches
5grammar-specific details. Node types below are the real ones exposed by
6the two grammars (verified against the installed grammar versions).
7
8Key grammar facts relied upon:
9
10- ``function_definition`` fields: ``type``, ``declarator``, ``body``.
11- ``function_declarator`` fields: ``declarator`` (name or nested
12 declarator / qualified_identifier / destructor_name), ``parameters``.
13- ``declaration`` / ``field_declaration`` fields: ``type``, ``declarator``
14 (``declarator`` is a repeated field for ``int a, b;``).
15- C++ ``class_specifier`` covers both ``class`` and ``struct`` (the first
16 anonymous child is the keyword); C uses ``struct_specifier``.
17 Fields: ``name``, ``body``; inheritance lives in a ``base_class_clause``
18 named child containing ``access_specifier`` markers plus
19 ``type_identifier`` / ``qualified_identifier`` / ``template_type`` bases.
20- ``namespace_definition`` fields: ``name`` (``namespace_identifier`` or
21 ``nested_namespace_specifier`` for ``namespace A::B {}``), ``body``.
22- ``call_expression`` fields: ``function``, ``arguments``.
23- ``field_expression`` fields: ``argument``, ``field`` (``.`` / ``->`` are
24 anonymous operator children).
25"""

75def class_kind(node) -> str:
76 """Return ``"class"`` or ``"struct"`` for a class/struct specifier."""
77 if node.type == "struct_specifier":
78 return "struct"
79 # class_specifier: the first anonymous child is the keyword itself.
80 for child in node.children:
81 if not child.is_named:
82 text = node_text(child)
83 if text in ("class", "struct"):
84 return text
85 break
86 return "class"

101def get_function_declarator(function_definition):
102 """Return the ``function_declarator`` of a function definition.
103
104 The declared ``declarator`` field may be wrapped in pointer/reference/
105 parenthesized declarators (e.g. ``void (*signal(int, void(*)(int)))(int)``
106 or ``Device &Device::operator=``); peel wrappers until the
107 ``function_declarator`` core is found.
108 """
109 node = function_definition.child_by_field_name("declarator")
110 while node is not None:
111 if node.type == FUNCTION_DECLARATOR:
112 return node
113 if node.type in _WRAPPER_DECLARATORS:
114 node = node.child_by_field_name("declarator")
115 continue
116 return None
117 return None

133def declarator_qualified_text(function_declarator) -> Optional[str]:
134 """Qualified name text of a function declarator, e.g. ``Device::start``.
135
136 Returns text possibly carrying a leading ``::`` for explicitly global
137 qualifications; callers handle that. Returns None when no usable name
138 exists (e.g. anonymous function-pointer declarators).
139 """
140 node = function_declarator.child_by_field_name("declarator")
141 while node is not None and node.type in _WRAPPER_DECLARATORS:
142 node = node.child_by_field_name("declarator")
143 if node is None:
144 return None
145 if node.type in _IDENTIFIER_TYPES:
146 return compact_text(node)
147 if node.type in ("qualified_identifier", "destructor_name", "operator_function_id"):
148 return compact_text(node)
149 if node.type == "template_function":
150 name = node.child_by_field_name("name")
151 return compact_text(name) if name is not None else compact_text(node)
152 if node.type == "dependent_name":
153 inner = node.child_by_field_name("name")
154 return compact_text(inner) if inner is not None else compact_text(node)
155 # Last resort: exact source text (still AST-derived, not regex).
156 return compact_text(node)

167def namespace_segments(name_node) -> List[str]:
168 """Namespace name as a list of segments.
169
170 Handles ``namespace A`` (``namespace_identifier``/``identifier``) and
171 ``namespace A::B`` (``nested_namespace_specifier``). Anonymous
172 namespaces map to ``["<anonymous>"]``.
173 """
174 if name_node is None:
175 return ["<anonymous>"]
176 text = compact_text(name_node)
177 if not text:
178 return ["<anonymous>"]
179 return [segment for segment in text.split("::") if segment]

182def get_base_type_names(class_node) -> List[str]:
183 """Base class names from a class/struct ``base_class_clause``.
184
185 Skips access specifiers (``public``/``protected``/``private``) and the
186 ``virtual`` keyword (anonymous node). Template bases contribute their
187 template name only (``Container<T>`` -> ``Container``).
188 """
189 bases: List[str] = []
190 for child in class_node.named_children:
191 if child.type != "base_class_clause":
192 continue
193 for item in child.named_children:
194 if item.type == "access_specifier":
195 continue
196 if item.type in ("type_identifier", "qualified_identifier"):
197 bases.append(compact_text(item))
198 elif item.type == "template_type":
199 name = item.child_by_field_name("name")
200 if name is not None:
201 bases.append(compact_text(name))
202 elif item.type == "dependent_name":
203 name = item.child_by_field_name("name")
204 if name is not None:
205 bases.append(compact_text(name))
206 return bases

209def type_name_of(type_node) -> Optional[str]:
210 """Best-effort simple class/type name for a declaration's type node.
211
212 Returns None for primitive/unknown types. Used only for best-effort
213 member-call resolution (``obj.start()``), never for fabrication.
214 """
215 if type_node is None:
216 return None
217 kind = type_node.type
218 if kind == "type_identifier":
219 return compact_text(type_node)
220 if kind == "qualified_identifier":
221 name = type_node.child_by_field_name("name")
222 return compact_text(name) if name is not None else compact_text(type_node)
223 if kind in ("struct_specifier", "class_specifier", "union_specifier", "enum_specifier"):
224 name = type_node.child_by_field_name("name")
225 return compact_text(name) if name is not None else None
226 if kind == "template_type":
227 name = type_node.child_by_field_name("name")
228 return compact_text(name) if name is not None else None
229 if kind == "dependent_name":
230 name = type_node.child_by_field_name("name")
231 return compact_text(name) if name is not None else None
232 return None
