31void dw_fini(void *const hdl)
32{
33 dw_spi_t *const spi = hdl;
34 dw_spi_enable(spi, DW_SPI_DISABLE);
35
36#ifdef DW_DMA
37 /* Clean up DMA */
38 dw_spi_fini_dma(spi);
39#endif
40
41 /* Reset interrupt status */
42 dw_read32(spi, DW_SPI_TXOICR);
43 dw_read32(spi, DW_SPI_RXOICR);
44 dw_read32(spi, DW_SPI_RXUICR);
45
46 dw_spi_deselect_slave(spi);
47
48 /* Detach the interrupt */
49 if (spi->iid != -1) {
50 InterruptDetach(spi->iid);
51 }
52
53 /* Unmap SPI register */
54 if (spi->vbase != (uintptr_t)MAP_FAILED) {
55 munmap_device_memory((void *)spi->vbase, spi->map_size);
56 }
57
58 /* Free SPI structure */
59 free(spi);
60}
