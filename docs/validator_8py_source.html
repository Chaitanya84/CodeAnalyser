1"""Graph validation.
2
3Verifies structural integrity of nodes and links before serialization:
4mandatory node fields, link endpoint existence, allowed relationship
5types, and exact-duplicate link removal. Recursive-call self-edges are
6explicitly preserved.
7"""
8from __future__ import annotations
9
10import logging
11from typing import List, Tuple
12
13from .models import CodeLink, CodeNode
14
15log = logging.getLogger(__name__)
16
17ALLOWED_NODE_TYPES = {"function", "class", "variable"}
18ALLOWED_LINK_TYPES = {"calls", "inherits", "defines"}
19
20
21def validate_graph(
22 nodes: List[CodeNode], links: List[CodeLink]
23) -> Tuple[List[CodeNode], List[CodeLink], List[str]]:
24 issues: List[str] = []
25
26 # ---- nodes ---------------------------------------------------------
27 valid_nodes: List[CodeNode] = []
28 seen_ids = set()
29 for node in nodes:
30 if not node.id or not node.name or not node.type or not node.file:
31 issues.append(
32 f"invalid node (missing mandatory field): "
33 f"id={node.id!r} name={node.name!r} type={node.type!r} file={node.file!r}"
34 )
35 continue
36 if node.type not in ALLOWED_NODE_TYPES:
37 issues.append(f"node {node.id}: unknown type {node.type!r}")
38 continue
39 if node.id in seen_ids:
40 issues.append(f"duplicate node id removed: {node.id}")
41 continue
42 seen_ids.add(node.id)
43 valid_nodes.append(node)
44
45 # ---- links ---------------------------------------------------------
46 valid_links: List[CodeLink] = []
47 seen_links = set()
48 for link in links:
49 if link.type not in ALLOWED_LINK_TYPES:
50 issues.append(f"link {link.source} -> {link.target}: invalid type {link.type!r}")
51 continue
52 if link.source not in seen_ids:
53 issues.append(f"link dropped, missing source node: {link.source}")
54 continue
55 if link.target not in seen_ids:
56 issues.append(f"link dropped, missing target node: {link.target}")
57 continue
58 key = (link.source, link.target, link.type)
59 if key in seen_links:
60 continue # exact duplicate; self-edges are NOT filtered out
61 seen_links.add(key)
62 valid_links.append(link)
63
64 if issues:
65 log.warning("graph validation reported %d issue(s)", len(issues))
66 log.info(
67 "graph validated: %d nodes, %d links", len(valid_nodes), len(valid_links)
68 )
69 return valid_nodes, valid_links, issues
