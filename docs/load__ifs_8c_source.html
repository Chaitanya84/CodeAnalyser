29void
30load_ifs(paddr_t ifs_paddr) {
31 int comp;
32 paddr_t src;
33
34 if(shdr == NULL) {
35 crash("NULL shdr");
36 }
37 if(!full_image_paddr || !shdr->startup_size) {
38 crash("startup: shdr: %V, image_paddr: %P, startup_size: %x\nImage header corrupt\n",
39 shdr, (paddr_t)full_image_paddr, shdr->startup_size);
40 }
41
42 if (debug_flag > 0) kprintf("Loading IFS...");
43
44 board_enable_caches();
45 comp = shdr->flags1 & STARTUP_HDR_FLAGS1_COMPRESS_MASK;
46 if(comp != 0) {
47 src = full_imagefs_paddr;
48 if(src == 0) {
49 /* If imagefs_paddr is zero, then we were loaded by some code
50 * that did not do an image_setup(). This was probably because
51 * we were loaded by a non-neutrino IPL (or an emulator).
52 *
53 * We'll copy the compressed FS out of the way and then
54 * uncompress it back to where it belongs. Since we just
55 * need the temporary memory for a short period of time,
56 * we won't bother actually allocating it.
57 *
58 */
59 src = find_ram(shdr->stored_size, sizeof(uint64_t), 0, 0);
60 copy_memory(src, ifs_paddr, shdr->stored_size);
61 }
62
63 // If the uncompressed image-size is GREATER than allowed space of expansion,
64 // vital memory space will land up being corrupted/overwritten.
65 if ((full_imagefs_paddr - full_image_paddr) < shdr->imagefs_size)
66 crash("\n\t *** Warning! Uncompressing this image will exceed allotted space & cause memory corruption! *** \n");
67
68 uncompress(comp, ifs_paddr, src);
69 } else if((full_imagefs_paddr != 0) &&
70 (full_imagefs_paddr != ifs_paddr)) {
71 copy_memory(ifs_paddr, shdr->imagefs_paddr, shdr->imagefs_size);
72 }
73
74 board_disable_caches();
75 if (debug_flag > 0) kprintf("done\n");
76}
