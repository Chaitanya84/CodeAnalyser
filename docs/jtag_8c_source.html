30void
31jtag_reserve_memory(paddr_t resmem_addr, size_t resmem_size, uint8_t resmem_flag) {
32 void *p;
33
34 // Reserve user specified block of memory
35 alloc_ram(resmem_addr, resmem_size, 1);
36 p = startup_memory_map(resmem_size, resmem_addr, PROT_READ|PROT_WRITE);
37 // Determine if reserved memory should be cleared
38 if(!resmem_flag) {
39 memset(p, 0, resmem_size);
40 }
41 startup_memory_unmap(p);
42}

44void
45jtag_reserve_syspage_addr(paddr_t jtag_syspage_addr) {
46
47 // Save the address to be used later
48 jtag_syspage_address = jtag_syspage_addr;
49
50 // Reserve a block of memory to store the address of the system page
51 alloc_ram(jtag_syspage_address, sizeof(syspage_paddr), 1);
52}

54void
55jtag_store_syspage_addr(void) {
56 // Save the address of the system page at a known location to be
57 // read by a JTAG.
58 // Only save the system page address if some memory has been reserved
59 if(jtag_syspage_address != NULL_PADDR) {
60 paddr_t *p;
61
62 p = startup_memory_map(sizeof(syspage_paddr), jtag_syspage_address, PROT_READ|PROT_WRITE);
63 *p = syspage_paddr;
64 startup_memory_unmap(p);
65 }
