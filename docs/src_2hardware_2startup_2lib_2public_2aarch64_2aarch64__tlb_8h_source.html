22/* For 64KiB translation granule
23* 512 MiB blocks
24* TLB level 2 block descriptor format:
25* bit[0]: valid
26* bit[1]: 0-block,1-table
27* bit[5:2]: MemAttr
28* bit[10]: Access flag
29* bit[47:29]: addr
30*
31* For 4KiB translation granule
32* 1 GiB blocks
33* TLB level 1 block descriptor format:
34* bit[0]: valid
35* bit[1]: 0-block,1-table
36* bit[5:2]: MemAttr
37* bit[10]: Access flag
38* bit[47:30]: addr
39*/

70/* Construct a translation table to cover 42 bit address space
71* Please NOTE that the tlb needs to be 64K aligned (0x10000)
72*/
73static __inline__ void aarch64_setup_tlb(aarch64_tlb_t *table, uint64_t *tlb)
74{
75 uint64_t base;
76 uint64_t offset = 0;
77
78 while (table && (table->start != (uint64_t)-1)) {
79 base = table->start;
80
81 /* uncacheable */
82 for (; offset < (base >> PAGE_SHIFT); offset++) {
83 tlb[offset] = NX | (offset << PAGE_SHIFT) | VALID | AF;
84 }
85
86 for (; (base >= table->start) && (table->len > base - table->start) && (offset < TLB_SIZE); base += PAGE_SIZE, offset++) {
87 tlb[offset] = (base & BA_MASK)
88 | table->attr
89 | VALID | AF;
90 }
91 table++;
92 }
93
94 /* uncacheable */
95 for (; offset < TLB_SIZE; offset++) {
96 tlb[offset] = NX | (offset << PAGE_SHIFT) | VALID | AF;
97 }
98}
