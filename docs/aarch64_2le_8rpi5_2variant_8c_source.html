25/*
26 * Raspberry PI5 PCIe RP1 supports 64 MSIX interrupt, these interrupts
27 * are mapped to GICv2 interrupt 0xa0~0xdf.
28 *
29 * According RP1 data sheet, each UART0~5 MSIX vector has its MSIn_CFG.IACK_EN set
30 * and requires its MSIn_CFG.IACK set before interrupt is unmasked.
31 */
32#define BCM2712_RP1_PCIE_MSIX_IRQ (160U)
33#define RP1_PCIE_MSIX_UART0_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 25U)
34#define RP1_PCIE_MSIX_UART1_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 42U)
35#define RP1_PCIE_MSIX_UART2_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 43U)
36#define RP1_PCIE_MSIX_UART3_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 44U)
37#define RP1_PCIE_MSIX_UART4_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 45U)
38#define RP1_PCIE_MSIX_UART5_IRQ (BCM2712_RP1_PCIE_MSIX_IRQ + 46U)
39
40#define RP1_PCIE_MSIX_ADDR (0x1f00108000ULL)
41#define RP1_PCIE_MSIX_SIZE (0x8000UL)
42
43#define RP1_PCIE_MSIX_IRQ_25_UART0_SET_REG (0x800U + 0x6cU)
44#define RP1_PCIE_MSIX_IRQ_42_UART1_SET_REG (0x800U + 0xb0U)
45#define RP1_PCIE_MSIX_IRQ_43_UART2_SET_REG (0x800U + 0xb4U)
46#define RP1_PCIE_MSIX_IRQ_44_UART3_SET_REG (0x800U + 0xb8U)
47#define RP1_PCIE_MSIX_IRQ_45_UART4_SET_REG (0x800U + 0xbcU)
48#define RP1_PCIE_MSIX_IRQ_46_UART5_SET_REG (0x800U + 0xc0U)
49#define RP1_PCIE_MSIX_CFG_N_IACK (1U << 2U)
50
51static uintptr_t rpi_msix_cfg_base = (uintptr_t)MAP_FAILED;
52static uint32_t rpi_msix_cfg_set_reg = 0U;
53
54void variant_intr_init(DEV_PL011 *dev) {
55
56 if (dev->intr == RP1_PCIE_MSIX_UART0_IRQ) {
57 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_25_UART0_SET_REG;
58 }
59 else if (dev->intr == RP1_PCIE_MSIX_UART1_IRQ) {
60 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_42_UART1_SET_REG;
61 }
62 else if (dev->intr == RP1_PCIE_MSIX_UART2_IRQ) {
63 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_43_UART2_SET_REG;
64 }
65 else if (dev->intr == RP1_PCIE_MSIX_UART3_IRQ) {
66 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_44_UART3_SET_REG;
67 }
68 else if (dev->intr == RP1_PCIE_MSIX_UART4_IRQ) {
69 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_45_UART4_SET_REG;
70 }
71 else if (dev->intr == RP1_PCIE_MSIX_UART5_IRQ) {
72 rpi_msix_cfg_set_reg = RP1_PCIE_MSIX_IRQ_46_UART5_SET_REG;
73 }
74 else {
75 /* do nothing */
76 rpi_msix_cfg_set_reg = 0U;
77 }
78
79 if (rpi_msix_cfg_set_reg != 0U) {
80 rpi_msix_cfg_base = (uintptr_t)mmap_device_memory (NULL, RP1_PCIE_MSIX_SIZE,
81 (PROT_READ | PROT_WRITE | PROT_NOCACHE), 0, RP1_PCIE_MSIX_ADDR);
82 if (rpi_msix_cfg_base == (uintptr_t)MAP_FAILED) {
83 (void)slogf (_SLOG_SETCODE (_SLOGC_CHAR, 0), _SLOG_NOTICE, "Failed to map RP1 PCIe");
84 }
85 }
86
87 if (dev->is_debug_console != 0U) {
88 const int32_t ret = fdt_debug_console_irq_fix(dev);
89 /* Retrieve IRQ from runtime FDT for default debug console */
90 if (ret != 0) {
91 (void)slogf (_SLOG_SETCODE (_SLOGC_CHAR, 0), _SLOG_ERROR, "Unable to get debug console IRQ from FDT, ret: %d", ret);
92 }
93 }
94
95 return;
96}

98void variant_intr_unmask(void)
99{
100 if ((rpi_msix_cfg_base != (uintptr_t)MAP_FAILED)) {
101 out32(rpi_msix_cfg_base + rpi_msix_cfg_set_reg, RP1_PCIE_MSIX_CFG_N_IACK);
102 }
103}
