34void
35init_qtime_v8gt(unsigned const vcnt_intr, unsigned const hvcnt_intr) {
36 struct qtime_entry *qtime;
37
38 qtime = alloc_qtime();
39
40 timer_start = timer_start_v8gt;
41 timer_diff = timer_diff_v8gt;
42
43 /*
44 * Set up qtime timer rate/scale
45 */
46 if (timer_freq == 0) {
47 timer_freq = aa64_sr_rd32(cntfrq_el0);
48 }
49 invert_timer_freq(qtime, timer_freq);
50
51 /*
52 * Each core has its own generic timer registers, so we force the
53 * system timer to cpu0's timer.
54 */
55 qtime->cycles_per_sec = timer_freq;
56 if (hypervisor_get_required_flags() == (HYP_FLAG_ENABLED | HYP_FLAG_EL2_HOST)) {
57 qtime->intr = hvcnt_intr;
58 } else {
59 qtime->intr = vcnt_intr;
60 }
61 qtime->timer_load_max = 0x7fffffffffffffffull;
62
63 /*
64 * Disable CNTV interrupt
65 */
66 aa64_sr_wr32(cntv_ctl_el0, 0);
67}
