24static int32_t parse_scl_timing(struct scl_timing_param *param, char *popt)
25{
26 int32_t ret = -1;
27 uint32_t high, low, fall;
28
29 errno = 0; /* CERT ERR30-C */
30 high = (uint32_t)strtoul(popt, &popt, 0);
31
32 if ((errno == EOK) && (*popt == ':')) {
33 errno = 0; /* CERT ERR30-C */
34 low = (uint32_t)strtoul(++popt, &popt, 0);
35
36 if ((errno == EOK) && (*popt == ':')) {
37 errno = 0; /* CERT ERR30-C */
38 fall = (uint32_t)strtoul(++popt, &popt, 0);
39
40 if (errno == EOK) {
41 param->high = high;
42 param->low = low;
43 param->fall = fall;
44 ret = 0;
45 }
46 }
47 }
48
49 return ret;
50}

52int32_t dwc_i2c_handle_common_option(dwc_dev_t *dev, int32_t opt)
53{
54 int32_t ret = 0;
55
56 switch (opt) {
57 case (int32_t)'c':
58 errno = 0; /* CERT ERR30-C */
59 dev->clock_khz = (uint32_t)strtoul(optarg, NULL, 0) / 1000U;
60 if (errno != EOK) {
61 ret = -1;
62 break;
63 }
64 break;
65 case (int32_t)'f':
66 if (parse_scl_timing(&dev->fast, optarg) != 0) {
67 logerr("failed to parse SCL timing parameter for fast mode.\n");
68 ret = -1;
69 }
70 break;
71 case (int32_t)'h':
72 errno = 0; /* CERT ERR30-C */
73 dev->sda_hold_time = (uint32_t)strtoul(optarg, NULL, 0);
74 if (errno != EOK) {
75 ret = -1;
76 break;
77 }
78 break;
79#ifdef DWC_SUPPORT_FIXED_SCL
80 case (int32_t)'I':
81 dev->fixed_scl = 1;
82 break;
83#endif
84 case (int32_t)'s':
85 if (parse_scl_timing(&dev->std, optarg) != 0) {
86 logerr("failed to parse SCL timing parameter for std mode.\n");
87 ret = -1;
88 }
89 break;
90 case (int32_t)'v':
91 dev->verbose++;
92 break;
93 default:
94 ret = -1;
95 break;
96 }
97
98 return ret;
99}
