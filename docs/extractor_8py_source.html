1"""AST entity extraction with scope/context tracking.
2
3Walks each parsed AST once, maintaining an explicit scope stack
4(file -> namespaces -> classes). Produces:
5
6- entity nodes (functions, methods, classes, structs, global/member vars)
7- pending call sites and pending inheritance records, resolved later
8 against the global symbol index (never during extraction).
9
10No regex is used anywhere: every decision is driven by Tree-sitter node
11types and named fields via the grammar adapter.
12"""

68 def _walk(self, node, scope: _Scope) -> None:
69 node_type = node.type
70 if node_type == "namespace_definition":
71 self._process_namespace(node, scope)
72 elif node_type in ga.CLASS_SPECIFIER_TYPES:
73 if ga.is_class_definition(node):
74 self._process_class(node, scope)
75 # forward declarations are intentionally not indexed as nodes
76 elif ga.is_function_definition(node):
77 self._process_function(node, scope)
78 elif ga.is_variable_declaration(node):
79 self._process_declaration(node, scope)
80 else:
81 # template_declaration, linkage_specification ("extern \"C\""),
82 # using-declarations, preproc regions, ... : recurse so wrapped
83 # entities are still found. Function bodies are never reached
84 # here because function_definition is handled above and does
85 # not recurse (locals are not graph nodes).
86 for child in node.named_children:
87 self._walk(child, scope)

93 def _process_namespace(self, node, scope: _Scope) -> None:
94 segments = ga.namespace_segments(node.child_by_field_name("name"))
95 child_scope = scope
96 for segment in segments:
97 child_scope = child_scope.child(segment)
98 body = node.child_by_field_name("body")
99 if body is not None:
100 for child in body.named_children:
101 self._walk(child, child_scope)
102 else:
103 for child in node.named_children:
104 self._walk(child, child_scope)

110 def _process_class(self, node, scope: _Scope) -> None:
111 name_node = node.child_by_field_name("name")
112 if name_node is not None:
113 name = compact_text(name_node)
114 else:
115 name = f"<anonymous_{start_line(node)}>"
116 kind = ga.class_kind(node)
117 qualified = f"{scope.qualified}::{name}" if scope.qualified else name
118 class_id = f"{scope.file}::{qualified}"
119
120 self.nodes.append(
121 CodeNode(
122 id=class_id,
123 name=name,
124 qualified_name=qualified,
125 type="class",
126 kind=kind,
127 scope=scope.qualified or None,
128 file=scope.file,
129 line=start_line(node),
130 column=start_column(node),
131 is_definition=True,
132 metadata={"kind": kind},
133 )
134 )
135
136 # inheritance (resolved later against the symbol index)
137 for base in ga.get_base_type_names(node):
138 self.pending_inheritance.append(
139 PendingInheritance(
140 derived_id=class_id,
141 derived_qualified_name=qualified,
142 base_raw=base,
143 file=scope.file,
144 line=start_line(node),
145 scope_segments=scope.segments,
146 )
147 )
148
149 body = node.child_by_field_name("body")
150 if body is not None:
151 child_scope = scope.child(name, class_id=class_id)
152 for child in body.named_children:
153 if child.type == "access_specifier":
154 continue # "public:" / "private:" labels carry no entity
155 self._walk(child, child_scope)

161 def _make_function_node(
162 self, fn_declarator, scope: _Scope, is_definition: bool
163 ) -> Optional[CodeNode]:
164 qtext = ga.declarator_qualified_text(fn_declarator)
165 if not qtext:
166 return None
167 absolute = qtext.startswith("::")
168 qtext = qtext.strip(":")
169 segments = [s for s in qtext.split("::") if s]
170 if not segments:
171 return None
172 name = segments[-1]
173 qualifier = segments[:-1]
174 if absolute:
175 full = segments
176 else:
177 full = list(scope.segments) + qualifier + [name]
178 qualified = "::".join(full)
179 scope_qname = "::".join(full[:-1]) or None
180
181 params = ga.parameters_text(fn_declarator)
182 signature = f"{name}{params}"
183 node_id = f"{scope.file}::{qualified}::{params}"
184
185 is_method = bool(qualifier) or bool(scope.class_ids)
186 kind = "method" if is_method else "function"
187 class_hint = qualifier[-1] if qualifier else (
188 scope.segments[-1] if scope.class_ids else None
189 )
190 if name.startswith("~"):
191 kind = "destructor"
192 elif class_hint and name == class_hint:
193 kind = "constructor"
194
195 return CodeNode(
196 id=node_id,
197 name=name,
198 qualified_name=qualified,
199 type="function",
200 kind=kind,
201 scope=scope_qname,
202 signature=signature,
203 file=scope.file,
204 line=start_line(fn_declarator),
205 column=start_column(fn_declarator),
206 is_definition=is_definition,
207 metadata={
208 "parameters": params,
209 "is_method": is_method,
210 "class_qualifier": qualifier[-1] if qualifier else None,
211 },
212 )

214 def _process_function(self, node, scope: _Scope) -> None:
215 fn_declarator = ga.get_function_declarator(node)
216 if fn_declarator is None:
217 log.debug("unusable function declarator in %s line %d",
218 scope.file, start_line(node))
219 return
220 fn_node = self._make_function_node(fn_declarator, scope, is_definition=True)
221 if fn_node is None:
222 return
223 self.nodes.append(fn_node)
224
225 body = ga.get_function_body(node)
226 if body is not None:
227 local_types = self._parameter_types(fn_declarator)
228 self._collect_calls(body, fn_node, local_types)
229 # Deliberately no recursion into the body: local variables are not
230 # graph nodes and nested entity definitions inside functions are
231 # out of scope for the primary graph.

237 def _process_declaration(self, node, scope: _Scope) -> None:
238 type_node = node.child_by_field_name("type")
239
240 # `class C { ... };` and `struct S { ... } s;` carry the definition
241 # in the type slot of a declaration.
242 if (
243 type_node is not None
244 and type_node.type in ga.CLASS_SPECIFIER_TYPES
245 and ga.is_class_definition(type_node)
246 ):
247 self._process_class(type_node, scope)
248
249 type_name = ga.type_name_of(type_node)
250 is_field = node.type == "field_declaration"
251
252 for raw_declarator in ga.get_declarators(node):
253 declarator = ga.unwrap_declarator(raw_declarator)
254 if declarator is None:
255 continue
256 if declarator.type == ga.FUNCTION_DECLARATOR:
257 # function / method declaration without a body
258 fn_node = self._make_function_node(declarator, scope, is_definition=False)
259 if fn_node is not None:
260 fn_node.kind = "method" if fn_node.metadata["is_method"] else "function"
261 self.nodes.append(fn_node)
262 continue
263 if declarator.type not in ("identifier", "field_identifier"):
264 continue
265 name = compact_text(declarator)
266 if not name:
267 continue
268 qualified = f"{scope.qualified}::{name}" if scope.qualified else name
269 var_id = f"{scope.file}::{qualified}"
270 self.nodes.append(
271 CodeNode(
272 id=var_id,
273 name=name,
274 qualified_name=qualified,
275 type="variable",
276 kind="field" if is_field else "variable",
277 scope=scope.qualified or None,
278 file=scope.file,
279 line=start_line(declarator),
280 column=start_column(declarator),
281 is_definition=True,
282 metadata={"type_name": type_name},
283 )
284 )

290 def _parameter_types(self, fn_declarator) -> Dict[str, str]:
291 """Map parameter names to best-effort type names for a function."""
292 result: Dict[str, str] = {}
293 params = fn_declarator.child_by_field_name("parameters")
294 if params is None:
295 return result
296 for param in params.named_children:
297 if param.type != "parameter_declaration":
298 continue
299 type_name = ga.type_name_of(param.child_by_field_name("type"))
300 declarator = ga.unwrap_declarator(param.child_by_field_name("declarator"))
301 if type_name and declarator is not None and declarator.type == "identifier":
302 result[compact_text(declarator)] = type_name
303 return result

305 def _collect_calls(self, body, caller: CodeNode, local_types: Dict[str, str]) -> None:
306 """Walk a function body once, recording call sites and local types."""
307 scope_segments = tuple(caller.qualified_name.split("::")[:-1])
308 class_scope = caller.scope if caller.metadata.get("is_method") else None
309
310 stack = [body]
311 while stack:
312 node = stack.pop()
313 for i in range(len(node.named_children) - 1, -1, -1):
314 stack.append(node.named_children[i])
315
316 if node.type == "declaration":
317 # track simple local variable types for member-call resolution
318 type_name = ga.type_name_of(node.child_by_field_name("type"))
319 if type_name:
320 for raw in ga.get_declarators(node):
321 d = ga.unwrap_declarator(raw)
322 if d is not None and d.type == "identifier":
323 local_types.setdefault(compact_text(d), type_name)
324 continue
325
326 if not ga.is_call_expression(node):
327 continue
328
329 pending = self._classify_call(
330 node, caller, scope_segments, class_scope, dict(local_types)
331 )
332 if pending is not None:
333 self.pending_calls.append(pending)

335 def _classify_call(
336 self, call_node, caller: CodeNode, scope_segments, class_scope, local_types
337 ) -> Optional[PendingCall]:
338 fn = ga.call_function_node(call_node)
339 if fn is None:
340 return None
341
342 base = dict(
343 caller_id=caller.id,
344 caller_qualified_name=caller.qualified_name,
345 file=caller.file,
346 line=start_line(call_node),
347 scope_segments=scope_segments,
348 class_scope=class_scope,
349 local_types=local_types,
350 )
351
352 fn_type = fn.type
353 if fn_type == "identifier":
354 return PendingCall(kind="direct", name=compact_text(fn), **base)
355
356 if fn_type == "qualified_identifier":
357 raw = compact_text(fn)
358 name = raw.split("::")[-1]
359 return PendingCall(kind="qualified", name=name, raw=raw, **base)
360
361 if fn_type == "template_function":
362 name_node = fn.child_by_field_name("name")
363 if name_node is None:
364 return None
365 raw = compact_text(name_node)
366 name = raw.split("::")[-1]
367 kind = "qualified" if "::" in raw else "direct"
368 return PendingCall(kind=kind, name=name, raw=raw if "::" in raw else None, **base)
369
370 if fn_type == "field_expression":
371 argument, field = ga.field_expression_parts(fn)
372 if field is None:
373 return None
374 name = compact_text(field)
375 object_name = None
376 object_is_this = False
377 if argument is not None:
378 if argument.type == "this":
379 object_is_this = True
380 elif argument.type == "identifier":
381 object_name = compact_text(argument)
382 # any other receiver expression (call result, cast, ...) is
383 # left as unknown: better unresolved than fabricated.
384 return PendingCall(
385 kind="member",
386 name=name,
387 object_name=object_name,
388 object_is_this=object_is_this,
389 **base,
390 )
391
392 # parenthesized callees, function pointers, etc.
393 return PendingCall(
394 kind="unknown", name=compact_text(fn), raw=compact_text(fn), **base
