49static struct co_info *
50get_co_info(void) {
51 struct co_info *next;
52 unsigned i;
53
54 #define INFO_BLOCK 20
55
56 if(co_free_list == NULL) {
57 next = ws_alloc(sizeof(*next)*INFO_BLOCK);
58 if(next == NULL) {
59 crash("No memory for callout information structure.\n");
60 }
61 for(i = 0; i < INFO_BLOCK; ++i) {
62 next->next = co_free_list;
63 co_free_list = next;
64 ++next;
65 }
66 }
67 next = co_free_list;
68 co_free_list = next->next;
69 return(next);
70}

72unsigned
73output_callouts(unsigned *const rwsize) {
74 output_callout_t *one_callout;
75 unsigned size;
76 int sizing = (rwsize) ? 1 : 0;
77
78 one_callout = callout_output_rtn;
79 do {
80 one_callout = (output_callout_t *)one_callout(sizing);
81 } while(one_callout != NULL);
82
83 rw_size = ROUND(rw_size, sizeof(uint64_t));
84 size = ROUND(rtn_size, sizeof(uint64_t));
85 if(sizing) {

87 *rwsize = rw_size;
88 }

92void
93callout_output_one(int sizing, void *rp) {
94 const struct callout_rtn *rtn;
95 callout_fp_t *fp;
96 struct co_info *rw;
97 unsigned rw_off;
98 uintptr_t vaddr;
99
100 fp = rp;
101 rtn = (struct callout_rtn *)*fp;
102 if(rtn == NULL) {
103 return;
104 }
105 if(sizing) {
106 rtn_size += rtn->rtn_size;
107 if(rtn->rw_size != NULL) {
108 rw = rw_addrs;
109 for( ;; ) {
110 if(rw == NULL) break;
111 if(rw->data == rtn->rw_size) return;
112 rw = rw->next;
113 }
114 rw = get_co_info();
115 rw->data = rtn->rw_size;
116 rw->u.rw_off = rw_size;
117 rw->next = rw_addrs;
118 rw_addrs = rw;
119 rw_size += *rtn->rw_size;
120 }
121 return;
122 }
123
124 rw_off = lsp.syspage.p->total_size;
125 if(rtn_size == 0) {
126 memset((uint8_t *)_syspage_ptr + rw_off, 0, rw_size);
127 rtn_size = ROUNDPG(rw_off + rw_size);
128 }
129 memmove((uint8_t *)_syspage_ptr + rtn_size, rtn->rtn_code, rtn->rtn_size);
130 if(rtn->rw_size != NULL) {
131 for(rw = rw_addrs; rw->data != rtn->rw_size; rw = rw->next) {
132 // nothing to do.
133 }
134 rw_off += rw->u.rw_off;
135 }
136 vaddr = (uintptr_t)lsp.system_private.p->kern_syspageptr;
137 if(rtn->patcher != NULL) {
138 void *data;
139 struct co_info *pd;
140
141 data = NULL;
142 pd = patch_data;
143 for( ;; ) {
144 if(pd == NULL) break;
145 if(pd->u.owner == rp) {
146 data = pd->data;
147 break;
148 }
149 pd = pd->next;
150 }
151 rtn->patcher(syspage_paddr, vaddr, rtn_size, rw_off, data, rtn);
152 }
153 vaddr += rtn_size;
154 rtn_size += rtn->rtn_size;
155 *fp = (callout_fp_t)vaddr;
156}

158void
159callout_register_data(void *rp, void *data) {
160 struct co_info *pd;
161 const struct callout_rtn *rtn;
162 callout_fp_t *fp;
163
164 fp = rp;
165 rtn = (struct callout_rtn *)*fp;
166 if(data != NULL && (rtn != NULL) && (rtn->patcher != NULL)) {
167 // Make sure that we're not overriding something already given.
168 for(pd = patch_data; pd != NULL; pd = pd->next) {
169 if(pd->u.owner == rp) return;
170 }
171 pd = get_co_info();
172 pd->next = patch_data;
173 patch_data = pd;
174 pd->u.owner = rp;
175 pd->data = data;
176 }
177}

179void
180callout_reloc_data(void *base, unsigned len, ptrdiff_t diff) {
181 struct co_info *pd;
182
183 // Temp syspage storage has moved around. Adjust registered
184 // addresses.
185 for(pd = patch_data; pd != NULL; pd = pd->next) {
186 if(((uintptr_t)base <= (uintptr_t)pd->u.owner)
187 && (((uintptr_t)base + len) > (uintptr_t)pd->u.owner)) {
188 pd->u.owner = (void *)((uintptr_t)pd->u.owner + diff);
189 }
190 }
191}

194void *
195add_callout(unsigned offset, const struct callout_rtn *callout) {
196 void (**call)(void) = (void (**)(void))lsp.callout.p;
197
198 call += offset / sizeof(*call);
199 *call = (void (*)(void))callout;
200
201 if( offset >= offsetof(struct callout_entry, debug[0])
202 && offset < offsetof(struct callout_entry, debug[1])) {
203 callout_register_data(call, &dbg_device[0]);
204 } else if( offset >= offsetof(struct callout_entry, debug[1])
205 && offset < offsetof(struct callout_entry, debug[2])) {
206 callout_register_data(call, &dbg_device[1]);
207 }
208
209 return (void *)call;
210}

223static output_callout_t *
224callout_output_section(int sizing) {
225 void (**call)(void) = (void (**)(void))lsp.callout.p;
226 unsigned i;
227 unsigned nelts;
228
229 nelts = sizeof(*lsp.callout.p) / sizeof(*call);
230 for(i = 0; i < nelts; ++i, ++call) {
231 if(!sizing) {
232 //
233 // No longer required because of callout_register_data().
234 // Can be removed once we get all the callouts converted.
235 //
236 if(i >= (offsetof(struct callout_entry, debug[1]) / sizeof(callout_fp_t))) {
237 // Kludge: the patcher routines have to know which debug
238 // channel they're working on.
239 patch_channel = 1;
240 }
241 }
242 callout_output_one(sizing, call);
243 }
244 return(NULL);
245}
