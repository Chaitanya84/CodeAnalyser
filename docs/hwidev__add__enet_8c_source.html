27/*******************************************************************************
28* hwidev_add_enet
29*
30* See sys/hwinfo.h for interface description
31*
32* Returns: the offset in the hwinfo section of the added device or HWI_NULL_OFF
33* on error
34*/
35unsigned hwidev_add_enet(const char *device_name, hwiattr_enet_t *attr, unsigned bus_hwi_off)
36{
37 /* we use the (unused) 'device.pnpid' as the device class identifier */
38 unsigned hwi_off = hwidev_add(device_name, hwi_devclass_ENET, bus_hwi_off);
39
40 if ((hwi_off != HWI_NULL_OFF) && (attr != NULL))
41 {
42 const uint8_t zeros[] = {0,0,0,0,0,0};
43 unsigned i;
44 hwitag_add_common(hwi_off, &attr->common);
45 for (i=0; i<attr->num_phys; i++)
46 hwitag_add_nicphyaddr(hwi_off, i);
47 for (i=0; i<attr->num_clks; i++)
48 hwitag_add_inputclk(hwi_off, 0, 1); // make divisor 1, just in case ;)
49
50 /* a non zero NIC address indicates that a tag should be created */
51 if (memcmp(attr->mac, zeros, sizeof(attr->mac)) != 0)
52 hwitag_add_nicaddr(hwi_off, attr->mac, sizeof(attr->mac));
53 }
54 return hwi_off;
55}
