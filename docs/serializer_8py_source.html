1"""Deterministic JSON serialization of the code graph.
2
3All collections are sorted before writing so the same source tree always
4produces byte-identical output (modulo the omitted timestamps, which are

21def build_payload(
22 root: str,
23 nodes: List[CodeNode],
24 links: List[CodeLink],
25 statistics: Dict,
26 diagnostics: Dict,
27) -> dict:
28 sorted_nodes = sorted(nodes, key=lambda n: n.id)
29 sorted_links = sorted(links, key=lambda l: (l.source, l.target, l.type))
30 return {
31 "metadata": {
32 "schema_version": SCHEMA_VERSION,
33 "generator": "code_graph_analyzer",
34 "source_root": root,
35 },
36 "statistics": statistics,
37 "diagnostics": diagnostics,
38 "nodes": [n.to_dict() for n in sorted_nodes],
39 "links": [l.to_dict() for l in sorted_links],
40 }

43def write_json(payload: dict, output_path: Path) -> None:
44 output_path = Path(output_path)
45 output_path.parent.mkdir(parents=True, exist_ok=True)
46 text = json.dumps(payload, indent=2, sort_keys=True, ensure_ascii=False)
47 output_path.write_text(text + "\n", encoding="utf-8")
48 log.info("wrote %s (%d nodes, %d links)",
49 output_path,
50 len(payload["nodes"]),
51 len(payload["links"]))
