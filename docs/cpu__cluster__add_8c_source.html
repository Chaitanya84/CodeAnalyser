36cpu_cluster_add(const char * const name, uint64_t const cpumask)
37{
38 // Users may not create names in the system cluster name space
39 if (*name == SYSTEM_CLUSTER_NAME_PREFIX) return 0;
40
41 if (cpumask == 0) return 0;
42 // The total number of CPUs in the system is not know at this point
43
44 // Add the name to the string table
45 unsigned const name_off = add_string(name);
46
47 struct cluster_entry * const cluster_array = lsp.cluster.p;
48 size_t const num_clusters = lsp.cluster.size / sizeof (*cluster_array);
49
50 size_t i;
51 for (i = 0; i < num_clusters; ++i) {
52 if (cluster_array[i].name == name_off) break;
53 }
54
55 if (i == num_clusters) {
56 grow_syspage_section(&lsp.cluster, sizeof(*cluster_array));
57 }
58
59 cluster_array[i].cpumask = cpumask;
60 cluster_array[i].name = (_Uint16t)name_off;
61 return 1;
62}
