58/* Defines for the XPT function codes, Table 8-2 in the CAM spec. */
59
60/* Common function commands, 0x00 - 0x0F */
61#define XPT_NOOP 0x00 /* Execute Nothing */
62#define XPT_SCSI_IO 0x01 /* Execute the requested SCSI IO */
63#define XPT_GDEV_TYPE 0x02 /* Get the device type information */
64#define XPT_PATH_INQ 0x03 /* Path Inquiry */
65#define XPT_REL_SIMQ 0x04 /* Release the SIM queue that is frozen */
66#define XPT_SASYNC_CB 0x05 /* Set Async callback parameters */
67#define XPT_SDEV_TYPE 0x06 /* Set the device type information */
68#define XPT_SCAN_BUS 0x07 /* Scan SCSI bus */
69
70/* XPT SCSI control functions, 0x10 - 0x1F */
71#define XPT_ABORT 0x10 /* Abort the selected CCB */
72#define XPT_RESET_BUS 0x11 /* Reset the SCSI bus */
73#define XPT_RESET_DEV 0x12 /* Reset the SCSI device, BDR */
74#define XPT_TERM_IO 0x13 /* Terminate the I/O process */
75#define XPT_SCAN_LUN 0x14 /* Scan LUN */
76
77/* HBA engine commands, 0x20 - 0x2F */
78#define XPT_ENG_INQ 0x20 /* HBA engine inquiry */
79#define XPT_ENG_EXEC 0x21 /* HBA execute engine request */
80
81/* Target mode commands, 0x30 - 0x3F */
82#define XPT_EN_LUN 0x30 /* Enable LUN, Target mode support */
83#define XPT_TARGET_IO 0x31 /* Execute the target IO request */
84
85#define XPT_FUNC 0x7F /* TEMPLATE */
86#define XPT_VUNIQUE 0x80 /* All the rest are vendor unique commands */
87#define XPT_DEVCTL 0x81 /* devctl */
