30static output_callout_t *
31callout_output_cache(int sizing) {
32 unsigned i;
33 unsigned num_cache;
34
35 num_cache = lsp.cacheattr.size / sizeof(*lsp.cacheattr.p);
36 for(i = 0; i < num_cache; ++i) {
37 struct cacheattr_entry *cache = &lsp.cacheattr.p[i];
38
39 callout_output_one(sizing, &cache->control);
40 }
41 return(cache_prev_output_rtn);
42}

44int
45add_cache_ways(int next, unsigned flags, unsigned line_size, unsigned num_lines,
46 unsigned ways, const struct callout_rtn *rtn) {
47 struct cacheattr_entry *cache = lsp.cacheattr.p;
48 int num = lsp.cacheattr.size / sizeof(*cache);
49 int i;
50
51 if(cache_prev_output_rtn == NULL) {
52 //Hook into the sizing/writing list
53 cache_prev_output_rtn = callout_output_rtn;
54 callout_output_rtn = (output_callout_t *)callout_output_cache;
55 }
56 //Look for an already existing entry that matches
57 for(i = 0; i < num; ++i, ++cache) {
58 if( cache->next == next
59 && cache->flags == flags
60 && cache->line_size == line_size
61 && cache->ways == ways
62 && cache->num_lines == num_lines
63 && cache->control == (cache_rtn *)rtn) return(i);
64 }
65 //Have to add a new entry
66 cache = grow_syspage_section(&lsp.cacheattr, sizeof(*cache));
67 cache += i;
68 cache->next = next;
69 cache->flags = flags;
70 cache->line_size = line_size;
71 cache->num_lines = num_lines;
72 cache->ways = ways;
73 cache->control = (cache_rtn *)rtn;
74 return(i);
75}
