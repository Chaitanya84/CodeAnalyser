30 * Drive an 8250 style serial chip.

33/*
34 * aspace[]
35 * this array allows us to record, on a per 'dbg_dvice[]' basis, the address
36 * space to use for the access.
37 * It is intentionally not part of the 'chip_info' structure. Refer to the
38 * code review for discussion
39 */
40static unsigned aspace[NUM_ELTS(dbg_device)] = {0};

42static void
43parse_line(unsigned channel, const char *line, unsigned long *baud, unsigned long *clk, unsigned long *div) {
44
45 if(*line != '.' && *line != '\0') {
46 dbg_device[channel].base = strtopaddr(line, (char **)&line, 16);
47 if(*line == '^') dbg_device[channel].shift = strtoul(line+1, (char **)&line, 0);
48 }
49 if(*line == '.') ++line;
50 if(*line != '.' && *line != '\0') *baud = strtoul(line, (char **)&line, 0);
51 if(*line == '.') ++line;
52 if(*line != '.' && *line != '\0') *clk = strtoul(line, (char **)&line, 0);
53 if(*line == '.') ++line;
54 if(*line != '.' && *line != '\0') *div = strtoul(line, (char **)&line, 0);
55}

70void
71init_8250_common(unsigned const channel, unsigned long const baud, unsigned long const clk,
72 unsigned long const div)
73{
74 paddr_t base;
75 unsigned shift;
76
77 /*
78 * FIX ME
79 * For now, a base address >= 64K is considered an MMIO space access
80 * otherwise, we default to I/O accesses
81 */
82 aspace[channel] = (dbg_device[channel].base >= (paddr_t)(64 * 1024));
83
84 base = dbg_device[channel].base;
85 shift = dbg_device[channel].shift;
86
87 /*
88 * FIX ME
89 * For some reason the remainder of this function doesn't work for the
90 * memory mapped UART devices on the Apollo Lake/Broxton platforms that
91 * use MMIO accesses. This needs more investigation
92 */
93 if (aspace[channel] == 0)
94 {
95 chip_access(base, shift, aspace[channel], REG_MS);
96
97 // Wait for all pending characters to be output...
98 do {
99 } while(!(chip_read8(REG_LS) & LSR_TSRE));
100
101 if(baud != 0) {
102 unsigned count = clk / (baud * div);
103
104 // Program divisor latch
105 chip_write8(REG_LC, LCR_DLAB);
106 chip_write8(REG_DL0, count & 0xff);
107 chip_write8(REG_DL1, count >> 8);
108 chip_write8(REG_LC, 0x03);
109 }
110
111 // 8 bits no parity
112 chip_write8(REG_MC, (chip_read8(REG_MC) & 0xE4) | MCR_DTR|MCR_RTS|MCR_OUT2);
113 chip_done();
114 }
115}

117void
118put_8250(int c) {
119 const paddr_t base = dbg_device[0].base;
120 const unsigned shift = dbg_device[0].shift;
121
122 chip_access(base, shift, aspace[0], REG_MS);
123 do {
124 } while(!(chip_read8(REG_LS) & LSR_TXRDY));
125 chip_write8(REG_TX, c);
126 chip_done();
127}
