40static int sdio_soc_config( sdio_hc_t *hc, int idx, unsigned offset )
41{
42 sdio_hc_cfg_t *cfg;
43 hwi_tag *tag;
44 char *name;
45 char *opts;
46
47 cfg = &hc->cfg;
48 opts = NULL;
49 cfg->idx = idx;
50
51 while( ( offset = hwi_next_tag( offset, 1 ) ) != HWI_NULL_OFF ) {
52 tag = hwi_off2tag( offset );
53 name = __hwi_find_string( tag->prefix.name );
54
55 if( !strcmp( name, HWI_TAG_NAME_busattr ) ) {
56 cfg->bus_width = tag->busattr.width;
57 }
58 else if( !strcmp( name, HWI_TAG_NAME_location ) ) {
59 cfg->base_addr[cfg->base_addrs] = tag->location.base;
60 cfg->base_addr_size[cfg->base_addrs] = tag->location.len;
61 cfg->base_addrs++;
62 }
63 else if( !strcmp( name, HWI_TAG_NAME_irq ) ) {
64 cfg->irq[cfg->irqs++] = tag->irq.vector;
65 }
66 else if( !strcmp( name, HWI_TAG_NAME_inputclk ) ) {
67 cfg->clk = tag->inputclk.clk;
68 }
69 else if( !strcmp( name, HWI_TAG_NAME_dll ) ) {
70 strlcpy( cfg->name, __hwi_find_string( tag->dll.name ), sizeof( cfg->name ) );
71 }
72 else if( !strcmp( name, HWI_TAG_NAME_optstr ) ) {
73 if( opts ) {
74 free( opts );
75 }
76 opts = strdup( __hwi_find_string( tag->optstr.string ) );
77 }
78 else if( !strcmp( name, HWI_TAG_NAME_dma ) ) {
79 cfg->dma_chnl[cfg->dma_chnls++] = tag->dma.chnl;
80 }
81 else {
82 // nothing
83 }
84 }
85
86 if( opts ) {
87 sdio_options( hc, opts );
88 free( opts );
89 }
90
91 if( cfg->name[0] == '\0' ) {
92 sdio_slogf( _SLOGC_SDIODI, _SLOG_ERROR, 1, 1, "%s: skipping SOC %d, base 0x%"PRIx64"", __FUNCTION__, idx, cfg->base_addr[0] );
93 return( ENODEV );
94 }
95
96 return( EOK );
97}

99int sdio_soc_scan( void )
100{
101 sdio_hc_t *hc;
102 unsigned item;
103 int occurence;
104
105 item = HWI_NULL_OFF;
106 occurence = 0;
107 hc = TAILQ_FIRST( &sdio_ctrl.hlist );
108 while( 1 ) {
109 item = sdio_soc_hwi( occurence );
110 if( item == HWI_NULL_OFF ) {
111 break;
112 }
113 if( hc == NULL ) {
114 hc = sdio_hc_alloc( );
115 if( hc == NULL ) {
116 return( ENOMEM );
117 }
118 }
119 if( sdio_soc_config( hc, occurence, item ) != EOK ) {
120 sdio_hc_free( hc );
121 }
122 hc = NULL;
123 occurence++;
124 }
125
126 // if we don't find any SOCs in HWI, allocate a
127 // dumby entry so we will try the default product
128 if( TAILQ_FIRST( &sdio_ctrl.hlist ) == NULL ) {
129 if( sdio_hc_alloc( ) == NULL ) {
130 return( ENOMEM );
131 }
132 }
133
134 return( EOK );
135}

137int sdio_soc_device( sdio_hc_t *hc )
138{
139 sdio_hc_cfg_t *cfg;
140 unsigned item;
141 int status;
142
143 cfg = &hc->cfg;
144 status = EOK;
145
146 if( ( cfg->name[0] == '\0' ) && cfg->base_addrs ) {
147 return( ENODEV );
148 }
149
150 if( cfg->idx != -1 ) {
151 item = sdio_soc_hwi( cfg->idx );
152 if( item != HWI_NULL_OFF ) {
153 status = sdio_soc_config( hc, cfg->idx, item );
154 }
155 }
156
157 return( status );
158}
