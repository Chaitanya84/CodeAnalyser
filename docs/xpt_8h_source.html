192/*
193 * The SIM calls this routine to inform the XPT that an async event has occured
194 * and that there may be peripheral drivers which need to be informed.
195 *
196 * - The opcode, path_id, target_id, lun, and data_cnt arguments are 32-bit
197 * values.
198 * - The path_id, target_id, and lun define a nexus for the Async Callback.
199 * - The opcode contains the value for what has happened.
200 * - The buffer_ptr and data_cnt are used to inform the XPT where and how much
201 * data is associated with the opcode.
202 */
203void xpt_async(int opcode, path_id_t path_id, target_id_t target_id, lun_id_t lun,
204 void *buffer_ptr, int data_cnt);

207/*

226/*
227 * This routine is called by the peripheral driver to request that the XPT and
228 * sub-layers be initialized. Once the sub-layers are initialized any subsequent
229 * calls by other peripheral drivers shall quickly return.
230 *
231 * There are no arguments and the CAM Status is either CAM_SUCCESS of CAM_FAILURE.
232 */
233int xpt_init(void);

236/*
237 * This routine is used whenever a peripheral driver needs a CCB (the common data
238 * structure for processing SCSI requests). It returns a pointer to the allocated
239 * CCB which the peripheral driver can now use as the CCB for it's SCSI/XPT
240 * requests. The returned CCB shall be properly initialized for use as a SCSI I/O
241 * Request CCB. The SIM Private Data area shall have been already set up to be
242 * used by the XPT and SIM, and shall not be modified by the peripheral driver.
243 *
244 * There are no arguments and the return value is a pointer to an initialized
245 * CCB.
246 */
247void *xpt_ccb_alloc(void);

250/*
251 * This routine takes a pointer to the CCB that the peripheral driver has
252 * finished with so it can be returned to the CAM subsystem CCB pool.
253 *
254 * The argument is the pointer to the CCB to be freed, there is no CAM Status.
255 */
256void xpt_ccb_free(CCB *freeccb);

265/*
