33static void
34load_bootstraps(struct image_header *ifs_hdr,
35 struct system_private_entry *private) {
36 unsigned i;
37 union image_dirent *dir;
38 paddr_t base_addr;
39 uintptr_t start_vaddr;
40 struct ifs_bootstrap_head *head;
41 struct ifs_bootstrap_data *prev = NULL;
42
43 //
44 // Load all the boot images
45 //
46 dir = (void *)((uint8_t *)ifs_hdr + ifs_hdr->dir_offset);
47 i = 0;
48 if(!(ifs_hdr->flags & IMAGE_FLAGS_INO_BITS)) {
49 crash("Image unsupported - too old");
50 }
51 for(;;) {
52 for( ;; ) {
53 if(dir->attr.size == 0) goto done;
54 if(dir->attr.ino & IFS_INO_BOOTSTRAP_EXE) break;
55 dir = (void *)((uint8_t *)dir + dir->attr.size);
56 }
57 base_addr = full_image_paddr + shdr->startup_size + dir->file.offset;
58 start_vaddr = load_elf(base_addr);
59 if(start_vaddr == ~0UL) {
60 crash("Unable to load boot process '/%s'\n", dir->file.path);
61 /*NOTREACHED*/
62 }
63 if((rifs_flag & RIFS_FLAG_ENABLE) && rifs_save_elf32data(base_addr, dir, i) == -1) {
64 crash("Unable to save boot process elf data\n");
65 }
66 private = lsp.system_private.p;
67 if(prev != NULL) {
68 prev->next_entry = start_vaddr;
69 startup_memory_unmap(prev);
70 } else {
71 first_bootstrap_start_vaddr = start_vaddr;
72 }
73 head = bootstrap_map(start_vaddr-sizeof(*head), sizeof(*head));
74 if(head->signature != IFS_BOOTSTRAP_SIGNATURE) {
75 crash("missing bootstrap signature");
76 }
77 prev = bootstrap_map(head->bootstrap, sizeof(*prev));
78 if(prev->args != 0) {
79 // newer style bootstrap_data
80 struct bootargs_entry *args;
81
82 // Give board specific code a change to modify the
83 // command line arguments for this bootstrap executable
84 args = bootstrap_map(prev->args, 0x2000);
85 tweak_cmdline(args, dir->file.path);
86 startup_memory_unmap(args);
87 }
88 // prev will get unmapped next time around or at end
89 startup_memory_unmap(head);
90 ++i;
91 dir = (void *)((uint8_t *)dir + dir->attr.size);
92 }
93done:
94 if(prev == NULL) {
95 crash("No bootstrap executables found");
96 }
97 startup_memory_unmap(prev);
98}

100void
101init_system_private() {
102 struct system_private_entry *private = set_syspage_section(&lsp.system_private, sizeof(*lsp.system_private.p));
103 unsigned opt;
104 paddr_t start_paddr;
105 unsigned mem;
106 struct image_header *ifs_hdr;
107 paddr_t ifs_paddr;
108 unsigned flags_on = 0;
109 unsigned flags_off = 0;
110 struct cpuinfo_entry *cpu;
111 unsigned i;
112
113 // Propagate the -F switches to cpu 0 first. On x86, if we are disabling PSE
114 // we need to do it before calling load_bootstraps(), otherwise it creates 1-1
115 // mapped area using large pages, see PR 127992
116 optind = 0;
117 // need to keep the full option list because getopt() need the list to parse
118 // the options properly
119 // pass NULL for the option list to ensure that all getopt() calls use the same list of options in main()
120 while((opt = getopt(_argc, _argv, NULL)) != -1) {
121 unsigned flags;
122 switch(opt) {
123 case 'F':
124 if(*optarg == '~') {
125 flags = strtoul(optarg+1, NULL, 0);
126 flags_off |= flags;
127 flags_on &= ~flags;
128 } else {
129 flags = strtoul(optarg, NULL, 0);
130 flags_off &= ~flags;
131 flags_on |= flags;
132 }
133 break;
134 default:
135 break;
136 }
137 }
138
139 lsp.cpuinfo.p[0].flags |= flags_on;
140 lsp.cpuinfo.p[0].flags &= ~flags_off;
141
142 if(shdr->flags1 & STARTUP_HDR_FLAGS1_VIRTUAL) {
143 private->pagesize = CPU_VIRT_PAGESIZE;
144 } else {
145 private->pagesize = CPU_PHYS_PAGESIZE;
146 }
147
148 //
149 // Set the hostname if it hasn't been set yet
150 //
151 if(find_typed_string(_CS_HOSTNAME) < 0) {
152 add_typed_string(_CS_HOSTNAME, "localhost");
153 }
154
155 //
156 // Get the image file system into it's proper position
157 //
158 ifs_paddr = full_image_paddr + shdr->startup_size;
159 ifs_hdr = MAKE_1TO1_PTR(ifs_paddr);
160 // Try and restore the IFS, if enabled
161 if(!(rifs_flag & RIFS_FLAG_ENABLE) || rifs_restore_ifs(ifs_paddr) == -1) {
162 // Normal (full) load of the IFS
163 load_ifs(ifs_paddr);
164 }
165
166 //
167 // Get the second (non-bootable) image file system into it's proper position
168 //
169 if(rifs_flag & RIFS_FLAG_IFS2) {
170 load_ifs2_nonbootable();
171 }
172
173 //
174 // Tell Neutrino where the image file system is located
175 //
176 mem = as_find(AS_NULL_OFF, "memory", NULL);
177 as_add(ifs_paddr, ifs_paddr+ifs_hdr->image_size-1, AS_ATTR_ROM,
178 "imagefs", mem);
179
180 //
181 // Tell Neutrino where the second (non-bootable) image file system is located
182 //
183 if((rifs_flag & RIFS_FLAG_IFS2)) {
184 as_add(ifs2_paddr_dst, ifs2_paddr_dst+ifs2_size-1, AS_ATTR_ROM, "imagefs", mem);
185 }
186
187 //
188 // Tell Neutrino where startup is (was)
189 //
190 as_add(full_ram_paddr, full_ram_paddr+shdr->startup_size-1, AS_ATTR_RAM,
191 "startup", mem);
192
193 //
194 // Tell Neutrino where RAM is reserved for processes in the
195 // image file system.
196 //
197 start_paddr = full_ram_paddr + shdr->startup_size;
198 as_add(start_paddr, full_ram_paddr+shdr->ram_size-1, AS_ATTR_RAM,
199 "bootram", mem);
200
201 load_bootstraps(ifs_hdr, private);
202
203 // Calculate checksum as last thing we do
204 if((rifs_flag & RIFS_FLAG_ENABLE)) {
205 rifs_set_cksum(ifs_hdr);
206 }
207
208 // Update syspage entries with the given fdt only when -u startup option is specified
209 if (fdt_boot_option && (fdt != NULL)) {
210 as_add(fdt_paddr, fdt_paddr + fdt_size - 1, AS_ATTR_ROM, "fdt", mem);
211 alloc_ram(fdt_paddr, fdt_size, 1);
212 }
213
214 //
215 // Add the additional memory blocks specified by the -M option(s) now.
216 //
217 optind = 0;
218 // pass NULL for the option list to ensure that all getopt() calls use the same list of options in main()
219 while((opt = getopt(_argc, _argv, NULL)) != -1) {
220 char *p;
221 paddr_t start;
222 size_t size;
223 paddr_t resmem_addr;
224 size_t resmem_size;
225 uint8_t resmem_flag = 0;
226 paddr_t jtag_syspage_addr;
227
228 switch(opt) {
229 case 'A':
230 private->private_flags |= SYSTEM_PRIVATE_FLAG_ABNORMAL_REBOOT;
231 break;
232 case 'M':
233#if __PTR_BITS__ != 64
234 start = getsize(optarg, &p);
235 if(*p != ',') {
236 crash("missing comma in -M option near '%s'.\n", p);
237 }
238 size = getsize(p + 1, &p);
239 unsigned type = MEMTYPE_RAM;
240 if(*p == ',') {
241 type = strtoul(p + 1, NULL, 0);
242 }
243 if(type == MEMTYPE_RAM) {
244 add_ram(start, size);
245 } else {
246 kprintf("-M option memory type %d is no longer supported - ignored\n", type);
247 }
248#else
249 start = getsize(optarg, &p);
250 if(*p != ',') {
251 crash("missing comma in -M option near '%s'.\n", p);
252 }
253 size = getsize(p + 1, &p);
254 if(*p == ',') {
255 kprintf("-M option: memory types not supported - ignored (%s)\n", p+1);
256 }
257 add_ram(start, size);
258#endif
259 break;
260 case 'r':
261 // reserve user-specified memory from asinfo
262 resmem_addr = getsize(optarg, &p);
263 if(*p == ',') {
264 resmem_size = getsize(p + 1, &p);
265
266 // user can choose to not clear reserved memory
267 if(*p == ',') {
268 resmem_flag = strtoul(p + 1, &p, 2);
269 }
270 jtag_reserve_memory(resmem_addr, resmem_size, resmem_flag);
271 }
272 // else: ignore the option - a size is required
273 break;
274 case 'j':
275 // save syspage address to be read by a JTAG
276 jtag_syspage_addr = getsize(optarg, &p);
277 jtag_reserve_syspage_addr(jtag_syspage_addr);
278 break;
279 case 'Z':
280 lsp.qtime.p->flags |= QTIME_FLAG_TICKLESS;
281 break;
282 }
283 }
284
285 //
286 // If the user wanted to reserve some memory, allocate it now.
287 // Unless specified, get memory from the top of available memory
288 // so that we use as little of any one-to-one mapping area
289 // as possible.
290 //
291
292 while (reserved_ram_list != NULL) {
293 if(reserved_ram_list->size > 0) {
294 reserve_ram(reserved_ram_list->size, reserved_ram_list->align,reserved_ram_list->name, private->pagesize, reserved_ram_list->low_region);
295 }
296 reserved_ram_list = reserved_ram_list->next;
297 }
298
299 // Add memory allocation specific to hardware mitigation
300 init_mitigation_mem();
301
302 //
303 // Start any AP's.
304 //
305 smp_hook_rtn();
306
307 // Propagate the -F switches. Has to be done down here because the
308 // smp_hook_rtn() call just above is the one that sets the flags
309 // field for the AP's.
310 cpu = lsp.cpuinfo.p;
311 for(i = 0; i < lsp.syspage.p->num_cpu; ++i, ++cpu) {
312 cpu->flags |= flags_on;
313 cpu->flags &= ~flags_off;
314 }
315
316 //
317 // Add the asinfo "sysram" entries.
318 //
319 add_sysram();
320
321 hypervisor_syspage_init();
322
323 //
324 // Allocate the real storage location for the system & cpu pages.
325 // We can't call set_syspage_section()/grow_syspage_section() after this.
326 //
327 alloc_syspage_memory();
328
329 // Save the address of the system page to be retrieved by a JTAG.
330 // NOTE: Must be stored after alloc_syspage_memory.
331 jtag_store_syspage_addr();
332}
