26/*******************************************************************************
27* hwitag_set_avail_ivec
28*
29* if the vector tag <ivec_idx> is free for the device corresponding to <hwi_off>,
30* then set that tag to <vector>. If <ivec_idx> < 0, then the next available
31* HWI_TAG_NAME_irq tag that is free will be used. This function will only add
32* the vector to a free tag (ie. current value == HWI_ILLEGAL_VECTOR). If no free
33* tags are found, the function will fail. You can use hwi_set_ivec() to force a
34* known value for a specific tag
35*
36* Returns: -1 on error or the vector tag index that was set (base 0)
37*
38*/
39int hwitag_set_avail_ivec(unsigned hwi_off, int ivec_idx, uint32_t vector)
40{
41 unsigned tag_idx = (ivec_idx < 0) ? 0 : ivec_idx;
42 hwi_tag *tag;
43
44 while ((tag = hwi_tag_find(hwi_off, HWI_TAG_NAME_irq, &tag_idx)) != NULL)
45 {
46 /* if we find a free tag, use it */
47 if (tag->irq.vector == HWI_ILLEGAL_VECTOR) {
48 tag->irq.vector = vector;
49 if (debug_flag >= 2) {
50 hwi_tag *device_tag = hwi_off2tag(hwi_off);
51 kprintf("%s%d, ivec idx %d == %d\n", __hwi_find_string(device_tag->item.itemname),
52 hwi_find_unit(hwi_off), tag_idx - 1, tag->irq.vector);
53 }
54 return (tag_idx - 1);
55 }
56 /* if the caller wanted to add at a specific index, we have failed
57 * otherwise keep searching for a free tag */
58 if (ivec_idx >= 0) break;
59 }
60 return -1;
61}
