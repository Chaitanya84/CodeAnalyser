73 /*
74 * Initialize the devinit to raw mode
75 */
76 ttc(TTC_INIT_RAW, &devinit, 0);

78 while (optind < argc) {
79 /*
80 * Process dash options.
81 */
82 while ((opt = getopt(argc, argv, IO_CHAR_SERIAL_OPTIONS "t:T:D")) != -1) {
83 switch (opt) {
84 case 't':
85 errno = EOK;
86 val = strtoul(optarg, NULL, 0);
87 if (errno != EOK) {
88 (void)fprintf(stderr, "Invalid receive FIFO trigger level, errno: %d\n", errno);
89 break;
90 }
91 rx_fifo = (unsigned)val;
92 if (rx_fifo > 4)
93 rx_fifo = 2;
94 break;
95
96 case 'T':
97 errno = EOK;
98 val = strtoul(optarg, NULL, 0);
99 if (errno != EOK) {
100 (void)fprintf(stderr, "Invalid number of characters to send to transmit FIFO, errno: %d\n", errno);
101 break;
102 }
103 tx_fifo = (unsigned)val;
104 if (tx_fifo > 4)
105 tx_fifo = 2;
106 break;
107
108 case 'D':
109 devinit.is_debug_console = 1;
110 break;
111
112 default:
113 /* the other options has parsed by ttc function */
114 (void)ttc(TTC_SET_OPTION, &devinit, opt);
115 break;
116 }
117 }
118
119 devinit.tty.fifo = (rx_fifo << 3) | tx_fifo;
120
121 /*
122 * Process ports and interrupts.
123 */
124 while (optind < argc && *(optarg = argv[optind]) != '-') {
125 errno = EOK;
126 val = strtoul(optarg, &optarg, 16);
127 if (errno != EOK) {
128 (void)fprintf(stderr, "Invalid serial port base address, errno: %d\n", errno);
129 port_error = 1L;
130 break;
131 }
132 devinit.tty.port = val;
133 if (*optarg == '^') {
134 errno = EOK;
135 val = strtoul(optarg + 1, &optarg, 0);
136 if (errno != EOK) {
137 (void)fprintf(stderr, "Invalid serial port shift, errno: %d\n", errno);
138 port_error = 1L;
139 break;
140 }
141 devinit.tty.port_shift = (unsigned)val;
142 }
143 if (*optarg == ',') {
144 errno = EOK;
145 val = strtoul(optarg + 1, NULL, 0);
146 if (errno != EOK) {
147 (void)fprintf(stderr, "Invalid serial port interrupt, errno: %d\n", errno);
148 port_error = 1L;
149 break;
150 }
151 devinit.tty.intr = (unsigned)val;
152 }
153 create_device(&devinit);
154 devinit.tty.unit++;
155 ++numports;
156 ++optind;
157 }
158 if (port_error != 0L) {
159 break;
160 }
161 }
162
163 if (numports == 0) {
164 link = NULL;
165 devinit.tty.fifo = (rx_fifo << 3) | tx_fifo;
166 while (1) {
167 link = query_default_device(&devinit, link);
168 if (link == NULL)
169 break;
170 create_device(&devinit);
171 devinit.tty.unit++;
172 ++numports;
173 }
174 }
175
176 return numports;
177}
