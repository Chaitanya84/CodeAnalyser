26void write_port(DEV_PL011 *dev, int reg, unsigned val)
27{
28 switch (dev->port_size) {
29 case 2:
30 out32(dev->base + reg, val);
31 break;
32 case 1:
33 out16(dev->base + reg, val);
34 break;
35 default:
36 out8 (dev->base + reg, val);
37 break;
38 }
39}

41unsigned read_port(DEV_PL011 *dev, int reg)
42{
43 switch (dev->port_size) {
44 case 2:
45 return in32(dev->base + reg);
46 case 1:
47 return in16(dev->base + reg);
48 default:
49 return in8 (dev->base + reg);
50 }
51}

57query_default_device(TTYINIT_PL011 *dip, void *link)
58{
59 switch ((uintptr_t)link) {
60 case 0:
61 dip->tty.port = 0x10009000;
62 dip->tty.intr = 36;
63 return (void *)1;
64 case 1:
65 dip->tty.port = 0x1000A000;
66 dip->tty.intr = 37;
67 return (void *)2;
68 }
69
70 return NULL;
71}

73void
74create_device(TTYINIT_PL011 *dip)
75{
76 DEV_PL011 *dev;
77
78 /*
79 * Get a device entry and the input/output buffers for it.
80 */
81 dev = (void *) malloc(sizeof(*dev));
82
83 if (dev == NULL) {
84 printf("%s(%d): Memory allocation failed\n", __func__, __LINE__);
85 exit(EXIT_FAILURE);
86 }
87
88 memset(dev, 0, sizeof(*dev));
89
90 /*
91 * Get buffers.
92 */
93 dev->tty.ibuf.buff = malloc(dev->tty.ibuf.size = dip->tty.isize);
94 if (dev->tty.ibuf.buff == NULL) {
95 perror("PL011 UART: input buffer MAP_FAILED\n");
96 exit(1);
97 }
98 dev->tty.ibuf.head = dev->tty.ibuf.buff;
99 dev->tty.ibuf.tail = dev->tty.ibuf.buff;
100
101 dev->tty.obuf.buff = malloc(dev->tty.obuf.size = dip->tty.osize);
102 if (dev->tty.obuf.buff == NULL) {
103 perror("PL011 UART: output buffer MAP_FAILED\n");
104 exit(1);
105 }
106 dev->tty.obuf.head = dev->tty.obuf.buff;
107 dev->tty.obuf.tail = dev->tty.obuf.buff;
108
109 dev->tty.cbuf.buff = malloc(dev->tty.cbuf.size = dip->tty.csize);
110 if (dev->tty.cbuf.buff == NULL) {
111 perror("PL011 UART: canonical buffer MAP_FAILED\n");
112 exit(1);
113 }
114 dev->tty.cbuf.head = dev->tty.cbuf.buff;
115 dev->tty.cbuf.tail = dev->tty.cbuf.buff;
116
117 dev->tty.highwater = dev->tty.ibuf.size - (dev->tty.ibuf.size < 128 ? dev->tty.ibuf.size/4 : 100);
118
119 strlcpy(dev->tty.name, dip->tty.name, sizeof(dev->tty.name));
120
121 dev->tty.baud = dip->tty.baud;
122 dev->fifo = dip->tty.fifo;
123 dev->tty.fifo = dip->tty.fifo;
124
125 dev->intr = dip->tty.intr;
126 dev->clk = dip->tty.clk;
127 dev->div = dip->tty.div;
128
129 dev->tty.flags = EDIT_INSERT | LOSES_TX_INTR;
130 dev->tty.c_cflag = dip->tty.c_cflag;
131 dev->tty.c_iflag = dip->tty.c_iflag;
132 dev->tty.c_lflag = dip->tty.c_lflag;
133 dev->tty.c_oflag = dip->tty.c_oflag;
134 dev->tty.verbose = dip->tty.verbose;
135 dev->fifosize = FIFOSIZE;
136 dev->loopback = dip->loopback;
137 dev->drt = dip->drt;
138 dev->prio = dip->prio;
139 dev->fifo_reg = dip->fifo_reg;
140 dev->dma_enable = dip->dma_enable;
141 dev->dma_request_rx = dip->dma_request_rx;
142 dev->dma_request_tx = dip->dma_request_tx;
143 dev->dma_xfer_size = dip->dma_xfer_size;
144 dev->chan_rx = dip->chan_rx;
145 dev->chan_tx = dip->chan_tx;
146 dev->port_size = dip->tty.port_shift;
147 dev->is_debug_console = dip->is_debug_console;
148
149#ifdef USE_DMA
150 if(dev->dma_enable){
151 pthread_mutex_init (&dev->dma_lock, NULL);
152 if(dev->tty.ibuf.size < dev->dma_xfer_size*2){
153 printf("input buffer size has to be equal or greater than 2 times of the dma buffer size\n");
154 pthread_mutex_destroy (&dev->dma_lock);
155 free(dev);
156 return ;
157 }
158 dev->tty.highwater = dev->tty.ibuf.size - max(dev->tty.ibuf.size/4, dev->dma_xfer_size);
159
160 nanospin_calibrate(0);
161 //Initialize DMA libs
162 if (get_dmafuncs (&dev->dmafuncs, sizeof (dev->dmafuncs)) == -1){
163 printf("failed to get DMA lib functions\n");
164 pthread_mutex_destroy (&dev->dma_lock);
165 free(dev);
166 return ;
167 }
168 if (dev->dmafuncs.init (NULL) == -1) {
169 printf("failed to init DMA lib functions\n");
170 pthread_mutex_destroy (&dev->dma_lock);
171 free(dev);
172 return ;
173 }
174 if(dev->dma_enable & DMA_TX_ENABLE){
175
176 if ((dev->tx_chid = ChannelCreate(_NTO_CHF_DISCONNECT | _NTO_CHF_UNBLOCK)) == -1 ||
177 (dev->tx_coid = ConnectAttach(0, 0, dev->tx_chid, _NTO_SIDE_CHANNEL, 0)) == -1)
178 {
179 fprintf(stderr, "Channel create failed\n");
180 pthread_mutex_destroy (&dev->dma_lock);
181 free(dev);
182 return ;
183 }
184
185 dev->event.sigev_notify = SIGEV_PULSE;
186 dev->event.sigev_coid = dev->tx_coid;
187 dev->event.sigev_code = TX_DMA_PULSE;
188 dev->event.sigev_priority = dev->prio+1;
189 if(dev->chan_tx==-1){
190 dev->dma_chn_tx = dev->dmafuncs.channel_attach (NULL, &dev->event, &dev->dma_request_tx, 0,
191 DMA_ATTACH_EVENT_ON_COMPLETE|DMA_ATTACH_ANY_CHANNEL);
192 }else{
193 dev->dma_chn_tx = dev->dmafuncs.channel_attach (NULL, &dev->event, &dev->dma_request_tx, dev->chan_tx,
194 DMA_ATTACH_EVENT_ON_COMPLETE|DMA_ATTACH_PRIORITY_STRICT);
195 }
196 if (dev->dma_chn_tx == NULL){
197 printf("Unable to attach to DMA Channel dev->chan_tx %d\n", dev->chan_tx);
198 pthread_mutex_destroy (&dev->dma_lock);
199 free(dev);
200 return ;
201 }
202 dev->dmafuncs.xfer_abort(dev->dma_chn_tx);
203 if(dev->dmafuncs.alloc_buffer(dev->dma_chn_tx, &dev->buf_tx, dev->dma_xfer_size, 0)==-1){
204 printf("%s(%d): alloc tx dma buffer failed\n", __func__, __LINE__);
205 pthread_mutex_destroy (&dev->dma_lock);
206 free(dev);
207 return ;
208 }
209
210 dev->tinfo_tx.src_flags = DMA_ADDR_FLAG_MEMORY;
211 dev->tinfo_tx.dst_flags = DMA_ADDR_FLAG_NO_INCREMENT|DMA_ADDR_FLAG_DEVICE;
212 dev->tinfo_tx.dst_fragments = 1;
213 dev->tinfo_tx.src_fragments = 1;
214 dev->tinfo_tx.xfer_unit_size = 1;
215 dev->tinfo_tx.mode_flags = 0;
216 dev->dst_tx.paddr = dev->fifo_reg;
217 dev->dst_tx.len = 0;
218 dev->tinfo_tx.src_addrs = &dev->buf_tx;
219 dev->tinfo_tx.dst_addrs = &dev->dst_tx;
220 }
221
222 if(dev->dma_enable & DMA_RX_ENABLE){
223 if(dev->chan_rx==-1){
224 dev->dma_chn_rx = dev->dmafuncs.channel_attach (NULL, NULL, &dev->dma_request_rx, 0,
225 DMA_ATTACH_ANY_CHANNEL);
226 }else{
227 dev->dma_chn_rx = dev->dmafuncs.channel_attach (NULL, NULL, &dev->dma_request_rx, dev->chan_rx,
228 DMA_ATTACH_PRIORITY_STRICT);
229 }
230 if (dev->dma_chn_rx == NULL){
231 printf("Unable to attach to DMA Channel dev->chan_rx %d\n", dev->chan_rx);
232 pthread_mutex_destroy (&dev->dma_lock);
233 free(dev);
234 return ;
235 }
236 dev->dmafuncs.xfer_abort(dev->dma_chn_rx);
237
238 if(dev->dmafuncs.alloc_buffer(dev->dma_chn_rx, &dev->buf_rx[0], dev->dma_xfer_size, 0)==-1){
239 printf("%s(%d): alloc rx dma buffer failed\n", __func__, __LINE__);
240 pthread_mutex_destroy (&dev->dma_lock);
241 free(dev);
242 return ;
243 }
244 if(dev->dmafuncs.alloc_buffer(dev->dma_chn_rx, &dev->buf_rx[1], dev->dma_xfer_size, 0)==-1){
245 printf("%s(%d): alloc rx dma buffer failed\n", __func__, __LINE__);
246 pthread_mutex_destroy (&dev->dma_lock);
247 free(dev);
248 return ;
249 }
250
251 dev->tinfo_rx.dst_flags = DMA_ADDR_FLAG_MEMORY;
252 dev->tinfo_rx.src_flags = DMA_ADDR_FLAG_NO_INCREMENT|DMA_ADDR_FLAG_DEVICE;
253 dev->tinfo_rx.dst_fragments = 1;
254 dev->tinfo_rx.src_fragments = 1;
255 dev->tinfo_rx.xfer_unit_size = 1;
256 dev->tinfo_rx.mode_flags = 0;
257 }
258 }
259#endif
260 /*
261 * Map device registers
262 */
263 dev->base = (uintptr_t)mmap_device_memory (0, PL011_SIZE,
264 PROT_READ | PROT_WRITE | PROT_NOCACHE, 0, dip->tty.port);
265 if (dev->base == (uintptr_t)MAP_FAILED) {
266 perror("PL011 UART: MAP_FAILED\n");
267#ifdef USE_DMA
268 if(dev->dma_enable)
269 pthread_mutex_destroy (&dev->dma_lock);
270#endif
271 exit(1);
272 }
273
274 /*
275 * Initialize termios cc codes to an ANSI terminal.
276 */
277 ttc(TTC_INIT_CC, &dev->tty, 0);
278
279 /*
280 * Initialize the device's name.
281 * Assume that the basename is set in device name. This will attach
282 * to the path assigned by the unit number/minor number combination
283 */
284 ttc(TTC_INIT_TTYNAME, &dev->tty, (int)(SET_NAME_NUMBER(dip->tty.unit) | NUMBER_DEV_FROM_USER));
285
286 /*
287 * Only setup IRQ handler for non-pcmcia devices.
288 * Pcmcia devices will have this done later when card is inserted.
289 */
290 if (dip->tty.port != 0 && dip->tty.intr != _NTO_INTR_SPARE) {
291 ser_attach_intr(dev);
292 ser_stty(dev);
293 }
294
295 /*
296 * Attach the resource manager
297 */
298 ttc(TTC_INIT_ATTACH, &dev->tty, 0);
299
300#ifdef USE_DMA
301 if ((dev->dma_enable & DMA_RX_ENABLE) && dev->drt != 0){
302 pthread_attr_t tattr;
303 struct sched_param param;
304 pthread_attr_init (&tattr);
305 pthread_attr_setschedpolicy (&tattr, SCHED_RR);
306 param.sched_priority = dev->prio;
307 pthread_attr_setschedparam (&tattr, &param);
308 pthread_attr_setinheritsched (&tattr, PTHREAD_EXPLICIT_SCHED);
309 pthread_attr_setdetachstate (&tattr, PTHREAD_CREATE_DETACHED);
310 if (pthread_create (NULL, &tattr, (void *) timer_thread, (void *)dev) != EOK)
311 {
312 printf("devc-serpl0o11: Cleanup thread create failed\n");
313 ser_detach_intr(dev);
314 pthread_mutex_destroy (&dev->dma_lock);
315 free(dev);
316 exit(1);
317 }
318 }
319
320 if ((dev->dma_enable & DMA_TX_ENABLE) ){
321 pthread_attr_t tattr;
322 struct sched_param param;
323 pthread_attr_init (&tattr);
324 pthread_attr_setschedpolicy (&tattr, SCHED_RR);
325 param.sched_priority = dev->prio;
326 pthread_attr_setschedparam (&tattr, &param);
327 pthread_attr_setinheritsched (&tattr, PTHREAD_EXPLICIT_SCHED);
328 pthread_attr_setdetachstate (&tattr, PTHREAD_CREATE_DETACHED);
329 if (pthread_create (NULL, &tattr, (void *) tx_dma_thread, (void *)dev) != EOK)
330 {
331 printf("devc-serpl0o11: Cleanup thread create failed\n");
332 ser_detach_intr(dev);
333 pthread_mutex_destroy (&dev->dma_lock);
334 free(dev);
335 exit(1);
336 }
337 }
338#endif
339
340}
