24i2c_status_t dwc_i2c_send(void *hdl, void *buf, uint32_t len, uint32_t stop)
25{
26 dwc_dev_t *dev = hdl;
27 i2c_status_t ret = I2C_STATUS_ERROR;
28
29 (void)stop;
30
31 if (len <= 0U) {
32 return I2C_STATUS_DONE;
33 }
34
35 dev->txlen = len; // how many bytes for slave transmit (isend)
36 dev->rxlen = 0; // how many bytes for slave receive (irecv)
37 dev->xlen = dev->txlen + dev->rxlen; // how many bytes for total transaction (isend and irecv)
38 dev->wrlen = 0; // how many cmds have been write to TxFIFO
39 dev->rdlen = 0; // how many data have been read from RxFIFO
40 dev->txbuf = buf;
41 dev->status = 0;
42
43 /* Active I2C bus */
44 if (0 != dwc_i2c_bus_active(dev)) {
45 return I2C_STATUS_BUSY;
46 }
47
48 /* Clear and enable interrupts */
49 (void)i2c_reg_read32(dev, DW_IC_CLR_INTR);
50 i2c_reg_write32(dev, DW_IC_INTR_MASK, DW_IC_INTR_DEFAULT_MASK);
51
52 /* Wait for transaction complete event */
53 ret = dwc_i2c_wait_complete(dev);
54
55 /* Disabled interrupts */
56 i2c_reg_write32(dev, DW_IC_INTR_MASK, 0);
57
58 /* Disable the I2C adapter */
59 (void)dwc_i2c_enable(dev, DW_IC_ENABLE_STATUS_DISABLE);
60
61 return ret;
62}
