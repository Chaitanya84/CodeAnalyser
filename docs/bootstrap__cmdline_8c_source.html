28static int
29do_adjust(struct bootargs_entry *bap, const char *old, const char *new,
30 unsigned start, int8_t *countp) {
31 unsigned total_len;
32 unsigned add_len;
33 unsigned sub_len;
34 int diff;
35 char *src;
36 char *p;
37 unsigned num = *countp;
38
39 p = bap->args;
40 while(start != 0) {
41 p += strlen(p) + 1;
42 --start;
43 };
44 while(num != 0) {
45 sub_len = strlen(p) + 1;
46 if((old != NULL) && (strcmp(old, p) == 0)) goto do_operation;
47 p += sub_len;
48 --num;
49 }
50 if(old == NULL) {
51 // adding new one
52 sub_len = 0;
53 goto do_operation;
54 }
55 return 0;
56
57do_operation:
58 if(new == NULL) {
59 // deleting entry
60 *countp -= 1;
61 add_len = 0;
62 } else {
63 // adding/replacing entry
64 add_len = strlen(new) + 1;
65 if(old == NULL) {
66 // adding entry
67 *countp += 1;
68 }
69 }
70 total_len = bap->size_lo + (bap->size_hi << 8);
71 diff = add_len - sub_len;
72 src = p - diff;
73 memmove(p, src, &bap->args[total_len - offsetof(struct bootargs_entry, args)] - src);
74 if(add_len > 0U) {
75 memcpy(p, new, add_len);
76 }
77 total_len += diff;
78 bap->size_lo = total_len & 0xff;
79 bap->size_hi = total_len >> 8;
80 return 1;
81}
