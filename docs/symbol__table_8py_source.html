34 def add(self, node: CodeNode) -> bool:
35 """Index one node. Returns False for duplicate ids.
36
37 When a declaration and a definition share an id (declared inside a
38 class body and defined out-of-class in the same file), the
39 definition replaces the declaration so the index and defines edges
40 point at the definition.
41 """
42 if node.id in self.nodes:
43 existing = self.nodes[node.id]
44 self.duplicate_ids.append(node.id)
45 if existing.is_definition or not node.is_definition:
46 log.debug("duplicate symbol id skipped: %s", node.id)
47 return False
48 # replace declaration with the definition in every index
49 for index, key in self._index_entries(existing):
50 entries = index.get(key)
51 if entries and existing in entries:
52 entries.remove(existing)
53 del self.nodes[node.id]
54 self._index(node)
55 return True

57 def _index_entries(self, node: CodeNode):
58 entries = [
59 (self.by_name, node.name),
60 (self.by_qualified_name, node.qualified_name),
61 (self.by_file, node.file),
62 ]
63 if node.type == "function":
64 entries.append((self.functions_by_name, node.name))
65 entries.append((self.functions_by_qualified_name, node.qualified_name))
66 elif node.type == "class":
67 entries.append((self.classes_by_name, node.name))
68 entries.append((self.classes_by_qualified_name, node.qualified_name))
69 elif node.type == "variable":
70 entries.append((self.variables_by_name, node.name))
71 return entries

87 def _prefer_definition(candidates: List[CodeNode]) -> Optional[CodeNode]:
88 """Pick the unique definition among same-named candidates.
89
90 Returns the single definition when exactly one exists; otherwise
91 the single candidate; otherwise None (ambiguous).
92 """
93 definitions = [n for n in candidates if n.is_definition]
94 pool = definitions or candidates
95 if len(pool) == 1:

97 return None

107 def lookup_class(self, name: str, scope_segments=()):
108 """Resolve a (possibly relative) class name to a unique class node.
109
110 Tries longest enclosing-scope prefix first, then the bare qualified
111 name, then a unique unqualified match. Returns (node, ambiguous).
112 """
113 segments = list(scope_segments)
114 for i in range(len(segments), -1, -1):
115 prefix = "::".join(segments[:i])
116 candidate = f"{prefix}::{name}" if prefix else name
117 found = self.classes_by_qualified_name.get(candidate)
118 if found:
119 if len(found) == 1:
120 return found[0], False
121 return None, True
122 found = self.classes_by_name.get(name, [])
123 if len(found) == 1:
124 return found[0], False
125 if found:
126 return None, True
127 return None, False

129 def variable_type_name(self, name: str, scope_segments=()) -> Optional[str]:
130 """Best-effort type name of a variable visible from a scope."""
131 candidates = self.variables_by_name.get(name, [])
132 typed = [v for v in candidates if v.metadata.get("type_name")]
133 if not typed:
134 return None
135 # Prefer the variable whose scope is the longest prefix of the
136 # caller's scope chain (nearest enclosing scope wins).
137 caller_scope = "::".join(scope_segments)
138
139 def scope_distance(var: CodeNode) -> int:
140 scope = var.scope or ""
141 if caller_scope == scope:
142 return 0
143 if caller_scope.startswith(scope + "::") or not scope:
144 return 1
145 return 2
146
147 typed.sort(key=lambda v: (scope_distance(v), -len(v.scope or "")))
148 return typed[0].metadata["type_name"]
