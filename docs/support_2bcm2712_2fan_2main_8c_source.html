49{

81 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "Failed to detaching process as a daemon, errno: %d", errno);
82 status = errno;
83 goto done;
84 }
85
86 /* Drop the abilities now that the resource manager is set up */
87 status = procmgr_ability(0, PROCMGR_ADN_ROOT | PROCMGR_ADN_NONROOT | PROCMGR_AOP_DENY |

89 if (status != EOK) {
90 (void)slogf(_SLOGC_PWM, _SLOG_ERROR, "Dropping procmgr abilities failed, status: %d", status);

95
96 if (verbose > 0) {
97 (void)slogf(_SLOGC_PWM, _SLOG_DEBUG1, "Exited resource manager loop");
98 }
99
100done:

102 resmgr_deinit();
103 pwm_fan_deinit(&dev);
104 return ((status == EOK) ? EXIT_SUCCESS : EXIT_FAILURE);
